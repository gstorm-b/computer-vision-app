92 bool fromJson(const QJsonObject &obj)
93 {
94 m_role = taskDeviceRoleFromString(obj["role"].toString());
95 m_deviceId = obj["deviceId"].toString();
96 m_cameraNumber = obj["cameraNumber"].toInt(0);
97
98 if (m_role == TaskDeviceRole::Unknown)
99 return false;
100
101 // Reject malformed/hostile imports: over-long device ids and out-of-range
102 // camera numbers. Invalid bindings are dropped by the bool-return
103 // convention in TaskDeviceBindings::fromJson().
104 if (m_deviceId.size() > kMaxDeviceIdLength)
105 return false;
106
107 if (m_role == TaskDeviceRole::CameraNumber &&
108 (m_cameraNumber < kMinCameraNumber || m_cameraNumber > kMaxCameraNumber))
109 return false;
110
111 return true;
112 }
