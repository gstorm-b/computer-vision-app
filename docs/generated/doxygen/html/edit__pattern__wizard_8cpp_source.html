372QWidget *EditPatternWizard::buildStepBox() {
373 auto *page = new QWidget;
374 auto *lay = new QHBoxLayout(page);
375 lay->setSpacing(18); lay->setContentsMargins(0, 0, 0, 0);
376
377 m_boxCanvas = new AddPatternImageCanvas;
378 m_boxCanvas->setMode(AddPatternImageCanvas::Box);
379 m_boxCanvas->setMinimumSize(CW, CH);
380 m_boxCanvas->setImage(m_old.image);
381 m_boxCanvas->setPick({m_new.pickX, m_new.pickY});
382 m_boxCanvas->setBoxConfig(m_new.pickBoxW, m_new.pickBoxH,
383 m_new.pickBoxDist, m_new.pickBoxAngle);
384 // Canvas-driven box edits (drag body / corners / rotation handle) feed the
385 // spin boxes, which then echo into m_new via onBoxChanged().
386 connect(m_boxCanvas, &AddPatternImageCanvas::boxChanged,
387 this, [this](double w, double h, double d, double a) {
388 if (m_boxWSpin) {
389 QSignalBlocker b1(m_boxWSpin), b2(m_boxHSpin),
390 b3(m_boxDistSpin), b4(m_boxAngleSpin);
391 m_boxWSpin->setValue(w);
392 m_boxHSpin->setValue(h);
393 m_boxDistSpin->setValue(d);
394 m_boxAngleSpin->setValue(a);
395 }
396 m_new.pickBoxW = w; m_new.pickBoxH = h;
397 m_new.pickBoxDist = d; m_new.pickBoxAngle = a;
398 updateFooterStatus();
399 });
400 // The picking centre is draggable on the box canvas too (image is locked,
401 // so coords are plain image pixels). Echo the Pick-step spin boxes.
402 connect(m_boxCanvas, &AddPatternImageCanvas::pickChanged,
403 this, [this](const QPoint &p, const QPoint &) {
404 m_new.pickX = p.x(); m_new.pickY = p.y();
405 if (m_pickXSpin) {
406 QSignalBlocker b1(m_pickXSpin), b2(m_pickYSpin);
407 m_pickXSpin->setValue(p.x());
408 m_pickYSpin->setValue(p.y());
409 }
410 updateFooterStatus();
411 });
412 lay->addWidget(m_boxCanvas, 1);
413
414 auto *col = new QWidget; col->setFixedWidth(280);

417 right->addWidget(makeFieldLabel(tr("Box Size (shared)")));

537void EditPatternWizard::goToStep(int step) {
538 if (step < 0 || step > 3) return;
539 if (step > m_currentStep && !currentStepValid()) return;
540
541 m_currentStep = step;
542 m_stack->setCurrentIndex(step);
543
544 if (step == 1 && m_pickCanvas) {
545 m_pickCanvas->setPick({m_new.pickX, m_new.pickY});
546 }
547 if (step == 2 && m_boxCanvas) {
548 m_boxCanvas->setPick({m_new.pickX, m_new.pickY});
549 m_boxCanvas->setBoxConfig(m_new.pickBoxW, m_new.pickBoxH,
550 m_new.pickBoxDist, m_new.pickBoxAngle);
551 }
552 if (step == 3 && m_finishCanvas) {
553 m_finishCanvas->setPick({m_new.pickX, m_new.pickY});
554 m_finishCanvas->setBoxConfig(m_new.pickBoxW, m_new.pickBoxH,
555 m_new.pickBoxDist, m_new.pickBoxAngle);
556 refreshDiff();
557 }
558
559 updateStepRail();
560 updateFooterStatus();
561 m_btnBack->setEnabled(step > 0);
562
563 if (step == 3) {
564 m_btnNext->setText("✓ " + tr("Apply Changes"));
565 disconnect(m_btnNext, nullptr, this, nullptr);
566 connect(m_btnNext, &QPushButton::clicked, this, &EditPatternWizard::onApply);
567 } else {
568 m_btnNext->setText(tr("Next") + " →");
569 disconnect(m_btnNext, nullptr, this, nullptr);
570 connect(m_btnNext, &QPushButton::clicked, this, &EditPatternWizard::onNext);
571 }
572
573 if (m_subtitleLabel) {
574 const QStringList subs = {tr("Image locked · edit name/number"),
575 tr("Adjust pick position"),
576 tr("Tweak gripper bounds"),
577 tr("Review changes")};
578 m_subtitleLabel->setText(QString("%1 · %2 %3 of 4 — %4")
579 .arg(tr("Group:"), m_groupName)
580 .arg(step + 1)
581 .arg(subs[step]));
582 }

592 m_stepBubbles[i]->setStyleSheet(ptn::stepBubbleStyle(done, current));

605bool EditPatternWizard::currentStepValid() const {
606 if (m_currentStep == 0) {
607 const bool nameOk = !m_new.name.trimmed().isEmpty()
608 && (m_new.name == m_old.name
609 || !m_usedNames.contains(m_new.name.trimmed()));
610 const bool numOk = m_new.number >= 1
611 && (m_new.number == m_old.number
612 || !m_usedNumbers.contains(m_new.number));

650void EditPatternWizard::refreshDiff() {
651 if (!m_diffSummary) return;
652
653 auto row = [&](const QString &label, const QString &oldV, const QString &newV) {
654 const bool changed = (oldV != newV);
655 const QString labelColor = changed ? ptn::WARN : ptn::TXT3;
656 const QString valueLine = changed
657 ? QString("<span style='color:%1;text-decoration:line-through;'>%2</span> "
658 "<span style='color:%3;'>→</span> "
659 "<span style='color:%4;font-weight:700;'>%5</span>")
660 .arg(ptn::TXT4, oldV, ptn::WARN, ptn::OK, newV)
661 : QString("<span style='color:%1;'>%2</span>").arg(ptn::TXT2, newV);
662 return QString(
663 "<div style='padding: 4px 0; border-bottom: 1px dashed %1;'>"
664 "<b style='color:%2; font-size: 8pt; letter-spacing: 1px;'>%3</b><br/>"
665 "<span style='font-family: \"JetBrains Mono\";'>%4</span></div>"
666 ).arg(ptn::BD, labelColor, label, valueLine);
667 };
668
669 QString html;
670 html += row(tr("NAME"), m_old.name, m_new.name);
671 html += row(tr("NUMBER"), QString("#%1").arg(m_old.number),
672 QString("#%1").arg(m_new.number));
673 html += row(tr("PICK"), QString("(%1, %2)").arg(m_old.pickX).arg(m_old.pickY),
674 QString("(%1, %2)").arg(m_new.pickX).arg(m_new.pickY));
675 html += row(tr("BOX W×H"), QString("%1×%2").arg(m_old.pickBoxW).arg(m_old.pickBoxH),
676 QString("%1×%2").arg(m_new.pickBoxW).arg(m_new.pickBoxH));
677 html += row(tr("OFFSET"), QString("d=%1 · %2°")
678 .arg(m_old.pickBoxDist).arg(m_old.pickBoxAngle),
679 QString("d=%1 · %2°")
680 .arg(m_new.pickBoxDist).arg(m_new.pickBoxAngle));
681 m_diffSummary->setText(html);
682}
