19bool parseMcTag(const QString &tag, QChar expectedPrefix, int *address)
20{
21 if (!address) {
22 return false;
23 }
24
25 static const QRegularExpression re(QStringLiteral("^([MD])(\\d+)$"),
26 QRegularExpression::CaseInsensitiveOption);
27 const auto match = re.match(tag.trimmed());
28 if (!match.hasMatch()) {
29 return false;
30 }
31
32 const QChar prefix = match.captured(1).at(0).toUpper();
33 if (prefix != expectedPrefix) {
34 return false;
35 }
36
37 bool ok = false;
38 const int parsed = match.captured(2).toInt(&ok, 10);
39 if (!ok || parsed < 0) {
40 return false;
41 }
42
43 *address = parsed;
44 return true;
45}

84bool McProtocolDevice::deviceConnect() {
85 QMutexLocker locker(&m_mutex);
86
87 // Connect requests are idempotent. Re-running the full init while already
88 // connected would build a second transport over the live one, leaking the
89 // open socket and desyncing the single-connection MC session — exactly the
90 // "old connection not closed, cannot reconnect" failure on phase re-entry.
91 // Re-publish Connected so a runner that issued the redundant request still
92 // sees a status event and clears its busy flag.
93 if (connectStatus() == ConnectStatus::Connected) {
94 this->setConnectionStatus(ConnectStatus::Connected);
95 return true;
96 }
97
98 if (!this->initialize_mc_device()) {
99 return false;
100 }
101
102 McMsgInterface::MsgIfState msg_port_state = m_msg_interface->ConnectToPort();
103 if (msg_port_state != McMsgInterface::MsgIfState::Connected) {
104 const QString err = m_msg_interface->GetErrorDescription();
105 LOG_DEV_INFO << "connect error occurred," << err;
106 // Free the half-open transport so a later retry starts from a clean
107 // state instead of leaking the failed socket.
108 releaseConnectionResources();
109 this->setConnectionStatus(ConnectStatus::ConnectFailed, err);
110 return false;
111 }
112
113 LOG_USER_INFO << "MC Device connected";
114 this->setConnectionStatus(ConnectStatus::Connected);
115 m_polling_timer->start();
116 return true;
117}

132void McProtocolDevice::releaseConnectionResources() {
133 // Caller holds m_mutex. The polling timer is created once and reused across
134 // connect cycles; only stop it here (never delete) because this can run
135 // from inside the timer's own slot chain (e.g. send/receive failure paths).
136 if (m_polling_timer && m_polling_timer->isActive()) {
137 m_polling_timer->stop();
138 }
139
140 if (m_msg_interface) {
141 // DestroyMsgPort() closes and deletes the socket synchronously on the
142 // device's own thread (see McEthernetTcpPort). reset() then frees the
143 // interface; both are deterministic and run on the correct thread.
144 m_msg_interface->DestroyMsgPort();
145 m_msg_interface.reset();
146 }
147 m_frame.reset();
148}

329bool McProtocolDevice::initialize_mc_device() {
330 // Clear any transport left over from a previous (possibly failed) session
331 // before building a fresh one, so connect attempts never stack sockets.
332 releaseConnectionResources();
333
334 McFrameType frame_type = m_config.context()->frameType();
335 McMsgItfType msg_type = m_config.context()->msgConfig()->type();
336
337 // setup frame builder
338 switch (frame_type) {
339 case vc::device::mc::McFrameType::Frame_3E:
340 m_config.context()->setDeviceMap(&m_device_map);
341 m_frame = std::make_unique<Frame3E>();
342 LOG_DEV_INFO << "Initialize MC Protocol Frame 3E.";
343 break;
344 default:
345 return false;
346 }
347
348 // setup msg interface
349 switch (msg_type) {
350 case vc::device::mc::McMsgItfType::EthernetTCPIP:
351 m_msg_interface = std::make_unique<McEthernetTcpPort>();
352 m_msg_interface->SetConfig(m_config.context()->msgConfig());
353 LOG_DEV_INFO << "Initialize Ethernet TCP/IP message interface.";
354 break;
355 default:
356 return false;
357 }
358 connect(m_msg_interface->ioDevice(), &QIODevice::readyRead,
359 this, &McProtocolDevice::onMsgInterfaceReadReady);
360
361
362 McContext *ctx = m_config.context();
363 // Setup polling timer. Created once and reused across connect cycles so it
364 // is never deleted from inside its own slot chain (see
365 // releaseConnectionResources). Only the interval is refreshed per session.
366 if (!m_polling_timer) {
367 m_polling_timer = new QTimer(this);
368 m_polling_timer->setTimerType(Qt::PreciseTimer);
369 connect(m_polling_timer, &QTimer::timeout,
370 this, &McProtocolDevice::onPollingTimerTimeOut);
371 }
372 m_polling_timer->setInterval(ctx->m_refreshInterval);
373 LOG_DEV_INFO << "MC Protocol Device refresh interval:" << ctx->m_refreshInterval;
374
375 // setup polling device map
376 optimizeDeviceMap();
377
378 // setup state flag
379 m_wait_for_response = false;
380 is_first_time_polling = true;
381 m_update_command_index = 0;
382 m_data_update_state = DataQueryState::QueryTriggerByTimer;
383 // m_retry_by_timeout = false;
384 m_retry_count = 0;
385
386 m_current_request.reset();
387 m_request_queue.clear();
388 m_read_buffer.clear();
389
390 m_comm_active_m_device = m_config.context()->m_activeMDevice;
391
392 return true;
393}

401void McProtocolDevice::polling_query() {
402 if (m_wait_for_response) {
403 auto current_time_point = std::chrono::high_resolution_clock::now();
404 auto time_lasp = std::chrono::duration_cast<std::chrono::milliseconds>(current_time_point - m_sent_time_point);
405 if (time_lasp.count() > m_config.context()->msgConfig()->m_responseTimeout) {
406 // response timeout handle;
407 retry_request_handle();
408 }
409 return;
410 }
411
412 m_mutex.lock();
413 if (!m_request_queue.empty()) {
414 m_current_request = m_request_queue.takeFirst();
415 m_mutex.unlock();
416 // Ad-hoc commands (the heartbeat M-bit toggle) are queued exactly once
417 // per completed round, so the round-robin reads must resume right after
418 // this response, in the same tick, instead of waiting for the next
419 // timer tick.
420 m_data_update_state = DataQueryState::QueryContinue;
421 } else {
422 m_mutex.unlock();
423
424 if (m_polling_request_queue.isEmpty()) {
425 return;
426 }
427
428 m_current_request = m_polling_request_queue.at(m_update_command_index++);
429 if (m_update_command_index >= m_polling_request_queue.count()) {
430 m_update_command_index = 0;
431 m_data_update_state = DataQueryState::QueryFinished;
432 is_first_time_polling = false;
433 // emit device's value change signal
434 // LOG_DEV_DEBUG << "Device map updated";
435 emit pollingUpdate(m_device_map.clone());
436 onSetCommActiveDevice();
437 } else {
438 m_data_update_state = DataQueryState::QueryContinue;
439 }
440 }
441
442 request_handle();
443}

448void McProtocolDevice::request_handle() {
449 if (m_current_request != nullptr) {
450 QByteArray send_frame;
451
452 m_mutex.lock();
453 MCFrameAbstract::FrameReturnCode rt_code = m_frame->makeSendFrame(m_current_request.get(), m_config.context(), send_frame);
454 m_mutex.unlock();
455
456 if (rt_code == MCFrameAbstract::RequestFrameOK) {
457 McMsgInterface::MsgErrorState send_state = m_msg_interface->SendMsg(send_frame);
458 // LOG_DEV_DEBUG << "MC Protocol Device sent message.";
459
460 if (send_state != McMsgInterface::MsgErrorState::NoError) {
461
462 LOG_USER_ERR << tr("MC Device error, cannot send request. Retry %1.").arg(m_retry_count);
463 m_current_request.reset();
464 m_retry_count += 1;
465
466 if (m_retry_count >= 5) {
467 LOG_USER_ERR << tr("Mc Device error, retry send request over 5 times, disconnect.");
468 deviceDisconnect();
469 }
470 return;
471 }
472
473 m_sent_time_point = std::chrono::high_resolution_clock::now();
474 m_wait_for_response = true;
475 // LOG_USER_INFO << "Send frame to PLC:" << send_frame.toHex(' ');
476 } else {
477 m_current_request.reset();
478 LOG_USER_ERR << "make request frame error";
479 }
480 }

489void McProtocolDevice::response_handle() {
490 if (!m_wait_for_response) {
491 return;
492 }
493
494 if (m_current_request == nullptr) {
495 return;
496 }
497
498 QByteArray receive_bytes = m_read_buffer; // carry over any partial frame from a prior split read
499 McMsgInterface::MsgErrorState recieve_state = m_msg_interface->ReceiveMsg(receive_bytes, 1);
500 // OLOG_INFO << "Received frame from PLC:" << receive_bytes.toHex(' ');
501
502 if (recieve_state != McMsgInterface::MsgErrorState::NoError) {
503 if (recieve_state == McMsgInterface::MsgErrorState::BufferEmpty) {
504 LOG_USER_ERR << "empty frame";
505 return;
506 }
507 LOG_USER_ERR << "error while receive frame, error code" << recieve_state;
508 return;
509 }
510
511 m_mutex.lock();
512 MCFrameAbstract::FrameReturnCode rt_code = m_frame->parseReceiveFrame(m_current_request.get(), m_config.context(), receive_bytes);
513 m_mutex.unlock();
514
515 if (rt_code == MCFrameAbstract::ResponseOk) {
516 // OLOG_INFO << "reponse frame status error," << m_frame->lastErrorDescription();
517 check_device_changed();
518
519 } else if (rt_code == MCFrameAbstract::ResponseError) {
520 LOG_USER_ERR << "reponse frame status error," << m_frame->lastErrorDescription();
521 } else if (rt_code == MCFrameAbstract::ResponseInvalid) {
522 LOG_USER_ERR << "reponse frame invalid," << m_frame->lastErrorDescription();
523 } else if (rt_code == MCFrameAbstract::WaitingReceive) {
524 m_read_buffer = receive_bytes; // still incomplete; keep the accumulated bytes for the next read
525 return;
526 }
527
528 m_read_buffer.clear(); // frame fully consumed (ok/error/invalid) - start clean next time
529 m_retry_count = 0;
530 // m_retry_by_timeout = false;
531 m_wait_for_response = false;
532 m_current_request.reset();
533
534 if (m_data_update_state == DataQueryState::QueryContinue) {
535 // LOG_DEV_DEBUG << "Continue query ...";
536 polling_query();
537 }
538}

543void McProtocolDevice::retry_request_handle() {
544 m_retry_count += 1;
545 if (m_retry_count > 5) {
546 // LOG_USER_ERR << "MC Device Error, disconnect to device";
547 setDeviceLostConnect();
548 return;
549 }
550
551 // if (m_current_request) {
552 // LOG_USER_ERR << "MC Device Error, request timeout" << m_current_request->type();
553 // }
554 LOG_USER_INFO << "MC Device Error, timeout, retry" << m_retry_count;
555 m_msg_interface->clearBuffer();
556 m_read_buffer.clear(); // discard any partial frame from the abandoned attempt
557 m_wait_for_response = false;
558 request_handle();
559}

648void McProtocolDevice::check_device_changed() {
649 QMap<QString, QVariant> changedValues;
650 if ((!m_device_map.device_map_m.empty()) && (!m_last_device_M_map.empty())) {
651
652 if (m_device_map.device_map_m.size() != m_last_device_M_map.size()) {
653 update_last_m_map();
654 LOG_DEV_INFO << "Device M map size change:" << m_last_device_M_map.size();
655 }
656
657 std::map<int, quint8>::iterator it_new_m = m_device_map.device_map_m.begin();
658 std::map<int, quint8>::iterator it_last_m = m_last_device_M_map.begin();
659 for (; it_new_m != m_device_map.device_map_m.end(); ++it_new_m, ++it_last_m) {
660 if (it_new_m->second != it_last_m->second) {
661 if (!is_first_time_polling) {
662 // append to changed maps;
663 changedValues.insert(QString("M%1").arg(it_new_m->first, 4, 10, QChar('0')), it_new_m->second);
664
665 emit deviceMChanged(it_new_m->first, it_last_m->second, it_new_m->second);
666 }
667 it_last_m->second = it_new_m->second;
668 }
669 }
670 }
671
672 if ((m_device_map.device_map_d.empty()) || (m_last_device_D_map.empty())) {
673 return;
674 }
675 if (m_device_map.device_map_d.size() != m_last_device_D_map.size()) {
676 update_last_d_map();
677 LOG_DEV_INFO << "Device D map size change:" << m_last_device_D_map.size();
678 }
679
680 std::map<int, qint16>::iterator it_new_d = m_device_map.device_map_d.begin();
681 std::map<int, qint16>::iterator it_last_d = m_last_device_D_map.begin();
682 for (; it_new_d != m_device_map.device_map_d.end(); ++it_new_d, ++it_last_d) {
683 if (it_new_d->second != it_last_d->second) {
684 if (!is_first_time_polling) {
685 // append to changed maps;
686 changedValues.insert(QString("D%1").arg(it_new_d->first, 4, 10, QChar('0')), it_new_d->second);
687
688 emit deviceDChanged(it_new_d->first, it_last_d->second, it_new_d->second);
689 }
690 it_last_d->second = it_new_d->second;
691 }
692 }
693
694 if (changedValues.size() > 0) {
695 emit valueChanged(changedValues);
696 }
