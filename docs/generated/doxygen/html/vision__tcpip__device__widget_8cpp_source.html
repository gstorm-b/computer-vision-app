18static QtVariantProperty* addPropertyToBrowser(const QMetaObject &meta, QMetaProperty &prop, QVariant &value,
19 QtVariantPropertyManager *manager, QtTreePropertyBrowser *browser) {
20
21 QString propName = prop.name();
22 if (propName == "objectName") {
23 // ignore objectName
24 return nullptr;
25 }
26
27 QtVariantProperty *variantProp = nullptr;
28 // enum type Check
29 if (prop.isEnumType()) {
30 variantProp = manager->addProperty(QtVariantPropertyManager::enumTypeId(), propName);
31
32 // get enum names
33 QStringList enumNames;
34 QMetaEnum metaEnum = prop.enumerator();
35 for (int j = 0; j < metaEnum.keyCount(); ++j) {
36 const char* key = metaEnum.key(j);
37 QString translatedName = QCoreApplication::translate(meta.className(), key);
38 enumNames << translatedName;
39 }
40 variantProp->setAttribute(QLatin1String("enumNames"), enumNames);
41 // set enum value
42 variantProp->setValue(value.toInt());
43
44 } else {
45 // normal data types handle (int, QString, bool, QColor, v.v.)
46 int typeId = value.userType();
47 variantProp = manager->addProperty(typeId, propName);
48 if (variantProp) {
49 variantProp->setValue(value);
50 }
51 }
52
53 if (!variantProp) {
54 return nullptr;
55 }
56
57 int displayNameIdx = meta.indexOfClassInfo(QString("%1_name").arg(propName).toUtf8());
58 if (displayNameIdx != -1) {
59 const QString displayName = meta.classInfo(displayNameIdx).value();
60 if (!displayName.isEmpty()) {
61 variantProp->setDisplayName(displayName);
62 }
63 }
64
65 // --- set attributes---
66 int minIdx = meta.indexOfClassInfo(QString("%1_min").arg(propName).toUtf8());
67 if (minIdx != -1) {
68 variantProp->setAttribute("minimum", QString(meta.classInfo(minIdx).value()).toInt());
69 }
70
71 int maxIdx = meta.indexOfClassInfo(QString("%1_max").arg(propName).toUtf8());
72 if (maxIdx != -1) {
73 variantProp->setAttribute("maximum", QString(meta.classInfo(maxIdx).value()).toInt());
74 }
75
76 if (!prop.isWritable()) {
77 variantProp->setEnabled(false);
78 }
79
80 return variantProp;
81}

474void VisionTcpipDeviceWidget::onSendResult() {
475 if (!m_runner || !m_output_device || !m_output_device->isMainClientConnected()) {
476 return;
477 }
478 QVector<vc::device::VisionOutputPosition> positions;
479 auto *tbl = ui->tbl_positions;
480 for (int row = 0; row < tbl->rowCount(); ++row) {
481 vc::device::VisionOutputPosition pos;
482 auto *sx = qobject_cast<QDoubleSpinBox*>(tbl->cellWidget(row, 0));
483 auto *sy = qobject_cast<QDoubleSpinBox*>(tbl->cellWidget(row, 1));
484 auto *sz = qobject_cast<QDoubleSpinBox*>(tbl->cellWidget(row, 2));
485 auto *sr = qobject_cast<QDoubleSpinBox*>(tbl->cellWidget(row, 3));
486 if (sx) pos.x = sx->value();
487 if (sy) pos.y = sy->value();
488 if (sz) pos.z = sz->value();
489 if (sr) pos.r = sr->value();
490 positions.append(pos);
491 }
492 m_runner->requestSendResult(positions);
493}
