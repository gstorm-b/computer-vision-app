12QPointF rotatePoint(const QPointF &point, const QPointF &center, double angleDeg)
13{
14 const double radians = qDegreesToRadians(angleDeg);
15 const double s = std::sin(radians);
16 const double c = std::cos(radians);
17 const QPointF translated = point - center;
18 return QPointF(translated.x() * c - translated.y() * s,
19 translated.x() * s + translated.y() * c) + center;
20}

26QPixmap pixmapFromMat(const cv::Mat &mat)
27{
28 if (mat.empty()) return {};

30 QImage image;

32 image = QImage(mat.data, mat.cols, mat.rows, static_cast<int>(mat.step),

45 return {};

48 return QPixmap::fromImage(image);

70VisionRoi clampRoiToImage(const VisionRoi &roi, const QSize &imageSize)
71{
72 VisionRoi clamped = normalizedRoi(roi);
73 if (imageSize.isEmpty()) return clamped;
74
75 clamped.center.setX(qBound(0.0, clamped.center.x(),
76 static_cast<double>(imageSize.width())));
77 clamped.center.setY(qBound(0.0, clamped.center.y(),
78 static_cast<double>(imageSize.height())));
79
80 clamped.size.setWidth(qBound(1.0, clamped.size.width(),
81 static_cast<double>(imageSize.width())));
82 clamped.size.setHeight(qBound(1.0, clamped.size.height(),
83 static_cast<double>(imageSize.height())));
84
85 if (clamped.shape == VisionRoiShape::AxisAlignedRect) {
86 QRectF rect = rectFromRoi(clamped);
87 rect.setLeft(qMax(0.0, rect.left()));
88 rect.setTop(qMax(0.0, rect.top()));
89 rect.setRight(qMin(rect.right(), static_cast<double>(imageSize.width())));
90 rect.setBottom(qMin(rect.bottom(), static_cast<double>(imageSize.height())));
91 clamped.center = rect.center();
92 clamped.size = rect.size();
93 }
94
95 return clamped;
96}

98QVector<QPointF> roiCorners(const VisionRoi &roi)
99{
100 const VisionRoi normalized = normalizedRoi(roi);
101 const QRectF rect = rectFromRoi(normalized);
102 QVector<QPointF> corners{
103 rect.topLeft(),
104 rect.topRight(),
105 rect.bottomRight(),
106 rect.bottomLeft(),
107 };
108
109 if (normalized.shape == VisionRoiShape::RotatedRect && !qFuzzyIsNull(normalized.angleDeg)) {
110 for (QPointF &point : corners) {
111 point = rotatePoint(point, normalized.center, normalized.angleDeg);
112 }
113 }
114
115 return corners;
116}
