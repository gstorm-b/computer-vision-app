66void AddPatternImageCanvas::setImage(const cv::Mat &mat) {
67 if (mat.empty()) { m_pix = QPixmap(); resetView(); update(); return; }
68
69 QImage img;
70 if (mat.type() == CV_8UC1) {
71 img = QImage(mat.data, mat.cols, mat.rows,
72 static_cast<int>(mat.step),
73 QImage::Format_Grayscale8).copy();
74 } else if (mat.type() == CV_8UC3) {
75 cv::Mat rgb;
76 cv::cvtColor(mat, rgb, cv::COLOR_BGR2RGB);
77 img = QImage(rgb.data, rgb.cols, rgb.rows,
78 static_cast<int>(rgb.step),
79 QImage::Format_RGB888).copy();
80 } else if (mat.type() == CV_8UC4) {
81 cv::Mat rgba;
82 cv::cvtColor(mat, rgba, cv::COLOR_BGRA2RGBA);
83 img = QImage(rgba.data, rgba.cols, rgba.rows,
84 static_cast<int>(rgba.step),
85 QImage::Format_RGBA8888).copy();
86 }
87 m_pix = QPixmap::fromImage(img);
88 resetView();
89 update();
90}

184void AddPatternImageCanvas::paintEvent(QPaintEvent *) {
185 QPainter p(this);
186 // Antialiasing for overlay shapes (crop/box outlines, handles).
187 // SmoothPixmapTransform is OFF so the underlying image renders with
188 // nearest-neighbour sampling — at high zoom levels users see real pixel
189 // boundaries, which matches the pattern-inspection use case.
190 p.setRenderHint(QPainter::Antialiasing, true);
191 p.setRenderHint(QPainter::SmoothPixmapTransform, false);
192
193 // ── Background ──────────────────────────────────────────────────────
194 p.fillRect(rect(), QColor("#181818"));
195
196 if (m_pix.isNull()) {
197 // Placeholder grid + "no image" text
198 p.setPen(QPen(QColor(ptn::BD), 0.5));
199 for (int x = 0; x < width(); x += 20) p.drawLine(x, 0, x, height());
200 for (int y = 0; y < height(); y += 20) p.drawLine(0, y, width(), y);
201
202 p.setPen(QColor(ptn::TXT3));
203 p.setFont(QFont("Segoe UI", 10));
204 p.drawText(rect(), Qt::AlignCenter, tr("No image"));
205 return;
206 }
207
208 // ── Draw image through view transform ───────────────────────────────
209 const QRectF imgWidget = imageRectInWidget();
210 p.drawPixmap(imgWidget, m_pix, QRectF(QPointF(0, 0), QSizeF(m_pix.size())));
211
212 // ── Mode-specific overlay ───────────────────────────────────────────
213 switch (m_mode) {
214 case None: break;
215
216 case Crop: {
217 if (m_crop.isNull() || m_crop.isEmpty()) break;
218
219 const QRectF cropW(imageToWidget(m_crop.topLeft()),
220 QSizeF(m_crop.width() * m_zoom, m_crop.height() * m_zoom));
221
222 // Dim scrim outside the crop, inside the image rect
223 QPainterPath outer; outer.addRect(imgWidget);
224 QPainterPath inner; inner.addRect(cropW);
225 p.setBrush(QColor(0, 0, 0, 170));
226 p.setPen(Qt::NoPen);
227 p.drawPath(outer.subtracted(inner));
228
229 // Crop rect outline
230 p.setBrush(Qt::NoBrush);
231 p.setPen(QPen(QColor(ptn::ACC), 1.5));
232 p.drawRect(cropW);
233
234 // Rule-of-thirds (subdued)
235 QColor tc(ptn::ACC); tc.setAlpha(85);
236 p.setPen(QPen(tc, 1));
237 const double w3 = cropW.width() / 3.0;
238 const double h3 = cropW.height() / 3.0;
239 p.drawLine(QPointF(cropW.left() + w3, cropW.top()),
240 QPointF(cropW.left() + w3, cropW.bottom()));
241 p.drawLine(QPointF(cropW.left() + 2 * w3, cropW.top()),
242 QPointF(cropW.left() + 2 * w3, cropW.bottom()));
243 p.drawLine(QPointF(cropW.left(), cropW.top() + h3),
244 QPointF(cropW.right(), cropW.top() + h3));
245 p.drawLine(QPointF(cropW.left(), cropW.top() + 2 * h3),
246 QPointF(cropW.right(), cropW.top() + 2 * h3));
247
248 // Corner handles (widget-pixel sized — readable at any zoom)
249 const QPointF corners[4] = {
250 cropW.topLeft(), cropW.topRight(),
251 cropW.bottomLeft(), cropW.bottomRight()
252 };
253 p.setBrush(QColor(ptn::ACC));
254 p.setPen(QPen(QColor("white"), 1.5));
255 for (const QPointF &c : corners)
256 p.drawRect(QRectF(c.x() - kHandleVisSize / 2,
257 c.y() - kHandleVisSize / 2,
258 kHandleVisSize, kHandleVisSize));
259 break;
260 }
261
262 case Pick: {
263 // Read-only crop overlay (so user knows the crop region)
264 if (!m_crop.isNull() && !m_crop.isEmpty()) {
265 const QRectF cropW(imageToWidget(m_crop.topLeft()),
266 QSizeF(m_crop.width() * m_zoom,
267 m_crop.height() * m_zoom));
268 QColor scrim(0, 0, 0, 110);
269 QPainterPath outer; outer.addRect(imgWidget);
270 QPainterPath inner; inner.addRect(cropW);
271 p.setBrush(scrim); p.setPen(Qt::NoPen);
272 p.drawPath(outer.subtracted(inner));
273
274 // Crop outline (dashed, subdued — signals read-only)
275 QColor lock(ptn::ACC); lock.setAlpha(180);
276 QPen pen(lock, 1.2, Qt::DashLine);
277 p.setBrush(Qt::NoBrush);
278 p.setPen(pen);
279 p.drawRect(cropW);
280 }
281
282 // Pick crosshair (widget coords)
283 const QPointF pw = imageToWidget(m_pick);
284 QColor warn(ptn::WARN);
285 QColor warn2(ptn::WARN); warn2.setAlpha(204);
286 p.setPen(QPen(warn2, 1));
287 p.drawLine(QPointF(0, pw.y()), QPointF(width(), pw.y()));
288 p.drawLine(QPointF(pw.x(), 0), QPointF(pw.x(), height()));
289
290 // Picking-orientation gizmo, interactive in this mode (drag knob shown). Drawn
291 // BEFORE the ring: both axes run right through the pick point, and the ring is what
292 // covers the crossing so the centre stays a clean marker rather than a knot of lines.
293 drawPickOrientation(p, pw, true);
294
295 QColor ringFill(ptn::WARN); ringFill.setAlpha(34);
296 p.setBrush(ringFill);
297 p.setPen(QPen(warn, 1.5));
298 p.drawEllipse(pw, 9, 9);
299
300 // Label — shows crop-relative coords when a crop is set, otherwise
301 // raw image coords. Matches the spec the host returns to callers.
302 QPoint displayPick = m_pick;
303 if (!m_crop.isNull() && !m_crop.isEmpty())
304 displayPick -= m_crop.topLeft();
305 p.setPen(warn);
306 p.setFont(QFont("JetBrains Mono", 8, QFont::Bold));
307 p.drawText(QPointF(pw.x() + 12, pw.y() - 12),
308 QString("PICK (%1, %2)").arg(displayPick.x()).arg(displayPick.y()));
309 break;
310 }
311
312 case Box:
313 case Finish: {
314 const QPointF pw = imageToWidget(m_pick);
315 const QColor jawColor(ptn::OK);
316 QColor jawFill(ptn::OK); jawFill.setAlpha(34);
317
318 auto drawBox = [&](double angleDeg, const QString &label, bool interactive) {
319 const double r = qDegreesToRadians(angleDeg);
320 const QPointF cImg(m_pick.x() + std::cos(r) * m_boxDist,
321 m_pick.y() + std::sin(r) * m_boxDist);
322 const QPointF cW = imageToWidget(cImg);
323
324 // Connector
325 p.setPen(QPen(jawColor, 1, Qt::DashLine));
326 p.drawLine(pw, cW);
327
328 // Box body — rotated by angleDeg around its centre. We rotate in
329 // widget space, scaling W/H by m_zoom so the box stays "in image
330 // pixels" without clipping at the widget edges (boxes are allowed
331 // to spill outside the image).
332 p.save();
333 p.translate(cW);
334 p.rotate(angleDeg);
335 const double ww = m_boxW * m_zoom;
336 const double hh = m_boxH * m_zoom;
337 p.setBrush(jawFill);
338 p.setPen(QPen(jawColor, 1.5));
339 p.drawRect(QRectF(-ww / 2, -hh / 2, ww, hh));
340 p.setBrush(jawColor); p.setPen(Qt::NoPen);
341 p.drawEllipse(QPointF(0, 0), 2.5, 2.5);
342
343 if (interactive && m_mode == Box) {
344 // Corner handles (widget-pixel size)
345 p.setBrush(QColor(ptn::ACC));
346 p.setPen(QPen(QColor("white"), 1.2));
347 const QPointF corners[4] = {
348 QPointF(-ww / 2, -hh / 2),
349 QPointF( ww / 2, -hh / 2),
350 QPointF(-ww / 2, hh / 2),
351 QPointF( ww / 2, hh / 2),
352 };
353 for (const QPointF &c : corners)
354 p.drawRect(QRectF(c.x() - kHandleVisSize / 2,
355 c.y() - kHandleVisSize / 2,
356 kHandleVisSize, kHandleVisSize));
357 }
358 p.restore();
359
360 // Label
361 p.setPen(jawColor);
362 p.setFont(QFont("JetBrains Mono", 8, QFont::Bold));
363 p.drawText(QPointF(cW.x() - 4, cW.y() - hh / 2 - 4), label);
364 };
365 drawBox(m_boxAngle, "A", /*interactive=*/true);
366 drawBox(m_boxAngle + 180, "B", /*interactive=*/false);
367
368 // Rotation handle — small filled disc offset perpendicular to the
369 // connector from box A centre, drawn in widget space.
370 if (m_mode == Box) {
371 const QPointF rotImg = boxRotationHandle();
372 const QPointF rotW = imageToWidget(rotImg);
373 const QPointF aW = imageToWidget(boxACentre());
374 QColor rot(ptn::WARN);
375 p.setPen(QPen(rot, 1.2, Qt::DotLine));
376 p.drawLine(aW, rotW);
377 p.setBrush(rot);
378 p.setPen(QPen(QColor("white"), 1.2));
379 p.drawEllipse(rotW, kHandleVisSize / 2, kHandleVisSize / 2);
380 }
381
382 // Picking orientation, read-only on the review step. Left out of Box mode: the jaw
383 // pair already owns a rotation handle there, and two rotation gizmos around one
384 // pick point would be ambiguous about which angle the user is dragging.
385 // Drawn before the crosshair so the centre marker covers the axis crossing.
386 if (m_mode == Finish)
387 drawPickOrientation(p, pw, /*interactive=*/false);
388
389 // Pick crosshair on top
390 p.setPen(QPen(QColor("white"), 1));
391 p.drawLine(QPointF(pw.x() - 12, pw.y()), QPointF(pw.x() + 12, pw.y()));
392 p.drawLine(QPointF(pw.x(), pw.y() - 12), QPointF(pw.x(), pw.y() + 12));
393 p.setBrush(QColor(ptn::WARN)); p.setPen(Qt::NoPen);
394 p.drawEllipse(pw, 3, 3);
395 break;
396 }
397 }
398
399 // ── Locked scrim ────────────────────────────────────────────────────
400 if (m_locked) {
401 QColor scrim(ptn::LOCK); scrim.setAlpha(80);
402 p.fillRect(rect(), scrim);
403
404 const int bw = 86, bh = 22;
405 QRect badge(width() - bw - 12, 12, bw, bh);
406 p.setBrush(QColor(ptn::LOCK));
407 p.setPen(QPen(QColor("white"), 1));
408 p.drawRoundedRect(badge, 4, 4);
409 p.setPen(QColor("white"));
410 p.setFont(QFont("Segoe UI", 8, QFont::Bold));
411 p.drawText(badge, Qt::AlignCenter, "🔒 LOCKED");
412 }
413}

477QPointF AddPatternImageCanvas::boxRotationHandle() const {
478 // Offset perpendicular to the connector direction. Perpendicular vector
479 // for angle θ is (cos(θ+90°), sin(θ+90°)) = (-sin θ, cos θ).
480 const double r = qDegreesToRadians(m_boxAngle);
481 const QPointF c = boxACentre();
482 return QPointF(c.x() + (-std::sin(r)) * (m_boxH / 2 + kRotateHandleStandoff),
483 c.y() + ( std::cos(r)) * (m_boxH / 2 + kRotateHandleStandoff));
484}
485
488void AddPatternImageCanvas::boxACorners(QPointF out[4]) const {
489 const double r = qDegreesToRadians(m_boxAngle);
490 const double co = std::cos(r), si = std::sin(r);
491 const QPointF c = boxACentre();
492 const double w2 = m_boxW / 2, h2 = m_boxH / 2;
493 // local (lx, ly) → image: (c + R * (lx, ly))
494 const double lx[4] = { -w2, w2, -w2, w2 };
495 const double ly[4] = { -h2, -h2, h2, h2 };
496 for (int i = 0; i < 4; ++i) {
497 out[i] = QPointF(c.x() + co * lx[i] - si * ly[i],
498 c.y() + si * lx[i] + co * ly[i]);
499 }
500}

510AddPatternImageCanvas::CropHandle
511AddPatternImageCanvas::hitCropHandle(const QPoint &widgetPos) const {
512 if (m_crop.isNull() || m_crop.isEmpty()) return CH_None;
513
514 // Test corners first (widget-pixel slack so they're always reachable).
515 const QPointF cs[4] = {
516 imageToWidget(m_crop.topLeft()),
517 imageToWidget(m_crop.topRight()),
518 imageToWidget(m_crop.bottomLeft()),
519 imageToWidget(m_crop.bottomRight()),
520 };
521 for (int i = 0; i < 4; ++i) {
522 if (qAbs(widgetPos.x() - cs[i].x()) <= kHandleHitSlack &&
523 qAbs(widgetPos.y() - cs[i].y()) <= kHandleHitSlack)
524 return CropHandle(i);
525 }
526 // Body — cursor inside the crop rect (widget coords)
527 const QRectF cw(cs[0], cs[3]);
528 if (cw.normalized().contains(widgetPos))
529 return CH_Body;
530 return CH_None;
531}
532
538AddPatternImageCanvas::BoxHandle
539AddPatternImageCanvas::hitBoxHandle(const QPoint &widgetPos) const {
540 // Pick crosshair — highest priority so the picking centre can always be
541 // grabbed and dragged (both jaws follow it rigidly).
542 {
543 const QPointF pw = imageToWidget(m_pick);
544 if (QLineF(pw, widgetPos).length() <= kHandleHitSlack + 2)
545 return BH_Pick;
546 }
547 // Rotation handle — widget-pixel proximity to the centre of the knob.
548 {
549 const QPointF rotW = imageToWidget(boxRotationHandle());
550 if (QLineF(rotW, widgetPos).length() <= kHandleHitSlack + 2)
551 return BH_Rotate;
552 }
553 // Corner handles
554 QPointF corners[4];
555 boxACorners(corners);
556 for (int i = 0; i < 4; ++i) {
557 const QPointF cw = imageToWidget(corners[i]);
558 if (qAbs(widgetPos.x() - cw.x()) <= kHandleHitSlack &&
559 qAbs(widgetPos.y() - cw.y()) <= kHandleHitSlack)
560 return BoxHandle(i);
561 }
562 // Body — transform cursor to box-local coords and test against [-W/2, W/2] × [-H/2, H/2]
563 const QPointF imgPt = widgetToImage(widgetPos);
564 const QPointF c = boxACentre();
565 const double r = qDegreesToRadians(m_boxAngle);
566 const double co = std::cos(-r), si = std::sin(-r);
567 const double dx = imgPt.x() - c.x();
568 const double dy = imgPt.y() - c.y();
569 const double lx = co * dx - si * dy;
570 const double ly = si * dx + co * dy;
571 if (qAbs(lx) <= m_boxW / 2 && qAbs(ly) <= m_boxH / 2)
572 return BH_Body;
573 return BH_None;
574}

585void AddPatternImageCanvas::mousePressEvent(QMouseEvent *e) {
586 if (m_pix.isNull()) return;
587
588 // Middle button — pan (works in every mode, even when locked)
589 if (e->button() == Qt::MiddleButton) {
590 m_panning = true;
591 m_panStart = e->pos();
592 m_panStartOffset = m_offset;
593 setCursor(Qt::ClosedHandCursor);
594 return;
595 }
596 if (m_locked) return;
597
598 if (m_mode == Crop && e->button() == Qt::LeftButton) {
599 const CropHandle h = hitCropHandle(e->pos());
600 if (h != CH_None) {
601 m_cropDragging = true;
602 m_cropHandle = h;
603 m_cropDragStartImg = widgetToImage(e->pos());
604 m_cropDragStartRect = m_crop;
605 }
606 } else if (m_mode == Pick && e->button() == Qt::LeftButton) {
607 // The orientation knob wins over placing the pick point. Without this the click
608 // that grabs the knob would first teleport the pick point under the cursor, which
609 // is the opposite of what grabbing a rotation handle should do.
610 if (hitPickRotationHandle(e->pos())) {
611 m_pickAngleDragging = true;
612 setCursor(Qt::ClosedHandCursor);
613 update();
614 return;
615 }
616
617 // Place the pick immediately, then stay armed so the same gesture can be dragged —
618 // a press-and-drag is how you nudge a point into place, and click-only forced the
619 // user to re-click repeatedly to converge on it.
620 setPickClamped(widgetToImage(e->pos()));
621 m_pickDragging = true;
622 setCursor(Qt::ClosedHandCursor);
623 update();
624 emit pickChanged(m_pick, m_pickCurrentPoint);
625 } else if (m_mode == Box && e->button() == Qt::LeftButton) {
626 const BoxHandle h = hitBoxHandle(e->pos());
627 if (h != BH_None) {
628 m_boxDragging = true;
629 m_boxHandle = h;
630 m_boxDragStartImg = widgetToImage(e->pos());
631 m_boxDragStartW = m_boxW;
632 m_boxDragStartH = m_boxH;
633 m_boxDragStartDist = m_boxDist;
634 m_boxDragStartAngle = m_boxAngle;
635 }
636 }
637}

644void AddPatternImageCanvas::mouseMoveEvent(QMouseEvent *e) {
645 // Middle-button pan — handled in every mode.
646 if (m_panning) {
647 const QPoint d = e->pos() - m_panStart;
648 m_offset = m_panStartOffset + QPointF(d);
649 update();
650 return;
651 }
652 if (m_pix.isNull() || m_locked) return;
653
654 // ── Crop ────────────────────────────────────────────────────────────
655 if (m_mode == Crop) {
656 if (m_cropDragging) {
657 const QPointF imgNow = widgetToImage(e->pos());
658 const QPointF dImg = imgNow - m_cropDragStartImg;
659 QRect r = m_cropDragStartRect;
660
661 auto P = [](const QPointF &p) {
662 return QPoint(int(std::round(p.x())), int(std::round(p.y())));
663 };
664
665 switch (m_cropHandle) {
666 case CH_TL: r.setTopLeft (m_cropDragStartRect.topLeft() + P(dImg)); break;
667 case CH_TR: r.setTopRight (m_cropDragStartRect.topRight() + P(dImg)); break;
668 case CH_BL: r.setBottomLeft (m_cropDragStartRect.bottomLeft() + P(dImg)); break;
669 case CH_BR: r.setBottomRight(m_cropDragStartRect.bottomRight() + P(dImg)); break;
670 case CH_Body: r.translate(P(dImg)); break;
671 default: break;
672 }
673
674 // Enforce minimum size in image pixels and clamp to image bounds.
675 if (r.width() < 8) r.setWidth(8);
676 if (r.height() < 8) r.setHeight(8);
677 r = r.intersected(QRect(QPoint(0, 0), m_pix.size()));
678
679 if (r != m_crop) {
680 m_crop = r;
681 update();
682 emit cropChanged(m_crop);
683 }
684 } else {
685 const CropHandle h = hitCropHandle(e->pos());
686 switch (h) {
687 case CH_TL: case CH_BR: setCursor(Qt::SizeFDiagCursor); break;
688 case CH_TR: case CH_BL: setCursor(Qt::SizeBDiagCursor); break;
689 case CH_Body: setCursor(Qt::SizeAllCursor); break;
690 default: setCursor(Qt::ArrowCursor); break;
691 }
692 }
693 return;
694 }
695
696 // ── Pick ────────────────────────────────────────────────────────────
697 if (m_mode == Pick) {
698 if (m_pickAngleDragging) {
699 // Track the cursor's bearing from the pick point, then subtract the knob's own
700 // bearing: the knob sits on the X/Y bisector, not on the X axis, so the raw
701 // bearing runs kPickHandleBearing degrees ahead of the angle it represents.
702 // Without this the angle would jump by that amount the instant the knob is
703 // grabbed. Ignore a cursor sitting almost exactly on the pick point, where the
704 // bearing is numerically meaningless.
705 const QPointF imgNow = widgetToImage(e->pos());
706 const double dx = imgNow.x() - m_pick.x();
707 const double dy = imgNow.y() - m_pick.y();
708 if (std::hypot(dx, dy) > 0.5) {
709 double deg = qRadiansToDegrees(std::atan2(dy, dx))
710 - pick_overlay::kHandleBearing;
711 while (deg > 180.0) deg -= 360.0;
712 while (deg < -180.0) deg += 360.0;
713 m_pickAngle = deg;
714 update();
715 emit pickAngleChanged(m_pickAngle);
716 }
717 return;
718 }
719 if (m_pickDragging) {
720 const QPoint before = m_pick;
721 setPickClamped(widgetToImage(e->pos()));
722 if (m_pick != before) {
723 update();
724 emit pickChanged(m_pick, m_pickCurrentPoint);
725 }
726 return;
727 }
728 setCursor(hitPickRotationHandle(e->pos()) ? Qt::OpenHandCursor
729 : Qt::CrossCursor);
730 return;
731 }
732
733 // ── Box ─────────────────────────────────────────────────────────────
734 if (m_mode == Box) {
735 if (m_boxDragging) {
736 const QPointF imgNow = widgetToImage(e->pos());
737
738 switch (m_boxHandle) {
739 case BH_TL: case BH_TR: case BH_BL: case BH_BR: {
740 // Convert cursor to box-local axis-aligned coords and use the
741 // absolute values as half-sizes — symmetric resize around
742 // box A centre (which then propagates to box B).
743 const QPointF c = boxACentre();
744 const double rR = qDegreesToRadians(m_boxAngle);
745 const double co = std::cos(-rR), si = std::sin(-rR);
746 const double dx = imgNow.x() - c.x();
747 const double dy = imgNow.y() - c.y();
748 const double lx = co * dx - si * dy;
749 const double ly = si * dx + co * dy;
750 const double newW = qMax<double>(8.0, 2.0 * std::abs(lx));
751 const double newH = qMax<double>(8.0, 2.0 * std::abs(ly));
752 m_boxW = newW;
753 m_boxH = newH;
754 break;
755 }
756 case BH_Body: {
757 // Translate the box pair — m_boxDist + m_boxAngle re-derived
758 // from the cursor position relative to the pick point.
759 const double dx = imgNow.x() - m_pick.x();
760 const double dy = imgNow.y() - m_pick.y();
761 const double dist = std::hypot(dx, dy);
762 if (dist > 0.5) {
763 m_boxDist = dist;
764 m_boxAngle = qRadiansToDegrees(std::atan2(dy, dx));
765 }
766 break;
767 }
768 case BH_Rotate: {
769 // Rotate around the pick. We compute the cursor's angle from
770 // the pick and offset by 90° so dragging the perpendicular
771 // handle puts the box's natural angle along the connector.
772 const double dx = imgNow.x() - m_pick.x();
773 const double dy = imgNow.y() - m_pick.y();
774 if (std::hypot(dx, dy) > 0.5) {
775 m_boxAngle = qRadiansToDegrees(std::atan2(dy, dx)) - 90.0;
776 }
777 break;
778 }
779 case BH_Pick: {
780 // Drag the picking centre itself — both jaws follow rigidly
781 // since their centres are derived from (pick + dist·angle).
782 // Same clamping rule as Pick mode, hence the shared helper.
783 setPickClamped(imgNow);
784 break;
785 }
786 default: break;
787 }
788 update();
789 if (m_boxHandle == BH_Pick)
790 emit pickChanged(m_pick, m_pickCurrentPoint);
791 else
792 emit boxChanged(m_boxW, m_boxH, m_boxDist, m_boxAngle);
793 } else {
794 const BoxHandle h = hitBoxHandle(e->pos());
795 switch (h) {
796 case BH_TL: case BH_BR: setCursor(Qt::SizeFDiagCursor); break;
797 case BH_TR: case BH_BL: setCursor(Qt::SizeBDiagCursor); break;
798 case BH_Rotate: setCursor(Qt::CrossCursor); break;
799 case BH_Pick: setCursor(Qt::PointingHandCursor); break;
800 case BH_Body: setCursor(Qt::SizeAllCursor); break;
801 default: setCursor(Qt::ArrowCursor); break;
802 }
803 }
804 }
805}

844void AddPatternImageCanvas::wheelEvent(QWheelEvent *e) {
845 if (m_pix.isNull()) return;
846
847 // Zoom around the cursor: keep the image point under the cursor fixed.
848 const QPointF mouseW = e->position();
849 const QPointF imgPt = widgetToImage(mouseW);
850
851 const int delta = e->angleDelta().y();
852 const double factor = (delta > 0) ? kZoomStep : 1.0 / kZoomStep;
853 const double newZoom = qBound(kMinZoom, m_zoom * factor, kMaxZoom);
854 if (qFuzzyCompare(newZoom, m_zoom)) return;
855
856 m_zoom = newZoom;
857 m_offset = mouseW - imgPt * m_zoom;
858 update();
859}
