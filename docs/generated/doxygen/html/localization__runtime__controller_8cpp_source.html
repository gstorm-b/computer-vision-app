119void LocalizationRuntimeController::setActiveCameraNumber(int cameraNumber)
120{
121 if (m_cycleState == CycleState::Running) {
122 appendTaskLog(QStringLiteral("WARN"),
123 QStringLiteral("Camera change ignored while cycle is running."));
124 return;
125 }
126
127 if (!m_context.cameraRunners.value(cameraNumber)) {
128 appendTaskLog(QStringLiteral("WARN"),
129 QStringLiteral("Camera change ignored: camera runner is not available."));
130 publishBoolSignal(QStringLiteral("bCameraValid"), false);
131 publishBoolSignal(QStringLiteral("bTaskFault"), true);
132 publishNumberSignal(QStringLiteral("nFaultCode"),
133 localizationFaultCodeValue(LocalizationFaultCode::CameraLost));
134 return;
135 }
136
137 auto *previousRunner = activeCameraRunner();
138 m_activeCameraNumber = cameraNumber;
139 publishBoolSignal(QStringLiteral("bTaskReady"), false);
140 bindActiveCameraRole(m_activeCameraNumber);
141 publishNumberSignal(QStringLiteral("nActiveCamera"), cameraNumber);
142
143 auto *newRunner = activeCameraRunner();
144 if (previousRunner && previousRunner != newRunner) {
145 previousRunner->requestDisconnect();
146 }
147
148 const bool calibrated = validateActiveCameraCalibration();
149 // Rebuild the pickability checker against the new camera's calibration.
150 rebuildPickingChecker();
151 publishBoolSignal(QStringLiteral("bCameraValid"), calibrated);
152 if (!calibrated) {
153 publishBoolSignal(QStringLiteral("bTaskFault"), true);
154 publishNumberSignal(QStringLiteral("nFaultCode"),
155 localizationFaultCodeValue(LocalizationFaultCode::CalibrationInvalid));
156 m_cycleState = CycleState::Faulted;
157 emit runtimeFault(QStringLiteral("Active camera calibration is invalid."));
158 return;
159 }
160
161 if (newRunner) {
162 newRunner->requestConnect();
163 // Refresh the active workspace from the new camera, guarded like the
164 // requestConnect above so a degraded switch cannot deref a null runner.
165 m_context.activeCameraWorkspace =
166 m_config.cameraWorkspace(newRunner->device()->id());
167 m_activeCameraWorkspace = m_context.activeCameraWorkspace;
168 }
169 if (allRequiredRolesHealthy()) {
170 markRuntimeReady(QStringLiteral("Active camera changed. Runtime ready."));
171 }

219LocalizationRuntimeController::setup(const RuntimeContext &context)
220{
221 SetupResult result;
222 m_valid = false;
223 m_context = context;
224 m_config = context.config;
225 m_signalMapper.configure(m_config);
226 m_cycleState = CycleState::NotReady;
227 m_lastExecuteTrigger = false;
228 m_activeCycleId = 0;
229 m_pendingCycleResult = CycleResult();
230 resetRuntimeBindings();
231
232 result.primaryPlcDeviceId = context.primaryPlcDeviceId;
233 result.visionOutputDeviceId = context.visionOutputDeviceId;
234 const QMap<int, QString> cameraMap = context.cameraDeviceIds;
235
236 if (result.primaryPlcDeviceId.isEmpty() || !context.primaryPlcRunner) {
237 result.errors.append(QStringLiteral("Missing primary_plc role runner."));
238 }
239 if (result.visionOutputDeviceId.isEmpty() || !context.visionOutputRunner) {
240 result.errors.append(QStringLiteral("Missing vision_output role runner."));
241 }
242
243 if (cameraMap.isEmpty() || context.cameraRunners.isEmpty()) {
244 result.errors.append(QStringLiteral("Missing camera_number role binding."));
245 } else {
246 for (auto it = cameraMap.cbegin(); it != cameraMap.cend(); ++it) {
247 if (!context.cameraRunners.value(it.key())) {
248 result.errors.append(QStringLiteral("camera_number_%1 runner is not registered: %2")
249 .arg(it.key())
250 .arg(it.value()));
251 }
252 }
253 }
254
255 disconnect(m_plcValueConnection);
256 if (context.primaryPlcRunner) {
257 m_plcValueConnection = connect(
258 context.primaryPlcRunner.data(),
259 &vc::runtime::PlcRunner::valueChanged,
260 this,
261 &LocalizationRuntimeController::handlePlcValues,
262 Qt::UniqueConnection);
263 }
264
265 bindFixedRoleRunners();
266 m_activeCameraNumber = context.activeCameraNumber;
267 if (m_activeCameraNumber < 0 && !cameraMap.isEmpty()) {
268 m_activeCameraNumber = cameraMap.firstKey();
269 }
270 bindActiveCameraRole(m_activeCameraNumber);
271 m_activePatternGroupNumber = context.activePatternGroupNumber;
272 if (m_activePatternGroupNumber < 0 && !context.patternGroups.isEmpty()) {
273 m_activePatternGroupNumber = context.patternGroups.firstKey();
274 }
275
276 if (!validateActivePatternGroup(&result.errors)) {
277 publishBoolSignal(QStringLiteral("bPatternValid"), false);
278 publishBoolSignal(QStringLiteral("bTaskFault"), true);
279 publishNumberSignal(QStringLiteral("nFaultCode"),
280 localizationFaultCodeValue(LocalizationFaultCode::PatternInvalid));
281 }
282
283 if (!validateActiveCameraCalibration(&result.errors)) {
284 publishBoolSignal(QStringLiteral("bCameraValid"), false);
285 publishBoolSignal(QStringLiteral("bTaskFault"), true);
286 publishNumberSignal(QStringLiteral("nFaultCode"),
287 localizationFaultCodeValue(LocalizationFaultCode::CalibrationInvalid));
288 }
289
290 // Build the robot-pickability checker once for the runtime (active camera +
291 // robot config now established); rebuilt later only on active-camera change.
292 rebuildPickingChecker();
293
294 result.valid = result.errors.isEmpty();
295 m_valid = result.valid;
296
297 for (const QString &error : result.errors)
298 LOG_DEV_ERR << "LocalizationRuntimeController setup:" << error;
299
300 if (m_valid) {
301 requestRoleConnectNow(RunnerRole::PrimaryPlc);
302 requestRoleConnectNow(RunnerRole::VisionOutput);
303 requestRoleConnectNow(RunnerRole::Camera);
304 if (allRequiredRolesHealthy()) {
305 markRuntimeReady(QStringLiteral("Runtime ready."));
306 } else {
307 appendTaskLog(QStringLiteral("INFO"),
308 QStringLiteral("Runtime setup valid. Waiting for device connections."));
309 }
310 } else {
311 m_cycleState = CycleState::Faulted;
312 }
313
314 return result;
315}

342void LocalizationRuntimeController::handlePlcValues(const QMap<QString, QVariant> &values)
343{
344 if (values.isEmpty()) {
345 LOG_USER_ERR << "Communication device passed an empty values map.";
346 return;
347 }
348
349 const QList<LocalizationSignalEvent> events = m_signalMapper.mapValues(values);
350 for (const LocalizationSignalEvent &event : events) {
351 emit signalChanged(event.name, event.value);
352
353 if (event.name == QStringLiteral("nActiveCamera")) {
354 emit cameraNumberChanged(event.value);
355 const int cameraNumber = event.value.toInt();
356 if (cameraNumber > 0) {
357 setActiveCameraNumber(cameraNumber);
358 }
359 } else if (event.name == QStringLiteral("nActivePatternGroup")) {
360 emit patternNumberChanged(event.value);
361 bool ok = false;
362 const int patternGroupNumber = event.value.toInt(&ok);
363 if (ok && patternGroupNumber > 0) {
364 setActivePatternGroupNumber(patternGroupNumber);
365 }
366 } else if (event.name == QStringLiteral("bExecuteTrigger")) {
367 const bool trigger = event.value.toBool();
368 const bool risingEdge = trigger && !m_lastExecuteTrigger;
369 const bool fallingEdge = !trigger && m_lastExecuteTrigger;
370 m_lastExecuteTrigger = trigger;
371
372 if (risingEdge) {
373 if (m_cycleState == CycleState::ReadyForTrigger) {
374 startCycle();
375 } else {
376 appendTaskLog(QStringLiteral("WARN"),
377 QStringLiteral("Trigger ignored: task is not ready."));
378 }
379 } else if (fallingEdge &&
380 m_cycleState == CycleState::WaitingTriggerReset) {
381 publishBoolSignal(QStringLiteral("bMatchingFinished"), false);
382 if (m_pendingCycleResult.faulted) {
383 m_cycleState = CycleState::Recovering;
384 } else {
385 markRuntimeReady(QStringLiteral("Trigger reset. Runtime ready."));
386 }
387 }
388 }
389 }

458void LocalizationRuntimeController::bindRoleContext(
459 RunnerRole role,
460 vc::runtime::IDeviceRunner *runner,
461 const QString &deviceId,
462 const LocalizationRecoveryPolicy &policy)
463{
464 clearRoleContext(role);
465
466 if (!runner) {
467 return;
468 }
469
470 RoleRecoveryContext context;
471 context.roleName = policy.roleName;
472 context.deviceId = deviceId;
473 context.runner = runner;
474 context.policy = policy;
475
476 m_recoveryContexts.insert(roleKey(role), context);
477
478 switch (role) {
479 case RunnerRole::PrimaryPlc:
480 connect(runner, &vc::runtime::IDeviceRunner::connectStatusChanged,
481 this, &LocalizationRuntimeController::onPrimaryPlcStatusChanged,
482 Qt::UniqueConnection);
483 connect(runner, &vc::runtime::IDeviceRunner::errorOccurred,
484 this, &LocalizationRuntimeController::onPrimaryPlcError,
485 Qt::UniqueConnection);
486 break;
487 case RunnerRole::VisionOutput:
488 connect(runner, &vc::runtime::IDeviceRunner::connectStatusChanged,
489 this, &LocalizationRuntimeController::onVisionOutputStatusChanged,
490 Qt::UniqueConnection);
491 connect(runner, &vc::runtime::IDeviceRunner::errorOccurred,
492 this, &LocalizationRuntimeController::onVisionOutputError,
493 Qt::UniqueConnection);
494 break;
495 case RunnerRole::Camera:
496 connect(runner, &vc::runtime::IDeviceRunner::connectStatusChanged,
497 this, &LocalizationRuntimeController::onCameraStatusChanged,
498 Qt::UniqueConnection);
499 connect(runner, &vc::runtime::IDeviceRunner::errorOccurred,
500 this, &LocalizationRuntimeController::onCameraError,
501 Qt::UniqueConnection);
502 break;
503 }

764QVector<vc::device::VisionOutputPosition>
765LocalizationRuntimeController::buildVisionOutputPositions(
766 const mtc::MatchResult &matchResult,
767 QVector<ResultRow> *rows,
768 LocalizationFaultCode *faultCode) const
769{
770 QVector<vc::device::VisionOutputPosition> positions;
771 if (faultCode) {
772 *faultCode = LocalizationFaultCode::None;
773 }
774
775 const calib::Calibrator calibrator = m_context.cameraCalibrators.value(m_activeCameraNumber);
776 if (!calibrator.isCalibrated()) {
777 if (faultCode) *faultCode = LocalizationFaultCode::CalibrationInvalid;
778 return positions;
779 }
780
781 int index = 1;
782 for (const mtc::MatchedObject &object : matchResult.Objects) {
783 ResultRow row;
784 row.index = index++;
785 row.patternName = QString::fromStdWString(object.pattern_name);
786 row.score = object.matched_Score;
787 row.imageX = object.point_Center.x;
788 row.imageY = object.point_Center.y;
789 row.imageR = object.point_angle;
790
791 // image to real world position
792 cv::Point2f pick_center = object.point_Center;
793 if (m_context.activeCameraWorkspace.useWorkspace) {
794 pick_center += matchResult.cropOffsetPoint;
795 }
796
797 cv::Point3f worldPoint = calibrator.imageToRobot(pick_center);
798 const double worldR = calibrator.rotateImageToRobot(object.point_angle);
799 // row.world.r = worldR;
800 row.world.r = -worldR;
801
802 worldPoint = calibrator.translateWithZAxis(
803 worldPoint, object.point_offset, row.world.r, false);
804
805 row.world.x = worldPoint.x;
806 row.world.y = worldPoint.y;
807 row.world.z = worldPoint.z;
808
809 // if (object.hasCollision() && object.isOutsideConditionRoi()) {
810 // row.status = QStringLiteral("Skipped: collision, outside");
811 // } else if (object.hasCollision()) {
812 // row.status = QStringLiteral("Skipped: collision");
813 // } else if (object.isOutsideConditionRoi()) {
814 // row.status = QStringLiteral("Skipped: outside");
815 // } else {
816 // row.status = QStringLiteral("Sent");
817 // positions.append(row.world);
818 // }
819
820 if ((!object.hasCollision()) && (!object.isOutsideConditionRoi()) && object.isPossibleToPick()) {
821 if (positions.size() < 2) {
822 row.status = QStringLiteral("Sent");
823 positions.append(row.world);
824 } else {
825 row.status = QStringLiteral("Skipped");
826 }
827 } else {
828 QString tempstr = object.hasCollision() ? tr("Collision") : "";
829 tempstr += (object.isOutsideConditionRoi()
830 ? ((tempstr.isEmpty() ? "" : ", ") + tr("Outside"))
831 : "");
832 tempstr += (!object.isPossibleToPick()
833 ? ((tempstr.isEmpty() ? "" : ", ") + tr("Unpickable"))
834 : "");
835 row.status = QStringLiteral("Skipped: %1").arg(tempstr);
836 }
837
838 if (rows) {
839 rows->append(row);
840 }
841 }
842
843 return positions;
844}

979void LocalizationRuntimeController::handleRoleStatusChanged(RunnerRole role,
980 vc::device::ConnectStatus status)
981{
982 const int key = roleKey(role);
983 if (!m_recoveryContexts.contains(key)) {
984 return;
985 }
986
987 RoleRecoveryContext context = m_recoveryContexts.value(key);
988 if (!context.runner) {
989 m_recoveryContexts.remove(key);
990 return;
991 }
992
993 if (isHealthyStatus(status)) {
994 const int retryHistory = context.retryCount;
995 const bool wasRecovering = retryHistory > 0 || context.retryScheduled;
996 context.retryCount = 0;
997 context.retryScheduled = false;
998 context.faultRaised = false;
999 m_recoveryContexts.insert(key, context);
1000 if (m_cycleState == CycleState::Running ||
1001 m_cycleState == CycleState::WaitingTriggerReset) {
1002 return;
1003 }
1004 if (allRequiredRolesHealthy()) {
1005 const QString readyMessage = wasRecovering
1006 ? buildRecoveryReadyMessage(
1007 context.policy,
1008 context.deviceId,
1009 retryHistory)
1010 : QStringLiteral("Runtime ready.");
1011 markRuntimeReady(readyMessage);
1012 }
1013 return;
1014 }
1015
1016 if (m_cycleState == CycleState::Running) {
1017 switch (role) {
1018 case RunnerRole::Camera:
1019 abortCycle(LocalizationFaultCode::CameraLost,
1020 QStringLiteral("Active camera lost during localization cycle."));
1021 break;
1022 case RunnerRole::VisionOutput:
1023 abortCycle(LocalizationFaultCode::VisionOutputLost,
1024 QStringLiteral("Vision output lost during localization cycle."));
1025 break;
1026 case RunnerRole::PrimaryPlc:
1027 abortCycle(LocalizationFaultCode::PlcLost,
1028 QStringLiteral("Primary PLC lost during localization cycle."));
1029 break;
1030 }
1031 }
1032
1033 const LocalizationRecoveryAction action = decideRecoveryAction(
1034 context.policy,
1035 status,
1036 context.retryCount,
1037 context.retryScheduled);
1038
1039 if (action == LocalizationRecoveryAction::RetryScheduled) {
1040 context.retryScheduled = true;
1041 context.retryCount += 1;
1042 m_recoveryContexts.insert(key, context);
1043 emit runtimeRecovering(buildRecoveryProgressMessage(
1044 context.policy,
1045 context.deviceId,
1046 context.retryCount,
1047 status));
1048 scheduleRoleReconnect(role);
1049 return;
1050 }
1051
1052 if (action == LocalizationRecoveryAction::EscalateFault) {
1053 context.faultRaised = true;
1054 m_recoveryContexts.insert(key, context);
1055 raiseRoleFault(role, status);
1056 }
1057}
