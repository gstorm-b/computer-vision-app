88 DeviceCommandResult submitCommand(const DeviceCommand &command) override
89 {
90 if (command.targetDeviceId != m_device->id()) {
91 DeviceCommandResult result = DeviceCommandResult::rejected(
92 command,
93 DeviceCommandResultCode::InvalidTarget,
94 QStringLiteral("Camera command target does not match this runner."));
95 finishRejectedCommand(result);
96 return result;
97 }
98
99 if (!isSupportedCommand(command.kind)) {
100 DeviceCommandResult result = DeviceCommandResult::rejected(
101 command,
102 DeviceCommandResultCode::UnsupportedCommand,
103 QStringLiteral("Unsupported camera command."));
104 finishRejectedCommand(result);
105 return result;
106 }
107
108 const DeviceCommandQueuePolicy policy = queuePolicyFor(command.kind);
109 const DeviceCommandEnqueueDecision decision =
110 m_commandQueue.enqueue(command, policy, hasActiveCommand());
111 if (!decision.accepted) {
112 finishRejectedCommand(decision.rejection);
113 return decision.rejection;
114 }
115
116 if (decision.shouldRunNow) {
117 runCommand(command);
118 return DeviceCommandResult::accepted(command);
119 }
120
121 const QString queueMessage = decision.replacedPending
122 ? QStringLiteral("Camera command queued (replaced pending same-kind request).")
123 : QStringLiteral("Camera command queued.");
124 LOG_USER_INFO << "Camera command queued."
125 << "id=" << command.id
126 << "kind=" << deviceCommandKindToString(command.kind)
127 << "target=" << command.targetDeviceId
128 << "pending=" << decision.pendingCount;
129 return DeviceCommandResult::accepted(command, queueMessage);
130 }

334 static DeviceCommandQueuePolicy queuePolicyFor(DeviceCommandKind kind)
335 {
336 switch (kind) {
337 case DeviceCommandKind::Connect:
338 case DeviceCommandKind::Disconnect:

344 case DeviceCommandKind::Unknown:

351 bool hasActiveCommand() const

360 void runCommand(const DeviceCommand &command)
361 {
362 const TriggerSignal trigger = triggerFor(command.kind);
363 if (trigger == nullptr) {
364 DeviceCommandResult result = DeviceCommandResult::rejected(
365 command,
366 DeviceCommandResultCode::UnsupportedCommand,
367 QStringLiteral("Camera command trigger is not available."));
368 finishRejectedCommand(result);
369 return;
370 }
371
372 m_activeCommand = command;
373 m_activeCommandTimer.start(activeTimeoutMs(command));
374 LOG_USER_INFO << "Camera command started."
375 << "id=" << command.id
376 << "kind=" << deviceCommandKindToString(command.kind)
377 << "target=" << command.targetDeviceId
378 << "timeoutMs=" << activeTimeoutMs(command)
379 << "pending=" << m_commandQueue.pendingCount();
380 (this->*trigger)();
381 }
