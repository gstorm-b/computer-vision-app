128 m_browser->setFactoryForManager(m_pointFManager->subDoubleManager(), m_dblFactory);
129 m_browser->setFactoryForManager(m_pointManager->subIntManager(), m_intFactory);

286// ── Filter implementation ─────────────────────────────────────────────────────
287
291bool PropertyBrowserWidget::itemOrChildMatchesFilter(QtBrowserItem *item,
292 const QString &lowerText) const
293{
294 if (lowerText.isEmpty()) return true;
295
296 const QString name = item->property()->displayName().toLower();
297 if (name.contains(lowerText)) {
298 return true;
299 }
300
301 for (auto *child : item->children())
302 if (itemOrChildMatchesFilter(child, lowerText)) {
303 return true;
304 }
305
306 return false;
307}

311void PropertyBrowserWidget::filterTopLevel(const QString &lowerText)
312{
313 // not only shows matching items, but also shows same-level siblings
314 for (auto *item : m_browser->topLevelItems()) {
315 const bool visible = itemOrChildMatchesFilter(item, lowerText);
316 m_browser->setItemVisible(item, visible);
317 }
