139void TaskRunner::enterRuntime(bool mergeToTaskThread)
140{
141 if (m_phase == Phase::Runtime) return;
142
143 if (mergeToTaskThread) {
144 // Detach all devices from their per-device threads and move them into
145 // the single runtime thread, then stop the now-empty per-device threads.
146 for (IDeviceRunner *runner : m_runners) {
147 runner->detach(m_runtimeThread);
148 runner->stop();
149 }
150 m_runtimeThread->start(QThread::HighPriority);
151 LOG_DEV_DEBUG << "TaskRunner: → Runtime phase (merged to task thread)";
152 } else {
153 // Keep per-device threads running.
154 // Start any runner that isn't running yet (e.g. fresh Runtime without Commission).
155 for (IDeviceRunner *runner : m_runners) {
156 if (!runner->isRunning()) {
157 runner->start();
158 runner->attach();
159 }
160 }
161 // Task runtime thread runs as coordinator alongside device threads.
162 m_runtimeThread->start(QThread::NormalPriority);
163 LOG_DEV_DEBUG << "TaskRunner: → Runtime phase (per-device threads)";
164 }
165
166 m_phase = Phase::Runtime;
167 emit phaseChanged(m_phase);
168}

173void TaskRunner::enterIdle()
174{
175 if (m_phase == Phase::Idle) return;
176
177 for (IDeviceRunner *runner : m_runners) {
178 if (runner->isAttached()) {
179 // Close the device connection on its own worker thread before
180 // detaching/stopping. Moving a live socket across threads or
181 // stopping the worker while the link is open leaks the connection
182 // and can crash on the next phase entry. See
183 // docs/task_localization for the lifecycle contract.
184 runner->disconnectAndWait();
185 runner->detach(nullptr);
186 }
187
188 if (runner->isRunning()) {
189 runner->stop();
190 }
191 }
192
193 if (m_runtimeThread->isRunning()) {
194 m_runtimeThread->quit();
195 m_runtimeThread->wait(3000);
196 }
197
198 m_phase = Phase::Idle;
199 LOG_DEV_DEBUG << "TaskRunner: → Idle phase";
200 emit phaseChanged(m_phase);
201}

209IDeviceRunner *TaskRunner::createRunner(std::shared_ptr<vc::device::IDevice> device)
210{
211 using namespace vc::device;
212
213 switch (device->deviceType()) {
214 case DeviceType::Camera:
215 if (auto *cam = qobject_cast<CameraDevice *>(device.get()))
216 return new CameraRunner(cam, this);
217 break;
218
219 case DeviceType::PLC:
220 if (auto *plc = qobject_cast<PlcDevice *>(device.get()))
221 return new PlcRunner(plc, this);
222 break;
223
224 case DeviceType::VisionOutput:
225 if (auto *vision_output = qobject_cast<VisionOutputDevice *>(device.get()))
226 return new VisionOutputRunner(vision_output, this);
227 break;
228
229 default:
230 break;
231 }
232 return nullptr;
233}
