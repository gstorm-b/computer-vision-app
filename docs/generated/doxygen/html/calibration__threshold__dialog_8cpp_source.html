172void CalibrationThresholdDialog::recompute()
173{
174 if (!m_board || !m_binView || !m_overlayView || !m_status) {
175 return;
176 }
177
178 if (m_image.empty()) {
179 m_binView->clearCurrentImage();
180 m_overlayView->clearCurrentImage();
181 m_status->setText(tr("Grab a frame from the camera to preview."));
182 return;
183 }
184
185 const int trial = trialThreshold();
186 m_board->setBinarizeThreshold(trial);
187
188 double used = 0.0;
189 cv::Mat binary = m_board->binarize(m_image, &used);
190 // fitsize=false keeps the current zoom/pan while the user drags the slider.
191 m_binView->loadImageOpenCv(binary, false);
192
193 std::vector<cv::Point2f> allPts;
194 std::vector<cv::Point2f> corners;
195 cv::Mat overlay;
196 const bool ok = m_board->detect(m_image, allPts, &corners, &overlay);
197 cv::Mat overlayShow = overlay.empty() ? m_image : overlay;
198 m_overlayView->loadImageOpenCv(overlayShow, false);
199
200 const QString modeStr = (trial < 0)
201 ? tr("Auto (Otsu = %1)").arg(static_cast<int>(std::lround(used)))
202 : tr("Manual = %1").arg(trial);
203
204 if (ok) {
205 m_status->setText(tr("Threshold: %1 | Detect: OK (%2 / %3 dots)")
206 .arg(modeStr)
207 .arg(static_cast<int>(allPts.size()))
208 .arg(m_board->totalDots()));
209 } else {
210 m_status->setText(tr("Threshold: %1 | Detect: FAILED — adjust the threshold")
211 .arg(modeStr));
212 }
213}
