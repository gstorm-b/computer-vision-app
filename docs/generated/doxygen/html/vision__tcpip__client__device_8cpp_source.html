118void VisionTcpipClientDevice::onLinkLost() {
119 // A link dropped. If we were fully Connected, fall back to Connecting; a
120 // heartbeat-loss path already set LostConnected and must stay visible
121 // until the redial actually starts (handled in attemptConnect()).
122 if (connectStatus() == ConnectStatus::Connected) {
123 setConnectionStatus(ConnectStatus::Connecting);
124 }
125 scheduleReconnect();
126}

135void VisionTcpipClientDevice::attemptConnect() {
136 if (!isActive()) {
137 return;
138 }
139
140 // Redialing (e.g. after LostConnected): reflect the in-progress state.
141 if (connectStatus() != ConnectStatus::Connected
142 && connectStatus() != ConnectStatus::Connecting) {
143 setConnectionStatus(ConnectStatus::Connecting);
144 }
145
146 if (!m_mainSocket && !m_mainConnector) {
147 m_mainConnector = new QTcpSocket(this);
148 connect(m_mainConnector, &QTcpSocket::connected,
149 this, &VisionTcpipClientDevice::onMainConnected);
150 connect(m_mainConnector, &QAbstractSocket::errorOccurred,
151 this, &VisionTcpipClientDevice::onMainConnectError);
152 m_mainConnector->connectToHost(m_config.m_serverAddress,
153 static_cast<quint16>(m_config.m_mainPort));
154 }
155
156 if (!m_hbSocket && !m_hbConnector) {
157 m_hbConnector = new QTcpSocket(this);
158 connect(m_hbConnector, &QTcpSocket::connected,
159 this, &VisionTcpipClientDevice::onHeartbeatConnected);
160 connect(m_hbConnector, &QAbstractSocket::errorOccurred,
161 this, &VisionTcpipClientDevice::onHeartbeatConnectError);
162 m_hbConnector->connectToHost(m_config.m_serverAddress,
163 static_cast<quint16>(m_config.m_heartbeatPort));
164 }
165}

170void VisionTcpipClientDevice::onMainConnected() {
171 QTcpSocket *sock = m_mainConnector;
172 m_mainConnector = nullptr;
173 if (!sock) return;
174
175 // Drop the dialing-stage handlers; the base re-wires readyRead/disconnected.
176 sock->disconnect(this);
177 attachMainSocket(sock);
178 evaluateConnected();
179}

232void VisionTcpipClientDevice::scheduleReconnect() {
233 if (!isActive()) {
234 return;
235 }
236 if (!m_reconnectTimer) {
237 m_reconnectTimer = new QTimer(this);
238 m_reconnectTimer->setSingleShot(true);
239 connect(m_reconnectTimer, &QTimer::timeout,
240 this, &VisionTcpipClientDevice::attemptConnect);
241 }
242 if (!m_reconnectTimer->isActive()) {
243 m_reconnectTimer->start(m_config.m_reconnectIntervalMs);
244 }
245}

250void VisionTcpipClientDevice::evaluateConnected() {
251 // Connected is reported ONLY when both the main and heartbeat links are up.
252 if (isActive() && m_mainSocket && m_hbSocket
253 && m_mainSocket->state() == QAbstractSocket::ConnectedState
254 && m_hbSocket->state() == QAbstractSocket::ConnectedState) {
255 if (connectStatus() != ConnectStatus::Connected) {
256 setConnectionStatus(ConnectStatus::Connected);
257 }
258 }
