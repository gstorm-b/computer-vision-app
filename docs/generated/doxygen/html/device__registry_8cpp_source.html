210const DeviceRegistryEntry *DeviceRegistry::find(const QJsonObject &obj,
211 DeviceType deviceType)
212{
213 for (const DeviceRegistryEntry &entry : kEntries) {
214 if (entry.deviceType != deviceType) {

216 }

225QStringList DeviceRegistry::displayNamesFor(DeviceType deviceType,
226 bool includeUnavailable)
227{
228 QStringList names;
229 for (const DeviceRegistryEntry &entry : kEntries) {
230 if (entry.deviceType != deviceType) {
231 continue;
232 }
233 if (!includeUnavailable && !entry.available) {
234 continue;
235 }
236 names.append(entry.subTypeValue);
237 }
