244void AddDeviceWizard::onAddClicked() {
245 const QString name = ui->adwNameEdit->text().trimmed();
246
247 if (name.isEmpty()) {
248 QMessageBox::warning(this, tr("Add Device"), tr("Please enter a device name."));
249 ui->adwNameEdit->setFocus();
250 return;
251 }
252
253 if (m_manager->isNameExists(name)) {
254 QMessageBox::warning(this, tr("Add Device"), tr("Device name already exists."));
255 ui->adwNameEdit->selectAll();
256 return;
257 }
258
259 QJsonObject jobj = buildDeviceJson(m_selectedType);
260 jobj[DEVICE_JSK_ID] = m_pendingDeviceId;
261 jobj[DEVICE_JSK_NAME] = name;
262 jobj[DEVICE_JSK_TYPE] = vc::device::DeviceTypeToString(m_selectedType);
263
264 std::shared_ptr<vc::device::IDevice> dv(
265 vc::device::DeviceFactory::create(m_selectedType, jobj));
266
267 if (m_manager->commitDevice(m_pendingDeviceId, name, dv)) {
268 LOG_USER_INFO << QString("Device added: %1 (%2)").arg(name, m_pendingDeviceId);
269 accept();
270 } else {
271 QMessageBox::critical(this, tr("Add Device"),
272 tr("Failed to register device. Please close and try again."));
273 }
274}

291QJsonObject AddDeviceWizard::buildDeviceJson(vc::device::DeviceType type) {
292 QJsonObject obj;
293
294 switch (type) {
295 case vc::device::DeviceType::Camera:
296 if (ui->cbxCameraType->count() > 0)
297 obj[DEVICE_JSK_CAM_TYPE] = ui->cbxCameraType->currentText();
298 break;
299
300 case vc::device::DeviceType::PLC: {
301 // The PLC page had no sub-type combo at all until the virtual PLC needed one: it
302 // hard-coded a Mitsubishi MC config, which was correct while MC was the only PLC.
303 // The frame-type and data-code controls below belong to MC specifically and mean
304 // nothing for a virtual PLC, so the sub-type is read first and MC's fields are only
305 // consulted on the MC branch.
306 const auto subType = (ui->cbxPlcType->count() > 0)
307 ? vc::device::PlcTypeFromString(ui->cbxPlcType->currentText())
308 : vc::device::PlcType::MitsubishiMc;
309
310 if (subType == vc::device::PlcType::VirtualPlc) {
311 vc::device::VirtualPlcCfg config;
312 obj[DEVICE_JSK_CONFIG] = config.toJson();
313 break;
314 }
315
316 vc::device::McProtocolConfig config;
317 auto frameType = (ui->cbxMcFrameType->count() > 0)
318 ? vc::device::mc::McFrameTypeFromString(ui->cbxMcFrameType->currentText())
319 : vc::device::mc::McFrameType::Frame_3E;
320 auto dataCode = (ui->cbxMcCode->count() > 0)
321 ? vc::device::mc::McDataCodeFromString(ui->cbxMcCode->currentText())
322 : vc::device::mc::McDataCode::Binary;
323 config.configMcProtocol(frameType, dataCode);
324 obj[DEVICE_JSK_CONFIG] = config.toJson();
325 break;
326 }
327
328 case vc::device::DeviceType::VisionOutput: {
329 auto subType = (ui->cbxVisionType->count() > 0)
330 ? vc::device::VisionOutputTypeFromString(ui->cbxVisionType->currentText())
331 : vc::device::VisionOutputType::VisionTCPIP;
332 switch (subType) {
333 case vc::device::VisionOutputType::VisionTCPIP: {
334 vc::device::VisionTcpipDeviceCfg config;
335 obj[DEVICE_JSK_CONFIG] = config.toJson();
336 break;
337 }
338 case vc::device::VisionOutputType::VisionTcpipClient: {
339 vc::device::VisionTcpipClientDeviceCfg config;
340 obj[DEVICE_JSK_CONFIG] = config.toJson();
341 break;
342 }
343 case vc::device::VisionOutputType::VirtualVisionOutput: {
344 vc::device::VirtualVisionOutputCfg config;
345 obj[DEVICE_JSK_CONFIG] = config.toJson();
346 break;
347 }
348 case vc::device::VisionOutputType::VisionSerial:
349 case vc::device::VisionOutputType::VisionOutputTypeNone:
350 // No concrete impl yet — leave config empty; factory will reject.
351 break;
352 }
353 break;
354 }
355
356 default:
357 break;
358 }
359
360 return obj;
361}
