102 bool fromJson(const QJsonObject &obj)
103 {
104 m_role = taskDeviceRoleFromString(obj["role"].toString());
105 m_deviceId = obj["deviceId"].toString();
106 m_cameraNumber = obj["cameraNumber"].toInt(0);
107
108 if (m_role == TaskDeviceRole::Unknown)
109 return false;
110
111 // Reject malformed/hostile imports: over-long device ids and out-of-range
112 // camera numbers. Invalid bindings are dropped by the bool-return
113 // convention in TaskDeviceBindings::fromJson().
114 if (m_deviceId.size() > kMaxDeviceIdLength)
115 return false;
116
117 if (m_role == TaskDeviceRole::CameraNumber &&
118 (m_cameraNumber < kMinCameraNumber || m_cameraNumber > kMaxCameraNumber))
119 return false;
120
121 return true;
122 }
