19StartupDecision resolveSingleInstance(SingleInstanceGuard &guard, ShellKind self)
20{
21 const bool handoff = ShellHandoff::wasStartedForHandoff(QCoreApplication::arguments());
22
23 LOG_USER_INFO << "Application starting."
24 << "shell=" << ShellHandoff::applicationName(self)
25 << "version=" << QCoreApplication::applicationVersion()
26 << "handoff=" << handoff;
27
28 if (guard.tryAcquire(handoff ? ShellHandoff::kHandoffAcquireTimeoutMs : 0)) {
29 return StartupDecision::Proceed;
30 }
31
32 // Read the holder BEFORE notifying it: once it comes forward it may exit, and then
33 // there is nothing left to name in the message.
34 const QString holder = guard.runningInstanceName();
35 const QString ownName = ShellHandoff::applicationName(self);
36
37 if (handoff) {
38 // The sibling this process was launched to replace is still holding the lock after
39 // the full timeout. Do not exit silently — from the user's side that is
40 // indistinguishable from the switch doing nothing at all.
41 LOG_USER_ERR << "Shell hand-off failed: the previous application did not exit."
42 << "holder=" << holder;
43 QMessageBox::warning(
44 nullptr,
45 ownName,
46 QCoreApplication::translate(
47 "ShellStartup",
48 "The previous application is still shutting down, so %1 could not start.\n\n"
49 "Wait a few seconds and start it again.")
50 .arg(ownName));
51 return StartupDecision::ExitWithError;
52 }
53
54 if (!holder.isEmpty() && holder != ownName) {
55 // The OTHER shell holds the lock. Say so before raising it, or the user clicks one
56 // icon and a different application appears.
57 LOG_USER_INFO << "Launch blocked: the other shell is running." << "holder=" << holder;
58 QMessageBox::information(
59 nullptr,
60 ownName,
61 QCoreApplication::translate(
62 "ShellStartup",
63 "%1 is already running, and only one of the two applications can run at a "
64 "time — they share the camera, the PLC connection and the output port.\n\n"
65 "To switch, use Project → Open Editor (or Open Runtime) in the running "
66 "application.")
67 .arg(holder));
68 }
69
70 // Same shell, or an unreadable lock: bring the running window forward and go away.
71 guard.notifyExistingInstance();
72 return StartupDecision::ExitQuietly;
73}

76void presentShellWindow(QWidget *window)
77{
78 if (window == nullptr) {
79 return;
80 }
81
82 window->show();
83
84 if (!ShellHandoff::wasStartedForHandoff(QCoreApplication::arguments())) {
85 return; // A user-initiated launch comes forward on its own.
86 }
87
88 // Claim the foreground twice: once now, once on the first turn of the event loop.
89 //
90 // Not belt-and-braces. show() creates and maps the native window synchronously, so the
91 // immediate claim normally succeeds — and taking it here means the claim happens before
92 // anything else this process posts, rather than behind a queue that a shell with
93 // running device threads fills up. But a claim made before the window manager has
94 // finished with the new window can still be dropped, which is what the deferred turn
95 // covers. Asking twice costs nothing: the second call on an already-active window is a
96 // no-op, and Windows revokes the hand-off grant on user input, not on a spent request.
97 const auto claimForeground = [](QWidget *w) {
98 // Clear a minimised state first, or "restore" puts the window back on the taskbar
99 // rather than in front of the operator — the same ordering the single-instance
100 // raise path needs.
101 w->setWindowState((w->windowState() & ~Qt::WindowMinimized) | Qt::WindowActive);
102 w->raise();
103 w->activateWindow();
104 };
105
106 claimForeground(window);
107 QTimer::singleShot(0, window, [window, claimForeground]() {
108 claimForeground(window);
109 // Logged as "asked", not "succeeded": activateWindow() returns void and Windows
110 // silently downgrades a refused request to a flashing taskbar button, so this line
111 // prints in the failing case too. It marks the attempt, not the outcome.
112 LOG_DEV_INFO << "Hand-off launch: foreground claimed for the new shell window.";
113 });
114}

117bool requestShellSwitch(QWidget *parent,
118 ShellKind self,
119 const std::function<bool()> &releaseResources)
120{
121 const ShellKind target = ShellHandoff::siblingOf(self);
122 const QString targetName = ShellHandoff::applicationName(target);
123
124 // 1. Authorise. Only entering commissioning needs it: going the other way hands
125 // authority back, and nobody needs permission to give up permission.
126 if (target == ShellKind::Commissioning && !AdminLoginDialog::ensureAdmin(parent)) {
127 LOG_USER_INFO << "Shell switch cancelled: administrator access was not granted.";
128 return false;
129 }
130
131 // 2. Confirm — in both directions, and even when the user already holds Admin. This
132 // guards against a mis-tap, which authorisation does not.
133 const QMessageBox::StandardButton answer = QMessageBox::question(
134 parent,
135 QCoreApplication::translate("ShellStartup", "Switch to %1").arg(targetName),
136 QCoreApplication::translate(
137 "ShellStartup",
138 "This will stop all running tasks and release the camera, the PLC connection "
139 "and the output port, then close this application and start %1.\n\n"
140 "Continue?")
141 .arg(targetName),
142 QMessageBox::Yes | QMessageBox::No,
143 QMessageBox::No);
144
145 if (answer != QMessageBox::Yes) {
146 LOG_USER_INFO << "Shell switch cancelled by the user." << "target=" << targetName;
147 return false;
148 }
149
150 LOG_USER_INFO << "Shell switch requested." << "target=" << targetName;
151
152 // 3. Release. If this fails the switch is abandoned: the sibling would come up against
153 // devices this process has not let go of, and the symptoms of that look exactly
154 // like a hardware fault.
155 if (releaseResources && !releaseResources()) {
156 LOG_USER_ERR << "Shell switch abandoned: this application could not release its "
157 "devices.";
158 QMessageBox::critical(
159 parent,
160 QCoreApplication::translate("ShellStartup", "Switch to %1").arg(targetName),
161 QCoreApplication::translate(
162 "ShellStartup",
163 "This application could not stop its running work, so the switch was "
164 "cancelled. Nothing has changed."));
165 return false;
166 }
167
168 // 4. Launch the sibling. It waits for the instance lock this process still holds.
169 QString error;
170 if (!ShellHandoff::launchSibling(self, &error)) {
171 QMessageBox::critical(
172 parent,
173 QCoreApplication::translate("ShellStartup", "Switch to %1").arg(targetName),
174 error);
175 // Deliberately does NOT quit: this shell has stopped its work, but it is still the
176 // only thing on screen. Leaving the operator with a running application they can
177 // restart from beats leaving them with nothing.
178 return false;
179 }
180
181 // 5. Quit, which releases the lock the sibling is waiting on.
182 LOG_USER_INFO << "Shell switch: exiting so the sibling can take over.";
183 QCoreApplication::quit();
184 return true;
185}
