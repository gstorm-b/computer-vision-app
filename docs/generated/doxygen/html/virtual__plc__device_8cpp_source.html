13bool isMcTag(const QString &tag, QChar expectedPrefix)
14{
15 const QString trimmed = tag.trimmed();
16 if (trimmed.size() < 2 ||
17 trimmed.at(0).toUpper() != expectedPrefix.toUpper()) {
18 return false;
