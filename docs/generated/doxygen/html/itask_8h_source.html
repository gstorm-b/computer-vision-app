210 virtual bool fromJson(const QJsonObject& obj) {
211 if (!obj.contains("id") ||
212 !obj.contains("name") ||
213 !obj.contains("taskType") ||
214 !obj.contains("taskConfig")) {
215 LOG_DEV_ERR << "Cannot convert task from json, wrong json format.";
216 LOG_DEV_ERR << QJsonDocument(obj).toJson();
217 return false;
218 }
219
220 TaskType task_type = stringToQEnum(obj["taskType"].toString(), TaskType::UndefineTask);
221 if (task_type != this->taskType()) {
222 LOG_DEV_ERR << "Cannot convert task from json, wrong task type.";
223 LOG_DEV_ERR << obj["taskType"].toString();
224 return false;
225 }
226
227 m_id = obj["id"].toString();
228 m_name = obj["name"].toString();
229
230 if (obj.contains("assignedDeviceIds")) {
231 m_assignedDeviceIds.clear();
232 for (const QJsonValue &v : obj["assignedDeviceIds"].toArray())
233 m_assignedDeviceIds.append(v.toString());
234 }
235
236 if (m_abstract_cfg) {
237 return m_abstract_cfg->fromJson(obj["taskConfig"].toObject());
238 }
239 return true;
240 }
