29double jointLimitMarginCost(const SerialRobotConfig& config, const JointVector& joints)
30{
31 double cost = 0.0;
32 int index = 0;
33 for (const Joint& joint : config.joints) {
34 if (joint.type != JointType::Revolute && joint.type != JointType::Prismatic) {
35 continue;
36 }
37 if (joint.limits.has_value()) {
38 const double range = joint.limits->upper - joint.limits->lower;
39 const double margin = std::min(joints[index] - joint.limits->lower, joint.limits->upper - joints[index]);
40 if (range > 0.0) {
41 cost += 1.0 / std::max(margin / range, 1e-6);
42 }
43 }
44 ++index;
45 }
46 return cost;
47}

53double postureMismatchCost(const ArmPosture& candidate, const ArmPosture& requested)
54{
55 double cost = 0.0;
56 if (!optionalBranchMatches(candidate.shoulder, requested.shoulder)) {
57 cost += 1.0;
58 }
59 if (!optionalBranchMatches(candidate.elbow, requested.elbow)) {
60 cost += 1.0;
61 }
62 if (!optionalBranchMatches(candidate.wrist, requested.wrist)) {
63 cost += 1.0;
64 }
65 for (const auto& requestedVendor : requested.vendorSpecific) {
66 const auto candidateVendor = candidate.vendorSpecific.find(requestedVendor.first);
67 if (candidateVendor == candidate.vendorSpecific.end() || candidateVendor->second != requestedVendor.second) {
68 cost += 1.0;
69 }
70 }
71 return cost;
72}

89IKSolution IKSolutionRanker::score(const SerialRobotConfig& config,
90 const IKRequest& request,
91 JointVector joints,
92 double positionError_m,
93 double orientationError_rad,
94 double jointLimitAvoidanceWeight)
95{
96 IKSolution solution;
97 solution.joints = std::move(joints);
98 solution.positionError_m = positionError_m;
99 solution.orientationError_rad = orientationError_rad;
100
101 if (std::unique_ptr<PostureResolver> resolver = PostureResolverFactory::create(config)) {
102 const Result<ArmPosture> posture = resolver->classify(config, solution.joints);
103 if (posture.ok()) {
104 solution.posture = posture.value;
105 }
106 }
107
108 solution.score.jointLimitMarginCost = jointLimitAvoidanceWeight * jointLimitMarginCost(config, solution.joints);
109 if (request.seedJoint.has_value() && request.seedJoint->size() == solution.joints.size()) {
110 solution.score.seedDistanceCost = (solution.joints.values() - request.seedJoint->values()).norm();
111 }
112 if (request.previousJoint.has_value() && request.previousJoint->size() == solution.joints.size()) {
113 solution.score.motionContinuityCost = (solution.joints.values() - request.previousJoint->values()).norm();
114 }
115 if (request.posture.has_value()) {
116 solution.score.postureMismatchCost = 100.0 * postureMismatchCost(solution.posture, *request.posture);
117 }
118 solution.score.totalCost = solution.positionError_m + solution.orientationError_rad
119 + solution.score.seedDistanceCost + solution.score.motionContinuityCost
120 + solution.score.jointLimitMarginCost + solution.score.postureMismatchCost;
121 return solution;
122}
