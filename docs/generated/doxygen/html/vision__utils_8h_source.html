70static bool imwrite_wstring(const std::wstring& filename,
71 const cv::Mat& img,
72 const std::vector<int>& params = std::vector<int>()) {
73 // Encode the image into a buffer
74 std::vector<uchar> buffer;
75 // Extract file extension to determine format (e.g., ".png", ".jpg")
76 std::wstring extension = filename.substr(filename.find_last_of(L"."));
77 std::string narrowExtension(extension.begin(), extension.end());
78
79 if (!cv::imencode(narrowExtension, img, buffer, params)) {
80 std::wcerr << L"Error: Could not encode image to buffer" << std::endl;
81 return false;
82 }
83
84 // Write the buffer to a file with the wide string name in binary mode
85 std::ofstream file(filename, std::ios::binary);
86 if (!file.is_open()) {
87 std::wcerr << L"Error: Could not open file for writing " << filename << std::endl;
88 return false;
89 }
90
91 file.write(reinterpret_cast<const char*>(buffer.data()), buffer.size());
92 file.close();
93
94 return true;
95}

134static std::vector<cv::Point> tNeighborSimplify(const std::vector<cv::Point>& contour,
135 int T = 5, double tolerance = 2.0) {
136 const size_t N = contour.size();
137 if (N < 2 || T < 2) return contour; // Nothing to do
138
139 std::vector<cv::Point> simplified;
140 simplified.reserve(N);
141 simplified.push_back(contour.front()); // Always keep the first point
142
143 size_t i = 0;
144 while (i < N - 1) {
145 bool compressed = false;
146 // j goes from far neighbor (i+T) back to the immediate next (i+1)
147 const size_t j_max = std::min(i + static_cast<size_t>(T), N - 1);
148 for (size_t j = j_max; j > i + 0; --j) {
149 const cv::Point& p1 = simplified.back(); // start point
150 const cv::Point& p2 = contour[j]; // candidate end point
151 const cv::Point2f seg = p2 - p1;
152 const double seg_len = cv::norm(seg);
153 if (seg_len == 0.0) continue; // degenerate; skip
154
155 // Compute max distance of intermediate points k ∈ (i, j)
156 double dmax = 0.0;
157 for (size_t k = i + 1; k < j; ++k) {
158 const cv::Point& pk = contour[k];
159 // perpendicular distance from pk to line p1‑p2
160 double cross = std::abs(seg.x * (pk.y - p1.y) - seg.y * (pk.x - p1.x));
161 double dist = cross / seg_len;
162 if (dist > dmax) dmax = dist;
163 if (dmax > tolerance) break; // early exit
164 }
165
166 if (dmax <= tolerance) {
167 // All intermediate points are close enough -> compress them
168 simplified.push_back(p2);
169 i = j; // jump forward
170 compressed = true;
171 break;
172 }
173 }
174
175 if (!compressed) {
176 // Could not compress with any neighbor → keep next point
177 simplified.push_back(contour[i + 1]);
178 ++i;
179 }
180 }
181
182 return simplified;
183}

218static void computePcaOrientation(const std::vector<cv::Point>& pts, double& angleOutput, cv::Point2f& centerOutput) {
219 int sz = static_cast<int>(pts.size());
220 cv::Mat data_pts = cv::Mat(sz, 2, CV_64F);
221 for (int i = 0; i < data_pts.rows; i++)
222 {
223 data_pts.at<double>(i, 0) = pts[i].x;
224 data_pts.at<double>(i, 1) = pts[i].y;
225 }
226 // Perform PCA analysis
227 cv::PCA pca_analysis(data_pts, cv::Mat(), cv::PCA::DATA_AS_ROW);
228 // Store the center of the object
229 cv::Point cntr = cv::Point(static_cast<int>(pca_analysis.mean.at<double>(0, 0)),
230 static_cast<int>(pca_analysis.mean.at<double>(0, 1)));
231 // Store the eigenvalues and eigenvectors
232 std::vector<cv::Point2d> eigen_vecs(2);
233 std::vector<double> eigen_val(2);
234 for (int i = 0; i < 2; i++)
235 {
236 eigen_vecs[i] = cv::Point2d(pca_analysis.eigenvectors.at<double>(i, 0),
237 pca_analysis.eigenvectors.at<double>(i, 1));
238 eigen_val[i] = pca_analysis.eigenvalues.at<double>(i);
239 }
240
241 double angle = atan2(eigen_vecs[0].y, eigen_vecs[0].x); // orientation in radians
242 angleOutput = angle;
243 centerOutput = cntr;
244}

253static double median(cv::Mat Input) {
254 double m = (Input.rows*Input.cols) / 2;
255 int bin = 0;
256 double med = -1.0;
257
258 int histSize = 256;
259 float range[] = { 0, 256 };
260 const float* histRange = { range };
261 bool uniform = true;
262 bool accumulate = false;
263 cv::Mat hist;
264 cv::calcHist( &Input, 1, 0, cv::Mat(), hist, 1, &histSize, &histRange, uniform, accumulate );
265
266 for ( int i = 0; i < histSize && med < 0.0; ++i )
267 {
268 bin += cvRound( hist.at< float >( i ) );
269 if ( bin > m && med < 0.0 )
270 med = i;
271 }
272
273 return med;
274}

330static cv::Mat binarizeSourceImage(const cv::Mat& src, int threshold = -1,
331 int maxValue = 255, double* usedThreshold = nullptr) {
332 if (src.empty()) return cv::Mat();
333
334 cv::Mat gray;
335 if (src.channels() == 3)
336 cv::cvtColor(src, gray, cv::COLOR_BGR2GRAY);
337 else if (src.channels() == 4)
338 cv::cvtColor(src, gray, cv::COLOR_BGRA2GRAY);
339 else
340 gray = src.clone();
341
342 cv::GaussianBlur(gray, gray, cv::Size(5, 5), 0);
343
344 cv::Mat binary;
345 double applied = 0.0;
346 if (threshold < 0) {
347 applied = cv::threshold(gray, binary, 0, maxValue,
348 cv::THRESH_BINARY | cv::THRESH_OTSU);
349 } else {
350 applied = threshold;
351 cv::threshold(gray, binary, threshold, maxValue, cv::THRESH_BINARY);
352 }
353 if (usedThreshold) *usedThreshold = applied;
354 return binary;
355}
