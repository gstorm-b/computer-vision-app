83QPointF ItemGripperBox::pointAtAngleAndDistance(qreal angleDeg, qreal dist) const {
84 qreal a = qDegreesToRadians(angleDeg);
85 // Qt coordinate: +x sang phải, +y xuống dưới -> y = -sin(angle) để angle tính theo độ toán học (ccw)
86 return QPointF(dist * qCos(a), -dist * qSin(a));

110 QRectF unionRect = r1.united(r2);
