84bool VisionTcpipDevice::startTransport() {
85 m_mainServer = new QTcpServer(this);
86 m_hbServer = new QTcpServer(this);
87
88 connect(m_mainServer, &QTcpServer::newConnection,
89 this, &VisionTcpipDevice::onMainNewConnection);
90 connect(m_hbServer, &QTcpServer::newConnection,
91 this, &VisionTcpipDevice::onHeartbeatNewConnection);
92
93 QHostAddress addr(m_config.m_listenAddress);
94 if (addr.isNull()) {
95 addr = QHostAddress::Any;
96 }
97
98 if (!m_mainServer->listen(addr, static_cast<quint16>(m_config.m_mainPort))) {
99 m_last_msg = QString("Cannot listen on main port %1: %2")
100 .arg(m_config.m_mainPort)
101 .arg(m_mainServer->errorString());
102 LOG_DEV_ERR << m_last_msg;
103 return false;
104 }
105 if (!m_hbServer->listen(addr, static_cast<quint16>(m_config.m_heartbeatPort))) {
106 m_last_msg = QString("Cannot listen on heartbeat port %1: %2")
107 .arg(m_config.m_heartbeatPort)
108 .arg(m_hbServer->errorString());
109 LOG_DEV_ERR << m_last_msg;
110 return false;
111 }
112
113 setConnectionStatus(ConnectStatus::Connected);
114 LOG_DEV_INFO << "VisionTcpipDevice listening, main port:" << m_config.m_mainPort
115 << " heartbeat port:" << m_config.m_heartbeatPort;
116 return true;
117}
