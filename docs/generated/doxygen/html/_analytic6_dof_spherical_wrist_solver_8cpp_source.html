64bool closestPointToAxes(const std::vector<Eigen::Vector3d>& points,
65 const std::vector<Eigen::Vector3d>& dirs,
66 Eigen::Vector3d& out)
67{
68 Eigen::Matrix3d M = Eigen::Matrix3d::Zero();
69 Eigen::Vector3d b = Eigen::Vector3d::Zero();
70 for (std::size_t i = 0; i < points.size(); ++i) {
71 const Eigen::Vector3d u = dirs[i].normalized();
72 const Eigen::Matrix3d A = Eigen::Matrix3d::Identity() - u * u.transpose();
73 M += A;
74 b += A * points[i];
75 }
76 const Eigen::FullPivLU<Eigen::Matrix3d> lu(M);
77 if (!lu.isInvertible()) {
78 return false;
79 }
80 out = lu.solve(b);
81 return true;
82}

139ArmWristGeometry extractGeometry(const SerialRobotConfig& config)
140{
141 ArmWristGeometry g;
142 const std::vector<const Joint*> mv = movableJoints(config);
143 if (mv.size() != 6) {
144 return g;
145 }
146 g.link3Id = mv[2]->childLinkId;
147
148 const FkChain home = ForwardKinematics::computeChain(config, homeJoint(config));
149 if (home.joints.size() != 6) {
150 return g;
151 }
152 std::array<Eigen::Vector3d, 6> axis;
153 std::array<Eigen::Vector3d, 6> point;
154 for (int i = 0; i < 6; ++i) {
155 axis[i] = home.joints[i].axisInBase.normalized();
156 point[i] = home.joints[i].originInBase;
157 }
158
159 // Spherical wrist: axes 4,5,6 intersect.
160 Eigen::Vector3d wristCenter;
161 if (!closestPointToAxes({point[3], point[4], point[5]}, {axis[3], axis[4], axis[5]}, wristCenter)) {
162 return g;
163 }
164 for (int i = 3; i < 6; ++i) {
165 if (distanceToAxis(wristCenter, point[i], axis[i]) > kGeomTol) {
166 return g;
167 }
168 }
169
170 // Articulated arm: axes 1,2 intersect (no shoulder offset); axis 1 _|_ axis 2; axis 2 || axis 3.
171 if (!closestPointToAxes({point[0], point[1]}, {axis[0], axis[1]}, g.shoulder)) {
172 return g;
173 }
174 if (distanceToAxis(g.shoulder, point[1], axis[1]) > kGeomTol
175 || std::abs(axis[0].dot(axis[1])) > kGeomTol
176 || axis[1].cross(axis[2]).norm() > kGeomTol) {
177 return g;
178 }
179 g.u1 = axis[0];
180
181 const Pose link3Home = linkPose(config, Eigen::Matrix<double, 6, 1>::Zero(), g.link3Id);
182 const Eigen::Vector3d elbowHome = link3Home.translation_m();
183 const Eigen::Matrix3d R03Home = link3Home.isometry().rotation();
184
185 Eigen::Vector3d radial = (elbowHome - g.shoulder);
186 radial -= radial.dot(g.u1) * g.u1;
187 if (radial.norm() < kGeomTol) {
188 return g; // degenerate (elbow on the J1 axis)
189 }
190 g.radial0 = radial.normalized();
191
192 g.upperArm = (elbowHome - g.shoulder).norm();
193 g.foreArm = (wristCenter - elbowHome).norm();
194
195 const auto plane = [&](const Eigen::Vector3d& p) {
196 return Eigen::Vector2d((p - g.shoulder).dot(g.radial0), (p - g.shoulder).dot(g.u1));
197 };
198 const Eigen::Vector2d e0 = plane(elbowHome);
199 const Eigen::Vector2d w0 = plane(wristCenter);
200 g.beta1Home = std::atan2(e0.y(), e0.x());
201 g.elbowHome = std::atan2(w0.y() - e0.y(), w0.x() - e0.x()) - g.beta1Home;
202
203 // Sign calibration for q2 -> upper-arm angle and q3 -> elbow angle.
204 const Eigen::Vector3d wristInLink3 = R03Home.transpose() * (wristCenter - elbowHome);
205 Eigen::Matrix<double, 6, 1> qd2 = Eigen::Matrix<double, 6, 1>::Zero();
206 qd2[1] = 0.1;
207 const Eigen::Vector2d e2 = plane(linkPose(config, qd2, g.link3Id).translation_m());
208 g.s2 = ((std::atan2(e2.y(), e2.x()) - g.beta1Home) >= 0.0) ? 1.0 : -1.0;
209
210 Eigen::Matrix<double, 6, 1> qd3 = Eigen::Matrix<double, 6, 1>::Zero();
211 qd3[2] = 0.1;
212 const Pose link3d3 = linkPose(config, qd3, g.link3Id);
213 const Eigen::Vector3d elbow3 = link3d3.translation_m();
214 const Eigen::Vector3d wrist3 = link3d3.isometry().rotation() * wristInLink3 + elbow3;
215 const Eigen::Vector2d e3 = plane(elbow3);
216 const Eigen::Vector2d w3 = plane(wrist3);
217 const double elbow3Angle = std::atan2(w3.y() - e3.y(), w3.x() - e3.x()) - std::atan2(e3.y(), e3.x());
218 g.s3 = ((elbow3Angle - g.elbowHome) >= 0.0) ? 1.0 : -1.0;
219
220 // Wrist reduction: N = A4^T R36 = Rz(q4+mu) Ry(q5) Rz(q6+delta-mu).
221 g.A4 = mv[3]->origin.isometry().rotation();
222 const Eigen::Matrix3d A5 = mv[4]->origin.isometry().rotation();
223 const Eigen::Matrix3d A6 = mv[5]->origin.isometry().rotation();
224 const Eigen::Vector3d m = A5 * Eigen::Vector3d::UnitZ();
225 if (std::abs(m.z()) > kGeomTol) {
226 return g; // J5 axis not perpendicular to J4 -> unsupported reduction
227 }
228 g.mu = std::atan2(-m.x(), m.y());
229 const Eigen::Matrix3d D = A5 * A6;
230 if (std::abs(D(0, 2)) > kGeomTol || std::abs(D(1, 2)) > kGeomTol || std::abs(D(2, 2) - 1.0) > kGeomTol) {
231 return g; // A5*A6 is not a rotation about z -> unsupported reduction
232 }
233 g.delta = std::atan2(D(1, 0), D(0, 0));
234
235 // Wrist center in the flange frame (constant): T_home^-1 * wristCenter.
236 g.wristInFlange = home.flangeInBase.isometry().inverse() * wristCenter;
237
238 g.valid = true;
239 return g;
240}

249void wristAngles(const Eigen::Matrix3d& N, double mu, double delta,
250 std::array<Eigen::Vector3d, 2>& out)
251{
252 for (int b = 0; b < 2; ++b) {
253 const double sign = (b == 0) ? 1.0 : -1.0;
254 const double q5 = std::atan2(sign * std::hypot(N(0, 2), N(1, 2)), N(2, 2));
255 double A;
256 double B;
257 if (std::abs(std::sin(q5)) < 1e-8) {
258 A = 0.0;
259 B = std::atan2(N(1, 0), N(0, 0));
260 } else {
261 A = std::atan2(sign * N(1, 2), sign * N(0, 2));
262 B = std::atan2(sign * N(2, 1), -sign * N(2, 0));
263 }
264 out[b] = Eigen::Vector3d(A - mu, q5, B - delta + mu);
265 }
266}

274double wrapToLimits(double q, const std::optional<JointLimits>& limits, bool& ok)
275{
276 ok = true;
277 if (!limits.has_value()) {
278 return q;
279 }
280 // Prefer the natural value (k = 0); only shift by +-2*pi when it is out of range.
281 for (const int k : {0, -1, 1}) {
282 const double candidate = q + 2.0 * kPi * k;
283 if (candidate >= limits->lower - 1e-9 && candidate <= limits->upper + 1e-9) {
284 return candidate;
285 }
286 }
287 ok = false;
288 return q;
289}

324IKResult Analytic6DofSphericalWristSolver::solve(const SerialRobotConfig& config,
325 const IKSolveContext& context) const
326{
327 IKResult all = solveAll(config, context);
328 if (!all.ok()) {
329 return all;
330 }
331
332 // Single best: closest to seed (then previous) joints if provided, else first branch.
333 const IKRequest& request = context.request;
334 const std::optional<JointVector> reference = request.seedJoint.has_value() ? request.seedJoint
335 : request.previousJoint;
336 if (reference.has_value() && reference->size() == 6) {
337 std::size_t best = 0;
338 double bestCost = std::numeric_limits<double>::max();
339 for (std::size_t i = 0; i < all.solutions.size(); ++i) {
340 const double cost = (all.solutions[i].joints.values() - reference->values()).norm();
341 if (cost < bestCost) {
342 bestCost = cost;
343 best = i;
344 }
345 }
346 const IKSolution chosen = all.solutions[best];
347 all.solutions = {chosen};
348 } else {
349 all.solutions.resize(1);
350 }
351 return all;
352}

364IKResult Analytic6DofSphericalWristSolver::solveAll(const SerialRobotConfig& config,
365 const IKSolveContext& context) const
366{
367 const ArmWristGeometry g = extractGeometry(config);
368 if (!g.valid) {
369 return IKResult{IKStatus::UnsupportedSolver, {},
370 "analytic spherical-wrist solver does not support this model"};
371 }
372
373 const Eigen::Isometry3d target = context.targetFlangeInBase.isometry();
374 const Eigen::Matrix3d R = target.rotation();
375 const Eigen::Vector3d p = target.translation();
376 const Eigen::Vector3d wristCenter = R * g.wristInFlange + p;
377
378 const Eigen::Vector3d v = wristCenter - g.shoulder;
379 const double v1 = v.dot(g.u1);
380 const Eigen::Vector3d vh = v - v1 * g.u1;
381 const double rho = vh.norm();
382
383 const std::vector<const Joint*> mv = movableJoints(config);
384
385 IKResult result;
386 result.status = IKStatus::Ok;
387
388 if (rho < 1e-9) {
389 // Wrist center on the J1 axis: shoulder singularity, q1 undetermined.
390 return IKResult{IKStatus::Singularity, {}, "wrist center lies on the J1 axis (shoulder singularity)"};
391 }
392 const Eigen::Vector3d vhat = vh / rho;
393 const double q1a = std::atan2(g.radial0.cross(vhat).dot(g.u1), g.radial0.dot(vhat));
394
395 const std::array<std::pair<double, double>, 2> shoulderBranches = {{{q1a, 1.0}, {q1a + kPi, -1.0}}};
396 for (const auto& sb : shoulderBranches) {
397 const double q1 = sb.first;
398 const double X = sb.second * rho;
399 const double Y = v1;
400 const double D2 = X * X + Y * Y;
401 double cosE = (D2 - g.upperArm * g.upperArm - g.foreArm * g.foreArm)
402 / (2.0 * g.upperArm * g.foreArm);
403 if (cosE > 1.0 + 1e-9 || cosE < -1.0 - 1e-9) {
404 continue; // target unreachable for this branch
405 }
406 cosE = std::max(-1.0, std::min(1.0, cosE));
407 const double elbowMag = std::acos(cosE);
408
409 for (const double Eb : {elbowMag, -elbowMag}) {
410 const double beta1 = std::atan2(Y, X)
411 - std::atan2(g.foreArm * std::sin(Eb), g.upperArm + g.foreArm * std::cos(Eb));
412 const double q2 = g.s2 * (beta1 - g.beta1Home);
413 const double q3 = g.s3 * (Eb - g.elbowHome);
414
415 Eigen::Matrix<double, 6, 1> arm = Eigen::Matrix<double, 6, 1>::Zero();
416 arm[0] = q1;
417 arm[1] = q2;
418 arm[2] = q3;
419 const Eigen::Matrix3d R03 = linkPose(config, arm, g.link3Id).isometry().rotation();
420 const Eigen::Matrix3d N = g.A4.transpose() * (R03.transpose() * R);
421
422 std::array<Eigen::Vector3d, 2> wrist;
423 wristAngles(N, g.mu, g.delta, wrist);
424 for (const Eigen::Vector3d& w : wrist) {
425 std::array<double, 6> q = {q1, q2, q3, w.x(), w.y(), w.z()};
426
427 bool withinLimits = true;
428 for (int i = 0; i < 6 && withinLimits; ++i) {
429 bool ok = true;
430 q[i] = wrapToLimits(q[i], mv[i]->limits, ok);
431 withinLimits = ok;
432 }
433 if (!withinLimits) {
434 continue;
435 }
436
437 Eigen::VectorXd qv(6);
438 for (int i = 0; i < 6; ++i) {
439 qv[i] = q[i];
440 }
441 const JointVector joints(qv);
442 const Pose fk = ForwardKinematics::flangePose(config, joints);
443 const double posErr = (fk.translation_m() - p).norm();
444 const double oriErr =
445 fk.rotationQuaternion().angularDistance(context.targetFlangeInBase.rotationQuaternion());
446 if (posErr > kAcceptPos || oriErr > kAcceptOri) {
447 continue;
448 }
449
450 bool duplicate = false;
451 for (const IKSolution& existing : result.solutions) {
452 if ((existing.joints.values() - qv).norm() <= kDupTol) {
453 duplicate = true;
454 break;
455 }
456 }
457 if (duplicate) {
458 continue;
459 }
460
461 IKSolution solution;
462 solution.joints = joints;
463 solution.posture.shoulder = branchSign(q[0]);
464 solution.posture.elbow = branchSign(q[2]);
465 solution.posture.wrist = branchSign(q[4]);
466 solution.positionError_m = posErr;
467 solution.orientationError_rad = oriErr;
468 result.solutions.push_back(solution);
469 }
470 }
471 }
472
473 if (result.solutions.empty()) {
474 return IKResult{IKStatus::TargetUnreachable, {},
475 "analytic spherical-wrist solver found no in-limit solution"};
476 }
477 return result;
478}
