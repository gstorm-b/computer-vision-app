110 // Fault auto-recovery. The timer is owned by this controller, so a runtime that ends
111 // during the wait takes the pending recovery down with it — the same lifetime
112 // guarantee the reconnect retries rely on.
113 m_faultRecoverTimer.setSingleShot(true);
114 m_faultRecoverTimer.setInterval(kFaultAutoRecoverMs);
115 connect(&m_faultRecoverTimer, &QTimer::timeout, this, [this]() {
116 // Read the code before recovering: recoverFromFault() clears the pending result.
117 const LocalizationFaultCode code = m_pendingCycleResult.faultCode;
118 m_consecutiveAutoRecoveries += 1;
119
120 // A fault that keeps auto-clearing is a real problem, but reporting every
121 // occurrence would bury the very log this path exists to keep readable.
122 if (m_consecutiveAutoRecoveries % kAutoRecoverWarnStride == 0) {
123 LOG_USER_WARN << "Repeating auto-recovered localization fault."
124 << "count=" << m_consecutiveAutoRecoveries
125 << "lastFault=" << localizationFaultCodeName(code);
126 } else {
127 LOG_DEV_INFO << "Localization fault auto-cleared."
128 << "count=" << m_consecutiveAutoRecoveries
129 << "fault=" << localizationFaultCodeName(code);
130 }
131
132 recoverFromFault(
133 QStringLiteral("Fault auto-cleared after %1 ms (no bErrorReset received).")
134 .arg(kFaultAutoRecoverMs));
135 });
136}

153void LocalizationRuntimeController::setActiveCameraNumber(int cameraNumber)
154{
155 if (m_cycleState == CycleState::Running) {
156 appendTaskLog(QStringLiteral("WARN"),
157 QStringLiteral("Camera change ignored while cycle is running."));
158 return;
159 }
160
161 if (!m_context.cameraRunners.value(cameraNumber)) {
162 appendTaskLog(QStringLiteral("WARN"),
163 QStringLiteral("Camera change ignored: camera runner is not available."));
164 publishBoolSignal(QStringLiteral("bCameraValid"), false);
165 publishBoolSignal(QStringLiteral("bTaskFault"), true);
166 publishNumberSignal(QStringLiteral("nFaultCode"),
167 localizationFaultCodeValue(LocalizationFaultCode::CameraLost));
168 return;
169 }
170
171 auto *previousRunner = activeCameraRunner();
172 m_activeCameraNumber = cameraNumber;
173 publishBoolSignal(QStringLiteral("bTaskReady"), false);
174 bindActiveCameraRole(m_activeCameraNumber);
175 publishNumberSignal(QStringLiteral("nActiveCamera"), cameraNumber);
176
177 auto *newRunner = activeCameraRunner();
178 if (previousRunner && previousRunner != newRunner) {
179 previousRunner->requestDisconnect();
180 }
181
182 const bool calibrated = validateActiveCameraCalibration();
183 // Rebuild the pickability checker against the new camera's calibration.
184 rebuildPickingChecker();
185 publishBoolSignal(QStringLiteral("bCameraValid"), calibrated);
186 if (!calibrated) {
187 publishBoolSignal(QStringLiteral("bTaskFault"), true);
188 publishNumberSignal(QStringLiteral("nFaultCode"),
189 localizationFaultCodeValue(LocalizationFaultCode::CalibrationInvalid));
190 m_cycleState = CycleState::Faulted;
191 emit runtimeFault(QStringLiteral("Active camera calibration is invalid."));
192 return;
193 }
194
195 if (newRunner) {
196 newRunner->requestConnect();
197 // Refresh the active workspace from the new camera, guarded like the
198 // requestConnect above so a degraded switch cannot deref a null runner.
199 m_context.activeCameraWorkspace =
200 m_config.cameraWorkspace(newRunner->device()->id());
201 m_activeCameraWorkspace = m_context.activeCameraWorkspace;
202 }
203 if (allRequiredRolesHealthy()) {
204 markRuntimeReady(QStringLiteral("Active camera changed. Runtime ready."));
205 }

253LocalizationRuntimeController::setup(const RuntimeContext &context)
254{
255 SetupResult result;
256 m_valid = false;
257 m_context = context;
258 m_config = context.config;
259 m_signalMapper.configure(m_config);
260 m_cycleState = CycleState::NotReady;
261 m_lastExecuteTrigger = false;
262 m_activeCycleId = 0;
263 m_pendingCycleResult = CycleResult();
264 resetRuntimeBindings();
265
266 result.primaryPlcDeviceId = context.primaryPlcDeviceId;
267 result.visionOutputDeviceId = context.visionOutputDeviceId;
268 const QMap<int, QString> cameraMap = context.cameraDeviceIds;
269
270 if (result.primaryPlcDeviceId.isEmpty() || !context.primaryPlcRunner) {
271 result.errors.append(QStringLiteral("Missing primary_plc role runner."));
272 }
273 if (result.visionOutputDeviceId.isEmpty() || !context.visionOutputRunner) {
274 result.errors.append(QStringLiteral("Missing vision_output role runner."));
275 }
276
277 if (cameraMap.isEmpty() || context.cameraRunners.isEmpty()) {
278 result.errors.append(QStringLiteral("Missing camera_number role binding."));
279 } else {
280 for (auto it = cameraMap.cbegin(); it != cameraMap.cend(); ++it) {
281 if (!context.cameraRunners.value(it.key())) {
282 result.errors.append(QStringLiteral("camera_number_%1 runner is not registered: %2")
283 .arg(it.key())
284 .arg(it.value()));
285 }
286 }
287 }
288
289 disconnect(m_plcValueConnection);
290 if (context.primaryPlcRunner) {
291 m_plcValueConnection = connect(
292 context.primaryPlcRunner.data(),
293 &vc::runtime::PlcRunner::valueChanged,
294 this,
295 &LocalizationRuntimeController::handlePlcValues,
296 Qt::UniqueConnection);
297 }
298
299 bindFixedRoleRunners();
300 m_activeCameraNumber = context.activeCameraNumber;
301 if (m_activeCameraNumber < 0 && !cameraMap.isEmpty()) {
302 m_activeCameraNumber = cameraMap.firstKey();
303 }
304 bindActiveCameraRole(m_activeCameraNumber);
305 m_activePatternGroupNumber = context.activePatternGroupNumber;
306 if (m_activePatternGroupNumber < 0 && !context.patternGroups.isEmpty()) {
307 m_activePatternGroupNumber = context.patternGroups.firstKey();
308 }
309
310 if (!validateActivePatternGroup(&result.errors)) {
311 publishBoolSignal(QStringLiteral("bPatternValid"), false);
312 publishBoolSignal(QStringLiteral("bTaskFault"), true);
313 publishNumberSignal(QStringLiteral("nFaultCode"),
314 localizationFaultCodeValue(LocalizationFaultCode::PatternInvalid));
315 }
316
317 if (!validateActiveCameraCalibration(&result.errors)) {
318 publishBoolSignal(QStringLiteral("bCameraValid"), false);
319 publishBoolSignal(QStringLiteral("bTaskFault"), true);
320 publishNumberSignal(QStringLiteral("nFaultCode"),
321 localizationFaultCodeValue(LocalizationFaultCode::CalibrationInvalid));
322 }
323
324 // Build the robot-pickability checker once for the runtime (active camera +
325 // robot config now established); rebuilt later only on active-camera change.
326 rebuildPickingChecker();
327
328 result.valid = result.errors.isEmpty();
329 m_valid = result.valid;
330
331 for (const QString &error : result.errors)
332 LOG_DEV_ERR << "LocalizationRuntimeController setup:" << error;
333
334 if (m_valid) {
335 requestRoleConnectNow(RunnerRole::PrimaryPlc);
336 requestRoleConnectNow(RunnerRole::VisionOutput);
337 requestRoleConnectNow(RunnerRole::Camera);
338 if (allRequiredRolesHealthy()) {
339 markRuntimeReady(QStringLiteral("Runtime ready."));
340 } else {
341 appendTaskLog(QStringLiteral("INFO"),
342 QStringLiteral("Runtime setup valid. Waiting for device connections."));
343 }
344 } else {
345 m_cycleState = CycleState::Faulted;
346 }
347
348 return result;

376void LocalizationRuntimeController::handlePlcValues(const QMap<QString, QVariant> &values)
377{
378 if (values.isEmpty()) {
379 LOG_USER_ERR << "Communication device passed an empty values map.";
380 return;
381 }
382
383 const QList<LocalizationSignalEvent> events = m_signalMapper.mapValues(values);
384 for (const LocalizationSignalEvent &event : events) {
385 emit signalChanged(event.name, event.value);
386
387 if (event.name == QStringLiteral("nActiveCamera")) {
388 emit cameraNumberChanged(event.value);
389 const int cameraNumber = event.value.toInt();
390 if (cameraNumber > 0) {
391 setActiveCameraNumber(cameraNumber);
392 }
393 } else if (event.name == QStringLiteral("nActivePatternGroup")) {
394 emit patternNumberChanged(event.value);
395 bool ok = false;
396 const int patternGroupNumber = event.value.toInt(&ok);
397 if (ok && patternGroupNumber > 0) {
398 setActivePatternGroupNumber(patternGroupNumber);
399 }
400 } else if (event.name == QStringLiteral("bExecuteTrigger")) {
401 const bool trigger = event.value.toBool();
402 const bool risingEdge = trigger && !m_lastExecuteTrigger;
403 const bool fallingEdge = !trigger && m_lastExecuteTrigger;
404 m_lastExecuteTrigger = trigger;
405
406 if (risingEdge) {
407 if (m_cycleState == CycleState::ReadyForTrigger) {
408 startCycle();
409 } else {
410 appendTaskLog(QStringLiteral("WARN"),
411 QStringLiteral("Trigger ignored: task is not ready."));
412 }
413 } else if (fallingEdge &&
414 m_cycleState == CycleState::WaitingTriggerReset) {
415 publishBoolSignal(QStringLiteral("bMatchingFinished"), false);
416 if (m_pendingCycleResult.faulted) {
417 // The fault now comes to rest: the trigger is low, so nothing else
418 // will move the runtime out of this state on its own. This is the
419 // point the runtime used to stay stuck at until the task was
420 // restarted, and it is where the way out has to be armed.
421 m_cycleState = CycleState::Recovering;
422 armFaultAutoRecovery();
423 } else {
424 markRuntimeReady(QStringLiteral("Trigger reset. Runtime ready."));
425 }
426 }
427 } else if (event.name == QStringLiteral("bErrorReset")) {
428 const bool reset = event.value.toBool();
429 const bool risingEdge = reset && !m_lastErrorReset;
430 m_lastErrorReset = reset;
431
432 // Rising edge only. A level-triggered acknowledge would re-clear the fault
433 // on every PLC scan for as long as the operator holds the bit, which would
434 // make a genuinely repeating fault invisible.
435 if (risingEdge) {
436 acknowledgeFault();
437 }
438 }
439 }

511void LocalizationRuntimeController::bindRoleContext(
512 RunnerRole role,
513 vc::runtime::IDeviceRunner *runner,
514 const QString &deviceId,
515 const LocalizationRecoveryPolicy &policy)
516{
517 clearRoleContext(role);
518
519 if (!runner) {
520 return;
521 }
522
523 RoleRecoveryContext context;
524 context.roleName = policy.roleName;
525 context.deviceId = deviceId;
526 context.runner = runner;
527 context.policy = policy;
528
529 m_recoveryContexts.insert(roleKey(role), context);
530
531 switch (role) {
532 case RunnerRole::PrimaryPlc:
533 connect(runner, &vc::runtime::IDeviceRunner::connectStatusChanged,
534 this, &LocalizationRuntimeController::onPrimaryPlcStatusChanged,
535 Qt::UniqueConnection);
536 connect(runner, &vc::runtime::IDeviceRunner::errorOccurred,
537 this, &LocalizationRuntimeController::onPrimaryPlcError,
538 Qt::UniqueConnection);
539 break;
540 case RunnerRole::VisionOutput:
541 connect(runner, &vc::runtime::IDeviceRunner::connectStatusChanged,
542 this, &LocalizationRuntimeController::onVisionOutputStatusChanged,
543 Qt::UniqueConnection);
544 connect(runner, &vc::runtime::IDeviceRunner::errorOccurred,
545 this, &LocalizationRuntimeController::onVisionOutputError,
546 Qt::UniqueConnection);
547 break;
548 case RunnerRole::Camera:
549 connect(runner, &vc::runtime::IDeviceRunner::connectStatusChanged,
550 this, &LocalizationRuntimeController::onCameraStatusChanged,
551 Qt::UniqueConnection);
552 connect(runner, &vc::runtime::IDeviceRunner::errorOccurred,
553 this, &LocalizationRuntimeController::onCameraError,
554 Qt::UniqueConnection);
555 break;
556 }

817QVector<vc::device::VisionOutputPosition>
818LocalizationRuntimeController::buildVisionOutputPositions(
819 const mtc::MatchResult &matchResult,
820 QVector<ResultRow> *rows,
821 LocalizationFaultCode *faultCode) const
822{
823 QVector<vc::device::VisionOutputPosition> positions;
824 if (faultCode) {
825 *faultCode = LocalizationFaultCode::None;
826 }
827
828 const calib::Calibrator calibrator = m_context.cameraCalibrators.value(m_activeCameraNumber);
829 if (!calibrator.isCalibrated()) {
830 if (faultCode) *faultCode = LocalizationFaultCode::CalibrationInvalid;
831 return positions;
832 }
833
834 int index = 1;
835 for (const mtc::MatchedObject &object : matchResult.Objects) {
836 ResultRow row;
837 row.index = index++;
838 row.patternName = QString::fromStdWString(object.pattern_name);
839 row.score = object.matched_Score;
840 row.imageX = object.point_Center.x;
841 row.imageY = object.point_Center.y;
842 row.imageR = object.point_angle;
843
844 // image to real world position
845 cv::Point2f pick_center = object.point_Center;
846 if (m_context.activeCameraWorkspace.useWorkspace) {
847 pick_center += matchResult.cropOffsetPoint;
848 }
849
850 const cv::Point3f pickPoint = calibrator.imageToRobot(pick_center);
851 // Top-down pick rotation about Z, negated to match the robot's Z convention.
852 // This is the axis the pre-Phase-5 4-axis contract called `r`; it is emitted as
853 // `rz` now that the pattern can add its own Z rotation on top.
854 const double worldYaw = -calibrator.rotateImageToRobot(object.point_angle);
855
856 // Compose the pattern's 6-axis picking offset in the TOOL frame:
857 // world = pick * offset
858 // This is the same composition RobotKinematicPickingChecker::isPickable uses,
859 // so the advisory pickability verdict and the pose actually emitted cannot
860 // drift apart. With a zero rotation offset the product reduces exactly to the
861 // previous Rz-only Calibrator::translateWithZAxis() result (Rz rotates x/y and
862 // passes z through), so existing patterns are unaffected.
863 const RobotKinematics::Pose pickPose = RobotKinematics::Pose::fromXYZRPY_mm_deg(
864 pickPoint.x, pickPoint.y, pickPoint.z, 0.0, 0.0, worldYaw);
865 const RobotKinematics::Pose offsetPose = RobotKinematics::Pose::fromXYZRPY_mm_deg(
866 object.point_offset.x, object.point_offset.y, object.point_offset.z,
867 object.point_rotation_offset.x, object.point_rotation_offset.y,
868 object.point_rotation_offset.z);
869
870 const Eigen::Vector3d worldTranslation = (pickPose * offsetPose).translation_m();
871
872 row.world.x = RobotKinematics::units::toMm(worldTranslation.x());
873 row.world.y = RobotKinematics::units::toMm(worldTranslation.y());
874 row.world.z = RobotKinematics::units::toMm(worldTranslation.z());
875
876 // Orientation sent to the robot: the pattern's rotation offset, with the pick
877 // yaw folded into the Z axis because both act about Z. A pattern with no
878 // rotation offset therefore emits exactly the pre-Phase-5 pose (rx = ry = 0,
879 // rz = the old `r`).
880 row.world.rx = object.point_rotation_offset.x;
881 row.world.ry = object.point_rotation_offset.y;
882 row.world.rz = worldYaw + object.point_rotation_offset.z;
883
884 // if (object.hasCollision() && object.isOutsideConditionRoi()) {
885 // row.status = QStringLiteral("Skipped: collision, outside");
886 // } else if (object.hasCollision()) {
887 // row.status = QStringLiteral("Skipped: collision");
888 // } else if (object.isOutsideConditionRoi()) {
889 // row.status = QStringLiteral("Skipped: outside");
890 // } else {
891 // row.status = QStringLiteral("Sent");
892 // positions.append(row.world);
893 // }
894
895 if ((!object.hasCollision()) && (!object.isOutsideConditionRoi()) && object.isPossibleToPick()) {
896 if (positions.size() < 2) {
897 row.status = QStringLiteral("Sent");
898 positions.append(row.world);
899 } else {
900 row.status = QStringLiteral("Skipped");
901 }
902 } else {
903 QString tempstr = object.hasCollision() ? tr("Collision") : "";
904 tempstr += (object.isOutsideConditionRoi()
905 ? ((tempstr.isEmpty() ? "" : ", ") + tr("Outside"))
906 : "");
907 tempstr += (!object.isPossibleToPick()
908 ? ((tempstr.isEmpty() ? "" : ", ") + tr("Unpickable"))
909 : "");
910 row.status = QStringLiteral("Skipped: %1").arg(tempstr);
911 }
912
913 if (rows) {
914 rows->append(row);
915 }
916 }
917
918 return positions;

1126void LocalizationRuntimeController::handleRoleStatusChanged(RunnerRole role,
1127 vc::device::ConnectStatus status)
1128{
1129 const int key = roleKey(role);
1130 if (!m_recoveryContexts.contains(key)) {
1131 return;
1132 }
1133
1134 RoleRecoveryContext context = m_recoveryContexts.value(key);
1135 if (!context.runner) {
1136 m_recoveryContexts.remove(key);
1137 return;
1138 }
1139
1140 if (isHealthyStatus(status)) {
1141 const int retryHistory = context.retryCount;
1142 const QDateTime outageStartedAt = context.outageStartedAt;
1143 const bool wasRecovering = retryHistory > 0 || context.retryScheduled;
1144 context.retryCount = 0;
1145 context.retryScheduled = false;
1146 context.reportedStatus = vc::device::ConnectStatus::Connected;
1147 context.outageStartedAt = QDateTime();
1148 m_recoveryContexts.insert(key, context);
1149
1150 // Log the reconnect itself, separately from markRuntimeReady() below. The
1151 // runtime may not be able to re-arm yet (another role still down, or a cycle in
1152 // flight), and "the camera came back" is worth showing the operator either way.
1153 if (wasRecovering) {
1154 appendTaskLog(QStringLiteral("INFO"),
1155 buildRecoveryReadyMessage(context.policy,
1156 context.deviceId,
1157 retryHistory,
1158 outageStartedAt));
1159 }
1160
1161 if (m_cycleState == CycleState::Running ||
1162 m_cycleState == CycleState::WaitingTriggerReset) {
1163 return;
1164 }
1165 if (allRequiredRolesHealthy()) {
1166 // The reconnect detail was logged above; this line reports the separate
1167 // fact that the runtime actually re-armed, which only happens once every
1168 // role is healthy and the pattern/calibration still validate.
1169 markRuntimeReady(QStringLiteral("Runtime ready."));
1170 }
1171 return;
1172 }
1173
1174 if (m_cycleState == CycleState::Running) {
1175 switch (role) {
1176 case RunnerRole::Camera:
1177 abortCycle(LocalizationFaultCode::CameraLost,
1178 QStringLiteral("Active camera lost during localization cycle."));
1179 break;
1180 case RunnerRole::VisionOutput:
1181 abortCycle(LocalizationFaultCode::VisionOutputLost,
1182 QStringLiteral("Vision output lost during localization cycle."));
1183 break;
1184 case RunnerRole::PrimaryPlc:
1185 abortCycle(LocalizationFaultCode::PlcLost,
1186 QStringLiteral("Primary PLC lost during localization cycle."));
1187 break;
1188 }
1189 }
1190
1191 const LocalizationRecoveryAction action = decideRecoveryAction(
1192 context.policy,
1193 status,
1194 context.retryScheduled);
1195
1196 if (action != LocalizationRecoveryAction::RetryScheduled) {
1197 return;
1198 }
1199
1200 // A new outage: stamp its start so the recovery message can report how long it
1201 // lasted, which is what an operator asks after the fact now that attempts are
1202 // no longer bounded by a known maximum.
1203 if (context.retryCount == 0) {
1204 context.outageStartedAt = QDateTime::currentDateTime();
1205
1206 // Once per outage, on the operator's dashboard, not only in the app log. An
1207 // outage that shows up nowhere in the task log is indistinguishable from a
1208 // runtime that simply stopped being triggered.
1209 appendTaskLog(QStringLiteral("WARN"),
1210 QStringLiteral("Connection lost: role=%1 deviceId=%2 status=%3. "
1211 "Reconnecting every %4 ms until it returns.")
1212 .arg(context.policy.roleName,
1213 context.deviceId,
1214 connectStatusName(status),
1215 QString::number(context.policy.retryIntervalMs)));
1216
1217 // Withdraw readiness for the duration of the outage. Losing a role outside a
1218 // running cycle previously left CycleState::ReadyForTrigger and bTaskReady=true
1219 // standing: the escalation path published bTaskFault a minute later and covered
1220 // for it. With escalation removed there is no such backstop, so an unreachable
1221 // camera would advertise "ready, send me a trigger" indefinitely. The PLC
1222 // contract is that it triggers only while bTaskReady is true, so this is the
1223 // signal it needs. markRuntimeReady() restores both on reconnect.
1224 if (m_cycleState == CycleState::ReadyForTrigger) {
1225 m_cycleState = CycleState::Recovering;
1226 publishBoolSignal(QStringLiteral("bTaskReady"), false);
1227 }
1228 }
1229
1230 // Report only when the outage begins or when its status actually changes. Without
1231 // this, an outage lasting minutes emits one identical line every retryIntervalMs
1232 // and drowns out everything else the operator needs to see.
1233 const bool statusIsNew = context.reportedStatus != status;
1234
1235 context.retryScheduled = true;
1236 context.retryCount += 1;
1237 context.reportedStatus = status;
1238 m_recoveryContexts.insert(key, context);
1239
1240 if (statusIsNew) {
1241 emit runtimeRecovering(buildRecoveryProgressMessage(
1242 context.policy,
1243 context.deviceId,
1244 status));
1245 }
1246
1247 scheduleRoleReconnect(role);
1248}

1269{

1271 if (!m_recoveryContexts.contains(key)) {
1272 return;

1300 return;

1302 RoleRecoveryContext current = m_recoveryContexts.value(currentKey);
1303 current.retryScheduled = false;
1304 m_recoveryContexts.insert(currentKey, current);
1305 requestRoleConnectNow(role);
1306 });
