13inline cv::Mat QPixmapToCvMat(const QPixmap& pixmap) {
14 QImage qimg = pixmap.toImage();
15 if (qimg.isNull()) {
16 return cv::Mat();
17 }
18
19 switch (qimg.format()) {
20 case QImage::Format_Grayscale8: {
21 // 1 channel
22 cv::Mat mat(qimg.height(), qimg.width(), CV_8UC1,
23 const_cast<uchar*>(qimg.bits()),
24 static_cast<size_t>(qimg.bytesPerLine()));
25 return mat.clone();
26 }
27
28 case QImage::Format_RGB888: {
29 // Qt = RGB, OpenCV = BGR
30 cv::Mat mat(qimg.height(), qimg.width(), CV_8UC3,
31 const_cast<uchar*>(qimg.bits()),
32 static_cast<size_t>(qimg.bytesPerLine()));
33 cv::Mat matBGR;
34 cv::cvtColor(mat, matBGR, cv::COLOR_RGB2BGR);
35 return matBGR;
36 }
37
38 case QImage::Format_RGBA8888: {
39 // Qt = RGBA, OpenCV = BGRA
40 cv::Mat mat(qimg.height(), qimg.width(), CV_8UC4,
41 const_cast<uchar*>(qimg.bits()),
42 static_cast<size_t>(qimg.bytesPerLine()));
43 cv::Mat matBGRA;
44 cv::cvtColor(mat, matBGRA, cv::COLOR_RGBA2BGRA);
45 return matBGRA;
46 }
47
48 default:
49 QImage converted = qimg.convertToFormat(QImage::Format_RGBA8888);
50 cv::Mat mat(converted.height(), converted.width(), CV_8UC4,
51 const_cast<uchar*>(converted.bits()),
52 static_cast<size_t>(converted.bytesPerLine()));
53 cv::Mat matBGRA;
54 cv::cvtColor(mat, matBGRA, cv::COLOR_RGBA2BGRA);
55 return matBGRA;
56 }
