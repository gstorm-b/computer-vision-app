151 DeviceCommandResult submitCommand(const DeviceCommand &command) override
152 {
153 if (command.targetDeviceId != m_device->id()) {
154 DeviceCommandResult result = DeviceCommandResult::rejected(
155 command,
156 DeviceCommandResultCode::InvalidTarget,
157 QStringLiteral("Camera command target does not match this runner."));
158 finishRejectedCommand(result);
159 return result;
160 }
161
162 if (!isSupportedCommand(command.kind)) {
163 DeviceCommandResult result = DeviceCommandResult::rejected(
164 command,
165 DeviceCommandResultCode::UnsupportedCommand,
166 QStringLiteral("Unsupported camera command."));
167 finishRejectedCommand(result);
168 return result;
169 }
170
171 const DeviceCommandQueuePolicy policy = queuePolicyFor(command.kind);
172 const DeviceCommandEnqueueDecision decision =
173 m_commandQueue.enqueue(command, policy, hasActiveCommand());
174 if (!decision.accepted) {
175 finishRejectedCommand(decision.rejection);
176 return decision.rejection;
177 }
178
179 if (decision.shouldRunNow) {
180 runCommand(command);
181 return DeviceCommandResult::accepted(command);
182 }
183
184 const QString queueMessage = decision.replacedPending
185 ? QStringLiteral("Camera command queued (replaced pending same-kind request).")
186 : QStringLiteral("Camera command queued.");
187 LOG_USER_INFO << "Camera command queued."
188 << "id=" << command.id
189 << "kind=" << deviceCommandKindToString(command.kind)
190 << "target=" << command.targetDeviceId
191 << "pending=" << decision.pendingCount;
192 return DeviceCommandResult::accepted(command, queueMessage);
193 }

327 void onGrabFinished(vc::device::GrabResult result) {
328 QVariantMap payload;
329 payload.insert(QStringLiteral("message"), result.msg);
330 if (!hasActiveCommand()) {
331 emit grabFinished(result);
332 return;
333 }
334
335 if (m_activeCommand.kind == DeviceCommandKind::CameraSingleShot &&
336 result.isGrabSuccess) {
337 m_grabFailedCount = 0;
338 finishActiveCommand(DeviceCommandResult::succeeded(
339 m_activeCommand,
340 result.msg,
341 payload));
342 } else if (m_activeCommand.kind == DeviceCommandKind::CameraSingleShot) {
343 m_grabFailedCount++;
344 if (m_grabFailedCount >= kMaxGrabAttempts) {
345 LOG_USER_WARN << "Camera grab failed on every attempt."
346 << "target=" << m_activeCommand.targetDeviceId
347 << "attempts=" << m_grabFailedCount
348 << "msg=" << result.msg;
349 finishActiveCommand(DeviceCommandResult::failed(
350 m_activeCommand,
351 DeviceCommandResultCode::DeviceError,
352 result.msg,
353 payload));
354 } else {
355 // Give the retry its own watchdog window. The timer was started once in
356 // runCommand() and never restarted, so the whole retry chain shared a
357 // single timeout: a grab that fails slowly used it up on the first
358 // attempt and the later attempts never ran at all.
359 LOG_DEV_INFO << "Camera grab failed, retrying."
360 << "target=" << m_activeCommand.targetDeviceId
361 << "attempt=" << m_grabFailedCount
362 << "of=" << kMaxGrabAttempts;
363 m_activeCommandTimer.start(activeTimeoutMs(m_activeCommand));
364 emit sig_singleShot();
365 return;
366 }
367 }
368 emit grabFinished(result);
369 }

461 void runCommand(const DeviceCommand &command)
462 {
463 const TriggerSignal trigger = triggerFor(command.kind);
464 if (trigger == nullptr) {
465 DeviceCommandResult result = DeviceCommandResult::rejected(
466 command,
467 DeviceCommandResultCode::UnsupportedCommand,
468 QStringLiteral("Camera command trigger is not available."));
469 finishRejectedCommand(result);
470 return;
471 }
472
473 // Reset the retry budget per command. It used to clear only on a successful grab,
474 // so failures during commissioning carried into the next runtime cycle and could
475 // consume the budget before that cycle's first attempt.
476 if (command.kind == DeviceCommandKind::CameraSingleShot) {
477 m_grabFailedCount = 0;
478 }
479
480 m_activeCommand = command;
481 m_activeCommandTimer.start(activeTimeoutMs(command));
482 LOG_USER_INFO << "Camera command started."
483 << "id=" << command.id
484 << "kind=" << deviceCommandKindToString(command.kind)
485 << "target=" << command.targetDeviceId
486 << "timeoutMs=" << activeTimeoutMs(command)
487 << "pending=" << m_commandQueue.pendingCount();
488 (this->*trigger)();
489 }
