74QRect DeviceRowDelegate::bitButtonRect(const QRect &cell, int btnIndex) const {
75 const QRect inner = cell.adjusted(kCellPadX, kCellPadY,
76 -kCellPadX, -kCellPadY);
77 // 3 buttons share the available width; favour TOGGLE getting more so
78 // the label doesn't elide. Fixed proportions match the painted labels.
79 constexpr int kRatioOn = 26;
80 constexpr int kRatioOff = 30;
81 constexpr int kRatioToggle = 44;
82 const int totalRatio = kRatioOn + kRatioOff + kRatioToggle;
83 const int spacings = 2 * kBtnSpacing;
84 const int avail = inner.width() - spacings;
85 const int wOn = (avail * kRatioOn) / totalRatio;
86 const int wOff = (avail * kRatioOff) / totalRatio;
87 const int wToggle = avail - wOn - wOff;
88
89 const int y = inner.y() + (inner.height() - kBtnHeight) / 2;
90 int x = inner.x();
91
92 if (btnIndex == SubOn) return { x, y, wOn, kBtnHeight };
93 x += wOn + kBtnSpacing;
94 if (btnIndex == SubOff) return { x, y, wOff, kBtnHeight };
95 x += wOff + kBtnSpacing;
96 if (btnIndex == SubToggle) return { x, y, wToggle, kBtnHeight };
97 return {};
98}

332void DeviceRowDelegate::paint(QPainter *p,
333 const QStyleOptionViewItem &opt,
334 const QModelIndex &idx) const {
335 paintBackground(p, opt, idx);
336
337 const int col = idx.column();
338
339 if (col == ColAddress) {
340 const int address = idx.data(AddressRole).toInt();
341 paintAddress(p, opt.rect, address);
342 return;
343 }
344
345 if (col == ColDescription) {
346 // Default text drawing — the QTableWidgetItem already holds the
347 // user's editable description string under Qt::DisplayRole.
348 p->save();
349 p->setPen(tokenColor("text.primary"));
350 p->setFont(monoFont(10));
351 p->drawText(opt.rect.adjusted(10, 0, -10, 0),
352 Qt::AlignVCenter | Qt::AlignLeft,
353 idx.data(Qt::DisplayRole).toString());
354 p->restore();
355 return;
356 }
357
358 if (col == ColState) {
359 if (m_mode == Bit) {
360 paintChip(p, opt.rect, idx.data(BitStateRole).toBool());
361 } else {
362 paintWordVal(p, opt.rect, idx.data(WordValueRole).toInt());
363 }
364 return;
365 }
366
367 if (col == ColAction) {
368 if (m_mode == Bit) {
369 paintBitAction(p, opt.rect, idx.row());
370 } else {
371 paintWordAction(p, opt.rect, idx.row(),
372 idx.data(PendingWriteRole).toInt());
373 }
374 return;
375 }

392bool DeviceRowDelegate::editorEvent(QEvent *event,
393 QAbstractItemModel *model,
394 const QStyleOptionViewItem &opt,
395 const QModelIndex &idx) {
396 if (idx.column() != ColAction)
397 return QStyledItemDelegate::editorEvent(event, model, opt, idx);
398
399 auto *me = (event->type() == QEvent::MouseButtonPress
400 || event->type() == QEvent::MouseButtonRelease)
401 ? static_cast<QMouseEvent *>(event)
402 : nullptr;
403 if (!me) return QStyledItemDelegate::editorEvent(event, model, opt, idx);
404
405 const QPoint pos = me->pos();
406 const QRect cell = opt.rect;
407
408 auto repaint = [&]() {
409 if (auto *view = qobject_cast<QAbstractItemView *>(parent()))
410 view->viewport()->update(cell);
411 };
412
413 if (m_mode == Bit) {
414 const SubButton btn = hitTestBit(cell, pos);
415 if (event->type() == QEvent::MouseButtonPress) {
416 if (btn == SubNone) return false;
417 m_pressedRow = idx.row();
418 m_pressedBtn = btn;
419 repaint();
420 return true;
421 }
422 if (event->type() == QEvent::MouseButtonRelease) {
423 const int row = m_pressedRow;
424 const SubButton pressed = m_pressedBtn;
425 m_pressedRow = -1;
426 m_pressedBtn = SubNone;
427 repaint();
428 if (pressed == SubNone || row != idx.row()) return false;
429 if (pressed != btn) return false;
430
431 const int address = idx.sibling(idx.row(), ColAddress)
432 .data(AddressRole).toInt();
433 switch (btn) {
434 case SubOn: emit bitWriteRequested(address, 1); break;
435 case SubOff: emit bitWriteRequested(address, 0); break;
436 case SubToggle: {
437 const bool cur = idx.sibling(idx.row(), ColState)
438 .data(BitStateRole).toBool();
439 emit bitWriteRequested(address, cur ? 0 : 1);
440 break;
441 }
442 default: break;
443 }
444 return true;
445 }
446 } else {
447 const SubButton btn = hitTestWord(cell, pos);
448 if (event->type() == QEvent::MouseButtonPress) {
449 if (btn == SubNone) return false;
450 m_pressedRow = idx.row();
451 m_pressedBtn = btn;
452 repaint();
453 return true;
454 }
455 if (event->type() == QEvent::MouseButtonRelease) {
456 const int row = m_pressedRow;
457 const SubButton pressed = m_pressedBtn;
458 m_pressedRow = -1;
459 m_pressedBtn = SubNone;
460 repaint();
461 if (pressed == SubNone || row != idx.row()) return false;
462 if (pressed != btn) return false;
463
464 const int address = idx.sibling(idx.row(), ColAddress)
465 .data(AddressRole).toInt();
466 if (btn == SubWrite) {
467 const qint16 v = static_cast<qint16>(
468 idx.data(PendingWriteRole).toInt());
469 emit wordWriteRequested(address, v);
470 return true;
471 }
472 if (btn == SubValue) {
473 // Open the spinbox editor on the same cell.
474 if (auto *view = qobject_cast<QAbstractItemView *>(parent()))
475 view->edit(idx);
476 return true;
477 }
478 }
479 }
480
481 return QStyledItemDelegate::editorEvent(event, model, opt, idx);
482}
