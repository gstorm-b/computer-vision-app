145 if (obj.empty()) {

149 // Schema/version gate: a document written by a newer app (version above
150 // what this build understands) is refused rather than silently loaded
151 // with partial-default state. A missing "version" key is the legacy
152 // pre-versioning baseline (treated as version 0) and is accepted.
153 const int version = obj.value("version").toInt(0);
154 if (version > kSchemaVersion) {
155 LOG_USER_ERR << QStringLiteral(
156 "TaskLocalizeConfig: document schema version %1 is newer "
157 "than supported %2; refusing to load.")
158 .arg(version)
159 .arg(kSchemaVersion);
160 return false;
161 }

163 d->m_nActiveCamera = obj["nActiveCamera"].toString("");
164 d->m_nActivePatternGroup = obj["nActivePatternGroup"].toString("");
165 d->m_nDetectedNumber = obj["nDetectedNumber"].toString("");
