105void PositionPropertyManager::setValue(QtProperty *property, const QVector<double> &val)
106{

226// ── Private helpers ───────────────────────────────────────────────────────────
227
231void PositionPropertyManager::createSubProperties(QtProperty *parent, Data &data)
232{
233 const QStringList labels = modeLabels(data.mode);
234 const int n = modeComponentCount(data.mode);
235
236 data.subProps.clear();
237 for (int i = 0; i < n; ++i) {
238 auto *sub = m_doubleManager->addProperty(labels[i]);
239 m_doubleManager->setDecimals (sub, data.decimals);
240 m_doubleManager->setRange (sub, data.minimum, data.maximum);
241 m_doubleManager->setSingleStep(sub, data.singleStep);
242 m_doubleManager->setValue (sub, data.values.value(i, 0.0));
243 parent->addSubProperty(sub);
244 m_subToParent[sub] = parent;
245 data.subProps.append(sub);
246 }
247}
248
251void PositionPropertyManager::destroySubProperties(QtProperty *parent, Data &data)
252{
253 for (auto *sub : std::as_const(data.subProps)) {
254 m_subToParent.remove(sub);
255 parent->removeSubProperty(sub);
256 delete sub;
257 }
258 data.subProps.clear();
259}

1162void PointFPropertyManager::createSubProperties(QtProperty *parent, Data &data)
1163{
1164 const QStringList labels = modeLabels(data.mode);
1165 const int n = modeComponentCount(data.mode);
1166
1167 data.subProps.clear();
1168 for (int i = 0; i < n; ++i) {
1169 auto *sub = m_doubleManager->addProperty(labels[i]);
1170 m_doubleManager->setDecimals (sub, data.decimals);
1171 m_doubleManager->setRange (sub, data.minimum, data.maximum);
1172 m_doubleManager->setSingleStep(sub, data.singleStep);
1173 m_doubleManager->setValue (sub, data.values.value(i, 0.0));
1174 parent->addSubProperty(sub);
1175 m_subToParent[sub] = parent;
1176 data.subProps.append(sub);
1177 }
