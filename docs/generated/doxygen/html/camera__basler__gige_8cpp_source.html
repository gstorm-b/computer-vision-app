123bool BaslerGigECamera::deviceConnect() {
124 m_camera_ip_address = m_config.m_ipAddress;
125
126 try {
127 Pylon::DeviceInfoList cameraDeviceList;
128 Pylon::ITransportLayer* tl = Pylon::CTlFactory::GetInstance().CreateTl("BaslerGigE");
129 Pylon::IGigETransportLayer* gigetl = dynamic_cast<Pylon::IGigETransportLayer*>(tl);
130 if (!gigetl) {
131 LOG_DEV_ERR << "Camera connect error, Basler SDL cannot create transport layer";
132 m_last_msg = tr("SDK error cannot create transport layer");
133 setConnectionStatus(ConnectStatus::ConnectFailed, m_last_msg);
134 return false;
135 }
136 gigetl->EnumerateAllDevices(cameraDeviceList, false);
137
138
139 bool camera_found = false;
140 for (int idx=0;idx<cameraDeviceList.size();idx++) {
141 QString ip_address = QString::fromStdString(cameraDeviceList.at(idx).GetIpAddress().c_str());
142 if (m_camera_ip_address == ip_address) {
143 m_config.m_deviceInfo = cameraDeviceList.at(idx);
144 m_camera_info = m_config.m_deviceInfo;
145 camera_found = true;
146 break;
147 }
148 }
149
150 if (!camera_found) {
151 m_last_msg = tr("Not found camera, ip address: %1").arg(m_camera_ip_address);
152 LOG_USER_ERR << m_last_msg;
153 setConnectionStatus(ConnectStatus::ConnectFailed, m_last_msg);
154 return false;
155 }
156
157 m_str_camera_id = QString::fromUtf8(m_camera_info.GetSerialNumber().c_str());
158 LOG_USER_INFO << QString("Starting connect with camera %1, %2, %3, %4")
159 .arg(QString::fromUtf8(m_camera_info.GetModelName().c_str()))
160 .arg(QString::fromUtf8(m_camera_info.GetSerialNumber().c_str()))
161 .arg(QString::fromUtf8(m_camera_info.GetUserDefinedName().c_str()))
162 .arg(QString::fromUtf8(m_camera_info.GetIpAddress().c_str()));
163
164 // create new camera instance
165 if (m_camera_instance != nullptr) {
166 m_camera_instance.reset();
167 }
168 m_camera_instance = std::make_shared<Pylon::CInstantCamera>(Pylon::CTlFactory::GetInstance().CreateDevice(m_camera_info));
169 m_camera_instance->Open();
170
171 if (!m_camera_instance->IsOpen()) {
172 m_last_msg = tr("Camera %1 open fail.").arg(
173 QString::fromStdString(m_camera_instance->GetDeviceInfo().GetDeviceID().c_str()));
174 LOG_USER_ERR << m_last_msg;
175 setConnectionStatus(ConnectStatus::ConnectFailed, m_last_msg);
176 m_camera_instance->DestroyDevice();
177 m_camera_instance.reset();
178 return false;
179 }
180
181
182 LOG_USER_INFO << QString("Camera opened (%1)").arg(QString::fromUtf8(m_camera_info.GetSerialNumber().c_str()));
183 m_camera_instance->MaxNumBuffer = 64;
184 initializeIOPort();
185 m_config.m_ioCapabilities= m_io_lines;
186 m_config.m_ioCapabilities.detach();
187 readCameraSetting();
188
189 } catch (const Pylon::GenericException& e) {
190 LOG_DEV_ERR << tr("Camera error, An exception occurred while connecting to camera");
191 LOG_DEV_ERR << e.GetDescription();
192 m_last_msg = e.GetDescription();
193 setConnectionStatus(ConnectStatus::ConnectFailed, m_last_msg);
194 return false;
195 }
196
197 LOG_USER_INFO << tr("Camera connect successful.");
198 setConnectionStatus(ConnectStatus::Connected);
199 return true;
200}

445GrabResult BaslerGigECamera::grabSingleShot() {
446 GrabResult result;
447 result.isGrabSuccess = false;
448 if (m_camera_instance == nullptr) {
449 LOG_DEV_ERR << "grab single shot fail, camera instance is null";
450 result.msg = "grab single shot fail, camera instance is null";
451 return result;
452 }
453
454 if (m_camera_instance->IsGrabbing()) {
455 LOG_DEV_ERR << "grab single shot fail, camera already grabbing";
456 m_last_msg = "grab single shot fail, camera already grabbing";
457 result.msg = m_last_msg;
458 return result;
459 }
460
461 try {
462 // enable output before grabbing
463 if (m_config.m_autoBacklightControl) {
464 setOutputLineState(m_config.m_autoBacklightLine, true);
465 QThread::usleep(m_config.m_autoBacklightDelay);
466 }
467
468 // smart pointer receive the grab result data
469 Pylon::CInstantCamera::GrabResultPtr_t ptrGrabResult;
470 m_camera_instance->GrabOne(5000, ptrGrabResult, TimeoutHandling_ThrowException);
471
472 // retrieve result
473 if (ptrGrabResult->GrabSucceeded()) {
474 cv::Mat image;
475 pylon_image_to_mat(ptrGrabResult, image);
476 GrabResult result;
477 result.frame = image.clone();
478 result.isGrabSuccess= true;
479 result.msg = "Grab Succesfull";
480 emit grabFinished(result);
481 } else {\
482 LOG_DEV_ERR << "grab single shot error:"
483 << QString::number(ptrGrabResult->GetErrorCode(), 16)
484 << ptrGrabResult->GetErrorDescription();
485 m_last_msg = tr("grab single shot failed, %1, %2")
486 .arg(QString::number(ptrGrabResult->GetErrorCode(), 16))
487 .arg(ptrGrabResult->GetErrorDescription().c_str());
488
489 result.msg = "Error";
490 emit grabFinished(result);
491 }
492
493 if (m_config.m_autoBacklightControl) {
494 setOutputLineState(m_config.m_autoBacklightLine, false);
495 }
496
497 } catch (GenICam::GenericException &e) {
498 LOG_DEV_ERR << "An exception occurred while handle continuous shot";
499 LOG_DEV_ERR << e.GetDescription();
500 m_last_msg = e.GetDescription();
501 }
502
503 return result;
504}

571void BaslerGigECamera::initializeIOPort() {
572 m_io_lines.clear();
573 try {
574 LOG_DEV_INFO << QString("Start check IO configuration capabilities.");
575
576 // Get all IO Line information from camera node map
577 GenApi::INodeMap& node_map = m_camera_instance->GetNodeMap();
578 GenApi::CEnumerationPtr lineSelector(node_map.GetNode("LineSelector"));
579 if (GenApi::IsAvailable(lineSelector)) {
580
581 GenApi::StringList_t line_names;
582 lineSelector->GetSymbolics(line_names);
583
584 for (int idx=0;idx<line_names.size();idx++) {
585 // add to all line option
586 Pylon::String_t line_name = line_names[idx];
587 QString line_option = line_name.c_str();
588 lineSelector->FromString(line_name);
589 GenApi::CEnumerationPtr lineMode(node_map.GetNode("LineMode"));
590 BaslerIOLine line_capabilities;
591 if (GenApi::IsAvailable(lineMode)) {
592
593 // check input/output change able mode
594 line_capabilities.is_writable = GenApi::IsWritable(lineMode);
595 GenApi::NodeList_t modeEntries;
596 lineMode->GetEntries(modeEntries);
597
598 for (auto const& node : modeEntries) {
599 GenApi::CEnumEntryPtr pEntry(node);
600 if (GenApi::IsAvailable(pEntry)) {
601 std::string modeStr = pEntry->GetSymbolic().c_str();
602 if (modeStr == "Input") line_capabilities.can_be_input = true;
603 if (modeStr == "Output") line_capabilities.can_be_output = true;
604 }
605 }
606
607 if (line_capabilities.can_be_input && line_capabilities.can_be_output) {
608 LOG_DEV_INFO << QString("Camera basler: %1 can be Input or Output").arg(line_option);
609 } else if (line_capabilities.can_be_input) {
610 LOG_DEV_INFO << QString("Camera basler: %1 can be Input only").arg(line_option);
611 } else if ( line_capabilities.can_be_output) {
612 LOG_DEV_INFO << QString("Camera basler: %1 can be Output only").arg(line_option);
613 }
614
615 line_capabilities.name = line_option;
616 m_io_lines.append(line_capabilities);
617 } else {
618 LOG_DEV_INFO << QString("Camera basler: %1 Line Mode not available!").arg(line_option);
619 }
620 }
621 } else {
622 LOG_DEV_INFO << QString("Camera basler: Line selector is not available!");
623 return;
624 }
625 } catch (const Pylon::GenericException& e) {
626 // Handle error
627 }

727inline void BaslerGigECamera::pylon_image_to_mat(Pylon::CInstantCamera::GrabResultPtr_t &ptrGrabResult, cv::Mat &mat) {
728 Pylon::CPylonImage pylon_image;
729 Pylon::CImageFormatConverter formatConverter;
730 Pylon::EPixelType pixel_type = ptrGrabResult->GetPixelType();
731 bool is_color = Pylon::IsColorImage(pixel_type);
732 if (is_color) {
733 formatConverter.OutputPixelFormat = Pylon::PixelType_BGR8packed;
734 formatConverter.Convert(pylon_image, ptrGrabResult);
735 } else {
736 formatConverter.OutputPixelFormat = Pylon::PixelType_Mono8;
737 formatConverter.Convert(pylon_image, ptrGrabResult);
738 }
739
740 switch (formatConverter.OutputPixelFormat.GetValue()) {
741 case Pylon::PixelType_Mono8:
742 mat = cv::Mat(pylon_image.GetHeight(), pylon_image.GetWidth(), CV_8UC1, (uint8_t*)pylon_image.GetBuffer()).clone();
743 return;
744 case Pylon::PixelType_BGR8packed:
745 mat = cv::Mat(pylon_image.GetHeight(), pylon_image.GetWidth(), CV_8UC3, (uint8_t*)pylon_image.GetBuffer()).clone();
746 return;
747 default:
748 break;
749 }
750
751 LOG_DEV_ERR << "convert image fail, unsupported pixel format" << formatConverter.OutputPixelFormat.GetValue();
752 mat = cv::Mat();
753 return;
754}
