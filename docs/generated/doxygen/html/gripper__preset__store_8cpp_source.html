70void GripperPresetStore::fromJson(const QJsonArray &arr) {
71 m_presets.clear();
72 for (const QJsonValue &v : arr) {
73 const QJsonObject o = v.toObject();
74
75 GripperPreset p;
76 p.name = o["name"].toString();
77 p.boxes.size = cv::Size2f(static_cast<float>(o["w"].toDouble(0.0)),
78 static_cast<float>(o["h"].toDouble(0.0)));
79 // An "angle" key written by an earlier build is deliberately ignored: the jaw
80 // angle is per-pattern now, so a preset must not carry one back in.
81 p.boxes.distance = o["distance"].toDouble(0.0);
82
83 // add() enforces the non-blank/unique invariant; a rejected entry is dropped so a
84 // hand-edited file degrades to a shorter list instead of failing the task load.
85 add(p);
86 }
