54void ItemRoi::paint(QPainter *painter,
55 const QStyleOptionGraphicsItem *option,
56 QWidget *widget) {
57 Q_UNUSED(widget);
58 // Draw ROI
59 QPen pen(Qt::red);
60 pen.setWidth(2);
61 pen.setCosmetic(true);
62 painter->setPen(pen);
63 painter->setBrush(Qt::NoBrush);
64 // painter->drawEllipse(this->rect().center(), 5, 5);
65 // painter->drawEllipse(this->transformOriginPoint(), 10, 10);
66
67 if (m_is_draw_center) {
68 int centerX = this->rect().center().x();
69 int centerY = this->rect().center().y();
70 const int sizeX = this->rect().width() * 0.06;
71 const int sizeY = this->rect().height() * 0.06;
72 const int center_size = (sizeX > sizeY) ? sizeY : sizeX;
73 painter->drawLine(centerX - center_size, centerY, centerX + center_size, centerY);
74 painter->drawLine(centerX, centerY - center_size, centerX, centerY + center_size);
75 }
76
77 // if ROI is selected draw Handle point
78 if (option->state & QStyle::State_Selected) {
79 // bouding box draw
80 pen.setColor(m_bounding_color_selected);
81 pen.setStyle(Qt::DashLine);
82 painter->setPen(pen);
83 painter->drawRect(rect());
84
85 // draw handle point add 4 corner
86 QColor handler_color(0x8F87F1);
87 handler_color.setAlpha(100);
88 painter->setBrush(handler_color);
89 QPen handlePen(Qt::black);
90 handlePen.setCosmetic(true);
91 painter->setPen(handlePen);
92 painter->drawRect(handleRect(TopLeft));
93 painter->drawRect(handleRect(TopRight));
94 painter->drawRect(handleRect(BottomLeft));
95 painter->drawRect(handleRect(BottomRight));
96 } else {
97 // draw ROI in mode not selected
98 pen.setColor(m_bounding_color_normal);
99 pen.setStyle(Qt::DashLine);
100 painter->setPen(pen);
101 painter->drawRect(rect());
102 }
103}

143QVariant ItemRoi::itemChange(QGraphicsItem::GraphicsItemChange change, const QVariant &value) {
144
145 // only override for item position changed
146 if (change == QGraphicsItem::ItemPositionChange && parentItem()) {
147 QPointF newPos = value.toPointF();
148
149 // map bounding rect to parent coordiante, which is pixmap item
150 QRectF rectInParent = mapRectToParent(rect());
151 QRectF parentRect = parentItem()->boundingRect();
152
153 // vector offset when move
154 QPointF delta = newPos - pos();
155 QRectF movedRect = rectInParent.translated(delta);
156
157 // void move outside of parent's bounding box
158 QPointF corrected = newPos;
159
160 if (!parentRect.contains(movedRect)) {
161 if (movedRect.left() < parentRect.left())
162 corrected.rx() += parentRect.left() - movedRect.left();
163 if (movedRect.right() > parentRect.right())
164 corrected.rx() -= movedRect.right() - parentRect.right();
165 if (movedRect.top() < parentRect.top())
166 corrected.ry() += parentRect.top() - movedRect.top();
167 if (movedRect.bottom() > parentRect.bottom())
168 corrected.ry() -= movedRect.bottom() - parentRect.bottom();
169
170 return corrected;
171 }
172 }
173
174 return QGraphicsRectItem::itemChange(change, value);
175}

179QRectF ItemRoi::handleRect(HandlePosition pos) const {
180 QRectF r = rect();
181 QPointF point;
182 const qreal handleSize = effectiveHandleSize();
183 switch(pos) {
184 case TopLeft:
185 point = r.topLeft();
186 break;
187 case TopRight:
188 point = r.topRight();
189 break;
190 case BottomLeft:
191 point = r.bottomLeft();
192 break;
193 case BottomRight:
194 point = r.bottomRight();
195 break;
196 default:
197 point = QPointF();
198 break;
199 }
200 // draw rectangle handle with center at corner
201 return QRectF(point.x() - handleSize/2, point.y() - handleSize/2,
202 handleSize, handleSize);
203}

252void ItemRoi::mousePressEvent(QGraphicsSceneMouseEvent *event) {

280void ItemRoi::mouseMoveEvent(QGraphicsSceneMouseEvent *event) {
281 if (m_ignore != nullptr) {
282 if (*m_ignore == true) {
283 event->ignore();
284 return;
285 }
286 }
287
288 if (m_current_handle != None) {
289 QPointF delta = event->pos() - m_press_pos;
290 QRectF newRect = m_original_rect;
291 switch(m_current_handle) {
292 case TopLeft:
293 newRect.setTopLeft(newRect.topLeft() + delta);
294 break;
295 case TopRight:
296 newRect.setTopRight(newRect.topRight() + delta);
297 break;
298 case BottomLeft:
299 newRect.setBottomLeft(newRect.bottomLeft() + delta);
300 break;
301 case BottomRight:
302 newRect.setBottomRight(newRect.bottomRight() + delta);
303 break;
304 default:
305 break;
306 }
307
308 // check new position outside parrent
309 if (!isInsideParent(newRect)) {
310 event->accept();
311 return;
312 }
313
314 // make sure new rect always large than limited rect
315 if (newRect.width() < m_handle_size) {
316 event->accept();
317 return;
318 }
319
320 if (newRect.height() < m_handle_size) {
321 event->accept();
322 return;
323 }
324
325 setRect(newRect);
326 event->accept();
327 return;
328 }
329
330 event->accept();
331 QGraphicsRectItem::mouseMoveEvent(event);
332}
