204 static QtVariantProperty *addPropertyToBrowser(
205 const QMetaObject &meta,
206 QMetaProperty &prop,
207 QVariant &value,
208 QtVariantPropertyManager *manager,
209 QtTreePropertyBrowser *browser)
210 {
211 const QString propName = QString::fromLatin1(prop.name());
212 if (propName == QLatin1String("objectName")) return nullptr;
213
214 QtVariantProperty *variantProp = nullptr;
215
216 if (prop.isEnumType()) {
217 variantProp = manager->addProperty(QtVariantPropertyManager::enumTypeId(), propName);
218
219 QStringList enumNames;
220 const QMetaEnum metaEnum = prop.enumerator();
221 enumNames.reserve(metaEnum.keyCount());
222 for (int j = 0; j < metaEnum.keyCount(); ++j) {
223 enumNames << QCoreApplication::translate(meta.className(),
224 metaEnum.key(j));
225 }
226 variantProp->setAttribute(QLatin1String("enumNames"), enumNames);
227 variantProp->setValue(value.toInt());
228 } else {
229 variantProp = manager->addProperty(value.userType(), propName);
230 if (variantProp) variantProp->setValue(value);
231 }
232
233 if (!variantProp) return nullptr;
234
235 // Display name override
236 const int nameIdx = meta.indexOfClassInfo(
237 QStringLiteral("%1_name").arg(propName).toUtf8());
238 if (nameIdx != -1) {
239 const QString displayName = QString::fromUtf8(meta.classInfo(nameIdx).value());
240 if (!displayName.isEmpty())
241 variantProp->setDisplayName(displayName);
242 }
243
244 // Numeric range (stored as strings in Q_CLASSINFO)
245 const int minIdx = meta.indexOfClassInfo(
246 QStringLiteral("%1_min").arg(propName).toUtf8());
247 if (minIdx != -1)
248 variantProp->setAttribute(QStringLiteral("minimum"),
249 QString::fromUtf8(meta.classInfo(minIdx).value()).toDouble());
250
251 const int maxIdx = meta.indexOfClassInfo(
252 QStringLiteral("%1_max").arg(propName).toUtf8());
253 if (maxIdx != -1)
254 variantProp->setAttribute(QStringLiteral("maximum"),
255 QString::fromUtf8(meta.classInfo(maxIdx).value()).toDouble());
256
257 // Tooltip / description
258 const int tipIdx = meta.indexOfClassInfo(
259 QStringLiteral("%1_desc").arg(propName).toUtf8());
260 if (tipIdx != -1)
261 variantProp->setDescriptionToolTip(
262 QString::fromUtf8(meta.classInfo(tipIdx).value()));
263
264 if (!prop.isWritable())
265 variantProp->setEnabled(false);
266
267 if (browser)
268 browser->addProperty(variantProp);
269
270 return variantProp;
271 }
