130QVariant ItemRoiRotated::itemChange(QGraphicsItem::GraphicsItemChange change, const QVariant &value) {
131
132 // only override for item position changed
133 if (change == QGraphicsItem::ItemPositionChange && parentItem()) {
134 QPointF newPos = value.toPointF();
135
136 // map bounding rect to parent coordiante, which is pixmap item
137 QRectF rectInParent = mapRectToParent(rect());
138 QRectF parentRect = parentItem()->boundingRect();
139
140 // vector offset when move
141 QPointF delta = newPos - pos();
142 QRectF movedRect = rectInParent.translated(delta);
143
144 // void move outside of parent's bounding box
145 QPointF corrected = newPos;
146
147 if (!parentRect.contains(movedRect)) {
148 if (movedRect.left() < parentRect.left())
149 corrected.rx() += parentRect.left() - movedRect.left();
150 if (movedRect.right() > parentRect.right())
151 corrected.rx() -= movedRect.right() - parentRect.right();
152 if (movedRect.top() < parentRect.top())
153 corrected.ry() += parentRect.top() - movedRect.top();
154 if (movedRect.bottom() > parentRect.bottom())
155 corrected.ry() -= movedRect.bottom() - parentRect.bottom();
156
157 return corrected;
158 }
159 }
160
161 return QGraphicsRectItem::itemChange(change, value);
162}

166QRectF ItemRoiRotated::handleRect(HandlePosition pos) const {
167 QRectF r = rect();
168 QPointF point;
169 const qreal handleSize = effectiveHandleSize();
170 switch(pos) {
171 case TopLeft:
172 point = r.topLeft();
173 break;
174 case TopRight:
175 point = r.topRight();
176 break;
177 case BottomLeft:
178 point = r.bottomLeft();
179 break;
180 case BottomRight:
181 point = r.bottomRight();
182 break;
183 default:
184 point = QPointF();
185 break;
186 }
187 // draw rectangle handle with center at corner
188 return QRectF(point.x() - handleSize/2, point.y() - handleSize/2,
189 handleSize, handleSize);
190}

264void ItemRoiRotated::mousePressEvent(QGraphicsSceneMouseEvent *event) {
265 if (m_ignore != nullptr) {
266 if (*m_ignore == true) {
267 event->ignore();
268 return;
269 }
270 }
271
272 QPointF pos = event->pos();
273 m_last_center = this->transformOriginPoint();
274 m_current_handle = getHandleAt(pos);
275 // check if event emit inside handle area
276 if (m_current_handle == RotateHandle) {
277 // store original origin to calculate rotated angle later
278 m_original_rotation = rotation();
279 m_valid_rotation = m_original_rotation;
280 m_rotation_origin = mapToScene(transformOriginPoint());
281 // angle determine by center of ROI to mouse clicked position
282 m_press_angle = QLineF(m_rotation_origin, event->scenePos()).angle();
283 setTransformOriginPoint(rect().center());
284 event->accept();
285 return;
286 } else if (m_current_handle != None) {
287 // if clicked in corner handle, store original rect
288 m_original_rect = rect();
289 m_press_pos = pos;
290 event->accept();
291 return;
292 }
293
294 // if clicked inside, switch to position changed mode, which is processed in based class
295 event->accept();
296 QGraphicsRectItem::mousePressEvent(event);
297}

304void ItemRoiRotated::mouseMoveEvent(QGraphicsSceneMouseEvent *event) {
305 if (m_ignore != nullptr) {
306 if (*m_ignore == true) {
307 event->ignore();
308 return;
309 }
310 }
311
312 if (m_current_handle == RotateHandle) {
313 // angle determine by center of ROI to mouse clicked position
314 qreal currentAngle = QLineF(m_rotation_origin, event->scenePos()).angle();
315 qreal angleDiff = currentAngle - m_press_angle;
316 setRotation(m_original_rotation - angleDiff);
317
318 // avoid rotate out side of parrent
319 QRectF new_rect = mapRectToParent(rect());
320 if (!parentItem()->boundingRect().contains(new_rect)) {
321 setRotation(m_valid_rotation);
322 } else {
323 m_valid_rotation = m_original_rotation - angleDiff;
324 }
325
326 event->accept();
327 return;
328 } else if (m_current_handle != None) {
329 QPointF delta = event->pos() - m_press_pos;
330 QRectF newRect = m_original_rect;
331 switch(m_current_handle) {
332 case TopLeft:
333 newRect.setTopLeft(newRect.topLeft() + delta);
334 break;
335 case TopRight:
336 newRect.setTopRight(newRect.topRight() + delta);
337 break;
338 case BottomLeft:
339 newRect.setBottomLeft(newRect.bottomLeft() + delta);
340 break;
341 case BottomRight:
342 newRect.setBottomRight(newRect.bottomRight() + delta);
343 break;
344 default:
345 break;
346 }
347
348 // check new position outside parrent
349 if (!isInsideParent(newRect)) {
350 event->accept();
351 return;
352 }
353
354 // make sure new rect always large than limited rect
355 if (newRect.width() < m_handle_size*2) {
356 event->accept();
357 return;
358 }
359
360 if (newRect.height() < m_handle_size*2) {
361 event->accept();
362 return;
363 }
364
365 setRect(newRect);
366 event->accept();
367 return;
368 }
369
370 event->accept();
371 QGraphicsRectItem::mouseMoveEvent(event);
372}
