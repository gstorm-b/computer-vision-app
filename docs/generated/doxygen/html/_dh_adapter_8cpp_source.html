49Result<SerialRobotConfig> DhAdapter::fromStandardDh(const RobotIdentity& identity,
50 const std::vector<StandardDhParameter>& rows)
51{
52 if (rows.empty()) {
53 return Result<SerialRobotConfig>::failure(KinematicsStatus::InvalidRequest, "DH rows must not be empty");
54 }
55
56 SerialRobotConfig config;
57 config.identity = identity;
58 config.topology = RobotTopologyType::Serial;
59 config.dof = static_cast<int>(rows.size());
60 config.frames.baseLinkId = "base_link";
61 config.frames.flangeLinkId = "flange";
62 config.links.push_back({"base_link"});
63
64 std::string parent = "base_link";
65 for (std::size_t i = 0; i < rows.size(); ++i) {
66 const StandardDhParameter& row = rows[i];
67 const std::string motionLink = "dh_motion_" + std::to_string(i + 1);
68 const std::string nextLink = (i + 1 == rows.size()) ? "flange" : "dh_link_" + std::to_string(i + 1);
69 config.links.push_back({motionLink});
70 config.links.push_back({nextLink});
71
72 Joint joint;
73 joint.id = row.jointId.empty() ? "J" + std::to_string(i + 1) : row.jointId;
74 joint.type = row.type == DhJointType::Revolute ? JointType::Revolute : JointType::Prismatic;
75 joint.parentLinkId = parent;
76 joint.childLinkId = motionLink;
77 joint.origin = poseFromIsometry(rzTz(row.type == DhJointType::Revolute ? row.thetaOffset_rad : 0.0,
78 row.type == DhJointType::Prismatic ? 0.0 : row.d_m));
79 joint.axis = Eigen::Vector3d::UnitZ();
80 joint.limits = row.limits;
81 joint.home = row.home;
82 config.joints.push_back(joint);
83
84 Joint fixed;
85 fixed.id = joint.id + "_link";
86 fixed.type = JointType::Fixed;
87 fixed.parentLinkId = motionLink;
88 fixed.childLinkId = nextLink;
89 fixed.origin = poseFromIsometry(txRx(row.a_m, row.alpha_rad));
90 if (row.type == DhJointType::Prismatic) {
91 fixed.origin = poseFromIsometry(rzTz(row.thetaOffset_rad, row.d_m) * txRx(row.a_m, row.alpha_rad));
92 }
93 config.joints.push_back(fixed);
94 parent = nextLink;
95 }
96
97 const ModelValidationResult validation = RobotModelValidator::validateSerialRobotConfig(config);
98 if (!validation.ok()) {
99 return Result<SerialRobotConfig>::failure(validation.status(), validation.issues.front().message);
100 }
101 return Result<SerialRobotConfig>::success(config);
102}
