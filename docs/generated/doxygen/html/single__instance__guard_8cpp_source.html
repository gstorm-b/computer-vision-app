75bool SingleInstanceGuard::tryAcquire(int timeoutMs)
76{
77 m_lockFile = std::make_unique<QLockFile>(lockFilePath());
78
79 // Stale-lock handling is left at QLockFile's default on purpose. A lock whose recorded
80 // PID is no longer running is reclaimed by QLockFile regardless, which is the crash
81 // case that matters; the stale *time* is only the secondary guard for a recycled PID.
82 // setStaleLockTime(0) would read like "reclaim immediately" but means the opposite —
83 // never stale by age — and would give up that secondary guard.
84 //
85 // `timeoutMs` is the tryLock TIMEOUT, not the stale time. It is 0 for a normal launch
86 // — that decision is made while the user waits on a double-clicked icon, so it must
87 // not block — and a real duration only when this process was started to REPLACE an
88 // exiting sibling, whose lock is still held while it shuts its devices down.
89 // QLockFile does the waiting and the retrying itself; a hand-rolled retry loop here
90 // would duplicate it and get the stale-lock handling subtly different.
91 if (!m_lockFile->tryLock(timeoutMs)) {
92 qint64 pid = 0;
93 QString hostname;
94 QString appname;
95 if (m_lockFile->getLockInfo(&pid, &hostname, &appname)) {
96 LOG_USER_INFO << "Another instance is already running."
97 << "key=" << m_instanceKey
98 << "pid=" << pid
99 << "app=" << appname;
100 } else {
101 LOG_USER_INFO << "Another instance is already running." << m_instanceKey;
102 }
103 m_lockFile.reset();
104 return false;
105 }
106
107 startRaiseListener();
108 return true;
109}

115void SingleInstanceGuard::startRaiseListener()
116{
117 m_server = new QLocalServer(this);
118
119 // Clear a socket left behind by a crash; on platforms where the endpoint survives the
120 // process, listen() would otherwise fail for a name nobody is using.
121 QLocalServer::removeServer(serverName());
122
123 if (!m_server->listen(serverName())) {
124 LOG_DEV_ERR << "Single-instance raise listener could not start."
125 << "key=" << m_instanceKey
126 << "error=" << m_server->errorString();
127 m_server->deleteLater();
128 m_server = nullptr;
129 return;
130 }
131
132 connect(m_server, &QLocalServer::newConnection, this, [this]() {
133 while (QLocalSocket *client = m_server->nextPendingConnection()) {
134 connect(client, &QLocalSocket::disconnected, client, &QLocalSocket::deleteLater);
135 client->write(kRaiseRequest, static_cast<qint64>(sizeof(kRaiseRequest) - 1));
136 client->flush();
137 client->disconnectFromServer();
138 }

146bool SingleInstanceGuard::notifyExistingInstance()
147{
148#ifdef Q_OS_WIN
149 // Without this, Windows refuses to let a background process take the foreground and
150 // only flashes its taskbar button — which reads to the operator as "clicking the icon
151 // did nothing". The PID comes from the lock file the running instance wrote.
152 QLockFile probe(lockFilePath());
153 qint64 pid = 0;
154 QString hostname;
155 QString appname;
156 if (probe.getLockInfo(&pid, &hostname, &appname) && pid > 0) {
157 ::AllowSetForegroundWindow(static_cast<DWORD>(pid));
158 }
159#endif
160
161 QLocalSocket socket;
162 socket.connectToServer(serverName());
163 if (!socket.waitForConnected(kNotifyTimeoutMs)) {
164 LOG_DEV_ERR << "Could not reach the running instance to raise it."
165 << "key=" << m_instanceKey
166 << "error=" << socket.errorString();
167 return false;
168 }
169
170 // Wait for the acknowledgement rather than just the connection: on Windows the
171 // foreground handover needs the running process to have actually reacted before this
172 // process exits and gives up its own foreground claim.
173 const bool acknowledged = socket.waitForReadyRead(kNotifyTimeoutMs);
174 socket.disconnectFromServer();
175 return acknowledged;
176}
