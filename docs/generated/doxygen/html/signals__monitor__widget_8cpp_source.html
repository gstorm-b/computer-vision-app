203 void setBoolValue(bool v) {
204 const bool first = !m_hasBool;
205 const bool last = m_lastBool;
206 m_hasBool = true;
207 m_lastBool = v;
208
209 // Render chip.
210 m_lblOnOff->setText(v ? tr("ON") : tr("OFF"));
211 m_lblOnOff->setProperty("onOffState", v ? "on" : "off");
212 repolish(m_lblOnOff);
213
214 if (first) {
215 m_tRising.invalidate();
216 return;
217 }
218 if (last == v) return; // no transition
219
220 if (!last && v) { // OFF→ON
221 m_tRising.start();
222 } else if (last && !v) { // ON→OFF
223 if (m_tRising.isValid() &&
224 m_tRising.elapsed() < kRisingEdgeWindowMs) {
225 triggerRisingEdge();
226 }
227 m_tRising.invalidate();
228 }

373 ModifyValueDialog(Type type,

384 buildBool(root, initial.toBool());

391 root->addWidget(bb);
392 connect(bb, &QDialogButtonBox::accepted, this, &QDialog::accept);
393 connect(bb, &QDialogButtonBox::rejected, this, &QDialog::reject);

398 QVariant value() const {
399 if (m_type == Type::Bool) {
400 return QVariant(m_rbOn && m_rbOn->isChecked());
401 }
402 // Number — read the active page's spinbox.
403 return QVariant(m_stack->currentIndex() == 0
404 ? m_spSigned->value()
405 : m_spUnsigned->value());
406 }

408private:
411 void buildBool(QVBoxLayout *root, bool initial) {
412 m_rbOn = new QRadioButton(tr("ON"), this);
413 m_rbOff = new QRadioButton(tr("OFF"), this);
414 (initial ? m_rbOn : m_rbOff)->setChecked(true);
415 root->addWidget(m_rbOn);
416 root->addWidget(m_rbOff);
417 }
418
424 void buildNumber(QVBoxLayout *root, int initial) {
425 // Combobox picks interpretation; stacked widget swaps the spinbox.
426 // Default page: signed if initial fits signed range, else unsigned.
427 m_cbInterp = new QComboBox(this);
428 m_cbInterp->addItem(tr("Signed int16 (-32768..32767)"));
429 m_cbInterp->addItem(tr("Unsigned int16 (0..65535)"));
430
431 m_stack = new QStackedWidget(this);
432
433 m_spSigned = new QSpinBox(this);
434 m_spSigned->setRange(-32768, 32767);
435 m_spUnsigned = new QSpinBox(this);
436 m_spUnsigned->setRange(0, 65535);
437
438 m_stack->addWidget(m_spSigned);
439 m_stack->addWidget(m_spUnsigned);
440
441 // Default to unsigned only when the value cannot fit signed int16.
442 const int initialPage = (initial > 32767) ? 1 : 0;
443 if (initialPage == 0) {
444 m_spSigned->setValue(qBound(-32768, initial, 32767));
445 } else {
446 m_spUnsigned->setValue(qBound(0, initial, 65535));
447 }
448 m_cbInterp->setCurrentIndex(initialPage);
449 m_stack->setCurrentIndex(initialPage);
450
451 connect(m_cbInterp,
452 qOverload<int>(&QComboBox::currentIndexChanged),
453 this, [this](int i) {
454 // When switching, carry the visible value over, clamped to the
455 // destination range.
456 const int v = (m_stack->currentIndex() == 0)
457 ? m_spSigned->value()
458 : m_spUnsigned->value();
459 m_stack->setCurrentIndex(i);
460 if (i == 0) m_spSigned->setValue(qBound(-32768, v, 32767));
461 else m_spUnsigned->setValue(qBound(0, v, 65535));
462 });
463
464 root->addWidget(m_cbInterp);
465 root->addWidget(m_stack);
466 }
467
468 Type m_type;
469 QRadioButton *m_rbOn{nullptr};
470 QRadioButton *m_rbOff{nullptr};
471 QComboBox *m_cbInterp{nullptr};
472 QStackedWidget *m_stack{nullptr};
473 QSpinBox *m_spSigned{nullptr};
474 QSpinBox *m_spUnsigned{nullptr};
475};

548void SignalsMonitorWidget::insertRowAt(int row,
549 const QString &internalName,
550 const QString &displayName,
551 Type type) {
552 if (rowIndexOf(internalName) >= 0) {
553 LOG_USER_WARN << "SignalsMonitorWidget: duplicate internalName"
554 << internalName;
555 return;
556 }
557 row = qBound(0, row, m_rows.size());
558
559 auto *w = new RowWidget(internalName, displayName, type, m_list);
560 w->setDeviceConnected(m_deviceConnected);
561 connect(w, &RowWidget::modifyRequested, this, [this, w] {
562 openModifyDialog(w);
563 });
564
565 m_rows.insert(row, w);
566
567 auto *item = new QListWidgetItem();
568 m_list->insertItem(row, item);
569 m_list->setItemWidget(item, w);
570
571 relayoutColumns();
572}

653void SignalsMonitorWidget::relayoutColumns() {
654 if (m_rows.isEmpty()) return;
655 int wName = 0, wType = 0, wValue = 0, wModify = 0;
656 for (auto *r : m_rows) {
657 auto h = r->columnSizeHints();
658 wName = qMax(wName, h.name);
659 wType = qMax(wType, h.type);
660 wValue = qMax(wValue, h.value);
661 wModify = qMax(wModify, h.modify);
662 }
663 for (auto *r : m_rows) {
664 r->applyColumnWidths(wName, wType, wValue, wModify);
665 }
666 for (int i = 0; i < m_list->count(); ++i) {
667 QListWidgetItem *it = m_list->item(i);
668 QWidget *w = m_list->itemWidget(it);
669 if (w) it->setSizeHint(w->sizeHint());
670 }
671}
