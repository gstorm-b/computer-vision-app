235void DevicesMonitorWidget::buildRow(int row, int address) {
236 using R = DeviceRowDelegate;
237
238 // Column 0 — Address (read-only, painted by delegate). We still need
239 // a QTableWidgetItem so the model has somewhere to store the role.
240 auto *addrItem = new QTableWidgetItem;
241 addrItem->setFlags(addrItem->flags() & ~Qt::ItemIsEditable);
242 addrItem->setData(R::AddressRole, address);
243 m_table->setItem(row, 0, addrItem);
244
245 // Column 1 — Description (editable text via Qt::DisplayRole).
246 auto *descItem = new QTableWidgetItem;
247 auto descFlags = descItem->flags() & ~Qt::ItemIsSelectable;
248 if (m_commentEditable) descFlags |= Qt::ItemIsEditable;
249 else descFlags &= ~Qt::ItemIsEditable;
250 descItem->setFlags(descFlags);
251 m_table->setItem(row, 1, descItem);
252
253 // Column 2 — State (Bit) or Value (Word) — painted from role data.
254 auto *stateItem = new QTableWidgetItem;
255 stateItem->setFlags(stateItem->flags() & ~Qt::ItemIsEditable
256 & ~Qt::ItemIsSelectable);
257 if (m_mode == Mode::Bit) stateItem->setData(R::BitStateRole, false);
258 else stateItem->setData(R::WordValueRole, 0);
259 m_table->setItem(row, 2, stateItem);
260
261 // Column 3 — Action: painted by the delegate; the item only carries the
262 // pending-write value for Word mode.
263 auto *actItem = new QTableWidgetItem;
264 Qt::ItemFlags actFlags = actItem->flags() & ~Qt::ItemIsSelectable;
265 if (m_mode == Mode::Word) actFlags |= Qt::ItemIsEditable;
266 else actFlags &= ~Qt::ItemIsEditable;
267 actItem->setFlags(actFlags);
268 if (m_mode == Mode::Word) actItem->setData(R::PendingWriteRole, 0);
269 m_table->setItem(row, 3, actItem);
270}

275void DevicesMonitorWidget::setBitState(int address, quint8 value) {
276 const int row = rowOfAddress(address);
277 if (row < 0) return;
278 auto *item = m_table->item(row, DeviceRowDelegate::ColState);
279 if (!item) return;
280 const bool on = value != 0;
281 if (item->data(DeviceRowDelegate::BitStateRole).toBool() == on) return;
282 item->setData(DeviceRowDelegate::BitStateRole, on);
283 recountActive();
284}

346void DevicesMonitorWidget::recountActive() {
347 if (m_mode == Mode::Bit) {
348 int active = 0;
349 for (int i = 0; i < m_table->rowCount(); ++i) {
350 auto *item = m_table->item(i, DeviceRowDelegate::ColState);
351 if (item && item->data(DeviceRowDelegate::BitStateRole).toBool())
352 ++active;
353 }
354 m_countLabel->setText(QString("%1 / %2 %3")
355 .arg(active).arg(m_amount).arg(tr("ACTIVE")));
356 } else {
357 m_countLabel->setText(QString("%1 %2")
358 .arg(m_amount).arg(tr("WORDS")));
359 }

365void DevicesMonitorWidget::onFilterTextChanged(const QString &text) {
366 using R = DeviceRowDelegate;
367 const QString needle = text.trimmed();
368 for (int i = 0; i < m_table->rowCount(); ++i) {
369 if (needle.isEmpty()) {
370 m_table->setRowHidden(i, false);
371 continue;
372 }
373 const int address = m_table->item(i, R::ColAddress)
374 ? m_table->item(i, R::ColAddress)
375 ->data(R::AddressRole).toInt()
376 : -1;
377 const QString name = (address >= 0) ? formatName(address) : QString();
378 const QString desc = m_table->item(i, R::ColDescription)
379 ? m_table->item(i, R::ColDescription)->text()
380 : QString();
381 const bool match = name.contains(needle, Qt::CaseInsensitive)
382 || desc.contains(needle, Qt::CaseInsensitive);
383 m_table->setRowHidden(i, !match);
384 }
