42void RuntimeLayoutController::applyLayout(ads::CDockManager *manager,
43 const QList<ads::CDockWidget *> &docks,
44 int tileCount)
45{
46 if (manager == nullptr || docks.isEmpty()) {
47 return;
48 }
49
50 const QSize grid = gridFor(tileCount);
51 const int columns = grid.width();
52 const int rows = grid.height();
53
54 // Floating is an operator decision about where they want to watch a task. Re-docking
55 // one here would silently undo it, so a floated dock is left alone and does not
56 // consume a tile.
57 QList<ads::CDockWidget *> docked;
58 for (ads::CDockWidget *dock : docks) {
59 if (dock != nullptr && !dock->isFloating()) {
60 docked.append(dock);
61 }
62 }
63 if (docked.isEmpty()) {
64 return;
65 }
66
67 // One dock area per cell, indexed row-major. rowAnchors keeps the first area of each
68 // row so the next row can be opened below it rather than below the last column.
69 QList<ads::CDockAreaWidget *> rowAnchors;
70 ads::CDockAreaWidget *previousInRow = nullptr;
71
72 for (int index = 0; index < docked.size(); ++index) {
73 ads::CDockWidget *dock = docked.at(index);
74
75 const int cell = qMin(index, columns * rows - 1);
76 const int row = cell / columns;
77 const int column = cell % columns;
78
79 ads::CDockAreaWidget *area = nullptr;
80
81 if (index >= columns * rows) {
82 // Past the last cell: tab into it instead of hiding the dock. A running task
83 // nobody can see is worse than a crowded tab bar — nothing on screen would
84 // say it is still cycling.
85 area = manager->addDockWidget(ads::CenterDockWidgetArea, dock, previousInRow);
86 } else if (row == 0 && column == 0) {
87 area = manager->addDockWidget(ads::CenterDockWidgetArea, dock);
88 rowAnchors.append(area);
89 } else if (column == 0) {
90 // First cell of a new row: open it below the row above's anchor.
91 area = manager->addDockWidget(ads::BottomDockWidgetArea, dock,
92 rowAnchors.at(row - 1));
93 rowAnchors.append(area);
94 } else {
95 area = manager->addDockWidget(ads::RightDockWidgetArea, dock, previousInRow);
96 }
97
98 previousInRow = area;
99 }
100
101 LOG_DEV_INFO << "Runtime layout applied."
102 << "tiles=" << tileCount
103 << "grid=" << columns << "x" << rows
104 << "docked=" << docked.size()
105 << "floating=" << (docks.size() - docked.size());
106}
