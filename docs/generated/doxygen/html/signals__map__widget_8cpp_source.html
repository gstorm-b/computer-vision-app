127void SignalsMapWidget::populateEditor(int row) {
128 if (row < 0 || row >= m_rows.size()) return;
129 auto *edit = qobject_cast<QLineEdit *>(cellWidget(row, kColMapped));
130 if (!edit) return;
131
132 const RowState &st = m_rows[row];
133 const QStringList tags = tagsFor(st.type);
134
135 if (auto *c = edit->completer()) {
136 if (auto *m = qobject_cast<QStringListModel *>(c->model())) {
137 m->setStringList(tags);
138 }
139 }
140
141 {
142 QSignalBlocker block(edit);
143 edit->setText(st.currentTag);
144 }
145 const bool orphan = !st.currentTag.isEmpty() && !tags.contains(st.currentTag);
146 applyWarning(row, orphan);
147}

176void SignalsMapWidget::insertRowAt(int row,
177 const QString &internalName,
178 const QString &displayName,
179 Type type) {
180 if (rowIndexOf(internalName) >= 0) {
181 LOG_USER_WARN << "SignalsMapWidget: duplicate internalName" << internalName;
182 return;
183 }
184 row = qBound(0, row, m_rows.size());
185
186 RowState st{internalName, displayName, type, QString(), false};
187 m_rows.insert(row, st);
188
189 QTableWidget::insertRow(row);
190
191 auto *cellSignal = new QTableWidgetItem(displayName);
192 cellSignal->setFlags(cellSignal->flags() & ~Qt::ItemIsEditable);
193 cellSignal->setData(Qt::UserRole, internalName);
194 setItem(row, kColSignal, cellSignal);
195
196 auto *cellType = new QTableWidgetItem(type == Type::Bool ? tr("Bool")
197 : tr("Number"));
198 cellType->setFlags(cellType->flags() & ~Qt::ItemIsEditable);
199 cellType->setTextAlignment(Qt::AlignCenter);
200 setItem(row, kColType, cellType);
201
202 auto *edit = new TagLineEdit(this);
203 edit->setProperty(kPropInternalName, internalName);
204 edit->setClearButtonEnabled(true);
205
206 auto *model = new QStringListModel(tagsFor(type), edit);
207 auto *completer = new QCompleter(model, edit);
208 completer->setCaseSensitivity(Qt::CaseInsensitive);
209 completer->setFilterMode(Qt::MatchContains);
210 completer->setCompletionMode(QCompleter::PopupCompletion);
211 completer->popup()->setItemDelegate(new UsedTagDelegate(this, row, completer));
212 edit->setCompleter(completer);
213
214 setCellWidget(row, kColMapped, edit);
215
216 connect(edit, &QLineEdit::editingFinished,
217 this, &SignalsMapWidget::onLineEditingFinished);
218
219 populateEditor(row);
220}

299void SignalsMapWidget::onLineEditingFinished() {
300 if (m_suppressEdit) return;
301 auto *edit = qobject_cast<QLineEdit *>(sender());
302 if (!edit) return;
303 const QString internalName = edit->property(kPropInternalName).toString();
304 int row = rowIndexOf(internalName);
305 if (row < 0) return;
306
307 const QString newTag = edit->text().trimmed();
308 const QString oldTag = m_rows[row].currentTag;
309 if (newTag == oldTag) return;
310
311 // Conflict check applies only to non-empty selections.
312 if (!newTag.isEmpty()) {
313 const QString ownerDisplay = tagOwnerDisplay(newTag, row);
314 if (!ownerDisplay.isEmpty()) {
315 QMessageBox::StandardButton btn = QMessageBox::question(
316 this, tr("Reassign tag"),
317 tr("Tag \"%1\" is currently mapped to \"%2\".\n"
318 "Reassign it to \"%3\"?")
319 .arg(newTag, ownerDisplay, m_rows[row].displayName),
320 QMessageBox::Yes | QMessageBox::No,
321 QMessageBox::No);
322 if (btn != QMessageBox::Yes) {
323 m_suppressEdit = true;
324 edit->setText(oldTag);
325 m_suppressEdit = false;
326 return;
327 }
328 for (int j = 0; j < m_rows.size(); ++j) {
329 if (j == row) continue;
330 if (m_rows[j].currentTag == newTag) {
331 m_rows[j].currentTag.clear();
332 populateEditor(j);
333 applyWarning(j, true);
334 emit signalMappingChanged(m_rows[j].internalName, QString());
335 }
336 }
337 }
338 }
339
340 m_rows[row].currentTag = newTag;
341 // Free-typed values that aren't in the current list become orphan-warning;
342 // checkEmpty() will purge them at validation time.
343 const QStringList tags = tagsFor(m_rows[row].type);
344 const bool orphan = !newTag.isEmpty() && !tags.contains(newTag);
345 applyWarning(row, orphan);
346
347 refreshEditorDelegates();
348 emit signalMappingChanged(internalName, newTag);
349}
