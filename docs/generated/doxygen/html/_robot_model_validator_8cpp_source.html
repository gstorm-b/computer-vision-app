69ModelValidationResult RobotModelValidator::validateSerialRobotConfig(const SerialRobotConfig& config)
70{
71 ModelValidationResult result;
72
73 if (config.topology != RobotTopologyType::Serial) {
74 addIssue(result, "topology", "serial validator requires serial topology");
75 }
76
77 if (config.dof <= 0) {
78 addIssue(result, "dof", "serial robot dof must be positive");
79 }
80
81 if (config.joints.empty()) {
82 addIssue(result, "joints", "serial robot must contain ordered joints");
83 }
84
85 if (config.dof > 0 && movableJointCount(config) != config.dof) {
86 addIssue(result, "joints", "movable joint count must match configured dof");
87 }
88
89 std::set<std::string> linkIds;
90 for (std::size_t i = 0; i < config.links.size(); ++i) {
91 const Link& link = config.links[i];
92 const std::string field = "links[" + std::to_string(i) + "].id";
93 if (link.id.empty()) {
94 addIssue(result, field, "link id must not be empty");
95 continue;
96 }
97 if (!linkIds.insert(link.id).second) {
98 addIssue(result, field, "link id must be unique");
99 }
100 }
101
102 if (config.frames.baseLinkId.empty()) {
103 addIssue(result, "frames.base", "base link id must be set");
104 } else if (!containsLink(linkIds, config.frames.baseLinkId)) {
105 addIssue(result, "frames.base", "base link id must refer to a declared link");
106 }
107
108 if (config.frames.flangeLinkId.empty()) {
109 addIssue(result, "frames.flange", "flange link id must be set");
110 } else if (!containsLink(linkIds, config.frames.flangeLinkId)) {
111 addIssue(result, "frames.flange", "flange link id must refer to a declared link");
112 }
113
114 std::set<std::string> jointIds;
115 for (std::size_t i = 0; i < config.joints.size(); ++i) {
116 const Joint& joint = config.joints[i];
117 const std::string prefix = "joints[" + std::to_string(i) + "]";
118
119 if (joint.id.empty()) {
120 addIssue(result, prefix + ".id", "joint id must not be empty");
121 } else if (!jointIds.insert(joint.id).second) {
122 addIssue(result, prefix + ".id", "joint id must be unique");
123 }
124
125 if (!containsLink(linkIds, joint.parentLinkId)) {
126 addIssue(result, prefix + ".parent", "joint parent must refer to a declared link");
127 }
128 if (!containsLink(linkIds, joint.childLinkId)) {
129 addIssue(result, prefix + ".child", "joint child must refer to a declared link");
130 }
131
132 if (joint.type == JointType::Revolute || joint.type == JointType::Prismatic) {
133 const double axisNorm = joint.axis.norm();
134 if (!isFinite(joint.axis.x()) || !isFinite(joint.axis.y()) || !isFinite(joint.axis.z())) {
135 addIssue(result, prefix + ".axis", "joint axis values must be finite");
136 } else if (axisNorm <= kAxisNormTolerance) {
137 addIssue(result, prefix + ".axis", "joint axis must not be zero");
138 } else if (std::abs(axisNorm - 1.0) > kAxisNormTolerance) {
139 addIssue(result, prefix + ".axis", "joint axis must be unit length");
140 }
141
142 if (!joint.limits.has_value()) {
143 addIssue(result, prefix + ".limits", "moving joint must define position limits");
144 } else {
145 const JointLimits& limits = *joint.limits;
146 if (!isFinite(limits.lower) || !isFinite(limits.upper)) {
147 addIssue(result, prefix + ".limits", "joint limits must be finite");
148 } else if (limits.lower >= limits.upper) {
149 addIssue(result, prefix + ".limits", "joint lower limit must be less than upper limit");
150 } else if (joint.home < limits.lower - kLimitTolerance || joint.home > limits.upper + kLimitTolerance) {
151 addIssue(result, prefix + ".home", "joint home must be inside position limits");
152 }
153
154 if (limits.velocity.has_value() && (!isFinite(*limits.velocity) || *limits.velocity <= 0.0)) {
155 addIssue(result, prefix + ".limits.velocity", "joint velocity limit must be positive when set");
156 }
157 if (limits.acceleration.has_value() && (!isFinite(*limits.acceleration) || *limits.acceleration <= 0.0)) {
158 addIssue(result, prefix + ".limits.acceleration", "joint acceleration limit must be positive when set");
159 }
160 }
161 }
162 }
163
164 if (!config.joints.empty() && !config.frames.baseLinkId.empty() && !config.frames.flangeLinkId.empty()) {
165 if (config.joints.front().parentLinkId != config.frames.baseLinkId) {
166 addIssue(result, "joints[0].parent", "first serial joint parent must be the base link");
167 }
168 for (std::size_t i = 1; i < config.joints.size(); ++i) {
169 if (config.joints[i - 1].childLinkId != config.joints[i].parentLinkId) {
170 addIssue(result,
171 "joints[" + std::to_string(i) + "].parent",
172 "ordered serial joints must form a contiguous chain");
173 }
174 }
175 if (config.joints.back().childLinkId != config.frames.flangeLinkId) {
176 addIssue(result, "joints.back().child", "last serial joint child must be the flange link");
177 }
178 }
179
180 std::set<std::string> userFrameIds;
181 for (std::size_t i = 0; i < config.frames.userFrames.size(); ++i) {
182 const UserFrame& frame = config.frames.userFrames[i];
183 const std::string prefix = "frames.userFrames[" + std::to_string(i) + "]";
184 if (frame.id.empty()) {
185 addIssue(result, prefix + ".id", "user frame id must not be empty");
186 } else if (!userFrameIds.insert(frame.id).second) {
187 addIssue(result, prefix + ".id", "user frame id must be unique");
188 }
189 if (!containsLink(linkIds, frame.parentLinkId)) {
190 addIssue(result, prefix + ".parent", "user frame parent must refer to a declared link");
191 }
192 }
193
194 std::set<std::string> toolIds;
195 for (std::size_t i = 0; i < config.tools.size(); ++i) {
196 const Tool& tool = config.tools[i];
197 const std::string field = "tools[" + std::to_string(i) + "].id";
198 if (tool.id.empty()) {
199 addIssue(result, field, "tool id must not be empty");
200 } else if (!toolIds.insert(tool.id).second) {
201 addIssue(result, field, "tool id must be unique");
202 }
203 }
204
205 if (!config.defaultToolId.empty() && toolIds.find(config.defaultToolId) == toolIds.end()) {
206 addIssue(result, "defaultTool", "default tool must refer to a declared tool", KinematicsStatus::ToolNotFound);
207 }
208
209 return result;
210}
