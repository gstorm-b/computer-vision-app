10VirtualCameraDevice::VirtualCameraDevice(const QString &id, const QString &name,

34void VirtualCameraDevice::rebuildFrame()
35{
36 const QString path = m_config.imagePath();
37 if (!path.isEmpty()) {
38 if (QFileInfo::exists(path)) {
39 const cv::Mat loaded = cv::imread(path.toStdString(), cv::IMREAD_GRAYSCALE);
40 if (!loaded.empty()) {
41 m_frame = loaded;
42 return;
43 }
44 LOG_USER_WARN << "Virtual camera could not decode its still image." << path;
45 } else {
46 LOG_USER_WARN << "Virtual camera still image is missing." << path;
47 }
48 }
49
50 m_frame = cv::Mat(m_config.frameHeight(),
51 m_config.frameWidth(),
52 CV_8UC1,
53 cv::Scalar(m_config.frameGreyLevel()));
54}

58void VirtualCameraDevice::rebuildCalibration()
59{
60 const double mmPerPixel = m_config.millimetresPerPixel();
61 if (mmPerPixel <= 0.0 || m_frame.empty()) {
62 return; // Deliberately leaves the camera uncalibrated; the runtime will refuse it.
63 }
64
65 const double originX = m_config.originXMm();
66 const double originY = m_config.originYMm();
67 const double planeZ = static_cast<float>(m_config.workPlaneZMm());
68
69 // A 3x3 grid rather than the four corners: calibrate() needs at least 4 points, and a
70 // spread-out set keeps both the homography fit and the work-plane fit well conditioned
71 // instead of resting on the minimum.
72 std::vector<cv::Point2f> imagePts;
73 std::vector<cv::Point3f> robotPts;
74 for (int row = 0; row < 3; ++row) {
75 for (int col = 0; col < 3; ++col) {
76 const float px = static_cast<float>(col) * (m_frame.cols - 1) / 2.0f;
77 const float py = static_cast<float>(row) * (m_frame.rows - 1) / 2.0f;
78 imagePts.emplace_back(px, py);
79 robotPts.emplace_back(static_cast<float>(originX + px * mmPerPixel),
80 static_cast<float>(originY + py * mmPerPixel),
81 static_cast<float>(planeZ));
82 }
83 }
84
85 calib::Calibrator calibrator;
86 calibrator.addCorrespondences(imagePts, robotPts);
87 if (!calibrator.calibrate()) {
88 LOG_DEV_ERR << "Virtual camera could not fit its synthetic calibration.";
89 return;
90 }
91 m_config.setCalibrator(calibrator);
92}
