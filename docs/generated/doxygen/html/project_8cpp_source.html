83bool Project::addTask(ITask* task) {
84 if (!task) {
85 return false;
86 }
87
88 if (m_occupiedTaskNames.contains(task->name())){
89 return false;
90 }
91
92 std::shared_ptr<vc::model::ITask> task_ptr;
93 task_ptr.reset(task);
94 task_ptr->setProject(this);
95 m_tasks.insert(task_ptr->id(), task_ptr);
96 m_occupiedTaskNames.insert(task_ptr->name());
97
98 // connect signal — capture the task id by value, NOT the shared_ptr, so this
99 // connection does not keep the task alive past removeTask(). The connection
100 // auto-disconnects when the task (the sender) is destroyed.
101 const QString taskId = task_ptr->id();
102 connect(task_ptr.get(), &vc::model::ITask::configChanged, this, [this, taskId]() {
103 emit this->taskModified(taskId);
104 });
105
106 connect(task_ptr.get(), &vc::model::ITask::configChanged,
107 this, &Project::projectModificationOccurred);
108
109 emit tasksChanged();
110 emit taskCreated(task_ptr->id());
111 return true;
112}

247bool Project::fromJson(const QJsonObject &json) {
248 if (json.isEmpty()) {
249 return false;
250 }
251
252 m_name = json["name"].toString();
253 m_version = json["version"].toString();
254 QJsonArray taskArr = json["tasks"].toArray();
255 QJsonArray deviceArr = json["devices"].toArray();
256
257 // convert devices from json
258 for (int idx=0;idx<deviceArr.size();idx++) {
259 if (!deviceArr[idx].isObject()) {
260 continue;
261 }
262
263 QJsonObject obj = deviceArr[idx].toObject();
264 std::shared_ptr<device::IDevice> dv(device::DeviceFactory::fromJson(obj));
265 if (!dv) {
266 continue;
267 }
268
269 QString id = obj[DEVICE_JSK_ID].toString();
270 QString name = obj[DEVICE_JSK_NAME].toString();
271 m_deviceManager->reserveDevice(id, name, dv);
272 LOG_USER_INFO << "Device loaded from JSON. ID: " << id << ", Name: " << name;
273 }
274
275 // convert tasks from json
276 for (int idx=0;idx<taskArr.size();idx++) {
277 if (!taskArr[idx].isObject()) {
278 continue;
279 }
280
281 QJsonObject obj = taskArr[idx].toObject();
282 model::ITask* task = model::TaskFactory::fromJson(obj);
283 if (!task) {
284 continue;
285 }
286
287 this->addTask(task);
288 LOG_USER_INFO << "Task loaded from JSON. ID: " << task->id() << ", Name: " << task->name();
289 }
290
291 // assign devices to tasks
292 const QMap<QString, std::shared_ptr<vc::device::IDevice>>& devices = m_deviceManager->getCurrentDevices();
293 QStringList releaseDeviceIds;
294 for (const std::shared_ptr<vc::device::IDevice>& device : devices) {
295 qDebug() << device->id() << device->assignedTaskId();
296 QString assignedTaskId = device->assignedTaskId();
297 if (assignedTaskId.isEmpty()) {
298 releaseDeviceIds.append(device->id());
299 LOG_USER_ERR << "Device " << device->id() << " has no assigned task, deleting device.";
300 continue;
301 }
302
303 std::shared_ptr<vc::model::ITask> task = this->taskById(assignedTaskId);
304 if (task) {
305 task->assignDevice(device->id());
306 LOG_USER_INFO << "Assigned device " << device->id() << " to task " << assignedTaskId;
307 }
308 }
309
310 for (const QString &deviceId : releaseDeviceIds) {
311 m_deviceManager->releaseDevice(deviceId);
312 }
313
314 LOG_USER_INFO << "Project loaded from JSON successfully. Name: " << m_name << ", Version: " << m_version
315 << ", Devices: " << devices.size() << ", Tasks: " << m_tasks.size();
316 // convert succeesfully
317 return true;
318}

350bool Project::moveDeviceToTask(const QString &deviceId,
351 const QString &fromTaskId,
352 const QString &toTaskId)
353{
354 auto fromTask = taskById(fromTaskId);
355 auto toTask = taskById(toTaskId);
356 auto device = deviceById(deviceId);
357 if (!fromTask || !toTask) return false;
358 if (!fromTask->hasDevice(deviceId)) return false;
359 if (toTask->hasDevice(deviceId)) return false;
360
361 fromTask->unassignDevice(deviceId);
362 toTask->assignDevice(deviceId);
363 device->setAssignedTaskId(toTaskId);
364 emit projectModificationOccurred();
365 return true;
366}
