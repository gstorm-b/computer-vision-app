32FkChain ForwardKinematics::computeChain(const SerialRobotConfig& config, const JointVector& joints)
33{
34 FkChain chain;
35
36 Eigen::Isometry3d accumulated = Eigen::Isometry3d::Identity(); // base link pose in base
37 if (!config.frames.baseLinkId.empty()) {
38 chain.linkPosesInBase[config.frames.baseLinkId] = Pose::fromIsometry(accumulated);
39 }
40
41 int movableIndex = 0;
42 for (const Joint& joint : config.joints) {
43 // Joint frame in base, before applying the joint's own motion.
44 const Eigen::Isometry3d jointFrame = accumulated * joint.origin.isometry();
45
46 Eigen::Isometry3d motion = Eigen::Isometry3d::Identity();
47 if (joint.type == JointType::Revolute || joint.type == JointType::Prismatic) {
48 const double value = (movableIndex < joints.size()) ? joints[movableIndex] : 0.0;
49 const Eigen::Vector3d axisLocal = joint.axis.normalized();
50
51 JointFrameData data;
52 data.type = joint.type;
53 data.axisInBase = jointFrame.linear() * axisLocal;
54 data.originInBase = jointFrame.translation();
55 chain.joints.push_back(data);
56
57 if (joint.type == JointType::Revolute) {
58 motion.linear() = Eigen::AngleAxisd(value, axisLocal).toRotationMatrix();
59 } else {
60 motion.translation() = axisLocal * value;
61 }
62 ++movableIndex;
63 }
64
65 accumulated = jointFrame * motion; // child link pose in base
66 if (!joint.childLinkId.empty()) {
67 chain.linkPosesInBase[joint.childLinkId] = Pose::fromIsometry(accumulated);
68 }
69 }
70
71 chain.flangeInBase = Pose::fromIsometry(accumulated);
72 if (!config.frames.flangeLinkId.empty()) {
73 const auto it = chain.linkPosesInBase.find(config.frames.flangeLinkId);
74 if (it != chain.linkPosesInBase.end()) {
75 chain.flangeInBase = it->second;
76 }
77 }
78 return chain;
79}

100Result<Pose> ForwardKinematics::userFrameInBase(const FkChain& chain, const UserFrame& frame)
101{
102 const auto it = chain.linkPosesInBase.find(frame.parentLinkId);
103 if (it == chain.linkPosesInBase.end()) {
104 return Result<Pose>::failure(KinematicsStatus::FrameNotFound,
105 "user frame parent link not found: " + frame.parentLinkId);
106 }
107 return Result<Pose>::success(it->second * frame.transform);
108}
