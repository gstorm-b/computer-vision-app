38bool SettingsAdminCredentialProvider::verify(const QString &password) const
39{
40 if (!isConfigured()) {
41 // No credential to check against. Refuse rather than accept — an unconfigured
42 // gate that lets everyone through is worse than no gate, because it looks like one.
43 return false;
44 }
45
46 const AppSettings *settings = AppSettings::instance();
47 const QByteArray salt = QByteArray::fromHex(
48 settings->value(QLatin1String(AppKey::adminPasswordSalt)).toString().toLatin1());
49 const QByteArray stored = QByteArray::fromHex(
50 settings->value(QLatin1String(AppKey::adminPasswordHash)).toString().toLatin1());
51
52 const QByteArray candidate = hashWith(salt, password);
53
54 if (candidate.size() != stored.size()) {
55 return false;
56 }
57
58 // Constant-time over the digest. The timing signal on a local password dialog is not a
59 // realistic attack; a variable-time compare on a credential is the kind of detail that
60 // gets copied into somewhere it does matter.
61 quint8 diff = 0;
62 for (int i = 0; i < stored.size(); ++i) {
63 diff |= static_cast<quint8>(stored.at(i) ^ candidate.at(i));
64 }
65 return diff == 0;
66}

100QByteArray SettingsAdminCredentialProvider::makeSalt()
101{
102 static_assert(kSaltBytes % 4 == 0,

104 "multiple of 4");
105
106 QByteArray salt(kSaltBytes, Qt::Uninitialized);
