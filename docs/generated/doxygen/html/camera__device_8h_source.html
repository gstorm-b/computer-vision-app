45[[maybe_unused]] static QString CameraTypeToString(CameraType t) {
46 switch(t) {
47 case vc::device::CameraType::Realsense:
48 return CAM_TYPE_REALSENSE;
49 case vc::device::CameraType::BaslerGigE:
50 return CAM_TYPE_BASLER_GIGE;
51 case vc::device::CameraType::BaslerUSB:
52 return CAM_TYPE_BASLER_USB;
53 case vc::device::CameraType::VirtualCamera:
54 return CAM_TYPE_VIRTUAL;
55 case CamType:
56 return "";
57 }
58 return "";
59};

63[[maybe_unused]] static CameraType CameraTypeFromString(QString t) {

67 return CameraType::BaslerGigE;

69 return CameraType::BaslerUSB;
70 } else if (t == CAM_TYPE_VIRTUAL) {
71 return CameraType::VirtualCamera;
72 }
73 return CameraType::CamType;
