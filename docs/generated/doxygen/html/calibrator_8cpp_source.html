24bool fitPlane(const std::vector<cv::Point3f>& pts, cv::Vec4d& plane)
25{
26 if (pts.size() < 3) return false;
27
28 cv::Vec3d centroid(0.0, 0.0, 0.0);
29 for (const auto& p : pts) {
30 centroid[0] += p.x;
31 centroid[1] += p.y;
32 centroid[2] += p.z;
33 }
34 centroid *= (1.0 / static_cast<double>(pts.size()));
35
36 cv::Mat A(static_cast<int>(pts.size()), 3, CV_64F);
37 for (int i = 0; i < static_cast<int>(pts.size()); ++i) {
38 A.at<double>(i, 0) = pts[i].x - centroid[0];
39 A.at<double>(i, 1) = pts[i].y - centroid[1];
40 A.at<double>(i, 2) = pts[i].z - centroid[2];
41 }
42
43 cv::Mat w, u, vt;
44 cv::SVD::compute(A, w, u, vt, cv::SVD::MODIFY_A);
45 if (vt.rows < 3) return false;
46
47 const cv::Vec3d normal(vt.at<double>(2, 0),
48 vt.at<double>(2, 1),
49 vt.at<double>(2, 2));
50 const double nLen = std::sqrt(normal[0]*normal[0]
51 + normal[1]*normal[1]
52 + normal[2]*normal[2]);
53 if (nLen < 1e-9) return false;
54
55 plane[0] = normal[0] / nLen;
56 plane[1] = normal[1] / nLen;
57 plane[2] = normal[2] / nLen;
58 plane[3] = -(plane[0] * centroid[0]
59 + plane[1] * centroid[1]
60 + plane[2] * centroid[2]);
61
62 // Sign convention: keep the normal pointing toward +Z when possible so that
63 // c > 0 (a non-vertical work plane has a well-defined z = f(x, y)).
64 if (plane[2] < 0.0) {
65 plane[0] = -plane[0];
66 plane[1] = -plane[1];
67 plane[2] = -plane[2];
68 plane[3] = -plane[3];
69 }
70 return true;
71}

75double planeZ(const cv::Vec4d& plane, double x, double y)
76{
77 const double a = plane[0], b = plane[1], c = plane[2], d = plane[3];
78 if (std::abs(c) < 1e-12) return 0.0; // vertical plane: z undefined
79 return -(a * x + b * y + d) / c;
80}

164bool Calibrator::calibrate(double ransacReprojThreshold)
165{
166 m_calibrated = false;
167 m_H.release();
168 m_Hinv.release();
169
170 if (m_imagePts.size() < 4) return false;
171 if (m_imagePts.size() != m_robotPts.size()) return false;
172
173 // ---- 1. Homography from image px to robot XY ----
174 std::vector<cv::Point2f> robotXY;
175 robotXY.reserve(m_robotPts.size());
176 for (const auto& p : m_robotPts) robotXY.emplace_back(p.x, p.y);
177
178 cv::Mat H = cv::findHomography(m_imagePts, robotXY, cv::RANSAC, ransacReprojThreshold);
179 if (H.empty()) return false;
180
181 cv::Mat Hinv;
182 try { Hinv = H.inv(); } catch (const cv::Exception&) { return false; }
183 if (Hinv.empty()) return false;
184
185 m_H = H;
186 m_Hinv = Hinv;
187
188 // ---- 2. Work plane (3D) fit through robot 3D points ----
189 if (!fitPlane(m_robotPts, m_plane)) return false;
190
191 // ---- 3. Diagnostics ----
192 std::vector<cv::Point2f> projRobotXY, projImage;
193 cv::perspectiveTransform(m_imagePts, projRobotXY, m_H);
194 cv::perspectiveTransform(robotXY, projImage, m_Hinv);
195
196 double sumMm = 0.0, sumPx = 0.0, maxZErr = 0.0;
197 for (size_t i = 0; i < m_imagePts.size(); ++i) {
198 sumMm += cv::norm(projRobotXY[i] - robotXY[i]);
199 sumPx += cv::norm(projImage[i] - m_imagePts[i]);
200 const double zPredicted = planeZ(m_plane, m_robotPts[i].x, m_robotPts[i].y);
201 maxZErr = std::max(maxZErr, std::abs(zPredicted - m_robotPts[i].z));
202 }
203 const double n = static_cast<double>(m_imagePts.size());
204 m_reprojErrorMm = sumMm / n;
205 m_reprojErrorPx = sumPx / n;
206 m_planeFitMaxErr = maxZErr;
207
208 // ---- 4. Cache centroids used as anchors for angle conversions ----
209 cv::Point2f imgC(0.f, 0.f), robC(0.f, 0.f);
210 for (const auto& p : m_imagePts) imgC += p;
211 for (const auto& p : m_robotPts) robC += cv::Point2f(p.x, p.y);
212 m_imgCentroid = imgC * (1.0f / static_cast<float>(n));
213 m_robCentroid = robC * (1.0f / static_cast<float>(n));
214
215 m_calibrated = true;
216 return true;
217}

287double Calibrator::rotateImageToRobot(double angleImg, bool radians) const
288{
289 if (!m_calibrated) return 0.0;
290 const double theta = radians ? angleImg : angleImg * kDeg2Rad;
291
292 const float step = 100.0f; // arbitrary length; only angle of result matters
293 const cv::Point2f imgDir(static_cast<float>( std::cos(theta)),
294 static_cast<float>(-std::sin(theta)));
295 const cv::Point2f imgP0 = m_imgCentroid;
296 const cv::Point2f imgP1 = m_imgCentroid + imgDir * step;
297
298 std::vector<cv::Point2f> in{ imgP0, imgP1 }, out;
299 cv::perspectiveTransform(in, out, m_H);
300 const cv::Point2f robDir = out[1] - out[0];
301 const double angleRob = std::atan2(robDir.y, robDir.x);
302 return radians ? angleRob : angleRob * kRad2Deg;
303}

328cv::Point3f Calibrator::translateWithZAxis(const cv::Point3f &A, const cv::Point3f &offset, double theta, double isRad) const
329{
330 double theta_rad = theta;
331 if (!isRad) {
332 theta_rad = (CV_PI / 180.0) * theta;
333 }
334
335 double x_new = offset.x * cos(theta_rad) - offset.y * sin(theta_rad);
336 double y_new = offset.x * sin(theta_rad) + offset.y * cos(theta_rad);
337 double z_new = offset.z;
338
339 cv::Point3f B;
340 B.x = A.x + x_new;
341 B.y = A.y + y_new;
342 B.z = A.z + z_new;
343 return B;
344}

383bool Calibrator::readFromStorage(cv::FileStorage& fs)
384{
385 int calibFlag = 0;
386 if (!fs["calibrated"].empty()) fs["calibrated"] >> calibFlag;
387
388 cv::Mat H;
389 fs["homography_image_to_robot"] >> H;
390 if (H.empty() || H.rows != 3 || H.cols != 3) return false;
391
392 cv::Mat Hinv;
393 try { Hinv = H.inv(); } catch (const cv::Exception&) { return false; }
394 if (Hinv.empty()) return false;
395
396 cv::Mat planeMat;
397 fs["work_plane_abcd"] >> planeMat;
398 if (planeMat.empty() || planeMat.total() < 4) return false;
399 cv::Vec4d plane(planeMat.at<double>(0),
400 planeMat.at<double>(1),
401 planeMat.at<double>(2),
402 planeMat.at<double>(3));
403
404 cv::Mat imgC, robC;
405 cv::Point2f imgCentroid{}, robCentroid{};
406 fs["image_centroid"] >> imgC;
407 fs["robot_centroid"] >> robC;
408 if (!imgC.empty() && imgC.total() >= 2)
409 imgCentroid = cv::Point2f(imgC.at<float>(0), imgC.at<float>(1));
410 if (!robC.empty() && robC.total() >= 2)
411 robCentroid = cv::Point2f(robC.at<float>(0), robC.at<float>(1));
412
413 std::vector<cv::Point2f> imagePts;
414 std::vector<cv::Point3f> robotPts;
415 if (!fs["image_points"].empty()) fs["image_points"] >> imagePts;
416 if (!fs["robot_points"].empty()) fs["robot_points"] >> robotPts;
417
418 double reprojErrPx = -1.0;
419 double reprojErrMm = -1.0;
420 double planeFitMaxEr = -1.0;
421 fs["reprojection_error_px"] >> reprojErrPx;
422 fs["reprojection_error_mm"] >> reprojErrMm;
423 fs["plane_fit_max_error_mm"] >> planeFitMaxEr;
424
425 // Commit.
426 m_H = H;
427 m_Hinv = Hinv;
428 m_plane = plane;
429 m_imgCentroid = imgCentroid;
430 m_robCentroid = robCentroid;
431 m_imagePts = std::move(imagePts);
432 m_robotPts = std::move(robotPts);
433 m_reprojErrorPx = reprojErrPx;
434 m_reprojErrorMm = reprojErrMm;
435 m_planeFitMaxErr = planeFitMaxEr;
436 m_calibrated = (calibFlag != 0) || !m_H.empty();
437 return true;
438}
