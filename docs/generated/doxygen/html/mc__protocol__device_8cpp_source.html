19bool parseMcTag(const QString &tag, QChar expectedPrefix, int *address)
20{
21 if (!address) {
22 return false;
23 }
24
25 static const QRegularExpression re(QStringLiteral("^([MD])(\\d+)$"),
26 QRegularExpression::CaseInsensitiveOption);
27 const auto match = re.match(tag.trimmed());
28 if (!match.hasMatch()) {
29 return false;
30 }
31
32 const QChar prefix = match.captured(1).at(0).toUpper();
33 if (prefix != expectedPrefix) {
34 return false;
35 }
36
37 bool ok = false;
38 const int parsed = match.captured(2).toInt(&ok, 10);
39 if (!ok || parsed < 0) {
40 return false;
41 }
42
43 *address = parsed;
44 return true;
45}

84bool McProtocolDevice::deviceConnect() {
85 QMutexLocker locker(&m_mutex);
86
87 // Connect requests are idempotent. Re-running the full init while already
88 // connected would build a second transport over the live one, leaking the
89 // open socket and desyncing the single-connection MC session — exactly the
90 // "old connection not closed, cannot reconnect" failure on phase re-entry.
91 // Re-publish Connected so a runner that issued the redundant request still
92 // sees a status event and clears its busy flag.
93 if (connectStatus() == ConnectStatus::Connected) {
94 this->setConnectionStatus(ConnectStatus::Connected);
95 return true;
96 }
97
98 if (!this->initialize_mc_device()) {
99 // Must publish a status even here. PlcRunner::requestConnect() sets a busy flag
100 // that is only cleared by a status/failure signal, so returning false silently
101 // wedged the runner: every later reconnect request — automatic or manual — was
102 // dropped without a trace.
103 this->setConnectionStatus(ConnectStatus::ConnectFailed,
104 QStringLiteral("MC device initialization failed."));
105 return false;
106 }
107
108 McMsgInterface::MsgIfState msg_port_state = m_msg_interface->ConnectToPort();
109 if (msg_port_state != McMsgInterface::MsgIfState::Connected) {
110 const QString err = m_msg_interface->GetErrorDescription();
111 LOG_DEV_INFO << "connect error occurred," << err;
112 // Free the half-open transport so a later retry starts from a clean
113 // state instead of leaking the failed socket.
114 releaseConnectionResources();
115 this->setConnectionStatus(ConnectStatus::ConnectFailed, err);
116 return false;
117 }
118
119 LOG_USER_INFO << "MC Device connected";
120 this->setConnectionStatus(ConnectStatus::Connected);
121 m_polling_timer->start();
122 return true;
123}

138void McProtocolDevice::releaseConnectionResources() {
139 // Caller holds m_mutex. The polling timer is created once and reused across
140 // connect cycles; only stop it here (never delete) because this can run
141 // from inside the timer's own slot chain (e.g. send/receive failure paths).
142 if (m_polling_timer && m_polling_timer->isActive()) {
143 m_polling_timer->stop();
144 }
145
146 if (m_msg_interface) {
147 // DestroyMsgPort() closes and deletes the socket synchronously on the
148 // device's own thread (see McEthernetTcpPort). reset() then frees the
149 // interface; both are deterministic and run on the correct thread.
150 m_msg_interface->DestroyMsgPort();
151 m_msg_interface.reset();
152 }
153 m_frame.reset();
154}

335bool McProtocolDevice::initialize_mc_device() {
336 // Clear any transport left over from a previous (possibly failed) session
337 // before building a fresh one, so connect attempts never stack sockets.
338 releaseConnectionResources();
339
340 McFrameType frame_type = m_config.context()->frameType();
341 McMsgItfType msg_type = m_config.context()->msgConfig()->type();
342
343 // setup frame builder
344 switch (frame_type) {
345 case vc::device::mc::McFrameType::Frame_3E:
346 m_config.context()->setDeviceMap(&m_device_map);
347 m_frame = std::make_unique<Frame3E>();
348 LOG_DEV_INFO << "Initialize MC Protocol Frame 3E.";
349 break;
350 default:
351 return false;
352 }
353
354 // setup msg interface
355 switch (msg_type) {
356 case vc::device::mc::McMsgItfType::EthernetTCPIP:
357 m_msg_interface = std::make_unique<McEthernetTcpPort>();
358 m_msg_interface->SetConfig(m_config.context()->msgConfig());
359 LOG_DEV_INFO << "Initialize Ethernet TCP/IP message interface.";
360 break;
361 default:
362 return false;
363 }
364 connect(m_msg_interface->ioDevice(), &QIODevice::readyRead,
365 this, &McProtocolDevice::onMsgInterfaceReadReady);
366
367
368 McContext *ctx = m_config.context();
369 // Setup polling timer. Created once and reused across connect cycles so it
370 // is never deleted from inside its own slot chain (see
371 // releaseConnectionResources). Only the interval is refreshed per session.
372 if (!m_polling_timer) {
373 m_polling_timer = new QTimer(this);
374 m_polling_timer->setTimerType(Qt::PreciseTimer);
375 connect(m_polling_timer, &QTimer::timeout,
376 this, &McProtocolDevice::onPollingTimerTimeOut);
377 }
378 m_polling_timer->setInterval(ctx->m_refreshInterval);
379 LOG_DEV_INFO << "MC Protocol Device refresh interval:" << ctx->m_refreshInterval;
380
381 // setup polling device map
382 optimizeDeviceMap();
383
384 // setup state flag
385 m_wait_for_response = false;
386 is_first_time_polling = true;
387 m_update_command_index = 0;
388 m_data_update_state = DataQueryState::QueryTriggerByTimer;
389 // m_retry_by_timeout = false;
390 m_retry_count = 0;
391
392 m_current_request.reset();
393 m_request_queue.clear();
394 m_read_buffer.clear();
395
396 m_comm_active_m_device = m_config.context()->m_activeMDevice;
397
398 return true;
399}

407void McProtocolDevice::polling_query() {
408 if (m_wait_for_response) {
409 auto current_time_point = std::chrono::high_resolution_clock::now();
410 auto time_lasp = std::chrono::duration_cast<std::chrono::milliseconds>(current_time_point - m_sent_time_point);
411 if (time_lasp.count() > m_config.context()->msgConfig()->m_responseTimeout) {
412 // response timeout handle;
413 retry_request_handle();
414 }
415 return;
416 }
417
418 m_mutex.lock();
419 if (!m_request_queue.empty()) {
420 m_current_request = m_request_queue.takeFirst();
421 m_mutex.unlock();
422 // Ad-hoc commands (the heartbeat M-bit toggle) are queued exactly once
423 // per completed round, so the round-robin reads must resume right after
424 // this response, in the same tick, instead of waiting for the next
425 // timer tick.
426 m_data_update_state = DataQueryState::QueryContinue;
427 } else {
428 m_mutex.unlock();
429
430 if (m_polling_request_queue.isEmpty()) {
431 return;
432 }
433
434 m_current_request = m_polling_request_queue.at(m_update_command_index++);
435 if (m_update_command_index >= m_polling_request_queue.count()) {
436 m_update_command_index = 0;
437 m_data_update_state = DataQueryState::QueryFinished;
438 is_first_time_polling = false;
439 // emit device's value change signal
440 // LOG_DEV_DEBUG << "Device map updated";
441 emit pollingUpdate(m_device_map.clone());
442 onSetCommActiveDevice();
443 } else {
444 m_data_update_state = DataQueryState::QueryContinue;
445 }
446 }
447
448 request_handle();
449}

454void McProtocolDevice::request_handle() {
455 if (m_current_request != nullptr) {
456 QByteArray send_frame;
457
458 m_mutex.lock();
459 MCFrameAbstract::FrameReturnCode rt_code = m_frame->makeSendFrame(m_current_request.get(), m_config.context(), send_frame);
460 m_mutex.unlock();
461
462 if (rt_code == MCFrameAbstract::RequestFrameOK) {
463 McMsgInterface::MsgErrorState send_state = m_msg_interface->SendMsg(send_frame);
464 // LOG_DEV_DEBUG << "MC Protocol Device sent message.";
465
466 if (send_state != McMsgInterface::MsgErrorState::NoError) {
467
468 LOG_USER_ERR << tr("MC Device error, cannot send request. Retry %1.").arg(m_retry_count);
469 m_current_request.reset();
470 m_retry_count += 1;
471
472 if (m_retry_count >= 5) {
473 LOG_USER_ERR << tr("Mc Device error, retry send request over 5 times, connection lost.");
474 // LostConnected, not Disconnected. Disconnected means "an operator
475 // asked for this" and is deliberately NOT a recoverable status, so
476 // publishing it here made a pulled cable permanent: the localization
477 // recovery policy ignored it, never scheduled a reconnect, and never
478 // withdrew bTaskReady. A failed send is an unexpected loss.
479 setDeviceLostConnect();
480 }
481 return;
482 }
483
484 m_sent_time_point = std::chrono::high_resolution_clock::now();
485 m_wait_for_response = true;
486 // LOG_USER_INFO << "Send frame to PLC:" << send_frame.toHex(' ');
487 } else {
488 m_current_request.reset();
489 LOG_USER_ERR << "make request frame error";
490 }
491 }
492}

500void McProtocolDevice::response_handle() {
501 if (!m_wait_for_response) {
502 return;
503 }
504
505 if (m_current_request == nullptr) {
506 return;
507 }
508
509 QByteArray receive_bytes = m_read_buffer; // carry over any partial frame from a prior split read
510 McMsgInterface::MsgErrorState recieve_state = m_msg_interface->ReceiveMsg(receive_bytes, 1);
511 // OLOG_INFO << "Received frame from PLC:" << receive_bytes.toHex(' ');
512
513 if (recieve_state != McMsgInterface::MsgErrorState::NoError) {
514 if (recieve_state == McMsgInterface::MsgErrorState::BufferEmpty) {
515 LOG_USER_ERR << "empty frame";
516 return;
517 }
518 LOG_USER_ERR << "error while receive frame, error code" << recieve_state;
519 return;
520 }
521
522 m_mutex.lock();
523 MCFrameAbstract::FrameReturnCode rt_code = m_frame->parseReceiveFrame(m_current_request.get(), m_config.context(), receive_bytes);
524 m_mutex.unlock();
525
526 if (rt_code == MCFrameAbstract::ResponseOk) {
527 // OLOG_INFO << "reponse frame status error," << m_frame->lastErrorDescription();
528 check_device_changed();
529
530 } else if (rt_code == MCFrameAbstract::ResponseError) {
531 LOG_USER_ERR << "reponse frame status error," << m_frame->lastErrorDescription();
532 } else if (rt_code == MCFrameAbstract::ResponseInvalid) {
533 LOG_USER_ERR << "reponse frame invalid," << m_frame->lastErrorDescription();
534 } else if (rt_code == MCFrameAbstract::WaitingReceive) {
535 m_read_buffer = receive_bytes; // still incomplete; keep the accumulated bytes for the next read
536 return;
537 }
538
539 m_read_buffer.clear(); // frame fully consumed (ok/error/invalid) - start clean next time
540 m_retry_count = 0;
541 // m_retry_by_timeout = false;
542 m_wait_for_response = false;
543 m_current_request.reset();
544
545 if (m_data_update_state == DataQueryState::QueryContinue) {
546 // LOG_DEV_DEBUG << "Continue query ...";
547 polling_query();
548 }
549}

554void McProtocolDevice::retry_request_handle() {
555 m_retry_count += 1;
556 if (m_retry_count > 5) {
557 // LOG_USER_ERR << "MC Device Error, disconnect to device";
558 setDeviceLostConnect();
559 return;
560 }
561
562 // if (m_current_request) {
563 // LOG_USER_ERR << "MC Device Error, request timeout" << m_current_request->type();
564 // }
565 LOG_USER_INFO << "MC Device Error, timeout, retry" << m_retry_count;
566 m_msg_interface->clearBuffer();
567 m_read_buffer.clear(); // discard any partial frame from the abandoned attempt
568 m_wait_for_response = false;
569 request_handle();
570}

659void McProtocolDevice::check_device_changed() {
660 QMap<QString, QVariant> changedValues;
661 if ((!m_device_map.device_map_m.empty()) && (!m_last_device_M_map.empty())) {
662
663 if (m_device_map.device_map_m.size() != m_last_device_M_map.size()) {
664 update_last_m_map();
665 LOG_DEV_INFO << "Device M map size change:" << m_last_device_M_map.size();
666 }
667
668 std::map<int, quint8>::iterator it_new_m = m_device_map.device_map_m.begin();
669 std::map<int, quint8>::iterator it_last_m = m_last_device_M_map.begin();
670 for (; it_new_m != m_device_map.device_map_m.end(); ++it_new_m, ++it_last_m) {
671 if (it_new_m->second != it_last_m->second) {
672 if (!is_first_time_polling) {
673 // append to changed maps;
674 changedValues.insert(QString("M%1").arg(it_new_m->first, 4, 10, QChar('0')), it_new_m->second);
675
676 emit deviceMChanged(it_new_m->first, it_last_m->second, it_new_m->second);
677 }
678 it_last_m->second = it_new_m->second;
679 }
680 }
681 }
682
683 if ((m_device_map.device_map_d.empty()) || (m_last_device_D_map.empty())) {
684 return;
685 }
686 if (m_device_map.device_map_d.size() != m_last_device_D_map.size()) {
687 update_last_d_map();
688 LOG_DEV_INFO << "Device D map size change:" << m_last_device_D_map.size();
689 }
690
691 std::map<int, qint16>::iterator it_new_d = m_device_map.device_map_d.begin();
692 std::map<int, qint16>::iterator it_last_d = m_last_device_D_map.begin();
693 for (; it_new_d != m_device_map.device_map_d.end(); ++it_new_d, ++it_last_d) {
694 if (it_new_d->second != it_last_d->second) {
695 if (!is_first_time_polling) {
696 // append to changed maps;
697 changedValues.insert(QString("D%1").arg(it_new_d->first, 4, 10, QChar('0')), it_new_d->second);
698
699 emit deviceDChanged(it_new_d->first, it_last_d->second, it_new_d->second);
