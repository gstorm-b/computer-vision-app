43Eigen::Vector3d boundsCenter(const StlMeshStatistics& statistics)
44{
45 return Eigen::Vector3d(
46 0.5 * (statistics.minimumBounds_m[0] + statistics.maximumBounds_m[0]),
47 0.5 * (statistics.minimumBounds_m[1] + statistics.maximumBounds_m[1]),
48 0.5 * (statistics.minimumBounds_m[2] + statistics.maximumBounds_m[2]));
49}

52int longestAxisIndex(const StlMeshStatistics& statistics)
53{
54 int axisIndex = 0;
55 double best = statistics.axisLengths_m[0];
56 for (int i = 1; i < 3; ++i) {
57 if (statistics.axisLengths_m[static_cast<std::size_t>(i)] > best) {
58 best = statistics.axisLengths_m[static_cast<std::size_t>(i)];
59 axisIndex = i;
60 }
61 }
62 return axisIndex;
63}

67std::array<double, 3> rpyForAxisIndex(int axisIndex)
68{
69 if (axisIndex == 0) {
70 return {0.0, 0.5 * kPi, 0.0};
71 }
72 if (axisIndex == 1) {
73 return {-0.5 * kPi, 0.0, 0.0};
74 }
75 return {0.0, 0.0, 0.0};
76}

83ProposedGeometryData makeSphereProposal(const StlMeshStatistics& statistics,
84 const StlPrimitiveProposalRequest& request)
85{
86 const Eigen::Vector3d center = boundsCenter(statistics);
87 const Eigen::Vector3d axisLengths(
88 statistics.axisLengths_m[0],
89 statistics.axisLengths_m[1],
90 statistics.axisLengths_m[2]);
91
92 ProposedGeometryData proposal;
93 proposal.xyz_m = {center.x(), center.y(), center.z()};
94 proposal.rpy_rad = {0.0, 0.0, 0.0};
95 proposal.geometry.id = defaultIfEmpty(request.geometryId, "draft_geometry") + "_sphere";
96 proposal.geometry.linkId = defaultIfEmpty(request.linkId, "draft_link");
97 proposal.geometry.shape.type = CollisionShapeType::Sphere;
98 proposal.geometry.shape.sphere.radius_m = 0.5 * axisLengths.norm();
99 proposal.geometry.geometryToLink =
100 Pose::fromXYZRPY_m_rad(center.x(), center.y(), center.z(), 0.0, 0.0, 0.0);
101 proposal.geometry.margin_m = request.margin_m;
102 proposal.geometry.enabled = request.enabled;
103 return proposal;
104}

111ProposedGeometryData makeCapsuleProposal(const StlMeshStatistics& statistics,
112 const StlPrimitiveProposalRequest& request)
113{
114 const Eigen::Vector3d center = boundsCenter(statistics);
115 const int axisIndex = longestAxisIndex(statistics);
116
117 const double axisA = statistics.axisLengths_m[static_cast<std::size_t>((axisIndex + 1) % 3)];
118 const double axisB = statistics.axisLengths_m[static_cast<std::size_t>((axisIndex + 2) % 3)];
119 const double longestAxis = statistics.axisLengths_m[static_cast<std::size_t>(axisIndex)];
120 const double radius = 0.5 * std::sqrt(axisA * axisA + axisB * axisB);
121 const double length = std::max(0.0, longestAxis - 2.0 * radius);
122 const std::array<double, 3> rpy = rpyForAxisIndex(axisIndex);
123
124 ProposedGeometryData proposal;
125 proposal.xyz_m = {center.x(), center.y(), center.z()};
126 proposal.rpy_rad = rpy;
127 proposal.geometry.id = defaultIfEmpty(request.geometryId, "draft_geometry") + "_capsule";
128 proposal.geometry.linkId = defaultIfEmpty(request.linkId, "draft_link");
129 proposal.geometry.shape.type = CollisionShapeType::Capsule;
130 proposal.geometry.shape.capsule.radius_m = radius;
131 proposal.geometry.shape.capsule.length_m = length;
132 proposal.geometry.geometryToLink =
133 Pose::fromXYZRPY_m_rad(center.x(), center.y(), center.z(), rpy[0], rpy[1], rpy[2]);
134 proposal.geometry.margin_m = request.margin_m;
135 proposal.geometry.enabled = request.enabled;
136 return proposal;
137}

141QJsonObject toGeometryJson(const ProposedGeometryData& proposal)
142{
143 QJsonObject object;
144 object["id"] = QString::fromStdString(proposal.geometry.id);
145 object["link"] = QString::fromStdString(proposal.geometry.linkId);
146 object["enabled"] = proposal.geometry.enabled;
147 object["margin_m"] = proposal.geometry.margin_m;
148
149 QJsonObject geometryToLink;
150 geometryToLink["xyz_m"] =
151 QJsonArray{proposal.xyz_m[0], proposal.xyz_m[1], proposal.xyz_m[2]};
152 geometryToLink["rpy_rad"] =
153 QJsonArray{proposal.rpy_rad[0], proposal.rpy_rad[1], proposal.rpy_rad[2]};
154 object["geometryToLink"] = geometryToLink;
155
156 if (proposal.geometry.shape.type == CollisionShapeType::Sphere) {
157 object["shape"] = QStringLiteral("sphere");
158 QJsonObject sphere;
159 sphere["radius_m"] = proposal.geometry.shape.sphere.radius_m;
160 object["sphere"] = sphere;
161 } else {
162 object["shape"] = QStringLiteral("capsule");
163 QJsonObject capsule;
164 capsule["radius_m"] = proposal.geometry.shape.capsule.radius_m;
165 capsule["length_m"] = proposal.geometry.shape.capsule.length_m;
166 object["capsule"] = capsule;
167 }
168
169 return object;
170}

179std::string draftJsonFor(const ProposedGeometryData& proposal,
180 const StlPrimitiveProposalRequest& request,
181 const char* primitiveKind)
182{
183 QJsonObject root;
184 root["schema"] = QStringLiteral("robot-kinematics-collision/v1");
185
186 QJsonObject profile;
187 profile["id"] = QString::fromStdString(defaultIfEmpty(request.profileId, "draft_collision_profile"));
188 profile["robotModel"] = QString::fromStdString(defaultIfEmpty(request.robotModel, "draft_robot"));
189 QJsonObject units;
190 units["length"] = QStringLiteral("m");
191 units["angle"] = QStringLiteral("rad");
192 profile["units"] = units;
193 root["profile"] = profile;
194
195 root["geometries"] = QJsonArray{toGeometryJson(proposal)};
196 root["disabledPairs"] = QJsonArray{};
197
198 QJsonObject source;
199 source["type"] = QStringLiteral("stl_authoring_helper");
200 source["title"] = QStringLiteral("Draft primitive proposal from STL bounds");
201 source["reference"] = QStringLiteral("draft_manual_review_required");
202 source["appliesTo"] = QJsonArray{QStringLiteral("collision_geometry")};
203 root["sources"] = QJsonArray{source};
204
205 QJsonObject metadata;
206 metadata["reviewState"] = QStringLiteral("draft_manual_review_required");
207 metadata["helper"] = QStringLiteral("StlPrimitiveAuthoringHelper");
208 metadata["primitive"] = QString::fromLatin1(primitiveKind);
209 root["metadata"] = metadata;
210
211 return QJsonDocument(root).toJson(QJsonDocument::Indented).toStdString();
212}

219Result<StlPrimitiveProposal> StlPrimitiveAuthoringHelper::proposeFromFile(
220 const std::string& path,
221 const StlPrimitiveProposalRequest& request)
222{
223 if (request.margin_m < 0.0) {
224 return invalidRequest("STL primitive proposal margin must be non-negative");
225 }
226
227 const Result<TriangleMesh> loaded =
228 StlMeshLoader::loadFile(path, StlMeshLoadOptions{1.0, true});
229 if (!loaded.ok()) {
230 return invalidRequest(loaded.message);
231 }
232
233 Eigen::Vector3d centroid = Eigen::Vector3d::Zero();
234 for (const Eigen::Vector3d& vertex : loaded.value.vertices_m) {
235 centroid += vertex;
236 }
237 centroid /= static_cast<double>(loaded.value.vertices_m.size());
238
239 const TriangleMeshStatistics& meshStatistics = loaded.value.statistics;
240 StlMeshStatistics statistics;
241 statistics.triangleCount = meshStatistics.triangleCount;
242 statistics.minimumBounds_m = meshStatistics.minimumBounds_m;
243 statistics.maximumBounds_m = meshStatistics.maximumBounds_m;
244 statistics.centroid_m = {centroid.x(), centroid.y(), centroid.z()};
245 statistics.axisLengths_m = {
246 meshStatistics.maximumBounds_m[0] - meshStatistics.minimumBounds_m[0],
247 meshStatistics.maximumBounds_m[1] - meshStatistics.minimumBounds_m[1],
248 meshStatistics.maximumBounds_m[2] - meshStatistics.minimumBounds_m[2],
249 };
250
251 const ProposedGeometryData sphere = makeSphereProposal(statistics, request);
252 const ProposedGeometryData capsule = makeCapsuleProposal(statistics, request);
253
254 StlPrimitiveProposal proposal;
255 proposal.format = loaded.value.sourceFormat;
256 proposal.statistics = statistics;
257 proposal.sphere = sphere.geometry;
258 proposal.capsule = capsule.geometry;
259 proposal.sphereDraftJson = draftJsonFor(sphere, request, "sphere");
260 proposal.capsuleDraftJson = draftJsonFor(capsule, request, "capsule");
261 proposal.reviewNote = "draft_manual_review_required";
262
263 return Result<StlPrimitiveProposal>::success(proposal);
264}
