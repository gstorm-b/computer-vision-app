261QString subTypeValueFrom(const QJsonObject &obj, const QString &configJsonKey)
262{
263 if (configJsonKey.isEmpty()) {
264 return QString();
265 }
266
267 if (obj.contains(configJsonKey)) {
268 return obj[configJsonKey].toString();
269 }
270
271 const QJsonObject cfg = obj.value(DEVICE_JSK_CONFIG).toObject();
272 return cfg.value(configJsonKey).toString();
273}

293const DeviceRegistryEntry *DeviceRegistry::find(const QJsonObject &obj,
294 DeviceType deviceType)
295{
296 for (const DeviceRegistryEntry &entry : kEntries) {
297 if (entry.deviceType != deviceType) {
298 continue;
299 }
300
301 if (subTypeValueFrom(obj, entry.configJsonKey) == entry.subTypeValue) {
302 return &entry;
303 }
304 }
305 return nullptr;
306}

308QStringList DeviceRegistry::displayNamesFor(DeviceType deviceType,
309 bool includeUnavailable)
310{
311 QStringList names;
312 for (const DeviceRegistryEntry &entry : kEntries) {
313 if (entry.deviceType != deviceType) {
314 continue;
315 }
316 if (!includeUnavailable && !entry.available) {
317 continue;
318 }
319 names.append(entry.subTypeValue);
320 }
321 return names;
322}
