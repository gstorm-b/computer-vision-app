32Pose Pose::fromXYZRPY_m_rad(double x_m, double y_m, double z_m,
33 double roll_rad, double pitch_rad, double yaw_rad)
34{
35 Eigen::Isometry3d transform = Eigen::Isometry3d::Identity();
36 transform.translation() = Eigen::Vector3d(x_m, y_m, z_m);
37 transform.linear() =
38 (Eigen::AngleAxisd(yaw_rad, Eigen::Vector3d::UnitZ())
39 * Eigen::AngleAxisd(pitch_rad, Eigen::Vector3d::UnitY())
40 * Eigen::AngleAxisd(roll_rad, Eigen::Vector3d::UnitX()))
41 .toRotationMatrix();
42 return Pose(transform);
43}
