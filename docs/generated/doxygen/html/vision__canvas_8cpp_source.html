123QGraphicsRectItem *addOverlayChip(QGraphicsScene *scene,
124 const QPointF &scenePos,
125 const QString &text,
126 const QColor &textColor,
127 qreal zValue = 3.0)
128{
129 if (!scene || text.isEmpty()) return nullptr;
130
131 auto *textItem = new QGraphicsSimpleTextItem(text);
132 textItem->setBrush(QBrush(textColor));
133
134 const QRectF textBounds = textItem->boundingRect();
135 constexpr qreal kPaddingX = 6.0;
136 constexpr qreal kPaddingY = 3.0;
137
138 auto *chipItem = new QGraphicsRectItem(
139 QRectF(0.0, 0.0,
140 textBounds.width() + kPaddingX * 2.0,
141 textBounds.height() + kPaddingY * 2.0));
142 chipItem->setBrush(QColor(0, 0, 0, 176));
143 chipItem->setPen(cosmeticPen(QColor(255, 255, 255, 64), 1.0));
144 chipItem->setPos(scenePos);
145 chipItem->setFlag(QGraphicsItem::ItemIgnoresTransformations, true);
146 chipItem->setZValue(zValue);
147 scene->addItem(chipItem);
148
149 textItem->setParentItem(chipItem);
150 textItem->setPos(kPaddingX, kPaddingY);
151 return chipItem;
152}

166QPainterPath arrowPath(const QPointF &direction, qreal length, qreal headSize)
167{
168 const QPointF unit = normalizeVector(direction);
169 if (qFuzzyIsNull(unit.x()) && qFuzzyIsNull(unit.y())) {
170 return {};
171 }
172
173 const QPointF tip = unit * length;
174 const QPointF normal(-unit.y(), unit.x());
175
176 QPainterPath path;
177 path.moveTo(0.0, 0.0);
178 path.lineTo(tip);
179 path.moveTo(tip.x(), tip.y());
180 path.lineTo(tip.x() - unit.x() * headSize + normal.x() * headSize * 0.55,
181 tip.y() - unit.y() * headSize + normal.y() * headSize * 0.55);
182 path.moveTo(tip.x(), tip.y());
183 path.lineTo(tip.x() - unit.x() * headSize - normal.x() * headSize * 0.55,
184 tip.y() - unit.y() * headSize - normal.y() * headSize * 0.55);
185 return path;
186}

469void VisionCanvas::setImage(const QPixmap &pixmap)
470{
471 if (pixmap.isNull()) {
472 clearImage();
473 return;
474 }
475
476 // Only re-fit the view when the incoming image size changes. Successive
477 // frames of the same resolution (e.g. a live camera feed) keep the user's
478 // current zoom/pan instead of snapping back to fit on every frame.
479 const QSize newSize = pixmap.size();
480 const bool sizeChanged = (newSize != m_fittedImageSize);
481
482 if (!m_imageItem) {
483 m_imageItem = m_scene->addPixmap(pixmap);
484 m_imageItem->setTransformationMode(Qt::FastTransformation);
485 } else {
486 m_imageItem->setPixmap(pixmap);
487 }
488
489 m_scene->setSceneRect(pixmap.rect());
490 rebuildEditableRois(editableRois());
491 rebuildAuxiliaryRois();
492 rebuildOverlayItems();
493
494 if (sizeChanged) {
495 m_fittedImageSize = newSize;
496 fitImageToView();
497 }
498}

503void VisionCanvas::clearImage()
504{
505 for (vision_canvas_detail::RoiItemBase *item : m_editableItems) {
506 delete item;
507 }
508 m_editableItems.clear();
509
510 if (m_imageItem) {
511 m_scene->removeItem(m_imageItem);
512 delete m_imageItem;
513 m_imageItem = nullptr;
514 }
515 // Reset so the next image always re-fits, even if it happens to match the
516 // size of the one that was just cleared.
517 m_fittedImageSize = QSize();
518 if (m_drawRectItem) {
519 delete m_drawRectItem;
520 m_drawRectItem = nullptr;
521 }
522 m_history.clear();
523 m_history.append(QVector<VisionRoi>{});
524 m_historyIndex = 0;
525 emit undoAvailabilityChanged(false, false);
526 m_auxiliaryRois.clear();
527 m_overlay = VisionResultOverlay();
528 m_selectedResultObject = -1;
529 m_hoveredResultObject = -1;
530 rebuildAuxiliaryRois();
531 rebuildOverlayItems();
532}

641int VisionCanvas::resultObjectAtScenePoint(const QPointF &scenePoint) const
642{
643 auto matchIndex = [scenePoint](const QVector<VisionResultObject> &objects, int preferredIndex) {
644 if (preferredIndex > 0) {
645 for (const VisionResultObject &object : objects) {
646 if (object.index == preferredIndex
647 && QPolygonF(object.corners).containsPoint(scenePoint, Qt::OddEvenFill)) {
648 return object.index;
649 }
650 }
651 }
652
653 for (auto it = objects.crbegin(); it != objects.crend(); ++it) {
654 if (QPolygonF(it->corners).containsPoint(scenePoint, Qt::OddEvenFill)) {
655 return it->index;
656 }
657 }
658 return -1;
659 };
660
661 const int acceptedHit = matchIndex(m_overlay.acceptedObjects, m_selectedResultObject);
662 if (acceptedHit > 0) return acceptedHit;
663
664 if (m_overlayVisibility.showRejectedCandidates) {
665 return matchIndex(m_overlay.rejectedObjects, m_selectedResultObject);
666 }
667
668 return -1;
669}

780{
781 setFocus(Qt::MouseFocusReason);
782
783 if (event->modifiers() & Qt::ControlModifier) {

805 && event->button() == Qt::LeftButton
806 && m_imageItem) {
807 beginDraw(clampScenePointToImage(mapToScene(event->pos())));
808 return;
809 }
810
811 QGraphicsView::mousePressEvent(event);

928void VisionCanvas::rebuildAuxiliaryRois()
929{
930 for (QGraphicsItem *item : m_auxiliaryItems) {
931 delete item;
932 }
933 m_auxiliaryItems.clear();
934
935 if (!m_imageItem) return;
936
937 for (const VisionRoi &roi : m_auxiliaryRois) {
938 if (!roi.visible || !roi.isValid()) continue;
939 auto *item = m_scene->addPolygon(
940 vision::roiPolygon(roi),
941 cosmeticPen(roi.color.isValid() ? roi.color
942 : tokenColor(QStringLiteral("state.info")),
943 2.0,
944 Qt::DashLine));
945 m_auxiliaryItems.append(item);
946
947 if (!roi.label.isEmpty()) {
948 const QPolygonF polygon = vision::roiPolygon(roi);
949 const QRectF bounds = vision::boundingRectForPoints(polygon.toVector());
950 if (auto *chipItem = addOverlayChip(
951 m_scene,
952 labelAnchorForBounds(bounds),
953 roi.label,
954 roi.color.isValid() ? roi.color : QColor(Qt::white))) {
955 m_auxiliaryItems.append(chipItem);
956 }
957 }
958 }

967{
968 for (QGraphicsItem *item : m_overlayItems) {
969 delete item;
970 }
971 m_overlayItems.clear();
972
