138Result<MeshCollisionProfile> MeshCollisionProfileJsonLoader::loadFile(const std::string& path)
139{
140 QFile file(QString::fromStdString(path));
141 if (!file.open(QIODevice::ReadOnly)) {
142 return Result<MeshCollisionProfile>::failure(KinematicsStatus::InvalidRequest,
143 "mesh collision profile file could not be opened");
144 }
145
146 Result<MeshCollisionProfile> loaded = loadJson(QString::fromUtf8(file.readAll()).toStdString());
147 if (!loaded.ok()) {
148 return loaded;
149 }
150
151 const QDir baseDir = QFileInfo(QString::fromStdString(path)).absoluteDir();
152 for (MeshCollisionGeometry& mesh : loaded.value.meshes) {
153 QFileInfo meshPath(QString::fromStdString(mesh.path));
154 if (!meshPath.isAbsolute()) {
155 mesh.path = QDir::cleanPath(baseDir.absoluteFilePath(meshPath.filePath())).toStdString();
156 }
157 }
158
159 return loaded;
160}

168Result<MeshCollisionProfile> MeshCollisionProfileJsonLoader::loadJson(const std::string& json)
169{
170 QJsonParseError error;
171 const QJsonDocument document = QJsonDocument::fromJson(QByteArray::fromStdString(json), &error);
172 if (error.error != QJsonParseError::NoError || !document.isObject()) {
173 return invalid("mesh collision profile JSON is not a valid object");
174 }
175
176 const QJsonObject root = document.object();
177 if (!hasOnlyKnownTopLevelFields(root)) {
178 return invalid("mesh collision profile contains unknown top-level fields");
179 }
180 if (root.value("schema").toString() != "robot-kinematics-collision-mesh/v1") {
181 return invalid("unsupported mesh collision profile schema");
182 }
183
184 const QJsonObject profileObject = root.value("profile").toObject();
185 const QJsonObject units = profileObject.value("units").toObject();
186 if (units.value("length").toString() != "m" || units.value("angle").toString() != "rad") {
187 return invalid("mesh collision profile units must be m/rad");
188 }
189
190 MeshCollisionProfile profile;
191 profile.id = stringField(profileObject, "id");
192 profile.robotModel = stringField(profileObject, "robotModel");
193
194 for (const QJsonValue& backendValue : profileObject.value("backendPreference").toArray()) {
195 const Result<MeshCollisionBackendKind> backend =
196 meshCollisionBackendKindFromString(backendValue.toString().toStdString());
197 if (!backend.ok()) {
198 return invalid(backend.message);
199 }
200 profile.backendPreference.push_back(backend.value);
201 }
202
203 std::set<std::string> meshIds;
204 for (const QJsonValue& meshValue : root.value("meshes").toArray()) {
205 const QJsonObject meshObject = meshValue.toObject();
206
207 if (!meshObject.contains("meshToLink")) {
208 return invalid("mesh collision geometry must include meshToLink");
209 }
210 const QJsonObject meshToLinkObject = meshObject.value("meshToLink").toObject();
211 if (!hasFiniteNumericArray(meshToLinkObject, "xyz_m") ||
212 !hasFiniteNumericArray(meshToLinkObject, "rpy_rad")) {
213 return invalid("mesh collision geometry meshToLink must include xyz_m and rpy_rad arrays");
214 }
215 if (!hasFiniteNumericField(meshObject, "scaleToMeters")) {
216 return invalid("mesh collision geometry scaleToMeters must be a finite number");
217 }
218 if (meshObject.contains("margin_m") && !isFiniteNumber(meshObject.value("margin_m"))) {
219 return invalid("mesh collision geometry margin_m must be a finite number");
220 }
221 if (meshObject.contains("enabled") && !meshObject.value("enabled").isBool()) {
222 return invalid("mesh collision geometry enabled must be a boolean");
223 }
224
225 MeshCollisionGeometry mesh;
226 mesh.id = stringField(meshObject, "id");
227 mesh.linkId = stringField(meshObject, "link");
228 mesh.path = stringField(meshObject, "path");
229 mesh.scaleToMeters = meshObject.value("scaleToMeters").toDouble();
230 mesh.meshToLink = poseFromObject(meshToLinkObject);
231 mesh.margin_m = meshObject.value("margin_m").toDouble();
232 mesh.enabled = !meshObject.contains("enabled") || meshObject.value("enabled").toBool();
233
234 const Result<MeshFileFormat> format =
235 parseFormat(meshObject.value("format").toString());
236 if (!format.ok()) {
237 return invalid(format.message);
238 }
239 mesh.format = format.value;
240
241 const Result<MeshSourceUnits> sourceUnits =
242 parseSourceUnits(meshObject.value("sourceUnits").toString());
243 if (!sourceUnits.ok()) {
244 return invalid(sourceUnits.message);
245 }
246 mesh.sourceUnits = sourceUnits.value;
247
248 const QJsonObject qualityObject = meshObject.value("quality").toObject();
249 const Result<MeshQualityMode> qualityMode =
250 parseQualityMode(qualityObject.value("mode").toString());
251 if (!qualityMode.ok()) {
252 return invalid(qualityMode.message);
253 }
254 mesh.quality.mode = qualityMode.value;
255 if (qualityObject.contains("triangleCount") && !qualityObject.value("triangleCount").isNull()) {
256 const QJsonValue triangleCount = qualityObject.value("triangleCount");
257 const double rawCount = triangleCount.toDouble(-1.0);
258 if (!triangleCount.isDouble() || !std::isfinite(rawCount) || rawCount < 1.0 ||
259 std::floor(rawCount) != rawCount ||
260 rawCount > static_cast<double>((std::numeric_limits<std::size_t>::max)())) {
261 return invalid("mesh quality triangleCount must be a positive integer when provided");
262 }
263 mesh.quality.triangleCount = static_cast<std::size_t>(rawCount);
264 }
265 if (qualityObject.contains("simplifiedFrom") && !qualityObject.value("simplifiedFrom").isNull()) {
266 if (!qualityObject.value("simplifiedFrom").isString()) {
267 return invalid("mesh quality simplifiedFrom must be a string when provided");
268 }
269 mesh.quality.simplifiedFrom = qualityObject.value("simplifiedFrom").toString().toStdString();
270 }
271 if (qualityObject.contains("maxSimplificationError_m") &&
272 !qualityObject.value("maxSimplificationError_m").isNull()) {
273 if (!isFiniteNumber(qualityObject.value("maxSimplificationError_m"))) {
274 return invalid("mesh quality maxSimplificationError_m must be a finite number when provided");
275 }
276 mesh.quality.maxSimplificationError_m =
277 qualityObject.value("maxSimplificationError_m").toDouble();
278 }
279
280 if (!meshIds.insert(mesh.id).second) {
281 return invalid("mesh collision geometry ids must be unique");
282 }
283
284 profile.meshes.push_back(mesh);
285 }
286
287 for (const QJsonValue& pairValue : root.value("disabledPairs").toArray()) {
288 const QJsonObject pairObject = pairValue.toObject();
289 const std::string geometryA = stringField(pairObject, "a");
290 const std::string geometryB = stringField(pairObject, "b");
291
292 if (meshIds.find(geometryA) == meshIds.end() || meshIds.find(geometryB) == meshIds.end()) {
293 return invalid("mesh collision disabledPairs must refer to existing mesh ids");
294 }
295
296 profile.disabledPairs.push_back(DisabledCollisionPair{
297 geometryA,
298 geometryB,
299 stringField(pairObject, "reason"),
300 });
301 }
302
303 for (const QJsonValue& sourceValue : root.value("sources").toArray()) {
304 const QJsonObject sourceObject = sourceValue.toObject();
305 SourceReference source;
306 source.type = stringField(sourceObject, "type");
307 source.title = stringField(sourceObject, "title");
308 source.reference = stringField(sourceObject, "reference");
309 for (const QJsonValue& appliesTo : sourceObject.value("appliesTo").toArray()) {
310 source.appliesTo.push_back(appliesTo.toString().toStdString());
311 }
312 profile.sources.push_back(source);
313 }
314
315 const QJsonObject metadata = root.value("metadata").toObject();
316 for (auto it = metadata.begin(); it != metadata.end(); ++it) {
317 if (it.value().isString()) {
318 profile.metadata[it.key().toStdString()] = it.value().toString().toStdString();
319 }
320 }
321
322 return Result<MeshCollisionProfile>::success(profile);
323}
