48void LocalizationSignalMapper::configure(const TaskLocalizeConfig &config)
49{
50 m_tagToSignalName.clear();
51 m_signalNameToTag.clear();
52
53 for (const char *fieldName : kSignalFields) {
54 const QString tag = readSignalTag(config, fieldName).trimmed();
55 if (tag.isEmpty())
56 continue;
57
58 if (m_tagToSignalName.contains(tag)) {
59 LOG_DEV_ERR << "LocalizationSignalMapper: duplicate PLC tag"
60 << tag
61 << "existing signal"
62 << m_tagToSignalName.value(tag)
63 << "new signal"
64 << fieldName;
65 }
66
67 const QString signalName = QString::fromUtf8(fieldName);
68 m_tagToSignalName.insert(tag, signalName);
69 m_signalNameToTag.insert(signalName, tag);
70 }
71}
