17inline QPixmap cvMatToQPixmap(const cv::Mat& mat) {
18 QImage qimg;
19 if (mat.type() == CV_8UC1) {
20 // Grayscale
21 qimg = QImage(mat.data,
22 mat.cols,
23 mat.rows,
24 static_cast<int>(mat.step),
25 QImage::Format_Grayscale8);
26 } else if (mat.type() == CV_8UC3) {
27 // OpenCV uses BGR, Qt uses RGB
28 cv::Mat rgb;
29 cv::cvtColor(mat, rgb, cv::COLOR_BGR2RGB);
30 qimg = QImage(rgb.data,
31 rgb.cols,
32 rgb.rows,
33 static_cast<int>(rgb.step),
34 QImage::Format_RGB888).copy();
35 } else if (mat.type() == CV_8UC4) {
36 // BGRA → RGBA
37 cv::Mat rgba;
38 cv::cvtColor(mat, rgba, cv::COLOR_BGRA2RGBA);
39 qimg = QImage(rgba.data,
40 rgba.cols,
41 rgba.rows,
42 static_cast<int>(rgba.step),
43 QImage::Format_RGBA8888).copy();
44 } else {
45 return QPixmap();
46 }
47 return QPixmap::fromImage(qimg);
48}

61static QtVariantProperty* addPropertyToBrowser(const QMetaObject &meta, QMetaProperty &prop, QVariant &value,
62 QtVariantPropertyManager *manager, QtTreePropertyBrowser *browser) {
63
64 QString propName = prop.name();
65 if (propName == "objectName") {
66 // ignore objectName
67 return nullptr;
68 }
69
70 QtVariantProperty *variantProp = nullptr;
71 // enum type Check
72 if (prop.isEnumType()) {
73 variantProp = manager->addProperty(QtVariantPropertyManager::enumTypeId(), propName);
74
75 // get enum names
76 QStringList enumNames;
77 QMetaEnum metaEnum = prop.enumerator();
78 for (int j = 0; j < metaEnum.keyCount(); ++j) {
79 const char* key = metaEnum.key(j);
80 QString translatedName = QCoreApplication::translate(meta.className(), key);
81 enumNames << translatedName;
82 }
83 variantProp->setAttribute(QLatin1String("enumNames"), enumNames);
84 // set enum value
85 variantProp->setValue(value.toInt());
86
87 } else {
88 // normal data types handle (int, QString, bool, QColor, v.v.)
89 int typeId = value.userType();
90 variantProp = manager->addProperty(typeId, propName);
91 if (variantProp) {
92 variantProp->setValue(value);
93 }
94 }
95
96 if (!variantProp) {
97 return nullptr;
98 }
99
100 int displayNameIdx = meta.indexOfClassInfo(QString("%1_name").arg(propName).toUtf8());
101 if (displayNameIdx != -1) {
102 const QString displayName = meta.classInfo(displayNameIdx).value();
103 if (!displayName.isEmpty()) {
104 variantProp->setDisplayName(displayName);
105 }
106 }
107
108 // --- set attributes---
109 int typeId = value.userType();
110 int minIdx = meta.indexOfClassInfo(QString("%1_min").arg(propName).toUtf8());
111 if (minIdx != -1) {
112 if (typeId == QMetaType::Int) {
113 variantProp->setAttribute("minimum", QString(meta.classInfo(minIdx).value()).toInt());
114 } else if (typeId == QMetaType::Double) {
115 variantProp->setAttribute("minimum", QString(meta.classInfo(minIdx).value()).toDouble());
116 }
117 }
118
119 int maxIdx = meta.indexOfClassInfo(QString("%1_max").arg(propName).toUtf8());
120 if (maxIdx != -1) {
121 if (typeId == QMetaType::Int) {
122 variantProp->setAttribute("maximum", QString(meta.classInfo(maxIdx).value()).toInt());
123 } else if (typeId == QMetaType::Double) {
124 variantProp->setAttribute("maximum", QString(meta.classInfo(maxIdx).value()).toDouble());
125 }
126 }
127
128 if (!prop.isWritable()) {
129 variantProp->setEnabled(false);
130 }
131
132 return variantProp;
133}

313void BaslerCameraWidget::onPropertyValueChanged(QtProperty *property, const QVariant &variant) {
314 if (m_populating_browser) {
315 return;
316 }
317
318 QString propName = property->propertyName();
319
320 const QMetaObject *idevice_meta = m_device->metaObject();
321 const QMetaObject &meta_cfg = m_params.getMetaObject();
322
323 // abstract device
324 int index = idevice_meta->indexOfProperty(propName.toUtf8());
325 if (index != -1) {
326 if (propName == "name") {
327 QString new_name = variant.toString();
328 // avoid loop made by signal valueChanged
329 if (m_device->name() == new_name) {

333 if (!m_device->deviceManager()->changeDeviceName(m_device->id(), new_name)) {
334 LOG_USER_WARN << tr("Cannot rename device to \"%1\": the name is "
335 "already in use.").arg(new_name);
336 m_variantManager->setValue(property, m_device->name());
337 }
338 }

340 }
341
342 // parameters
343 index = meta_cfg.indexOfProperty(propName.toUtf8());
344 if (index != -1) {
345 QMetaProperty mProp = meta_cfg.property(index);
346 mProp.writeOnGadget(&m_params, variant);
347 qDebug() << "Updated context: " << propName << "to" << variant;
348 this->m_camera->setBaslerGigeConfig(m_params);
349 if (this->m_camera->isDeviceConnected() && m_runner) {
350 m_runner->requestApplyParams();

352 return;

459void BaslerCameraWidget::onCameraGrabFinished(vc::device::GrabResult result) {
460 if (result.isGrabSuccess) {
461 QPixmap new_frame = cvMatToQPixmap(result.frame);
462 ui->image_view->loadImage(new_frame);
463 }
464
465 // Detect request initiated by btn_calib_detect: run board detection on the
466 // freshly-grabbed frame and push the 4 corner points into the table.
467 if (m_pendingCalibDetect) {
468 m_pendingCalibDetect = false;
469
470 if (!result.isGrabSuccess) {
471 ui->label_calib_status->setText(tr("Calibration status: grab failed"));
472 return;
473 }
474 if (!m_board) {
475 ui->label_calib_status->setText(tr("Calibration status: no board set"));
476 return;
477 }
478
479 std::vector<cv::Point2f> allPts;
480 std::vector<cv::Point2f> cornerPts;
481 cv::Mat debug_image;
482 bool ok = m_board->detect(result.frame, allPts, &cornerPts, &debug_image);
483 if (!ok || cornerPts.empty()) {
484 ui->label_calib_status->setText(
485 tr("Calibration status: board detect failed"));
486 return;
487 }
488
489 ui->table_calibration_points->setImagePoints(cornerPts);
490 ui->btn_calib_apply->setEnabled(true);
491 ui->label_calib_status->setText(
492 tr("Calibration status: detected (point changed)"));
493 }
494
495 // Frame requested by the threshold-tuning dialog (initial open or Re-grab):
496 // push it into the open dialog so the preview refreshes.
497 if (m_pendingThresholdGrab) {
498 m_pendingThresholdGrab = false;
499 if (m_thresholdDlg && result.isGrabSuccess) {
500 m_thresholdDlg->setImage(result.frame);
501 }
502 }
503}

613void BaslerCameraWidget::btn_calib_threshold_clicked() {
614 if (!m_runner || !m_camera || !m_camera->isDeviceConnected()) {
615 QMessageBox::warning(this, tr("Calibration"),
616 tr("Camera is not connected."));
617 return;
618 }
619 if (!m_board) {
620 QMessageBox::warning(this, tr("Calibration"),
621 tr("No calibration board configured."));
622 return;
623 }
624
625 m_thresholdDlg =
626 new CalibrationThresholdDialog(m_board.get(), m_params.calibThreshold(), this);
627
628 // Re-grab button and the initial open both pull a fresh frame from the
629 // camera; the result is routed back via onCameraGrabFinished().
630 connect(m_thresholdDlg, &CalibrationThresholdDialog::regrabRequested,
631 this, [this]() {
632 if (m_runner && m_camera && m_camera->isDeviceConnected()) {
633 m_pendingThresholdGrab = true;
634 m_runner->requestSingleShot();
635 }
636 });
637
638 m_pendingThresholdGrab = true;
639 m_runner->requestSingleShot();
640
641 const int rc = m_thresholdDlg->exec();
642
643 if (rc == QDialog::Accepted) {
644 const int threshold = m_thresholdDlg->threshold();
645 m_params.setCalibThreshold(threshold);
646 m_board->setBinarizeThreshold(threshold);
647 m_camera->setBaslerGigeConfig(m_params);
648 refreshCalibrationStatusLabels(tr("(threshold changed)"));
649 } else {
650 // Preview mutated the board's threshold; restore the stored value.
651 m_board->setBinarizeThreshold(m_params.calibThreshold());
652 }
653
654 // Drop the dialog; clear the pointer first so a late grab is ignored.
655 m_pendingThresholdGrab = false;
656 CalibrationThresholdDialog *dlg = m_thresholdDlg;
657 m_thresholdDlg = nullptr;
658 dlg->deleteLater();
659}

676void BaslerCameraWidget::btn_calib_apply_clicked() {
677 if (!m_board) {
678 QMessageBox::warning(this, tr("Calibration"),
679 tr("No calibration board configured."));
680 return;
681 }
682
683 const auto imgPts = ui->table_calibration_points->imagePoints();
684 const auto worldPts = ui->table_calibration_points->worldPoints();
685
686 if (imgPts.size() < 4 || imgPts.size() != worldPts.size()) {
687 QMessageBox::warning(
688 this, tr("Calibration"),
689 tr("Need at least 4 valid image/world point pairs. "
690 "Run Detect and fill in the world coordinates first."));
691 return;
692 }
693
694 // Basic sanity: reject the all-zero image-points case that happens before
695 // a detect has populated the table.
696 bool anyImgNonZero = false;
697 for (const auto &p : imgPts) {
698 if (p.x != 0.f || p.y != 0.f) { anyImgNonZero = true; break; }
699 }
700 if (!anyImgNonZero) {
701 QMessageBox::warning(this, tr("Calibration"),
702 tr("Image points are empty — run Detect first."));
703 return;
704 }
705
706 calib::Calibrator calib;
707 calib.addCorrespondences(imgPts, worldPts);
708 if (!calib.calibrate()) {
709 ui->label_calib_status->setText(tr("Calibration status: calibrate failed"));
710 return;
711 }
712
713 m_params.setCalibrator(calib);
714 m_camera->setBaslerGigeConfig(m_params);
715
716 refreshCalibrationStatusLabels();
717 ui->btn_calib_apply->setEnabled(false);
718}
