36[[maybe_unused]] static QString CameraTypeToString(CameraType t) {
37 switch(t) {
38 case vc::device::CameraType::Realsense:
39 return CAM_TYPE_REALSENSE;
40 case vc::device::CameraType::BaslerGigE:
41 return CAM_TYPE_BASLER_GIGE;
42 case vc::device::CameraType::BaslerUSB:
43 return CAM_TYPE_BASLER_USB;
44 case CamType:
45 return "";
46 }
47 return "";

52[[maybe_unused]] static CameraType CameraTypeFromString(QString t) {
53 if (t == CAM_TYPE_REALSENSE) {
54 return CameraType::Realsense;

56 return CameraType::BaslerGigE;

58 return CameraType::BaslerUSB;
59 }
