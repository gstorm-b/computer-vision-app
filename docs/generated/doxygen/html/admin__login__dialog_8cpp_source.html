36bool AdminLoginDialog::ensureAdmin(QWidget *parent)
37{
38 AccessControl *access = AccessControl::instance();
39 if (access->isAdmin()) {
40 return true;
41 }
42
43 if (!access->canElevate()) {
44 // No credential source. Say so instead of showing a dialog whose only possible
45 // outcome is failure — an operator retyping a password against a gate that cannot
46 // open is the worst version of this.
47 QMessageBox::warning(
48 parent,
49 tr("Administrator Access"),
50 tr("No administrator credential is available on this machine, so access "
51 "cannot be granted."));
52 return false;
53 }
54
55 AdminLoginDialog dialog(parent);

57}
