71bool appendTriangle(TriangleMesh& mesh,
72 const Eigen::Vector3d& a,
73 const Eigen::Vector3d& b,
74 const Eigen::Vector3d& c,
75 const StlMeshLoadOptions& options,
76 std::string& error)
77{
78 if (!isFinite(a) || !isFinite(b) || !isFinite(c)) {
79 error = "STL mesh contains non-finite vertex values";
80 return false;
81 }
82 if (isDegenerateTriangle(a, b, c)) {
83 if (options.rejectDegenerateTriangles) {
84 error = "STL mesh contains a degenerate triangle";
85 return false;
86 }
87 return true;
88 }
89
90 const std::size_t baseIndex = mesh.vertices_m.size();
91 mesh.vertices_m.push_back(a);
92 mesh.vertices_m.push_back(b);
93 mesh.vertices_m.push_back(c);
94 mesh.faces.push_back(TriangleFace{baseIndex, baseIndex + 1, baseIndex + 2});
95 return true;
96}

102void finalizeStatistics(TriangleMesh& mesh, const double scaleToMeters)
103{
104 Eigen::Vector3d minimum = Eigen::Vector3d::Constant(std::numeric_limits<double>::max());
105 Eigen::Vector3d maximum = Eigen::Vector3d::Constant(std::numeric_limits<double>::lowest());
106
107 for (const Eigen::Vector3d& vertex : mesh.vertices_m) {
108 minimum = minimum.cwiseMin(vertex);
109 maximum = maximum.cwiseMax(vertex);
110 }
111
112 mesh.statistics.vertexCount = mesh.vertices_m.size();
113 mesh.statistics.triangleCount = mesh.faces.size();
114 mesh.statistics.minimumBounds_m = {minimum.x(), minimum.y(), minimum.z()};
115 mesh.statistics.maximumBounds_m = {maximum.x(), maximum.y(), maximum.z()};
116 mesh.statistics.scaleToMeters = scaleToMeters;
117}

126Result<TriangleMesh> parseBinary(const QByteArray& bytes, const StlMeshLoadOptions& options)
127{
128 if (bytes.size() < 84) {
129 return invalid("STL payload is too small to be a binary STL");
130 }
131
132 const std::uint32_t triangleCount = readLeUInt32(bytes, 80);
133 const std::uint64_t expectedSize = 84ull + static_cast<std::uint64_t>(triangleCount) * 50ull;
134 if (expectedSize != static_cast<std::uint64_t>(bytes.size())) {
135 return invalid("binary STL size does not match triangle count");
136 }
137
138 TriangleMesh mesh;
139 mesh.sourceFormat = StlFileFormat::Binary;
140 mesh.vertices_m.reserve(static_cast<std::size_t>(triangleCount) * 3);
141 mesh.faces.reserve(static_cast<std::size_t>(triangleCount));
142
143 std::string error;
144 int offset = 84;
145 for (std::uint32_t triangleIndex = 0; triangleIndex < triangleCount; ++triangleIndex) {
146 offset += 12;
147
148 const Eigen::Vector3d a(readLeFloat(bytes, offset) * options.scaleToMeters,
149 readLeFloat(bytes, offset + 4) * options.scaleToMeters,
150 readLeFloat(bytes, offset + 8) * options.scaleToMeters);
151 offset += 12;
152 const Eigen::Vector3d b(readLeFloat(bytes, offset) * options.scaleToMeters,
153 readLeFloat(bytes, offset + 4) * options.scaleToMeters,
154 readLeFloat(bytes, offset + 8) * options.scaleToMeters);
155 offset += 12;
156 const Eigen::Vector3d c(readLeFloat(bytes, offset) * options.scaleToMeters,
157 readLeFloat(bytes, offset + 4) * options.scaleToMeters,
158 readLeFloat(bytes, offset + 8) * options.scaleToMeters);
159 offset += 12;
160
161 if (!appendTriangle(mesh, a, b, c, options, error)) {
162 return invalid(error);
163 }
164
165 offset += 2;
166 }
167
168 if (mesh.faces.empty()) {
169 return invalid("binary STL contains no triangles");
170 }
171
172 finalizeStatistics(mesh, options.scaleToMeters);
173 return Result<TriangleMesh>::success(mesh);
174}

183Result<TriangleMesh> parseAscii(const QByteArray& bytes, const StlMeshLoadOptions& options)
184{
185 const QString text = QString::fromUtf8(bytes);
186 const QStringList lines = text.split(QRegularExpression("[\r\n]+"), Qt::SkipEmptyParts);
187
188 std::vector<Eigen::Vector3d> vertices;
189 vertices.reserve(static_cast<std::size_t>(lines.size()));
190
191 for (const QString& rawLine : lines) {
192 const QString line = rawLine.trimmed();
193 if (!line.startsWith(QStringLiteral("vertex "))) {
194 continue;
195 }
196
197 const QStringList parts = line.split(QRegularExpression("\\s+"), Qt::SkipEmptyParts);
198 if (parts.size() != 4) {
199 return invalid("ASCII STL vertex line must contain three numeric coordinates");
200 }
201
202 bool okX = false;
203 bool okY = false;
204 bool okZ = false;
205 const double x = parts[1].toDouble(&okX);
206 const double y = parts[2].toDouble(&okY);
207 const double z = parts[3].toDouble(&okZ);
208 if (!okX || !okY || !okZ) {
209 return invalid("ASCII STL contains a non-numeric vertex coordinate");
210 }
211
212 vertices.emplace_back(x * options.scaleToMeters,
213 y * options.scaleToMeters,
214 z * options.scaleToMeters);
215 }
216
217 if (vertices.empty() || (vertices.size() % 3) != 0) {
218 return invalid("ASCII STL does not contain a whole number of triangles");
219 }
220
221 TriangleMesh mesh;
222 mesh.sourceFormat = StlFileFormat::Ascii;
223 mesh.vertices_m.reserve(vertices.size());
224 mesh.faces.reserve(vertices.size() / 3);
225
226 std::string error;
227 for (std::size_t i = 0; i < vertices.size(); i += 3) {
228 if (!appendTriangle(mesh, vertices[i], vertices[i + 1], vertices[i + 2], options, error)) {
229 return invalid(error);
230 }
231 }
232
233 if (mesh.faces.empty()) {
234 return invalid("ASCII STL contains no triangles after filtering degenerate faces");
235 }
236
237 finalizeStatistics(mesh, options.scaleToMeters);
238 return Result<TriangleMesh>::success(mesh);
239}

247Result<TriangleMesh> StlMeshLoader::loadFile(const std::string& path,
248 const StlMeshLoadOptions& options)
249{
250 if (options.scaleToMeters <= 0.0) {
251 return invalid("STL mesh scaleToMeters must be positive");
252 }
253
254 QFile file(QString::fromStdString(path));
255 if (!file.open(QIODevice::ReadOnly)) {
256 return invalid("STL mesh file could not be opened");
257 }
258
259 const QByteArray bytes = file.readAll();
260
261 Result<TriangleMesh> result = parseBinary(bytes, options);
262 if (result.ok()) {
263 return result;
264 }
265
266 const Result<TriangleMesh> asciiResult = parseAscii(bytes, options);
267 if (asciiResult.ok()) {
268 return asciiResult;
269 }
270
271 return asciiResult;
272}
