33inline bool parseImageKey(const QString &key, int &groupNumber, int &patternNumber) {
34 // Expected form: "g{int}_p{int}"
35 static const QRegularExpression re(QStringLiteral("^g(-?\\d+)_p(-?\\d+)$"));
36 const auto m = re.match(key);
37 if (!m.hasMatch()) return false;
38 bool ok1 = false, ok2 = false;
39 groupNumber = m.captured(1).toInt(&ok1);
40 patternNumber = m.captured(2).toInt(&ok2);
41 return ok1 && ok2;
42}

109bool TaskLocalization::isReachLimitOfDeviceType(vc::device::DeviceType t) const {
110
111 QList<std::shared_ptr<vc::device::IDevice>> result;
112 if (!m_proj) return true;
113
114 auto dm = m_proj->deviceManager();
115 if (!dm) return true;
116
117 auto assigned_device_ids = this->assignedDeviceIds();
118 int device_counter = 0;
119 int limit = m_limitDeviceMap.value(t);
120
121 for (const QString &id : assigned_device_ids) {
122 auto dev = dm->deviceById(id);
123 if (dev && dev->deviceType() == t) {
124 device_counter++;
125 }
126
127 if (device_counter >= limit) {
128 return true;
129 }
130 }
131
132 return false;
133}

302QMap<QString, cv::Mat> TaskLocalization::getTaskImageMap() {
303 QMap<QString, cv::Mat> map;
304
305 if (m_patternManager) {
306 for (const auto &group : m_patternManager->groups()) {
307 if (!group) continue;
308 const int gn = group->number();
309 for (const auto &pattern : group->patterns()) {
310 if (!pattern) continue;
311 const cv::Mat &img = pattern->config().m_rawImage;
312 if (img.empty()) continue; // skip patterns without a training image
313 map.insert(imageKey(gn, pattern->number()), img);
314 }
315 }
316 }
317
318 // Camera workspace reference images (key form "ws_{cameraId}").
319 const QMap<QString, cv::Mat> wsImages = m_config.cameraWorkspaces().getImageMap();
320 for (auto it = wsImages.cbegin(); it != wsImages.cend(); ++it)
321 map.insert(it.key(), it.value());
322
323 return map;
324}

330bool TaskLocalization::loadTaskImageMap(QMap<QString, cv::Mat> &mapping) {
331 if (!m_patternManager) return false;
332 if (mapping.isEmpty()) return true; // nothing to inject
333
334 bool allOk = true;
335 for (auto it = mapping.cbegin(); it != mapping.cend(); ++it) {
336 // Camera workspace reference image (key form "ws_{cameraId}").
337 QString workspaceCameraId;
338 if (CameraWorkspaceMap::parseImageKey(it.key(), workspaceCameraId)) {
339 m_config.d->m_cameraWorkspaces.setReferenceImage(workspaceCameraId, it.value());
340 continue;
341 }
342
343 int gn = 0, pn = 0;
344 if (!parseImageKey(it.key(), gn, pn)) {
345 LOG_DEV_ERR << "TaskLocalization::loadTaskImageMap – bad key:" << it.key();
346 allOk = false;
347 continue;
348 }
349
350 auto group = m_patternManager->findGroupByNumber(gn);
351 if (!group) {
352 LOG_DEV_ERR << "TaskLocalization::loadTaskImageMap – group" << gn
353 << "not found for key" << it.key();
354 allOk = false;
355 continue;
356 }
357
358 auto *pattern = group->findPatternByNumber(pn);
359 if (!pattern) {
360 LOG_DEV_ERR << "TaskLocalization::loadTaskImageMap – pattern" << pn
361 << "not found in group" << gn;
362 allOk = false;
363 continue;
364 }
365
366 const QString groupName = QString::fromStdWString(group->name());
367 const QString patternName = QString::fromStdWString(pattern->name());
368 if (auto r = m_patternManager->setPatternImage(groupName, patternName, it.value()); !r) {
369 LOG_DEV_ERR << "TaskLocalization::loadTaskImageMap – setPatternImage failed:"
370 << QString::fromStdWString(r.error);
371 allOk = false;
372 }
373 }
374 return allOk;
375}

586void TaskLocalization::wireRuntimeControllerSignals()
587{
588 if (!m_runtimeController) {
589 return;
590 }
591
592 connect(m_runtimeController, &LocalizationRuntimeController::signalChanged,
593 this, &ITask::signalChanged, Qt::QueuedConnection);
594 connect(m_runtimeController, &LocalizationRuntimeController::cycleResultUpdated,
595 this, &TaskLocalization::cycleResultUpdated, Qt::QueuedConnection);
596 connect(m_runtimeController, &LocalizationRuntimeController::taskLogAppended,
597 this, &TaskLocalization::taskLogAppended, Qt::QueuedConnection);
598 connect(m_runtimeController, &LocalizationRuntimeController::runtimeCycleStarted,
599 this, &TaskLocalization::onRuntimeCycleStarted, Qt::QueuedConnection);
600 connect(m_runtimeController, &LocalizationRuntimeController::runtimeRecovering,
601 this, &TaskLocalization::onRuntimeRecovering, Qt::QueuedConnection);
602 connect(m_runtimeController, &LocalizationRuntimeController::runtimeReady,
603 this, &TaskLocalization::onRuntimeReady, Qt::QueuedConnection);
604 connect(m_runtimeController, &LocalizationRuntimeController::runtimeFault,
605 this, &TaskLocalization::onRuntimeFault, Qt::QueuedConnection);
606
607 if (m_matchingWorker) {
608 QPointer<LocalizationRuntimeController> controller(m_runtimeController);
609 connect(m_runtimeController,
610 &LocalizationRuntimeController::runtimeMatchingRequested,
611 m_matchingWorker,
612 [this, controller](int cycleId,
613 std::shared_ptr<mtc::MatchGroup> group,
614 CameraWorkspace workspace,
615 cv::Mat image,
616 std::shared_ptr<mtc::IRobotPickingChecker> pickingChecker) {
617
618 // Persistent matcher: reused across cycles (runs on the
619 // matchingRunner thread). The matcher + learned model are rebuilt
620 // only when the active pattern group changes, never per cycle.
621 if (!m_runtimeMatcher || m_loadedRuntimeGroup != group) {
622 m_runtimeMatcher = std::make_unique<mtc::ImageMatcher>();
623 m_loadedRuntimeGroup =
624 m_pipeline.loadModel(*m_runtimeMatcher, group) ? group : nullptr;
625 }
626
627 mtc::MatchResult result;
628 if (m_loadedRuntimeGroup) {
629 result = m_pipeline.runMatchOn(*m_runtimeMatcher, workspace, image,
630 pickingChecker.get(),
631 LocalizationPipeline::kRuntimeMaxObjects);
632 }
633 if (!controller) {
634 return;
635 }
636 QMetaObject::invokeMethod(controller.data(),
637 [controller, cycleId, result]() {
638 if (controller) {
639 controller->onRuntimeMatchingFinished(cycleId, result);
640 }
641 }, Qt::QueuedConnection);
642 }, Qt::QueuedConnection);
643 }

651std::shared_ptr<mtc::MatchGroup>
652TaskLocalization::snapshotPatternGroup(const std::shared_ptr<mtc::MatchGroup> &source)
653{
654 if (!source) {
655 return nullptr;
656 }
657
658 // Independent deep copy: own MatchGroupConfig (with cloned typeConfig) plus
659 // an own pattern vector whose configs carry the training image (m_rawImage).
660 // The worker can learn/match from this snapshot without touching — or racing
661 // — the live PatternGroupManager group on the GUI thread.
662 auto snapshot = std::make_shared<mtc::MatchGroup>();
663 source->cloneConfigTo(*snapshot);
664 for (const auto &pattern : source->patterns()) {
665 if (pattern) {
666 snapshot->addPattern(pattern->config());
667 }
668 }
669 return snapshot;
670}

679LocalizationRuntimeController::RuntimeContext
680TaskLocalization::buildRuntimeContext() const
681{
682 LocalizationRuntimeController::RuntimeContext context;
683 context.config = m_config;
684 context.primaryPlcDeviceId = m_config.d->m_deviceBindings.primaryPlcDeviceId();
685 context.visionOutputDeviceId = m_config.d->m_deviceBindings.visionOutputDeviceId();
686 context.primaryPlcRunner = plcRunner(context.primaryPlcDeviceId);
687 context.visionOutputRunner = qobject_cast<vc::runtime::VisionOutputRunner *>(
688 taskRunner() ? taskRunner()->runnerFor(context.visionOutputDeviceId) : nullptr);
689
690 // Snapshot the robot kinematic check settings from the assigned vision
691 // output device config (drives the per-object robotPossiblePickingCheck).
692 if (auto visionDevice = getTaskDevice(context.visionOutputDeviceId)) {
693 std::unique_ptr<device::IDeviceCfg> visionCfg(visionDevice->deviceConfig());
694 if (auto *voutCfg = dynamic_cast<device::VisionOutputDeviceCfg *>(visionCfg.get())) {
695 context.robotCheckConfig = voutCfg->m_kinematicCheck;
696 }
697 }
698
699 context.cameraDeviceIds = m_config.d->m_deviceBindings.cameraNumberMap();
700
701 for (auto it = context.cameraDeviceIds.cbegin(); it != context.cameraDeviceIds.cend(); ++it) {
702 context.cameraRunners.insert(it.key(), cameraRunner(it.value()));
703
704 auto device = getTaskDevice(it.value());
705 auto camera = std::dynamic_pointer_cast<device::CameraDevice>(device);
706 if (!camera) {
707 continue;
708 }
709 std::unique_ptr<device::IDeviceCfg> cfg(camera->deviceConfig());
710 auto *cameraCfg = dynamic_cast<device::CameraCfg *>(cfg.get());
711 if (cameraCfg) {
712 context.cameraCalibrators.insert(it.key(), cameraCfg->calibrator());
713 }
714 }
715
716 if (!context.cameraDeviceIds.isEmpty()) {
717 context.activeCameraNumber = context.cameraDeviceIds.firstKey();
718 }
719
720 if (m_patternManager) {
721 const auto groups = m_patternManager->groups();
722 for (const auto &source : groups) {
723 if (auto snapshot = snapshotPatternGroup(source)) {
724 context.patternGroups.insert(snapshot->number(), snapshot);
725 }
726 }
727 if (!context.patternGroups.isEmpty()) {
728 context.activePatternGroupNumber = context.patternGroups.firstKey();
729 }
730 }
731
732 return context;
733}
