144void MatchConfigPropertyAdapter::buildTypeGroup()
145{
146 const MatchingType t = m_cfg->matchingType();
147 m_grpType = m_mgr->addProperty(QtVariantPropertyManager::groupTypeId(),
148 tr(matchingTypeName(t)));
149
150 if (t == MatchingType::EdgeBased) {
151 const EdgeMatchConfig *ecfg = m_cfg->edgeConfig();
152 if (!ecfg) return;
153
154 auto sub = PropSpecHelper::buildGroup(m_mgr, m_grpType,
155 kEdgeSpecs, *ecfg, m_propKeys);
156 for (auto it = sub.cbegin(); it != sub.cend(); ++it)
157 m_props[it.key()] = it.value();
158 }
159 // Future: add branches for MatchingType::Correlation etc.
160}

182void MatchConfigPropertyAdapter::destroy()
183{
184 // The variant manager is SHARED with the host widget (group-level props,
185 // other adapters, …). m_mgr->clear() would wipe everyone's properties.
186 // Delete only the top-level groups this adapter created. Their child
187 // properties are owned by the QtProperty tree and are released together
188 // with the parent group.
189 m_props.clear();
190 m_propKeys.clear();
191
192 delete m_grpType; m_grpType = nullptr;
193
194 m_cfg = nullptr;
195}

202void MatchConfigPropertyAdapter::onPropertyValueChanged(QtProperty *prop,
203 const QVariant &val)
204{
205 if (!m_cfg) return;
206
207 const QString key = m_propKeys.value(prop);
208 if (key.isEmpty()) return;
209
210 if (m_cfg->matchingType() == MatchingType::EdgeBased) {
211 EdgeMatchConfig *ecfg = m_cfg->edgeConfig();
212 if (ecfg && PropSpecHelper::dispatch(kEdgeSpecs, key, val, *ecfg)) {
213 emit configModified();
214 }
215 }
