15QtVariantProperty *addRow(const QMetaObject &meta, const QMetaProperty &prop,
16 const QVariant &value, QtVariantPropertyManager *manager)
17{
18 const QString name = QString::fromUtf8(prop.name());
19 if (name == QLatin1String("objectName")) {
20 return nullptr;
21 }
22
23 QtVariantProperty *row = nullptr;
24 if (prop.isEnumType()) {
25 row = manager->addProperty(QtVariantPropertyManager::enumTypeId(), name);
26 QStringList names;
27 const QMetaEnum metaEnum = prop.enumerator();
28 for (int i = 0; i < metaEnum.keyCount(); ++i) {
29 names << QString::fromUtf8(metaEnum.key(i));
30 }
31 if (row != nullptr) {
32 row->setAttribute(QLatin1String("enumNames"), names);
33 row->setValue(value.toInt());
34 }
35 } else {
36 row = manager->addProperty(value.userType(), name);
37 if (row != nullptr) {
38 row->setValue(value);
39 }
40 }
41
42 if (row != nullptr) {
43 row->setPropertyName(vc::gadget_meta::displayName(meta, prop.name()));
44 }
45 return row;
46}

92void VirtualDeviceWidget::buildPropertyBrowser()
93{
94 if (!m_device || m_variantManager == nullptr || m_variantEditor == nullptr) {
95 return;
96 }
97
98 m_populating = true;
99
100 QtProperty *deviceGroup =
101 m_variantManager->addProperty(QtVariantPropertyManager::groupTypeId(), tr("Device"));
102 m_variantEditor->addProperty(deviceGroup);
103
104 const QMetaObject *deviceMeta = m_device->metaObject();
105 for (int i = 0; i < deviceMeta->propertyCount(); ++i) {
106 QMetaProperty prop = deviceMeta->property(i);
107 QtVariantProperty *row = addRow(*deviceMeta, prop, prop.read(m_device.get()),
108 m_variantManager);
109 if (row != nullptr) {
110 deviceGroup->addSubProperty(row);
111 }
112 }
113
114 QtProperty *configGroup = m_variantManager->addProperty(
115 QtVariantPropertyManager::groupTypeId(), tr("Configuration"));
116 m_variantEditor->addProperty(configGroup);
117
118 if (m_config) {
119 const QMetaObject &configMeta = m_config->getMetaObject();
120 for (int i = 0; i < configMeta.propertyCount(); ++i) {
121 QMetaProperty prop = configMeta.property(i);
122 QtVariantProperty *row = addRow(configMeta, prop,
123 prop.readOnGadget(m_config.get()),
124 m_variantManager);
125 if (row != nullptr) {
126 configGroup->addSubProperty(row);
127 }
128 }
129 }
130
131 m_populating = false;
132}

139void VirtualDeviceWidget::onPropertyValueChanged(QtProperty *property, const QVariant &value)
140{
141 if (m_populating || !m_device || property == nullptr) {
142 return;
143 }
144
145 const QString propName = property->propertyName();
146
147 if (propName == vc::gadget_meta::displayName(*m_device->metaObject(), "name")
148 || propName == QLatin1String("name")) {
149 const QString newName = value.toString();
150 if (m_device->name() == newName) {
151 return;
152 }
153 auto *manager = m_device->deviceManager();
154 if (manager != nullptr && !manager->changeDeviceName(m_device->id(), newName)) {
155 LOG_USER_WARN << tr("Cannot rename device to \"%1\": the name is already in use.")
156 .arg(newName);
157 loadConfigToWidget();
158 }
159 return;
160 }
161
162 if (!m_config) {
163 return;
164 }
165
166 // The browser row is labelled with the display name, so the property has to be found by
167 // matching that back to the underlying key — the gadget knows the key, the row does not.
168 const QMetaObject &configMeta = m_config->getMetaObject();
169 for (int i = 0; i < configMeta.propertyCount(); ++i) {
170 const QMetaProperty prop = configMeta.property(i);
171 if (propName != vc::gadget_meta::displayName(configMeta, prop.name())
172 && propName != QString::fromUtf8(prop.name())) {
173 continue;
174 }
175 if (prop.writeOnGadget(m_config.get(), value)) {
176 loadConfigToDevice();
177 }
178 return;
179 }
180}
