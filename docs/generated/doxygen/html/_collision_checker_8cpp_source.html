52double distancePointToSegment(const Eigen::Vector3d& point,
53 const Eigen::Vector3d& start,
54 const Eigen::Vector3d& end)
55{
56 const Eigen::Vector3d segment = end - start;
57 const double lengthSquared = segment.squaredNorm();
58 if (lengthSquared <= 1e-18) {
59 return (point - start).norm();
60 }
61
62 const double t = clamp01((point - start).dot(segment) / lengthSquared);
63 return (point - (start + t * segment)).norm();
64}

69double distanceSegmentToSegment(const Eigen::Vector3d& p1,
70 const Eigen::Vector3d& q1,
71 const Eigen::Vector3d& p2,
72 const Eigen::Vector3d& q2)
73{
74 const Eigen::Vector3d d1 = q1 - p1;
75 const Eigen::Vector3d d2 = q2 - p2;
76 const Eigen::Vector3d r = p1 - p2;
77 const double a = d1.dot(d1);
78 const double e = d2.dot(d2);
79 const double f = d2.dot(r);
80
81 double s = 0.0;
82 double t = 0.0;
83
84 if (a <= 1e-18 && e <= 1e-18) {
85 return (p1 - p2).norm();
86 }
87
88 if (a <= 1e-18) {
89 t = clamp01(f / e);
90 } else {
91 const double c = d1.dot(r);
92 if (e <= 1e-18) {
93 s = clamp01(-c / a);
94 } else {
95 const double b = d1.dot(d2);
96 const double denom = a * e - b * b;
97
98 if (std::abs(denom) > 1e-18) {
99 s = clamp01((b * f - c * e) / denom);
100 }
101
102 t = (b * s + f) / e;
103 if (t < 0.0) {
104 t = 0.0;
105 s = clamp01(-c / a);
106 } else if (t > 1.0) {
107 t = 1.0;
108 s = clamp01((b - c) / a);
109 }
110 }
111 }
112
113 const Eigen::Vector3d c1 = p1 + d1 * s;
114 const Eigen::Vector3d c2 = p2 + d2 * t;
115 return (c1 - c2).norm();
116}

122PlacedGeometry placeGeometry(const FkChain& chain, const CollisionGeometry& geometry)
123{
124 const Pose linkPose = chain.linkPosesInBase.at(geometry.linkId);
125
126 PlacedGeometry placed;
127 placed.geometry = &geometry;
128 placed.geometryInBase = linkPose * geometry.geometryToLink;
129 placed.center_m = placed.geometryInBase.translation_m();
130
131 switch (geometry.shape.type) {
132 case CollisionShapeType::Sphere:
133 placed.segmentStart_m = placed.center_m;
134 placed.segmentEnd_m = placed.center_m;
135 placed.boundingRadius_m = geometry.shape.sphere.radius_m;
136 break;
137 case CollisionShapeType::Capsule: {
138 const double halfLength = geometry.shape.capsule.length_m * 0.5;
139 const Eigen::Vector3d axisInBase =
140 placed.geometryInBase.isometry().linear() * Eigen::Vector3d::UnitZ();
141 placed.segmentStart_m = placed.center_m - axisInBase * halfLength;
142 placed.segmentEnd_m = placed.center_m + axisInBase * halfLength;
143 placed.boundingRadius_m = geometry.shape.capsule.radius_m + halfLength;
144 break;
145 }
146 }
147
148 return placed;
149}

155double primitiveDistance(const PlacedGeometry& a, const PlacedGeometry& b)
156{
157 const CollisionGeometry& geometryA = *a.geometry;
158 const CollisionGeometry& geometryB = *b.geometry;
159
160 if (geometryA.shape.type == CollisionShapeType::Sphere &&
161 geometryB.shape.type == CollisionShapeType::Sphere) {
162 return (a.center_m - b.center_m).norm() -
163 geometryA.shape.sphere.radius_m -
164 geometryB.shape.sphere.radius_m;
165 }
166
167 if (geometryA.shape.type == CollisionShapeType::Sphere &&
168 geometryB.shape.type == CollisionShapeType::Capsule) {
169 return distancePointToSegment(a.center_m, b.segmentStart_m, b.segmentEnd_m) -
170 geometryA.shape.sphere.radius_m -
171 geometryB.shape.capsule.radius_m;
172 }
173
174 if (geometryA.shape.type == CollisionShapeType::Capsule &&
175 geometryB.shape.type == CollisionShapeType::Sphere) {
176 return distancePointToSegment(b.center_m, a.segmentStart_m, a.segmentEnd_m) -
177 geometryA.shape.capsule.radius_m -
178 geometryB.shape.sphere.radius_m;
179 }
180
181 return distanceSegmentToSegment(a.segmentStart_m, a.segmentEnd_m,
182 b.segmentStart_m, b.segmentEnd_m) -
183 geometryA.shape.capsule.radius_m -
184 geometryB.shape.capsule.radius_m;
185}

190bool isBroadPhaseSeparated(const PlacedGeometry& a,
191 const PlacedGeometry& b,

193{
194 const double centerDistance = (a.center_m - b.center_m).norm();
195 const double threshold = a.boundingRadius_m + b.boundingRadius_m + extraMargin_m;

205CollisionCheckResult CollisionChecker::check(const SerialRobotConfig& config,
206 const CollisionProfile& profile,
207 const CollisionCheckRequest& request)
208{
209 const ModelValidationResult modelValidation = RobotModelValidator::validateSerialRobotConfig(config);
210 if (!modelValidation.ok()) {
211 return CollisionCheckResult{
212 modelValidation.status(),
213 false,
214 {},
215 modelValidation.issues.front().message,
216 };
217 }
218
219 const CollisionProfileValidationResult profileValidation =
220 CollisionProfileValidator::validate(config, profile);
221 if (!profileValidation.ok()) {
222 return CollisionCheckResult{
223 profileValidation.status(),
224 false,
225 {},
226 profileValidation.issues.front().message,
227 };
228 }
229
230 if (request.joints.size() != ForwardKinematics::movableJointCount(config)) {
231 return CollisionCheckResult{
232 KinematicsStatus::JointDimensionMismatch,
233 false,
234 {},
235 "joint vector dimension does not match robot dof",
236 };
237 }
238
239 const FkChain chain = ForwardKinematics::computeChain(config, request.joints);
240
241 std::vector<PlacedGeometry> placedGeometries;
242 placedGeometries.reserve(profile.geometries.size());
243 for (const CollisionGeometry& geometry : profile.geometries) {
244 if (!geometry.enabled) {
245 continue;
246 }
247
248 const auto linkIt = chain.linkPosesInBase.find(geometry.linkId);
249 if (linkIt == chain.linkPosesInBase.end()) {
250 return CollisionCheckResult{
251 KinematicsStatus::InvalidRobotConfig,
252 false,
253 {},
254 "collision geometry link pose not found in FK chain: " + geometry.linkId,
255 };
256 }
257
258 placedGeometries.push_back(placeGeometry(chain, geometry));
259 }
260
261 std::set<std::string> disabledPairs;
262 for (const DisabledCollisionPair& pair : profile.disabledPairs) {
263 disabledPairs.insert(normalizePairKey(pair.geometryA, pair.geometryB));
264 }
265
266 CollisionCheckResult result;
267 result.status = KinematicsStatus::Ok;
268
269 for (std::size_t i = 0; i < placedGeometries.size(); ++i) {
270 for (std::size_t j = i + 1; j < placedGeometries.size(); ++j) {
271 const PlacedGeometry& a = placedGeometries[i];
272 const PlacedGeometry& b = placedGeometries[j];
273 const CollisionGeometry& geometryA = *a.geometry;
274 const CollisionGeometry& geometryB = *b.geometry;
275
276 if (geometryA.id == geometryB.id) {
277 continue;
278 }
279 if (geometryA.linkId == geometryB.linkId) {
280 continue;
281 }
282 if (disabledPairs.find(normalizePairKey(geometryA.id, geometryB.id)) != disabledPairs.end()) {
283 continue;
284 }
285
286 const double pairMargin =
287 geometryA.margin_m + geometryB.margin_m + request.safetyMargin_m;
288
289 if (isBroadPhaseSeparated(a, b, pairMargin)) {
290 const double distance =
291 (a.center_m - b.center_m).norm() - (a.boundingRadius_m + b.boundingRadius_m);
292 result.pairs.push_back(CollisionPairResult{
293 geometryA.id,
294 geometryB.id,
295 geometryA.linkId,
296 geometryB.linkId,
297 false,
298 distance,
299 0,
300 });
301 continue;
302 }
303
304 const double distance = primitiveDistance(a, b);
305 const bool colliding = distance <= pairMargin;
306
307 result.pairs.push_back(CollisionPairResult{
308 geometryA.id,
309 geometryB.id,
310 geometryA.linkId,
311 geometryB.linkId,
312 colliding,
313 distance,
314 0,
315 });
316
317 if (colliding) {
318 result.hasCollision = true;
319 if (!request.returnAllPairs) {
320 return result;
321 }
322 }
323 }
324 }
325
326 return result;
327}
