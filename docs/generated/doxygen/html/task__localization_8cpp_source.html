32inline bool parseImageKey(const QString &key, int &groupNumber, int &patternNumber) {
33 // Expected form: "g{int}_p{int}"
34 static const QRegularExpression re(QStringLiteral("^g(-?\\d+)_p(-?\\d+)$"));
35 const auto m = re.match(key);
36 if (!m.hasMatch()) return false;
37 bool ok1 = false, ok2 = false;
38 groupNumber = m.captured(1).toInt(&ok1);
39 patternNumber = m.captured(2).toInt(&ok2);
40 return ok1 && ok2;
41}

111bool TaskLocalization::isReachLimitOfDeviceType(vc::device::DeviceType t) const {
112
113 QList<std::shared_ptr<vc::device::IDevice>> result;
114 if (!m_proj) return true;
115
116 auto dm = m_proj->deviceManager();
117 if (!dm) return true;
118
119 auto assigned_device_ids = this->assignedDeviceIds();
120 int device_counter = 0;
121 int limit = m_limitDeviceMap.value(t);
122
123 for (const QString &id : assigned_device_ids) {
124 auto dev = dm->deviceById(id);
125 if (dev && dev->deviceType() == t) {
126 device_counter++;
127 }
128
129 if (device_counter >= limit) {
130 return true;
131 }
132 }
133
134 return false;
135}

262QMap<QString, cv::Mat> TaskLocalization::getTaskImageMap() {
263 QMap<QString, cv::Mat> map;
264
265 if (m_patternManager) {
266 for (const auto &group : m_patternManager->groups()) {
267 if (!group) continue;
268 const int gn = group->number();
269 for (const auto &pattern : group->patterns()) {
270 if (!pattern) continue;
271 const cv::Mat &img = pattern->config().m_rawImage;
272 if (img.empty()) continue; // skip patterns without a training image
273 map.insert(imageKey(gn, pattern->number()), img);
274 }
275 }
276 }
277
278 // Camera workspace reference images (key form "ws_{cameraId}").
279 const QMap<QString, cv::Mat> wsImages = m_config.cameraWorkspaces().getImageMap();
280 for (auto it = wsImages.cbegin(); it != wsImages.cend(); ++it)
281 map.insert(it.key(), it.value());
282
283 return map;
284}

290bool TaskLocalization::loadTaskImageMap(QMap<QString, cv::Mat> &mapping) {
291 if (!m_patternManager) return false;
292 if (mapping.isEmpty()) return true; // nothing to inject
293
294 bool allOk = true;
295 for (auto it = mapping.cbegin(); it != mapping.cend(); ++it) {
296 // Camera workspace reference image (key form "ws_{cameraId}").
297 QString workspaceCameraId;
298 if (CameraWorkspaceMap::parseImageKey(it.key(), workspaceCameraId)) {
299 m_config.d->m_cameraWorkspaces.setReferenceImage(workspaceCameraId, it.value());
300 continue;
301 }
302
303 int gn = 0, pn = 0;
304 if (!parseImageKey(it.key(), gn, pn)) {
305 LOG_DEV_ERR << "TaskLocalization::loadTaskImageMap – bad key:" << it.key();
306 allOk = false;
307 continue;
308 }
309
310 auto group = m_patternManager->findGroupByNumber(gn);
311 if (!group) {
312 LOG_DEV_ERR << "TaskLocalization::loadTaskImageMap – group" << gn
313 << "not found for key" << it.key();
314 allOk = false;
315 continue;
316 }
317
318 auto *pattern = group->findPatternByNumber(pn);
319 if (!pattern) {
320 LOG_DEV_ERR << "TaskLocalization::loadTaskImageMap – pattern" << pn
321 << "not found in group" << gn;
322 allOk = false;
323 continue;
324 }
325
326 const QString groupName = QString::fromStdWString(group->name());
327 const QString patternName = QString::fromStdWString(pattern->name());
328 if (auto r = m_patternManager->setPatternImage(groupName, patternName, it.value()); !r) {
329 LOG_DEV_ERR << "TaskLocalization::loadTaskImageMap – setPatternImage failed:"
330 << QString::fromStdWString(r.error);
331 allOk = false;
332 }
333 }
334 return allOk;
335}

542void TaskLocalization::wireRuntimeControllerSignals()
543{
544 if (!m_runtimeController) {
545 return;
546 }
547
548 connect(m_runtimeController, &LocalizationRuntimeController::signalChanged,
549 this, &ITask::signalChanged, Qt::QueuedConnection);
550 connect(m_runtimeController, &LocalizationRuntimeController::cycleResultUpdated,
551 this, &TaskLocalization::cycleResultUpdated, Qt::QueuedConnection);
552 connect(m_runtimeController, &LocalizationRuntimeController::taskLogAppended,
553 this, &TaskLocalization::taskLogAppended, Qt::QueuedConnection);
554 connect(m_runtimeController, &LocalizationRuntimeController::runtimeCycleStarted,
555 this, &TaskLocalization::onRuntimeCycleStarted, Qt::QueuedConnection);
556 connect(m_runtimeController, &LocalizationRuntimeController::runtimeRecovering,
557 this, &TaskLocalization::onRuntimeRecovering, Qt::QueuedConnection);
558 connect(m_runtimeController, &LocalizationRuntimeController::runtimeReady,
559 this, &TaskLocalization::onRuntimeReady, Qt::QueuedConnection);
560 connect(m_runtimeController, &LocalizationRuntimeController::runtimeFault,
561 this, &TaskLocalization::onRuntimeFault, Qt::QueuedConnection);
562
563 if (m_matchingWorker) {
564 QPointer<LocalizationRuntimeController> controller(m_runtimeController);
565 connect(m_runtimeController,
566 &LocalizationRuntimeController::runtimeMatchingRequested,
567 m_matchingWorker,
568 [this, controller](int cycleId,
569 std::shared_ptr<mtc::MatchGroup> group,
570 CameraWorkspace workspace,
571 cv::Mat image,
572 std::shared_ptr<mtc::IRobotPickingChecker> pickingChecker) {
573
574 // Persistent matcher: reused across cycles (runs on the
575 // matchingRunner thread). The matcher + learned model are rebuilt
576 // only when the active pattern group changes, never per cycle.
577 if (!m_runtimeMatcher || m_loadedRuntimeGroup != group) {
578 m_runtimeMatcher = std::make_unique<mtc::ImageMatcher>();
579 m_loadedRuntimeGroup =
580 m_pipeline.loadModel(*m_runtimeMatcher, group) ? group : nullptr;
581 }
582
583 mtc::MatchResult result;
584 if (m_loadedRuntimeGroup) {
585 result = m_pipeline.runMatchOn(*m_runtimeMatcher, workspace, image,
586 pickingChecker.get(),
587 LocalizationPipeline::kRuntimeMaxObjects);
588 }
589 if (!controller) {
590 return;
591 }
592 QMetaObject::invokeMethod(controller.data(),
593 [controller, cycleId, result]() {
594 if (controller) {
595 controller->onRuntimeMatchingFinished(cycleId, result);
596 }
597 }, Qt::QueuedConnection);
598 }, Qt::QueuedConnection);
599 }

607std::shared_ptr<mtc::MatchGroup>
608TaskLocalization::snapshotPatternGroup(const std::shared_ptr<mtc::MatchGroup> &source)
609{
610 if (!source) {
611 return nullptr;
612 }
613
614 // Independent deep copy: own MatchGroupConfig (with cloned typeConfig) plus
615 // an own pattern vector whose configs carry the training image (m_rawImage).
616 // The worker can learn/match from this snapshot without touching — or racing
617 // — the live PatternGroupManager group on the GUI thread.
618 auto snapshot = std::make_shared<mtc::MatchGroup>();
619 source->cloneConfigTo(*snapshot);
620 for (const auto &pattern : source->patterns()) {
621 if (pattern) {
622 snapshot->addPattern(pattern->config());
623 }
624 }
625 return snapshot;
626}

635LocalizationRuntimeController::RuntimeContext
636TaskLocalization::buildRuntimeContext() const
637{
638 LocalizationRuntimeController::RuntimeContext context;
639 context.config = m_config;
640 context.primaryPlcDeviceId = m_config.d->m_deviceBindings.primaryPlcDeviceId();
641 context.visionOutputDeviceId = m_config.d->m_deviceBindings.visionOutputDeviceId();
642 context.primaryPlcRunner = plcRunner(context.primaryPlcDeviceId);
643 context.visionOutputRunner = qobject_cast<vc::runtime::VisionOutputRunner *>(
644 taskRunner() ? taskRunner()->runnerFor(context.visionOutputDeviceId) : nullptr);
645
646 // Snapshot the robot kinematic check settings from the assigned vision
647 // output device config (drives the per-object robotPossiblePickingCheck).
648 if (auto visionDevice = getTaskDevice(context.visionOutputDeviceId)) {
649 std::unique_ptr<device::IDeviceCfg> visionCfg(visionDevice->deviceConfig());
650 if (auto *voutCfg = dynamic_cast<device::VisionOutputDeviceCfg *>(visionCfg.get())) {
651 context.robotCheckConfig = voutCfg->m_kinematicCheck;
652 }
653 }
654
655 context.cameraDeviceIds = m_config.d->m_deviceBindings.cameraNumberMap();
656
657 for (auto it = context.cameraDeviceIds.cbegin(); it != context.cameraDeviceIds.cend(); ++it) {
658 context.cameraRunners.insert(it.key(), cameraRunner(it.value()));
659
660 auto device = getTaskDevice(it.value());
661 auto camera = std::dynamic_pointer_cast<device::CameraDevice>(device);
662 if (!camera) {
663 continue;
664 }
665 std::unique_ptr<device::IDeviceCfg> cfg(camera->deviceConfig());
666 auto *cameraCfg = dynamic_cast<device::CameraCfg *>(cfg.get());
667 if (cameraCfg) {
668 context.cameraCalibrators.insert(it.key(), cameraCfg->calibrator());
669 }
670 }
671
672 if (!context.cameraDeviceIds.isEmpty()) {
673 context.activeCameraNumber = context.cameraDeviceIds.firstKey();
674 }
675
676 if (m_patternManager) {
677 const auto groups = m_patternManager->groups();
678 for (const auto &source : groups) {
679 if (auto snapshot = snapshotPatternGroup(source)) {
680 context.patternGroups.insert(snapshot->number(), snapshot);
681 }
682 }
683 if (!context.patternGroups.isEmpty()) {
684 context.activePatternGroupNumber = context.patternGroups.firstKey();
685 }
686 }
687
688 return context;
689}
