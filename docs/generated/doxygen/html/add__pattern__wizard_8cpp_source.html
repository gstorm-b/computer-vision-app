474 m_boxCanvas->setMode(AddPatternImageCanvas::Box);
475 m_boxCanvas->setMinimumSize(CW, CH);
476 // Canvas-driven box edits (drag corner / body / rotation handle) push
477 // values back into the spin boxes; the existing onBoxChanged() slot
478 // then echoes them into m_box* state.
479 connect(m_boxCanvas, &AddPatternImageCanvas::boxChanged,
480 this, [this](double w, double h, double d, double a) {
481 if (m_boxWSpin) {
482 QSignalBlocker b1(m_boxWSpin), b2(m_boxHSpin),
483 b3(m_boxDistSpin), b4(m_boxAngleSpin);
484 m_boxWSpin->setValue(w);
485 m_boxHSpin->setValue(h);
486 m_boxDistSpin->setValue(d);
487 m_boxAngleSpin->setValue(a);
488 }
489 m_boxW = w; m_boxH = h; m_boxDist = d; m_boxAngle = a;
490 updateFooterStatus();
491 });
492 // The picking centre can also be dragged on the box canvas (for
493 // convenience while positioning the jaws). The box canvas shows the full
494 // frame, so `p` is full-image coords — mirror the Pick-step storage
495 // semantics (crop-relative when the user cropped) and echo the pick spins.
496 connect(m_boxCanvas, &AddPatternImageCanvas::pickChanged,
497 this, [this](const QPoint &p, const QPoint &) {
498 m_pick = m_keepOriginal ? p : (p - m_crop.topLeft());
499 if (m_pickXSpin) {
500 QSignalBlocker b1(m_pickXSpin), b2(m_pickYSpin);
501 m_pickXSpin->setValue(m_pick.x());
502 m_pickYSpin->setValue(m_pick.y());
503 }
504 updateFooterStatus();
505 });
506 lay->addWidget(m_boxCanvas, 1);
507

914void AddPatternWizard::onImageSizeChanged(int imageW, int imageH) {
915 if (imageW <= 0 || imageH <= 0) return;
916
917 // ── Spin box ranges ─────────────────────────────────────────────────
918 // Block per-spin signals while we widen — setRange() can clip the
919 // current value, which would otherwise re-fire onCropChanged / etc.
920 auto setR = [](QSpinBox *sb, int lo, int hi) {
921 if (!sb) return;
922 QSignalBlocker b(sb);
923 sb->setRange(lo, hi);
924 };
925 auto setRD = [](QDoubleSpinBox *sb, double lo, double hi) {
926 if (!sb) return;
927 QSignalBlocker b(sb);
928 sb->setRange(lo, hi);
929 };
930
931 setR(m_cropX, 0, imageW);
932 setR(m_cropY, 0, imageH);
933 setR(m_cropW, 1, imageW);
934 setR(m_cropH, 1, imageH);
935
936 setR(m_pickXSpin, 0, imageW - 1);
937 setR(m_pickYSpin, 0, imageH - 1);
938
939 // Pick-box sizes can in principle exceed the image (the canvas now lets
940 // jaws spill off-frame), so give a generous cap based on the image.
941 const double sizeCap = qMax(double(qMax(imageW, imageH)), 5000.0);
942 const double distCap = qMax(std::hypot(double(imageW), double(imageH)), 1000.0);
943 setRD(m_boxWSpin, 1.0, sizeCap);
944 setRD(m_boxHSpin, 1.0, sizeCap);
945 setRD(m_boxDistSpin, 0.0, distCap);
946 // Angle range stays as configured (-180..180).
947
948 // ── Clamp / re-centre defaults to fit the new image ─────────────────
949 QRect imgR(0, 0, imageW, imageH);
950 if (!imgR.contains(m_crop) || m_crop.isEmpty()) {
951 const int cw = qBound(20, m_crop.width(), imageW);
952 const int ch = qBound(20, m_crop.height(), imageH);
953 m_crop = QRect((imageW - cw) / 2, (imageH - ch) / 2, cw, ch);
954 }
955 if (m_pick.x() < 0 || m_pick.x() >= imageW ||
956 m_pick.y() < 0 || m_pick.y() >= imageH) {
957 m_pick = QPoint(imageW / 2, imageH / 2);
958 }
959
960 // Push the clamped values back into the spin boxes (still blocked above
961 // for the range set; values need their own block).
962 auto setV = [](QSpinBox *sb, int v) {
963 if (!sb) return;
964 QSignalBlocker b(sb);
965 sb->setValue(v);
966 };
967 setV(m_cropX, m_crop.x()); setV(m_cropY, m_crop.y());
968 setV(m_cropW, m_crop.width()); setV(m_cropH, m_crop.height());
969 setV(m_pickXSpin, m_pick.x()); setV(m_pickYSpin, m_pick.y());
970}
