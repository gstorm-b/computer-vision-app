50bool MatchPattern::learnPattern() {
51 m_hasPatternLearned = false;
52 m_image = m_config.m_rawImage.clone();
53 if (m_image.empty())
54 return false;
55
56 const EdgeMatchConfig* ecfg = groupEdgeConfig();
57 if (!ecfg)
58 return false; // only EdgeBased is implemented
59
60 clearPatternData();
61 this->findTopLayer();
62 cv::buildPyramid(m_image, m_pyramids, m_topLayer);
63 m_borderColor = mean(m_image).val[0] < 128 ? 255 : 0;
64
65 const int pyr_size = static_cast<int>(m_pyramids.size());
66 m_patterns.resize(pyr_size);
67
68 std::vector<std::vector<cv::Point>> top_layer_contours;
69
70 for (int index = 0; index < pyr_size; ++index) {
71 PatternLayer pattern;
72 pattern.patternPoints.clear();
73 pattern.contours.clear();
74 pattern.hierarchies.clear();
75
76 Mat imgGray = m_pyramids.at(index).clone();
77 GaussianBlur(imgGray, imgGray,
78 cv::Size(ecfg->blurWidth, ecfg->blurHeight), 0);
79 Canny(imgGray, pattern.image,
80 ecfg->threshLower, ecfg->threshUpper, ecfg->kernelSize);
81
82 // find contours on canny output
83 findContours(pattern.image, pattern.contours, pattern.hierarchies,
84 cv::RETR_EXTERNAL, cv::CHAIN_APPROX_NONE);
85
86 Mat gx, gy, magnitude, angle;
87 Sobel(imgGray, gx, CV_32F, 1, 0, 3);
88 Sobel(imgGray, gy, CV_32F, 0, 1, 3);
89 cartToPolar(gx, gy, magnitude, angle, true);
90
91 pattern.angle = angle;
92 pattern.Magnitude = magnitude;
93
94 if (index == 0)
95 m_maskPatternPoint = cv::Mat(imgGray.size(), CV_8UC1, cv::Scalar(0));
96
97 for (int ci = 0; ci < static_cast<int>(pattern.contours.size()); ++ci) {
98 std::vector<cv::Point> resampled =
99 vsu::tNeighborSimplify(pattern.contours[ci], ecfg->tSamples);
100
101 for (int pi = 0; pi < static_cast<int>(resampled.size()); ++pi) {
102 PatternPoint pt;
103 pt.Coordinates = resampled[pi];
104 pt.Derivative = Point2f(gx.at<float>(pt.Coordinates),
105 gy.at<float>(pt.Coordinates));
106 pt.Magnitude = magnitude.at<float>(pt.Coordinates);
107 pattern.patternPoints.push_back(pt);
108 }
109
110 if (index == 0)
111 top_layer_contours.push_back(resampled);
112 }
113
114 pattern.freeMemory();
115 const std::size_t numPts = pattern.patternPoints.size();
116 pattern.allocateMemory(numPts);
117 for (int count = 0; count < static_cast<int>(numPts); ++count) {
118 PatternPoint* tp = &pattern.patternPoints[count];
119 pattern.pGx[count] = tp->Derivative.x;
120 pattern.pGy[count] = tp->Derivative.y;
121 pattern.pMag[count] = tp->Magnitude;
122 }
123
124 pattern.image = m_pyramids.at(index).clone();
125 m_patterns[index] = pattern;
126 }
127
128 if (!m_maskPatternPoint.empty())
129 cv::polylines(m_maskPatternPoint, top_layer_contours, true, cv::Scalar(255), 1);
130
131 // ── Pattern low workpieces contour selection (used for area-based pre-filter) ────────────
132
133
134 Mat threshImage;
135 vector<vector<Point>> maxSizeContours;
136 vector<Vec4i> pcaHierarchy;
137 Mat imgGray = m_image.clone();
138 int binary_threshold_style = cv::THRESH_BINARY;
139
140 GaussianBlur(imgGray, imgGray, cv::Size(ecfg->blurWidth, ecfg->blurHeight), 0);
141
142 if (ecfg->invertBinaryThreshold)
143 binary_threshold_style |= cv::THRESH_BINARY_INV;
144
145 if(ecfg->binaryThreshold >= 0) {
146 threshold(imgGray, threshImage, ecfg->binaryThreshold, ecfg->binaryMaxValue, binary_threshold_style);
147 } else {
148 if (ecfg->invertBinaryThreshold) {
149 binary_threshold_style = cv::THRESH_BINARY_INV + cv::THRESH_OTSU;
150 } else {
151 binary_threshold_style = cv::THRESH_BINARY + cv::THRESH_OTSU;
152 }
153 threshold(imgGray, threshImage, 0, 255, binary_threshold_style);
154 }
155 // threshold(imgGray, threshImage, 0, 255, binary_threshold_style);
156 findContours(threshImage, maxSizeContours, pcaHierarchy,
157 cv::RETR_TREE, cv::CHAIN_APPROX_NONE);
158
159 const int limitArea = static_cast<int>(imgGray.cols * imgGray.rows * 0.98);
160 int maxArea = 0;
161 for (int ci = 0; ci < static_cast<int>(maxSizeContours.size()); ++ci) {
162 int area = static_cast<int>(contourArea(maxSizeContours[ci]));
163 if (area >= limitArea) continue;
164 if (area > maxArea) {
165 maxArea = area;
166 m_contoursSelectIndex = ci;
167 m_contoursSelectArea = maxArea;
168 m_contoursSelectRect = cv::minAreaRect(maxSizeContours[ci]);
169 m_contoursSelectRectArea = static_cast<int>(
170 m_contoursSelectRect.size.height * m_contoursSelectRect.size.width);
171 }
172 }
173 m_maskImage = threshImage.clone();
174
175 m_hasPatternLearned = true;
176 return true;
177}

247cv::Mat MatchPattern::getImageWithCannyThreshold() {
248 if (m_image.empty()) return cv::Mat();
249
250 const EdgeMatchConfig* ecfg = groupEdgeConfig();
251 const int kernelSize = ecfg ? ecfg->kernelSize : 3;
252
253 double cannyRange = 50;
254 double CANNY_RANGE_MAX = 100;
255 double cannyLow, cannyHigh;
256
257 cv::Mat cannyImg = m_image.clone();
258 GaussianBlur(cannyImg, cannyImg, cv::Size(5, 5), 0);
259 double med = vsu::median(cannyImg);
260 double sigma = (0.1 / CANNY_RANGE_MAX) * cannyRange;
261 cannyLow = std::max(0.0, (1.0 - sigma) * med);
262 cannyHigh = std::min(255.0, (1.0 + sigma) * med);
263 Canny(cannyImg, cannyImg, cannyLow, cannyHigh, kernelSize, true);
264 return cannyImg;
265}

277void MatchPattern::findTopLayer() {
278 const EdgeMatchConfig* ecfg = groupEdgeConfig();
279 const int minLen = ecfg ? ecfg->minReduceLength : 32;
280 m_minReduceArea = minLen * minLen;
281 m_topLayer = 0;
282 int area = m_image.cols * m_image.rows;
283 while (area > m_minReduceArea) {
284 area /= 4;
285 ++m_topLayer;
286 }
287}
