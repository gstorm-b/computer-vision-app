25inline bool canTransitionTaskState(TaskState from, TaskState to)
26{
27 if (from == to) {
28 return true;
29 }
30
31 switch (from) {
32 case TaskState::Idle:
33 return to == TaskState::CommissionStarting ||
34 to == TaskState::RuntimeStarting ||
35 to == TaskState::Faulted;
36
37 case TaskState::CommissionStarting:
38 return to == TaskState::Commission ||
39 to == TaskState::Stopping ||
40 to == TaskState::Faulted;
41
42 case TaskState::Commission:
43 return to == TaskState::RuntimeStarting ||
44 to == TaskState::Stopping ||
45 to == TaskState::Faulted;
46
47 case TaskState::RuntimeStarting:
48 return to == TaskState::Ready ||
49 to == TaskState::Recovering ||
50 to == TaskState::Stopping ||
51 to == TaskState::Faulted;
52
53 case TaskState::Ready:
54 return to == TaskState::RunningCycle ||
55 to == TaskState::Recovering ||
56 to == TaskState::Stopping ||
57 to == TaskState::Faulted;
58
59 case TaskState::RunningCycle:
60 return to == TaskState::Ready ||
61 to == TaskState::Recovering ||
62 to == TaskState::Stopping ||
63 to == TaskState::Faulted;
64
65 case TaskState::Recovering:
66 return to == TaskState::Ready ||
67 to == TaskState::RunningCycle ||
68 to == TaskState::Stopping ||
69 to == TaskState::Faulted;
70
71 case TaskState::Faulted:
72 // Ready is reachable because a fault can be acknowledged (bErrorReset) or clear
73 // itself after LocalizationRuntimeController::kFaultAutoRecoverMs. Without this
74 // edge the PLC flags would clear while the task stayed Faulted, which reads as
75 // "acknowledged" to the robot and "still broken" to the operator. RunningCycle
76 // stays unreachable: recovery re-arms the runtime, it does not start a cycle.
77 return to == TaskState::Ready ||
78 to == TaskState::Stopping;
79
80 case TaskState::Stopping:
81 return to == TaskState::Idle ||
82 to == TaskState::Faulted;
83 }
84
85 return false;
86}
