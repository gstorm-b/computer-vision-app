50Eigen::Vector3d rpyFromPose(const Pose& pose)
51{
52 const Eigen::Matrix3d r = pose.isometry().linear();
53 const double pitch = std::asin(std::max(-1.0, std::min(1.0, -r(2, 0))));
54 const double roll = std::atan2(r(2, 1), r(2, 2));
55 const double yaw = std::atan2(r(1, 0), r(0, 0));
56 return Eigen::Vector3d(roll, pitch, yaw);
57}

84Result<std::string> UrdfAdapter::exportSerialRobot(const SerialRobotConfig& config)
85{
86 const ModelValidationResult validation = RobotModelValidator::validateSerialRobotConfig(config);
87 if (!validation.ok()) {
88 return Result<std::string>::failure(validation.status(), validation.issues.front().message);
89 }
90
91 std::ostringstream xml;
92 xml << "<robot name=\"" << config.identity.model << "\">\n";
93 for (const Link& link : config.links) {
94 xml << " <link name=\"" << link.id << "\"/>\n";
95 }
96 for (const Joint& joint : config.joints) {
97 const Eigen::Vector3d rpy = rpyFromPose(joint.origin);
98 xml << " <joint name=\"" << joint.id << "\" type=\"" << jointTypeName(joint.type) << "\">\n";
99 xml << " <parent link=\"" << joint.parentLinkId << "\"/>\n";
100 xml << " <child link=\"" << joint.childLinkId << "\"/>\n";
101 xml << " <origin xyz=\"" << vec3(joint.origin.translation_m()) << "\" rpy=\"" << vec3(rpy) << "\"/>\n";
102 if (joint.type != JointType::Fixed) {
103 xml << " <axis xyz=\"" << vec3(joint.axis) << "\"/>\n";
104 if (joint.limits.has_value()) {
105 xml << " <limit lower=\"" << joint.limits->lower << "\" upper=\"" << joint.limits->upper << "\"/>\n";
106 }
107 }
108 xml << " </joint>\n";
109 }
110 if (!config.posture.resolver.empty() || !config.tools.empty() || !config.sources.empty()) {
111 xml << " <!-- RobotKinematics metadata not represented in URDF: tools, posture, solver, sources. -->\n";
112 }
113 xml << "</robot>\n";
114 return Result<std::string>::success(xml.str());
115}

123Result<SerialRobotConfig> UrdfAdapter::importSerialRobot(const std::string& urdf,
124 const std::string& baseLinkId,
125 const std::string& flangeLinkId)
126{
127 SerialRobotConfig config;
128 config.topology = RobotTopologyType::Serial;
129 config.frames.baseLinkId = baseLinkId;
130 config.frames.flangeLinkId = flangeLinkId;
131
132 std::vector<Joint> parsedJoints;
133 QXmlStreamReader reader(QString::fromStdString(urdf));
134 while (!reader.atEnd()) {
135 reader.readNext();
136 if (!reader.isStartElement()) {
137 continue;
138 }
139 if (reader.name() == QStringLiteral("robot")) {
140 config.identity.model = reader.attributes().value("name").toString().toStdString();
141 } else if (reader.name() == QStringLiteral("link")) {
142 config.links.push_back(Link{reader.attributes().value("name").toString().toStdString()});
143 } else if (reader.name() == QStringLiteral("joint")) {
144 Joint joint;
145 joint.id = reader.attributes().value("name").toString().toStdString();
146 joint.type = jointTypeFromName(reader.attributes().value("type").toString());
147 while (!(reader.isEndElement() && reader.name() == QStringLiteral("joint")) && !reader.atEnd()) {
148 reader.readNext();
149 if (!reader.isStartElement()) {
150 continue;
151 }
152 if (reader.name() == QStringLiteral("parent")) {
153 joint.parentLinkId = reader.attributes().value("link").toString().toStdString();
154 } else if (reader.name() == QStringLiteral("child")) {
155 joint.childLinkId = reader.attributes().value("link").toString().toStdString();
156 } else if (reader.name() == QStringLiteral("origin")) {
157 const Eigen::Vector3d xyz = parseVec3(reader.attributes().value("xyz").toString(), Eigen::Vector3d::Zero());
158 const Eigen::Vector3d rpy = parseVec3(reader.attributes().value("rpy").toString(), Eigen::Vector3d::Zero());
159 joint.origin = Pose::fromXYZRPY_m_rad(xyz.x(), xyz.y(), xyz.z(), rpy.x(), rpy.y(), rpy.z());
160 } else if (reader.name() == QStringLiteral("axis")) {
161 joint.axis = parseVec3(reader.attributes().value("xyz").toString(), Eigen::Vector3d::UnitZ());
162 } else if (reader.name() == QStringLiteral("limit")) {
163 JointLimits limits;
164 limits.lower = reader.attributes().value("lower").toDouble();
165 limits.upper = reader.attributes().value("upper").toDouble();
166 joint.limits = limits;
167 }
168 }
169 if (joint.type != JointType::Fixed && !joint.limits.has_value()) {
170 joint.limits = JointLimits{-3.141592653589793, 3.141592653589793, std::nullopt, std::nullopt};
171 }
172 parsedJoints.push_back(joint);
173 }
174 }
175 if (reader.hasError()) {
176 return Result<SerialRobotConfig>::failure(KinematicsStatus::InvalidRobotConfig, "URDF XML parse error");
177 }
178
179 std::string current = baseLinkId;
180 while (current != flangeLinkId) {
181 auto next = std::find_if(parsedJoints.begin(), parsedJoints.end(), [&](const Joint& joint) {
182 return joint.parentLinkId == current;
183 });
184 if (next == parsedJoints.end()) {
185 return Result<SerialRobotConfig>::failure(KinematicsStatus::InvalidRobotConfig,
186 "URDF serial chain is incomplete");
187 }
188 config.joints.push_back(*next);
189 current = next->childLinkId;
190 }
191
192 int movable = 0;
193 for (const Joint& joint : config.joints) {
194 if (joint.type == JointType::Revolute || joint.type == JointType::Prismatic) {
195 ++movable;
196 }
197 }
198 config.dof = movable;
199
200 const ModelValidationResult validation = RobotModelValidator::validateSerialRobotConfig(config);
201 if (!validation.ok()) {
202 return Result<SerialRobotConfig>::failure(validation.status(), validation.issues.front().message);
203 }
204 return Result<SerialRobotConfig>::success(config);
205}
