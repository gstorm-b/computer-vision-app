290void PatternTreeWidget::addGroup(const MatchGroupConfig &group)
291{
292 m_groups.append(group);
293 const int gi = m_groups.size() - 1;
294
295 // Remove footer, insert new group before it, re-add footer
296 if (m_footerItem) {
297 takeTopLevelItem(indexOfTopLevelItem(m_footerItem));
298 delete m_footerItem;
299 m_footerItem = nullptr;
300 }
301 appendGroupItem(m_groups[gi], gi);
302 appendFooterItem();

304 emit groupsChanged(m_groups);

311void PatternTreeWidget::removeGroup(int groupIndex)
312{
313 groupIndex = mapGroupNumber(groupIndex);
314
315 if (groupIndex < 0 || groupIndex >= m_groups.size()) return;
316 m_groups.removeAt(groupIndex);
317 // Defer rebuild — the call may come from a signal emitted by a child widget
318 // that is still on the call stack; deleting it now would be unsafe.
319 QTimer::singleShot(0, this, [this]() {
320 rebuildTree();
321 emit groupsChanged(m_groups);
322 });

445void PatternTreeWidget::rebuildTree()
446{
447 const QSet<int> expanded = expandedGroupIndices();
448
449 // Detach footer before clearing so we can delete it cleanly
450 if (m_footerItem) {
451 takeTopLevelItem(indexOfTopLevelItem(m_footerItem));
452 delete m_footerItem;
453 m_footerItem = nullptr;
454 }
455 clear();
456
457 for (int gi = 0; gi < m_groups.size(); ++gi)
458 appendGroupItem(m_groups[gi], gi);
459
460 appendFooterItem();
461 restoreExpanded(expanded);
462}

466void PatternTreeWidget::appendGroupItem(const MatchGroupConfig &group, int groupIndex)
467{
468 auto *treeItem = new QTreeWidgetItem();
469 treeItem->setData(0, Qt::UserRole, QStringLiteral("group"));
470 treeItem->setData(0, Qt::UserRole + 1, groupIndex);
471 treeItem->setSizeHint(0, QSize(0, 38));
472
473 // Insert before footer (footer is always last)
474 const int pos = m_footerItem
475 ? indexOfTopLevelItem(m_footerItem)
476 : topLevelItemCount();
477 insertTopLevelItem(pos, treeItem);
478
479 auto *w = new PatternGroupItemWidget(group, this);
480 setItemWidget(treeItem, 0, w);
481
482 // ── Group widget → tree signals ──────────────────────────────────────────
483 connect(w, &PatternGroupItemWidget::clicked, this,
484 [this, group, w]() {
485 emit groupClicked(group.number, w->config());
486 });
487
488 connect(w, &PatternGroupItemWidget::addPatternRequested, this,
489 [this, group]() {
490 emit addPatternRequested(group.number);
491 });
492
493 connect(w, &PatternGroupItemWidget::deleteRequested, this,
494 [this, group]() {
495 emit deleteGroupRequested(group.number);
496 });
497
498 // ── Patterns ─────────────────────────────────────────────────────────────
499 for (int pi = 0; pi < group.patterns.size(); ++pi)
500 appendPatternItem(treeItem, group.patterns[pi], groupIndex, pi);
501}
