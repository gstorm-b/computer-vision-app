109std::vector<cv::Point2f> CalibrationPointsTable::imagePoints() const {
110 std::vector<cv::Point2f> out;
111 out.reserve(rowCount());
112 for (int r = 0; r < rowCount(); ++r) {
113 bool okX = false, okY = false;
114 float x = item(r, kColImgX)->text().toFloat(&okX);
115 float y = item(r, kColImgY)->text().toFloat(&okY);
116 if (okX && okY) {
117 out.emplace_back(x, y);
118 } else {
119 out.emplace_back(0.f, 0.f);
120 }
121 }
122 return out;
