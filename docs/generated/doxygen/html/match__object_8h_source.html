161 bool checkCollisionObject(std::vector<std::vector<cv::Point>> &contours, std::vector<int> &indexes) {
162 m_hasCollision = false;
163 for(int idx=0;idx<indexes.size();idx++) {
164 std::vector<cv::Point>& con = contours.at(indexes[idx]);
165 for(int con_idx=0;con_idx<con.size();con_idx++) {
166 if(cv::pointPolygonTest(m_collisionGeometry.box_left_pts, con.at(con_idx), false) >= 0) {
167 m_hasCollision = true;
168 goto _ret_point;
169 }
170
171 if(cv::pointPolygonTest(m_collisionGeometry.box_right_pts, con.at(con_idx), false) >= 0) {
172 m_hasCollision = true;
173 goto _ret_point;
174 }
175 }
176 }
177
178 _ret_point:
179 return m_hasCollision;

198 CollisionState checkCollisionObject2(const cv::Mat &contourMask) {
199 m_collisionState = CollisionState::Outside;
200 m_hasCollision = false;
201 if (contourMask.empty()) {
202 return m_collisionState;
203 }
204
205 // Union bounding box of both jaw quads, clamped to the mask. Jaws may
206 // legitimately extend off-image; the off-image part cannot overlap any
207 // contour, so clamping is safe.
208 std::vector<cv::Point2f> allPts;
209 allPts.reserve(m_collisionGeometry.box_left_pts.size() + m_collisionGeometry.box_right_pts.size());
210 allPts.insert(allPts.end(), m_collisionGeometry.box_left_pts.begin(), m_collisionGeometry.box_left_pts.end());
211 allPts.insert(allPts.end(), m_collisionGeometry.box_right_pts.begin(), m_collisionGeometry.box_right_pts.end());
212 if (allPts.empty()) {
213 return m_collisionState;
214 }
215
216 cv::Rect bbox = cv::boundingRect(allPts) & cv::Rect(0, 0, contourMask.cols, contourMask.rows);
217 if (bbox.width <= 0 || bbox.height <= 0) {
218 return m_collisionState;
219 }
220
221 // Rasterise the two jaws into a bbox-sized mask, shifted into
222 // crop-local coordinates. Each jaw is a convex quad.
223 cv::Mat gripperCrop = cv::Mat::zeros(bbox.size(), CV_8UC1);
224 const cv::Point shift = bbox.tl();
225 fillJawInto(gripperCrop, m_collisionGeometry.box_left_pts, shift);
226 fillJawInto(gripperCrop, m_collisionGeometry.box_right_pts, shift);
227
228 // contourMask(bbox) is a view (no copy); AND it with the jaw mask.
229 cv::Mat inter;
230 cv::bitwise_and(gripperCrop, contourMask(bbox), inter);
231
232 const int interCount = cv::countNonZero(inter);
233 const int gripperCount = cv::countNonZero(gripperCrop);
234
235 if (interCount == 0) {
236 m_collisionState = CollisionState::Outside;
237 } else if (gripperCount > 0 && interCount == gripperCount) {
238 m_collisionState = CollisionState::Inside;
239 } else {
240 m_collisionState = CollisionState::Collision;
241 }
242
243 m_hasCollision = (m_collisionState != CollisionState::Outside);
244 return m_collisionState;
245 }

301 static void fillJawInto(cv::Mat &mask, const std::vector<cv::Point2f> &pts,
302 const cv::Point &shift) {
303 if (pts.size() < 3) {
304 return;

318 void drawMaskIntoImage(cv::Mat &image, cv::Mat &mask, cv::Scalar color) {
319 if (mask.empty() || image.empty()) {
320 return;
321 }
322
323 // wrap always counter clockwise
324 double angle = -matched_Angle;
325
326 // rotate mask
327 cv::Mat rotMat = cv::getRotationMatrix2D(cv::Point2f(0,0), angle, 1.0);
328
329 // get mask bounding box
330 std::vector<cv::Point2f> corners = {
331 cv::Point2f(0,0),
332 cv::Point2f(mask.cols,0),
333 cv::Point2f(mask.cols, mask.rows),
334 cv::Point2f(0,mask.rows)
335 };
336
337 // transform rotated box
338 std::vector<cv::Point2f> rotatedCorners;
339 cv::transform(corners, rotatedCorners, rotMat);
340
341 cv::Rect2f bbox = cv::boundingRect(rotatedCorners);
342
343 rotMat.at<double>(0,2) -= bbox.x;
344 rotMat.at<double>(1,2) -= bbox.y;
345
346 cv::Mat rotatedMask;
347 cv::warpAffine(mask, rotatedMask, rotMat, bbox.size(),
348 cv::INTER_NEAREST, cv::BORDER_CONSTANT, cv::Scalar(0));
349
350 // get raw bouding box
351 std::vector<cv::Point2f> raw_corners = {
352 cv::Point2f(point_LT),
353 cv::Point2f(point_LB),
354 cv::Point2f(point_RT),
355 cv::Point2f(point_RB)
356 };
357 cv::Rect raw_bouding = cv::boundingRect(raw_corners);
358
359 cv::Rect roi(raw_bouding.x, raw_bouding.y, rotatedMask.cols, rotatedMask.rows);
360 roi = roi & cv::Rect(0, 0, image.cols, image.rows);
361
362 cv::Mat roiImg = image(roi);
363 cv::Point mask_new_tf((raw_bouding.x < 0) ? -raw_bouding.x : 0,
364 (raw_bouding.y < 0) ? -raw_bouding.y : 0);
365 cv::Mat maskROI = rotatedMask(cv::Rect(mask_new_tf.x, mask_new_tf.y,roi.width,roi.height));
