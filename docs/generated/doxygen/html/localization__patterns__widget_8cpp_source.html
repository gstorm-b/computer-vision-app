60cv::Mat pixmapToMat(const QPixmap &pixmap) {
61 const QImage qimg = pixmap.toImage();
62 if (qimg.isNull()) return cv::Mat();
63
64 switch (qimg.format()) {
65 case QImage::Format_Grayscale8: {
66 cv::Mat mat(qimg.height(), qimg.width(), CV_8UC1,
67 const_cast<uchar*>(qimg.bits()),
68 static_cast<size_t>(qimg.bytesPerLine()));
69 return mat.clone();
70 }
71 case QImage::Format_RGB888: {
72 cv::Mat mat(qimg.height(), qimg.width(), CV_8UC3,
73 const_cast<uchar*>(qimg.bits()),
74 static_cast<size_t>(qimg.bytesPerLine()));
75 cv::Mat bgr;
76 cv::cvtColor(mat, bgr, cv::COLOR_RGB2BGR);
77 return bgr;
78 }
79 default: {
80 const QImage converted = qimg.convertToFormat(QImage::Format_RGBA8888);
81 cv::Mat mat(converted.height(), converted.width(), CV_8UC4,
82 const_cast<uchar*>(converted.bits()),
83 static_cast<size_t>(converted.bytesPerLine()));
84 cv::Mat bgra;
85 cv::cvtColor(mat, bgra, cv::COLOR_RGBA2BGRA);
86 return bgra;
87 }
88 }

166QPixmap LocalizationPatternsWidget::matToPixmap(const cv::Mat &mat) {
167 if (mat.empty()) return {};
168
169 QImage img;
170 if (mat.type() == CV_8UC1) {
171 img = QImage(mat.data, mat.cols, mat.rows,
172 static_cast<int>(mat.step),
173 QImage::Format_Grayscale8).copy();
174 } else if (mat.type() == CV_8UC3) {
175 cv::Mat rgb;
176 cv::cvtColor(mat, rgb, cv::COLOR_BGR2RGB);
177 img = QImage(rgb.data, rgb.cols, rgb.rows,
178 static_cast<int>(rgb.step),
179 QImage::Format_RGB888).copy();
180 } else if (mat.type() == CV_8UC4) {
181 cv::Mat rgba;
182 cv::cvtColor(mat, rgba, cv::COLOR_BGRA2RGBA);
183 img = QImage(rgba.data, rgba.cols, rgba.rows,
184 static_cast<int>(rgba.step),
185 QImage::Format_RGBA8888).copy();
186 } else {
187 return {};
188 }
189 return QPixmap::fromImage(img);
190}

563void LocalizationPatternsWidget::wireManagerSignals() {
564 if (!m_patternManager) return;
565
566 // ── Initial population ──────────────────────────────────────────────────
567 // The tree is empty when widget is constructed. Push current manager
568 // state into it so existing groups/patterns (loaded from project JSON)
569 // appear immediately — without this, the user sees an empty library.
570 rebuildTreeFromManager();
571
572 // ── Group added ────────────────────────────────────────────────────────
573 connect(m_patternManager, &mtc::PatternGroupManager::groupAdded,
574 this, [this] (std::shared_ptr<mtc::MatchGroup> group) {
575 if (!group) return;
576 MatchGroupConfig gr;
577 gr.name = QString::fromStdWString(group->name());
578 gr.number = group->number();
579 // // Mirror pre-existing patterns so the tree shows complete state.
580 for (const auto &p : group->patterns()) {
581 MatchPatternConfig pc;
582 pc.name = QString::fromStdWString(p->name());
583 pc.number = p->number();
584 pc.threshScore = p->config().m_minScore;
585 pc.thumbnail = matToPixmap(p->config().m_rawImage);
586 gr.patterns.append(pc);
587 }
588 ui->treeWidget_patternEditor->addGroup(gr);
589 rebuildGroupCombo();
590 updateGroupsCount();
591 });
592
593 // ── Group removed ──────────────────────────────────────────────────────
594 connect(m_patternManager, &mtc::PatternGroupManager::groupRemoved,
595 this, [this] (const mtc::MatchGroupConfig &removed) {
596 // Mirror the removal in the tree so the row disappears.
597 ui->treeWidget_patternEditor->removeGroup(removed.m_groupIndex);
598
599 // If the removed group was selected, clear selection panel.
600 if (m_selectedGroupIndex == removed.m_groupIndex) clearSelection();
601
602 rebuildGroupCombo();
603 updateGroupsCount();
604 });
605
606 // ── Group changed (rename/renumber/picking-box edit) ───────────────────
607 // Fires for any successful PatternGroupManager group mutation. The
608 // signal does not carry the *previous* identity of the group, so the
609 // safest refresh is a full tree rebuild (it preserves expansion state).
610 connect(m_patternManager, &mtc::PatternGroupManager::groupChanged,
611 this, [this] (std::shared_ptr<mtc::MatchGroup> group, const QString &) {
612 if (!group) return;
613
614 // Keep our cached selection in sync if the currently-bound group
615 // was the one that changed. The pointer is stable across edits
616 // (setConfig updates in place), so identity tracking via raw
617 // pointer is reliable here.
618 if (m_boundMatchGroup == group.get())
619 m_selectedGroupIndex = group->number();
620
621 rebuildTreeFromManager();
622 rebuildGroupCombo();
623
624 // rebuildGroupCombo() restores selection by previous currentData()
625 // (the old group number), which won't match after a renumber.
626 // Re-select explicitly by the cached (possibly updated) number.
627 const int idx = ui->comboBox_pattern_group->findData(m_selectedGroupIndex);
628 if (idx >= 0) {
629 QSignalBlocker b(ui->comboBox_pattern_group);
630 ui->comboBox_pattern_group->setCurrentIndex(idx);
631 }
632 });
633
634 // ── Pattern added ──────────────────────────────────────────────────────
635 connect(m_patternManager, &mtc::PatternGroupManager::patternAdded,
636 this, [this] (mtc::MatchGroup *group, mtc::MatchPattern *pattern) {

1797bool LocalizationPatternsWidget::commitWorkingPatternConfig() {
1798 // Commit m_workingPatternCfg (pattern "Common" params) through the manager
1799 // so the live MatchPattern picks up the change.
1800 if (!m_patternManager || !m_boundPattern) return false;
1801 if (m_selectedGroupIndex < 0) return false;
1802
1803 auto group = m_patternManager->findGroupByNumber(m_selectedGroupIndex);
1804 if (!group) return false;
1805
1806 const QString groupName = QString::fromStdWString(group->name());
1807 const QString patternKey = QString::fromStdWString(m_boundPattern->name());
1808
1809 auto r = m_patternManager->setPatternConfig(
1810 groupName, patternKey, m_workingPatternCfg);
1811 if (!r) {
1812 QMessageBox::warning(this, tr("Update failed"), resultMessage(r));
1813
1814 // Roll the working copy back to the live config and refresh the Common
1815 // editors. PropSpecHelper::refresh blocks manager signals internally.
1816 m_workingPatternCfg = m_boundPattern->config();
1817 PropSpecHelper::refresh(m_variantManager, kCommonSpecs,
1818 m_workingPatternCfg, m_commonProps);
1819 return false;
1820 }
1821
1822 // Re-resolve in case the rename/renumber changed identity. The tree
1823 // row and pattern thumb are refreshed by the patternChanged listener.
1824 m_boundPattern = group->findPatternByName(
1825 m_workingPatternCfg.m_patternName);
1826 return true;
1827}
1828
1832bool LocalizationPatternsWidget::commitWorkingGroupConfig() {
1833 // Commit m_workingGroupConfig through the manager. Shared by the Group
1834 // Settings + Edge-Based property groups and the binary-view controls.
1835 if (!m_patternManager || !m_boundMatchGroup) return false;
1836
1837 // Snapshot the current name — a dispatch may already have written a new
1838 // name into m_workingGroupConfig, but the live MatchGroup still carries the
1839 // old name, which is the lookup key for setGroupConfig.
1840 const QString oldName = QString::fromStdWString(m_boundMatchGroup->name());
1841
1842 auto r = m_patternManager->setGroupConfig(oldName, m_workingGroupConfig);
1843 if (!r) {
1844 QMessageBox::warning(this, tr("Update failed"), resultMessage(r));
1845
1846 // Roll back to the live state and refresh every editor bound to the
1847 // group config. refresh() blocks manager signals internally.
1848 m_workingGroupConfig = m_boundMatchGroup->config();
1849 PropSpecHelper::refresh(m_variantManager, kMatchGroupSpecs,
1850 m_workingGroupConfig, m_groupProps);
1851 if (m_configAdapter) m_configAdapter->refresh();
1852 seedBinaryControlsFromConfig();
1853 return false;
1854 }
1855
1856 // Success. Tree row, combo, and m_selectedGroupIndex are refreshed by the
1857 // PatternGroupManager::groupChanged listener (wireManagerSignals).
1858 return true;
1859}
