23bool FanucIRvisionBoard::Params::isValid(std::string* errorOut) const
24{
25 auto fail = [errorOut](const char* msg) {
26 if (errorOut) *errorOut = msg;
27 return false;
28 };
29
30 if (rows < 5) return fail("rows must be >= 5");
31 if (cols < 7) return fail("cols must be >= 7");
32 if (rows % 2 == 0 || cols % 2 == 0) return fail("rows and cols must be odd");
33 if (dotSpacingMm <= 0.0) return fail("dotSpacingMm must be > 0");
34 if (marginMm < 0.0) return fail("marginMm must be >= 0");
35 if (innerTargetDotSizeMm <= 0.0) return fail("innerTargetDotSizeMm must be > 0");
36 if (normalDotSizeMm <= innerTargetDotSizeMm)
37 return fail("normalDotSizeMm must be greater than innerTargetDotSizeMm");
38 if (coorDotSizeMm <= normalDotSizeMm)
39 return fail("coorDotSizeMm must be greater than normalDotSizeMm");
40 if (coorDotSizeMm >= dotSpacingMm)
41 return fail("coorDotSizeMm must be smaller than dotSpacingMm");
42 if (binarizeThreshold < -1 || binarizeThreshold > 255)
43 return fail("binarizeThreshold must be -1 (auto/Otsu) or in 0..255");
44 return true;
45}

151{
152 const auto centre = centerCell();

154 for (const auto& g : coordCells()) if (g.row == row && g.col == col) return DotRole::Coord;

245cv::Mat FanucIRvisionBoard::generateImage(double pixelsPerMm) const
246{
247 const double widthMm = (m_params.cols - 1) * m_params.dotSpacingMm + 2.0 * m_params.marginMm;
248 const double heightMm = (m_params.rows - 1) * m_params.dotSpacingMm + 2.0 * m_params.marginMm;
249
250 const int widthPx = static_cast<int>(std::round(widthMm * pixelsPerMm));
251 const int heightPx = static_cast<int>(std::round(heightMm * pixelsPerMm));
252
253 cv::Mat img(heightPx, widthPx, CV_8UC1, cv::Scalar(255));
254
255 const int rNormalPx = std::max(2,
256 static_cast<int>(std::round(m_params.normalDotSizeMm * pixelsPerMm * 0.5)));
257 const int rCoorPx = std::max(2,
258 static_cast<int>(std::round(m_params.coorDotSizeMm * pixelsPerMm * 0.5)));
259 const int rInnerPx = std::max(1,
260 static_cast<int>(std::round(m_params.innerTargetDotSizeMm * pixelsPerMm * 0.5)));
261
262 for (int r = 0; r < m_params.rows; ++r) {
263 for (int c = 0; c < m_params.cols; ++c) {
264 const double xMm = m_params.marginMm + c * m_params.dotSpacingMm;
265 const double yMm = m_params.marginMm + r * m_params.dotSpacingMm;
266 const cv::Point2f center(static_cast<float>(xMm * pixelsPerMm),
267 static_cast<float>(yMm * pixelsPerMm));
268
269 switch (roleOf(r, c)) {
270 case DotRole::Normal:
271 cv::circle(img, center, rNormalPx, cv::Scalar(0), cv::FILLED, cv::LINE_AA);
272 break;
273 case DotRole::Target:
274 cv::circle(img, center, rNormalPx, cv::Scalar(0), cv::FILLED, cv::LINE_AA);
275 cv::circle(img, center, rInnerPx, cv::Scalar(255), cv::FILLED, cv::LINE_AA);
276 break;
277 case DotRole::Center:
278 cv::circle(img, center, rCoorPx, cv::Scalar(0), cv::FILLED, cv::LINE_AA);
279 cv::circle(img, center, rInnerPx, cv::Scalar(255), cv::FILLED, cv::LINE_AA);
280 break;
281 case DotRole::Coord:
282 cv::circle(img, center, rCoorPx, cv::Scalar(0), cv::FILLED, cv::LINE_AA);
283 break;
284 }
285 }
286 }
287 return img;
288}

331bool identifyOrientationMarkers4(const std::vector<cv::Point2f>& m,
332 cv::Point2f& centre,
333 cv::Point2f& xMid,
334 cv::Point2f& xTip,
335 cv::Point2f& yTip)
336{
337 if (m.size() != 4) return false;
338
339 double bestResidual = std::numeric_limits<double>::max();
340 int yIdx = -1;
341 std::array<int, 3> triple{ -1, -1, -1 };
342
343 for (int yi = 0; yi < 4; ++yi) {
344 std::array<int, 3> other{};
345 int k = 0;
346 for (int i = 0; i < 4; ++i) if (i != yi) other[k++] = i;
347
348 const cv::Point2f A = m[other[0]];
349 const cv::Point2f B = m[other[1]];
350 const cv::Point2f C = m[other[2]];
351 const cv::Point2f AB = B - A;
352 const cv::Point2f AC = C - A;
353 const double denom = std::max(1e-3, cv::norm(AB));
354 const double residual = std::abs(crossZ(AB, AC)) / denom;
355 if (residual < bestResidual) {
356 bestResidual = residual;
357 yIdx = yi;
358 triple = other;
359 }
360 }
361 if (yIdx < 0) return false;
362 yTip = m[yIdx];
363
364 const cv::Point2f A = m[triple[0]];
365 const cv::Point2f B = m[triple[1]];
366 const cv::Point2f C = m[triple[2]];
367 const double dAB = cv::norm(A - B);
368 const double dBC = cv::norm(B - C);
369 const double dAC = cv::norm(A - C);
370
371 int midIdx = -1, end1Idx = -1, end2Idx = -1;
372 if (dAB >= dBC && dAB >= dAC) { midIdx = triple[2]; end1Idx = triple[0]; end2Idx = triple[1]; }
373 else if (dAC >= dBC) { midIdx = triple[1]; end1Idx = triple[0]; end2Idx = triple[2]; }
374 else { midIdx = triple[0]; end1Idx = triple[1]; end2Idx = triple[2]; }
375
376 xMid = m[midIdx];
377
378 const double d1 = cv::norm(m[end1Idx] - yTip);
379 const double d2 = cv::norm(m[end2Idx] - yTip);
380 if (d1 < d2) { centre = m[end1Idx]; xTip = m[end2Idx]; }
381 else { centre = m[end2Idx]; xTip = m[end1Idx]; }
382
383 const cv::Point2f mid = (centre + xTip) * 0.5f;
384 const double midErr = cv::norm(xMid - mid);
385 const double xSpan = cv::norm(xTip - centre);
386 if (xSpan < 1e-3 || midErr / xSpan > 0.25) return false;
387 return true;
388}

401bool FanucIRvisionBoard::detect(const cv::Mat& image,
402 std::vector<cv::Point2f>& imagePoints,
403 std::vector<cv::Point2f>* cornerImagePoints,
404 cv::Mat* debugOverlay) const
405{
406 imagePoints.clear();
407 if (cornerImagePoints) cornerImagePoints->clear();
408 if (image.empty()) return false;
409
410 const cv::Mat gray = ensureGray(image);
411
412 // Binarize with the configured threshold (auto/Otsu when binarizeThreshold
413 // < 0, fixed manual value otherwise). Shared with the tuning dialog preview.
414 const cv::Mat binary = binarize(gray);
415
416 std::vector<std::vector<cv::Point>> contours;
417 std::vector<cv::Vec4i> hierarchy;
418 cv::findContours(binary, contours, hierarchy, cv::RETR_CCOMP, cv::CHAIN_APPROX_SIMPLE);
419
420 std::vector<Blob> blobs;
421 blobs.reserve(contours.size());
422 for (int i = 0; i < static_cast<int>(contours.size()); ++i) {
423 if (hierarchy[i][3] != -1) continue;
424 const auto& c = contours[i];
425 if (c.size() < 5) continue;
426 const cv::Moments mm = cv::moments(c);
427 if (mm.m00 <= 4.0) continue;
428 const double area = mm.m00;
429 const double perim = cv::arcLength(c, true);
430 if (perim <= 1.0) continue;
431 const double circularity = 4.0 * PI_NUM * area / (perim * perim);
432 if (circularity < 0.55) continue;
433
434 // Reject non-dot contours by bounding-box squareness. The big surrounding
435 // background region (e.g. table outside the board) traces a rectangular
436 // outer boundary with image-wide bbox, so its w/h ratio is far from 1
437 // even when its circularity sneaks past 0.55.
438 const cv::Rect bbox = cv::boundingRect(c);
439 if (bbox.width < 3 || bbox.height < 3) continue;
440 const int wMax = std::max(bbox.width, bbox.height);
441 const int wMin = std::min(bbox.width, bbox.height);
442 if (static_cast<double>(wMax) / wMin > 1.4) continue;
443
444 const cv::Point2f centroid(static_cast<float>(mm.m10 / mm.m00),
445 static_cast<float>(mm.m01 / mm.m00));
446 blobs.push_back({ centroid, area });
447 }
448
449 const int expectedTotal = totalDots();
450 if (static_cast<int>(blobs.size()) < expectedTotal) return false;
451
452 std::sort(blobs.begin(), blobs.end(),
453 [](const Blob& a, const Blob& b) { return a.area > b.area; });
454
455 // Median-area defence: small dots dominate the lower half of the sorted
456 // area list. Any blob >> 12x the median is almost certainly background or
457 // a stray label; drop it before picking the 4 orientation markers.
458 {
459 const size_t mid = blobs.size() / 2 + (blobs.size() / 4); // ~75th-percentile-from-top -> small dots
460 const double medianSmallArea = blobs[std::min(mid, blobs.size() - 1)].area;
461 const double areaCap = medianSmallArea * 12.0;
462 blobs.erase(std::remove_if(blobs.begin(), blobs.end(),
463 [areaCap](const Blob& b) { return b.area > areaCap; }),
464 blobs.end());
465 }
466
467 if (static_cast<int>(blobs.size()) < expectedTotal) return false;
468 if (blobs[0].area > 3.0 * blobs[3].area) return false;
469
470 const std::vector<cv::Point2f> large4{
471 blobs[0].center, blobs[1].center, blobs[2].center, blobs[3].center
472 };
473
474 cv::Point2f centrePt, xMidPt, xTipPt, yTipPt;
475 if (!identifyOrientationMarkers4(large4, centrePt, xMidPt, xTipPt, yTipPt))
476 return false;
477
478 const int rmid = (m_params.rows - 1) / 2;
479 const int cmid = (m_params.cols - 1) / 2;
480
481 const cv::Point2f vCol = (xTipPt - centrePt) * 0.5f;
482 const cv::Point2f vRow = (centrePt - yTipPt);
483 if (cv::norm(vCol) < 1e-3 || cv::norm(vRow) < 1e-3) return false;
484
485 auto projectGridCell = [&](int row, int col) -> cv::Point2f {
486 return centrePt
487 + static_cast<float>(col - cmid) * vCol
488 + static_cast<float>(row - rmid) * vRow;
489 };
490
491 std::vector<cv::Point2f> expectedImg;
492 expectedImg.reserve(expectedTotal);
493 for (int r = 0; r < m_params.rows; ++r)
494 for (int c = 0; c < m_params.cols; ++c)
495 expectedImg.push_back(projectGridCell(r, c));
496
497 imagePoints.assign(expectedTotal, cv::Point2f(-1.f, -1.f));
498 std::vector<float> bestDist(expectedTotal, std::numeric_limits<float>::max());
499 const float maxAcceptDist = static_cast<float>(std::min(cv::norm(vCol), cv::norm(vRow)) * 0.45);
500
501 for (const auto& b : blobs) {
502 int bestIdx = -1;
503 float bestD = maxAcceptDist;
504 for (int i = 0; i < expectedTotal; ++i) {
505 const float dd = static_cast<float>(cv::norm(b.center - expectedImg[i]));
506 if (dd < bestD) { bestD = dd; bestIdx = i; }
507 }
508 if (bestIdx >= 0 && bestD < bestDist[bestIdx]) {
509 bestDist[bestIdx] = bestD;
510 imagePoints[bestIdx] = b.center;
511 }
512 }
513
514 for (const auto& p : imagePoints) {
515 if (p.x < 0.f) { imagePoints.clear(); return false; }
516 }
517
518 if (cornerImagePoints) {
519 const auto cells = cornerCells();
520 cornerImagePoints->reserve(4);
521 for (const auto& g : cells)
522 cornerImagePoints->push_back(imagePoints[g.row * m_params.cols + g.col]);
523 }
524
525 if (debugOverlay) {
526 cv::cvtColor(gray, *debugOverlay, cv::COLOR_GRAY2BGR);
527 for (const auto& p : imagePoints)
528 cv::circle(*debugOverlay, p, 3, cv::Scalar(0, 255, 0), 1, cv::LINE_AA);
529 cv::circle(*debugOverlay, centrePt, 14, cv::Scalar(0, 0, 255), 2);
530 cv::circle(*debugOverlay, xMidPt, 10, cv::Scalar(0, 165, 255), 2);
531 cv::circle(*debugOverlay, xTipPt, 10, cv::Scalar(0, 255, 0), 2);
532 cv::circle(*debugOverlay, yTipPt, 10, cv::Scalar(255, 0, 0), 2);
533 cv::putText(*debugOverlay, "O", centrePt + cv::Point2f(16, -12),
534 cv::FONT_HERSHEY_SIMPLEX, 0.6, cv::Scalar(0, 0, 255), 2);
535 cv::putText(*debugOverlay, "+X", xTipPt + cv::Point2f(14, -10),
536 cv::FONT_HERSHEY_SIMPLEX, 0.6, cv::Scalar(0, 255, 0), 2);
537 cv::putText(*debugOverlay, "+Y", yTipPt + cv::Point2f(14, -10),
538 cv::FONT_HERSHEY_SIMPLEX, 0.6, cv::Scalar(255, 0, 0), 2);
539
540 if (cornerImagePoints) {
541 const std::array<const char*, 4> lbl{ "TL", "TR", "BR", "BL" };
542 for (int i = 0; i < 4; ++i) {
543 cv::circle(*debugOverlay, (*cornerImagePoints)[i], 16,
544 cv::Scalar(255, 255, 0), 2);
545 cv::putText(*debugOverlay, lbl[i],
546 (*cornerImagePoints)[i] + cv::Point2f(18, 0),
547 cv::FONT_HERSHEY_SIMPLEX, 0.6, cv::Scalar(255, 255, 0), 2);
548 }
549 }
550 }
551
552 return true;
553}
