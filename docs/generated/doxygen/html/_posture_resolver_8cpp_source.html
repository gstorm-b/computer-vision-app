23bool applyLabel(const PostureMetadata& metadata,
24 const std::map<std::string, std::string>& labels,
25 const std::string& axis,
26 std::optional<int>& target)
27{
28 const auto requested = labels.find(axis);
29 if (requested == labels.end()) {
30 return true;
31 }
32 const auto configured = metadata.labels.find(axis);
33 if (configured == metadata.labels.end()) {
34 return false;
35 }
36 if (requested->second == configured->second.negative) {
37 target = -1;
38 return true;
39 }
40 if (requested->second == configured->second.positive) {
41 target = 1;
42 return true;
43 }
44 return false;
45}

53Result<ArmPosture> Serial6DofSignPostureResolver::classify(const SerialRobotConfig&,
54 const JointVector& joints) const
55{
56 if (joints.size() < 6) {
57 return Result<ArmPosture>::failure(KinematicsStatus::JointDimensionMismatch,
58 "serial 6dof posture classification requires at least 6 joints");
59 }
60
61 ArmPosture posture;
62 posture.shoulder = branchSign(joints[0]);
63 posture.elbow = branchSign(joints[2]);
64 posture.wrist = branchSign(joints[4]);
65 return Result<ArmPosture>::success(posture);
66}

75Result<ArmPosture> Serial6DofSignPostureResolver::fromLabels(
76 const PostureMetadata& metadata,
77 const std::map<std::string, std::string>& labels) const
78{
79 ArmPosture posture;
80 if (!applyLabel(metadata, labels, "shoulder", posture.shoulder)
81 || !applyLabel(metadata, labels, "elbow", posture.elbow)
82 || !applyLabel(metadata, labels, "wrist", posture.wrist)) {
83 return Result<ArmPosture>::failure(KinematicsStatus::PostureConstraintUnsatisfied,
84 "posture label is not configured");
85 }
86 return Result<ArmPosture>::success(posture);
