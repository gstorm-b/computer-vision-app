102std::string poseKey(const Pose& pose)
103{
104 const Eigen::Isometry3d& iso = pose.isometry();
105 std::ostringstream key;
106 for (int row = 0; row < 3; ++row) {
107 for (int col = 0; col < 3; ++col) {
108 appendDouble(key, iso.linear()(row, col));
109 key << ",";
110 }
111 }
112 for (int axis = 0; axis < 3; ++axis) {
113 appendDouble(key, iso.translation()(axis));
114 key << ",";
115 }
116 return key.str();

209Result<std::shared_ptr<CachedCoalGeometry>> loadCoalGeometry(const MeshCollisionGeometry& mesh)
210{
211 if (mesh.format != MeshFileFormat::Stl) {
212 return Result<std::shared_ptr<CachedCoalGeometry>>::failure(
213 KinematicsStatus::InvalidRequest,
214 "coal mesh backend currently supports STL meshes only");
215 }
216
217 const std::string resolvedPath = resolveMeshPath(mesh.path);
218 const std::string cacheKey = geometryCacheKey(mesh, resolvedPath);
219
220 auto& cache = geometryCache();
221 const auto cached = cache.find(cacheKey);
222 if (cached != cache.end()) {
223 return Result<std::shared_ptr<CachedCoalGeometry>>::success(cached->second);
224 }
225
226 // Real-world CAD STL exports often contain a handful of degenerate triangles.
227 // Skip them silently so production loads do not fail on otherwise-valid assets;
228 // Coal's BVH builder also expects non-degenerate faces.
229 const Result<TriangleMesh> loaded =
230 StlMeshLoader::loadFile(resolvedPath, StlMeshLoadOptions{mesh.scaleToMeters, false});
231 if (!loaded.ok()) {
232 return Result<std::shared_ptr<CachedCoalGeometry>>::failure(
233 loaded.status,
234 "failed to load mesh '" + mesh.id + "' from '" + resolvedPath + "': " + loaded.message);
235 }
236
237 std::vector<coal::Vec3s> vertices;
238 vertices.reserve(loaded.value.vertices_m.size());
239 for (const Eigen::Vector3d& vertex : loaded.value.vertices_m) {
240 vertices.push_back(toCoalVector(vertex));
241 }
242
243 std::vector<coal::Triangle32> triangles;
244 triangles.reserve(loaded.value.faces.size());
245 for (const TriangleFace& face : loaded.value.faces) {
246 triangles.emplace_back(static_cast<coal::Triangle32::IndexType>(face.a),
247 static_cast<coal::Triangle32::IndexType>(face.b),
248 static_cast<coal::Triangle32::IndexType>(face.c));
249 }
250
251 auto model = std::make_shared<CoalBvh>();
252 if (model->beginModel(static_cast<unsigned int>(triangles.size()),
253 static_cast<unsigned int>(vertices.size())) != 0) {
254 return Result<std::shared_ptr<CachedCoalGeometry>>::failure(
255 KinematicsStatus::NumericalError,
256 "coal beginModel failed for mesh '" + mesh.id + "'");
257 }
258 if (model->addSubModel(vertices, triangles) != 0) {
259 return Result<std::shared_ptr<CachedCoalGeometry>>::failure(
260 KinematicsStatus::NumericalError,
261 "coal addSubModel failed for mesh '" + mesh.id + "'");
262 }
263 if (model->endModel() != 0) {
264 return Result<std::shared_ptr<CachedCoalGeometry>>::failure(
265 KinematicsStatus::NumericalError,
266 "coal endModel failed for mesh '" + mesh.id + "'");
267 }
268
269 auto cachedGeometry = std::make_shared<CachedCoalGeometry>();
270 cachedGeometry->resolvedPath = resolvedPath;
271 cachedGeometry->statistics = loaded.value.statistics;
272 cachedGeometry->model = model;
273 cache.insert({cacheKey, cachedGeometry});
274 return Result<std::shared_ptr<CachedCoalGeometry>>::success(cachedGeometry);
275}

282Result<std::shared_ptr<CoalProfileRuntime>> loadProfileRuntime(const MeshCollisionProfile& profile)
283{
284 const std::string cacheKey = profileRuntimeKey(profile);
285
286 auto& cache = profileCache();
287 const auto cached = cache.find(cacheKey);
288 if (cached != cache.end()) {
289 return Result<std::shared_ptr<CoalProfileRuntime>>::success(cached->second);
290 }
291
292 auto runtime = std::make_shared<CoalProfileRuntime>();
293 runtime->cacheKey = cacheKey;
294 runtime->meshes.reserve(profile.meshes.size());
295
296 for (const MeshCollisionGeometry& mesh : profile.meshes) {
297 if (!mesh.enabled) {
298 continue;
299 }
300
301 const Result<std::shared_ptr<CachedCoalGeometry>> geometry = loadCoalGeometry(mesh);
302 if (!geometry.ok()) {
303 return Result<std::shared_ptr<CoalProfileRuntime>>::failure(geometry.status, geometry.message);
304 }
305
306 CoalMeshRuntimeEntry entry;
307 entry.mesh = mesh;
308 entry.geometry = geometry.value;
309 entry.object = std::make_unique<coal::CollisionObject>(entry.geometry->model);
310 runtime->meshes.push_back(std::move(entry));
311 }
312
313 cache.insert({cacheKey, runtime});
314 return Result<std::shared_ptr<CoalProfileRuntime>>::success(runtime);
315}

342CollisionCheckResult checkCoalMeshBackend(const SerialRobotConfig& config,
343 const MeshCollisionProfile& profile,
344 const MeshCollisionCheckRequest& request)
345{
346 const ModelValidationResult modelValidation = RobotModelValidator::validateSerialRobotConfig(config);
347 if (!modelValidation.ok()) {
348 return failure(modelValidation.status(), modelValidation.issues.front().message);
349 }
350
351 const MeshCollisionProfileValidationResult profileValidation =
352 MeshCollisionProfileValidator::validate(config, profile);
353 if (!profileValidation.ok()) {
354 return failure(profileValidation.status(), profileValidation.issues.front().message);
355 }
356
357 if (request.joints.size() != ForwardKinematics::movableJointCount(config)) {
358 return failure(KinematicsStatus::JointDimensionMismatch,
359 "joint vector dimension does not match robot dof");
360 }
361
362 const Result<std::shared_ptr<CoalProfileRuntime>> runtime = loadProfileRuntime(profile);
363 if (!runtime.ok()) {
364 return failure(runtime.status, runtime.message);
365 }
366
367 const FkChain chain = ForwardKinematics::computeChain(config, request.joints);
368 for (CoalMeshRuntimeEntry& mesh : runtime.value->meshes) {
369 const auto linkIt = chain.linkPosesInBase.find(mesh.mesh.linkId);
370 if (linkIt == chain.linkPosesInBase.end()) {
371 return failure(KinematicsStatus::InvalidRobotConfig,
372 "mesh collision link pose not found in FK chain: " + mesh.mesh.linkId);
373 }
374
375 const Pose meshInBase = linkIt->second * mesh.mesh.meshToLink;
376 mesh.object->setTransform(toCoalTransform(meshInBase));
377 mesh.object->computeAABB();
378 }
379
380 std::set<std::string> disabledPairs;
381 for (const DisabledCollisionPair& pair : profile.disabledPairs) {
382 disabledPairs.insert(normalizePairKey(pair.geometryA, pair.geometryB));
383 }
384
385 CollisionCheckResult result;
386 result.status = KinematicsStatus::Ok;
387
388 for (std::size_t i = 0; i < runtime.value->meshes.size(); ++i) {
389 for (std::size_t j = i + 1; j < runtime.value->meshes.size(); ++j) {
390 const CoalMeshRuntimeEntry& a = runtime.value->meshes[i];
391 const CoalMeshRuntimeEntry& b = runtime.value->meshes[j];
392
393 if (a.mesh.id == b.mesh.id) {
394 continue;
395 }
396 if (a.mesh.linkId == b.mesh.linkId) {
397 continue;
398 }
399 if (disabledPairs.find(normalizePairKey(a.mesh.id, b.mesh.id)) != disabledPairs.end()) {
400 continue;
401 }
402
403 coal::DistanceRequest distanceRequest(false, true);
404 coal::DistanceResult distanceResult;
405 coal::distance(a.object.get(), b.object.get(), distanceRequest, distanceResult);
406 double distance = static_cast<double>(distanceResult.min_distance);
407
408 const double pairMargin = a.mesh.margin_m + b.mesh.margin_m + request.safetyMargin_m;
409 coal::CollisionRequest collisionRequest;
410 collisionRequest.num_max_contacts = request.returnAllPairs ? std::size_t(16) : std::size_t(1);
411 collisionRequest.enable_contact = true;
412 collisionRequest.security_margin = pairMargin;
413 collisionRequest.distance_upper_bound = (std::numeric_limits<coal::Scalar>::max)();
414
415 coal::CollisionResult collisionResult;
416 coal::collide(a.object.get(), b.object.get(), collisionRequest, collisionResult);
417 const std::size_t contactCount = collisionResult.numContacts();
418 const bool colliding = contactCount > 0 || distance <= pairMargin;
419
420 if (colliding) {
421 for (std::size_t contactIndex = 0; contactIndex < contactCount; ++contactIndex) {
422 const double contactDistance =
423 static_cast<double>(collisionResult.getContact(contactIndex).penetration_depth);
424 if (std::isfinite(contactDistance) && (contactIndex == 0 || contactDistance < distance)) {
425 distance = contactDistance;
426 }
427 }
428 }
429
430 if (!std::isfinite(distance)) {
431 return failure(KinematicsStatus::NumericalError,
432 "coal returned a non-finite mesh distance for pair '" +
433 a.mesh.id + "'/'" + b.mesh.id + "'");
434 }
435
436 result.pairs.push_back(CollisionPairResult{
437 a.mesh.id,
438 b.mesh.id,
439 a.mesh.linkId,
440 b.mesh.linkId,
441 colliding,
442 distance,
443 contactCount,
444 });
445
446 if (colliding) {
447 result.hasCollision = true;
448 if (!request.returnAllPairs) {
449 return result;
450 }
451 }
452 }
453 }
454
455 return result;
456}
