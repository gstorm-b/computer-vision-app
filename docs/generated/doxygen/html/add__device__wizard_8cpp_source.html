233void AddDeviceWizard::onAddClicked() {
234 const QString name = ui->adwNameEdit->text().trimmed();
235
236 if (name.isEmpty()) {
237 QMessageBox::warning(this, tr("Add Device"), tr("Please enter a device name."));
238 ui->adwNameEdit->setFocus();
239 return;
240 }
241
242 if (m_manager->isNameExists(name)) {
243 QMessageBox::warning(this, tr("Add Device"), tr("Device name already exists."));
244 ui->adwNameEdit->selectAll();
245 return;
246 }
247
248 QJsonObject jobj = buildDeviceJson(m_selectedType);
249 jobj[DEVICE_JSK_ID] = m_pendingDeviceId;
250 jobj[DEVICE_JSK_NAME] = name;
251 jobj[DEVICE_JSK_TYPE] = vc::device::DeviceTypeToString(m_selectedType);
252
253 std::shared_ptr<vc::device::IDevice> dv(
254 vc::device::DeviceFactory::create(m_selectedType, jobj));
255
256 if (m_manager->commitDevice(m_pendingDeviceId, name, dv)) {
257 LOG_USER_INFO << QString("Device added: %1 (%2)").arg(name, m_pendingDeviceId);
258 accept();
259 } else {
260 QMessageBox::critical(this, tr("Add Device"),
261 tr("Failed to register device. Please close and try again."));
262 }
263}

269QJsonObject AddDeviceWizard::buildDeviceJson(vc::device::DeviceType type) {
270 QJsonObject obj;
271
272 switch (type) {
273 case vc::device::DeviceType::Camera:
274 if (ui->cbxCameraType->count() > 0)
275 obj[DEVICE_JSK_CAM_TYPE] = ui->cbxCameraType->currentText();
276 break;
277
278 case vc::device::DeviceType::PLC: {
279 vc::device::McProtocolConfig config;
280 auto frameType = (ui->cbxMcFrameType->count() > 0)
281 ? vc::device::mc::McFrameTypeFromString(ui->cbxMcFrameType->currentText())
282 : vc::device::mc::McFrameType::Frame_3E;
283 auto dataCode = (ui->cbxMcCode->count() > 0)
284 ? vc::device::mc::McDataCodeFromString(ui->cbxMcCode->currentText())
285 : vc::device::mc::McDataCode::Binary;
286 config.configMcProtocol(frameType, dataCode);
287 obj[DEVICE_JSK_CONFIG] = config.toJson();
288 break;
289 }
290
291 case vc::device::DeviceType::VisionOutput: {
292 auto subType = (ui->cbxVisionType->count() > 0)
293 ? vc::device::VisionOutputTypeFromString(ui->cbxVisionType->currentText())
294 : vc::device::VisionOutputType::VisionTCPIP;
295 switch (subType) {
296 case vc::device::VisionOutputType::VisionTCPIP: {
297 vc::device::VisionTcpipDeviceCfg config;
298 obj[DEVICE_JSK_CONFIG] = config.toJson();
299 break;
300 }
301 case vc::device::VisionOutputType::VisionTcpipClient: {
302 vc::device::VisionTcpipClientDeviceCfg config;
303 obj[DEVICE_JSK_CONFIG] = config.toJson();
304 break;
305 }
306 case vc::device::VisionOutputType::VisionSerial:
307 case vc::device::VisionOutputType::VisionOutputTypeNone:
308 // No concrete impl yet — leave config empty; factory will reject.
309 break;
310 }
311 break;
312 }
313
314 default:
315 break;
316 }
317
318 return obj;
319}
