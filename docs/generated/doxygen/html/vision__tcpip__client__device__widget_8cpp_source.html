19static QtVariantProperty* addPropertyToBrowser(const QMetaObject &meta, QMetaProperty &prop, QVariant &value,
20 QtVariantPropertyManager *manager, QtTreePropertyBrowser *browser) {
21
22 QString propName = prop.name();
23 if (propName == "objectName") {
24 // ignore objectName
25 return nullptr;
26 }
27
28 QtVariantProperty *variantProp = nullptr;
29 // enum type Check
30 if (prop.isEnumType()) {
31 variantProp = manager->addProperty(QtVariantPropertyManager::enumTypeId(), propName);
32
33 // get enum names
34 QStringList enumNames;
35 QMetaEnum metaEnum = prop.enumerator();
36 for (int j = 0; j < metaEnum.keyCount(); ++j) {
37 const char* key = metaEnum.key(j);
38 QString translatedName = QCoreApplication::translate(meta.className(), key);
39 enumNames << translatedName;
40 }
41 variantProp->setAttribute(QLatin1String("enumNames"), enumNames);
42 // set enum value
43 variantProp->setValue(value.toInt());
44
45 } else {
46 // normal data types handle (int, QString, bool, QColor, v.v.)
47 int typeId = value.userType();
48 variantProp = manager->addProperty(typeId, propName);
49 if (variantProp) {
50 variantProp->setValue(value);
51 }
52 }
53
54 if (!variantProp) {
55 return nullptr;
56 }
57
58 int displayNameIdx = meta.indexOfClassInfo(QString("%1_name").arg(propName).toUtf8());
59 if (displayNameIdx != -1) {
60 const QString displayName = meta.classInfo(displayNameIdx).value();
61 if (!displayName.isEmpty()) {
62 variantProp->setDisplayName(displayName);
63 }
64 }
65
66 // --- set attributes---
67 int minIdx = meta.indexOfClassInfo(QString("%1_min").arg(propName).toUtf8());
68 if (minIdx != -1) {
69 variantProp->setAttribute("minimum", QString(meta.classInfo(minIdx).value()).toInt());
70 }
71
72 int maxIdx = meta.indexOfClassInfo(QString("%1_max").arg(propName).toUtf8());
73 if (maxIdx != -1) {
74 variantProp->setAttribute("maximum", QString(meta.classInfo(maxIdx).value()).toInt());
75 }
76
77 if (!prop.isWritable()) {
78 variantProp->setEnabled(false);
79 }
80
81 return variantProp;
82}

362void VisionTcpipClientDeviceWidget::onConnectClicked() {
363 if (!m_runner) return;
364 // The client is "active" (dialing / connected / recovering) for any status
365 // other than the idle/failed ones, so the button can cancel an in-progress
366 // dial as well as disconnect an established link.
367 const vc::device::ConnectStatus st =
368 m_device ? m_device->connectStatus() : vc::device::ConnectStatus::Disconnected;
369 const bool active = (st == vc::device::ConnectStatus::Connected
370 || st == vc::device::ConnectStatus::Connecting
371 || st == vc::device::ConnectStatus::LostConnected);
372 if (active) {
373 m_runner->requestDisconnect();
374 } else {
375 m_runner->requestConnect();
376 }

490void VisionTcpipClientDeviceWidget::onSendResult() {

492 return;
493 }
494 QVector<vc::device::VisionOutputPosition> positions;
495 auto *tbl = ui->tbl_positions;
496 for (int row = 0; row < tbl->rowCount(); ++row) {
497 vc::device::VisionOutputPosition pos;
498 auto *sx = qobject_cast<QDoubleSpinBox*>(tbl->cellWidget(row, 0));
499 auto *sy = qobject_cast<QDoubleSpinBox*>(tbl->cellWidget(row, 1));
500 auto *sz = qobject_cast<QDoubleSpinBox*>(tbl->cellWidget(row, 2));
501 auto *sr = qobject_cast<QDoubleSpinBox*>(tbl->cellWidget(row, 3));
502 if (sx) pos.x = sx->value();
503 if (sy) pos.y = sy->value();
504 if (sz) pos.z = sz->value();
505 if (sr) pos.r = sr->value();
506 positions.append(pos);
507 }
508 m_runner->requestSendResult(positions);

516void VisionTcpipClientDeviceWidget::updateConnectionVisual(vc::device::ConnectStatus status) {
517 using vc::device::ConnectStatus;
518 const bool connected = status == ConnectStatus::Connected;
519 // "Active" = dialing / connected / recovering. Connecting and LostConnected
520 // both mean the client is running but not (fully) linked yet.
521 const bool active = connected
522 || status == ConnectStatus::Connecting
523 || status == ConnectStatus::LostConnected;
524
525 QByteArray state;
526 QString stateText;
527 if (connected) {
528 state = "connected";
529 stateText = tr("CONNECTED");
530 } else if (status == ConnectStatus::LostConnected) {
531 state = "connecting";
532 stateText = tr("RECONNECTING…");
533 } else if (status == ConnectStatus::Connecting) {
534 state = "connecting";
535 stateText = tr("CONNECTING…");
536 } else {
537 state = "disconnected";
538 stateText = tr("DISCONNECTED");
539 }
540
541 ui->lbl_conn_state->setText(stateText);
542 ui->btn_connect->setText(connected ? tr("Disconnect")
543 : active ? tr("Cancel")
544 : tr("Connect"));
545
546 // Config is locked while the client is active.
547 ui->ledit_ip->setEnabled(!active);
548 ui->spb_port_data->setEnabled(!active);
549 ui->spb_port_heartbeat->setEnabled(!active);

551 for (QWidget *w : {static_cast<QWidget*>(ui->lbl_conn_dot),
552 static_cast<QWidget*>(ui->lbl_conn_state),
553 static_cast<QWidget*>(ui->btn_connect)}) {
554 w->setProperty("connectionState", state);
