87inline void applyAttributes(QtVariantPropertyManager *mgr,
88 QtVariantProperty *prop,
89 const PropSpec<Config> &spec)
90{
91 if (spec.min.isValid()) mgr->setAttribute(prop, QStringLiteral("minimum"), spec.min);
92 if (spec.max.isValid()) mgr->setAttribute(prop, QStringLiteral("maximum"), spec.max);
93 if (spec.step.isValid()) mgr->setAttribute(prop, QStringLiteral("singleStep"), spec.step);
94 if (spec.decimals >= 0) mgr->setAttribute(prop, QStringLiteral("decimals"), spec.decimals);
95 if (spec.description) prop->setDescriptionToolTip(QString::fromUtf8(spec.description));
96 prop->setEnabled(!spec.readOnly);
97}

110template<typename Config>
111inline QtVariantProperty *buildOne(QtVariantPropertyManager *mgr,
112 const PropSpec<Config> &spec,
113 const Config &cfg,
114 QMap<QString, QtVariantProperty *> *propMap = nullptr,
115 QMap<QtProperty *, QString> *propKeyMap = nullptr)
116{
117 auto *prop = mgr->addProperty(spec.propType, QString::fromUtf8(spec.label));
118 if (!prop) return nullptr;
119
120 applyAttributes(mgr, prop, spec);
121 prop->setValue(spec.read(cfg));
122
123 const QString key = QString::fromUtf8(spec.key);
124 if (propMap) (*propMap)[key] = prop;
125 if (propKeyMap) (*propKeyMap)[prop] = key;
126 return prop;
127}

133inline QMap<QString, QtVariantProperty *>
134buildGroup(QtVariantPropertyManager *mgr,
135 QtVariantProperty *group,
136 const QList<PropSpec<Config>> &specs,
137 const Config &cfg,
138 QMap<QtProperty *, QString> &propKeyMap)
139{
140 QMap<QString, QtVariantProperty *> propMap;

151template<typename Config>
152inline QMap<QString, QtVariantProperty *>
153buildGroups(QtVariantPropertyManager *mgr,
154 QtAbstractPropertyBrowser *browser,
155 const QList<PropGroup<Config>> &groups,
156 const Config &cfg,
157 QMap<QtProperty *, QString> &propKeyMap)
158{
159 QMap<QString, QtVariantProperty *> combined;
160 for (const auto &grp : groups) {
161 auto *groupProp = mgr->addProperty(QtVariantPropertyManager::groupTypeId(),
162 QString::fromUtf8(grp.label));
163 auto sub = buildGroup(mgr, groupProp, grp.specs, cfg, propKeyMap);
164 combined.insert(sub);
165 browser->addProperty(groupProp);
166 }
167 return combined;
