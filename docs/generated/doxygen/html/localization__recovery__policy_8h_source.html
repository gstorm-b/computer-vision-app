85inline LocalizationRecoveryAction decideRecoveryAction(

87 vc::device::ConnectStatus status,
88 int currentRetryCount,
89 bool retryAlreadyScheduled)
90{
91 if (!policy.isRecoverableStatus(status)) {

94
95 if (retryAlreadyScheduled) {
