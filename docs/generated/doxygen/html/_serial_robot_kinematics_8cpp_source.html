24Result<Pose> referenceFrameInBase(const SerialRobotConfig& config, const FrameId& id)
25{
26 if (id.empty() || id.value == config.frames.baseLinkId || id.value == "base") {
27 return Result<Pose>::success(Pose::identity());
28 }
29
30 const FrameRegistry frames = FrameRegistry::fromConfig(config);
31 Result<UserFrame> frame = frames.get(id);
32 if (!frame.ok()) {
33 return Result<Pose>::failure(frame.status, frame.message);
34 }
35 if (frame.value.parentLinkId != config.frames.baseLinkId) {
36 return Result<Pose>::failure(KinematicsStatus::InvalidRequest,
37 "IK reference frame must be fixed to the base link");
38 }
39 return Result<Pose>::success(frame.value.transform);
40}

90IKResult SerialRobotKinematics::run(const IKRequest& request, bool allSolutions) const
91{
92 const ModelValidationResult model = RobotModelValidator::validateSerialRobotConfig(config_);
93 if (!model.ok()) {
94 return IKResult{model.status(), {}, model.issues.front().message};
95 }
96
97 const Result<Tool> tool = resolveTool(config_, request.tool);
98 if (!tool.ok()) {
99 return IKResult{tool.status, {}, tool.message};
100 }
101
102 const Result<Pose> baseFromReference = referenceFrameInBase(config_, request.referenceFrame);
103 if (!baseFromReference.ok()) {
104 return IKResult{baseFromReference.status, {}, baseFromReference.message};
105 }
106
107 IKSolveContext context;
108 context.request = request;
109 context.targetFlangeInBase = baseFromReference.value * request.targetPose * tool.value.flangeToTcp.inverse();
110
111 // Hybrid policy: use the analytic plugin when it supports the model and returns a
112 // solution; otherwise fall back to the numerical solver.
113 const Analytic6DofSphericalWristSolver analytic;
114 if (analytic.supportsModel(config_)) {
115 const IKResult analyticResult =
116 allSolutions ? analytic.solveAll(config_, context) : analytic.solve(config_, context);
117 if (analyticResult.ok()) {
118 return analyticResult;
119 }
120 }
121
122 const NumericalIKSolver solver;
123 return allSolutions ? solver.solveAll(config_, context) : solver.solve(config_, context);
124}
