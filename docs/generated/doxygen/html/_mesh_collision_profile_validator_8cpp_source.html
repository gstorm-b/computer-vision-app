48{
49 MeshCollisionProfileValidationResult result;
50
51 if (profile.id.empty()) {
52 addIssue(result, "profile.id", "mesh collision profile id must not be empty");
53 }
54 if (profile.robotModel.empty()) {
55 addIssue(result, "profile.robotModel", "mesh collision profile robot model must not be empty");
56 }
57
58 std::set<std::string> linkIds;
59 for (const Link& link : config.links) {
60 if (!link.id.empty()) {
61 linkIds.insert(link.id);
62 }
63 }
64
65 std::set<std::string> meshIds;
66 for (std::size_t i = 0; i < profile.meshes.size(); ++i) {
67 const MeshCollisionGeometry& mesh = profile.meshes[i];
68 const std::string prefix = "meshes[" + std::to_string(i) + "]";
69
70 if (mesh.id.empty()) {
71 addIssue(result, prefix + ".id", "mesh collision geometry id must not be empty");
72 } else if (!meshIds.insert(mesh.id).second) {
73 addIssue(result, prefix + ".id", "mesh collision geometry id must be unique");
74 }
75
76 if (!hasLinkId(linkIds, mesh.linkId)) {
77 addIssue(result, prefix + ".linkId", "mesh collision geometry link id must refer to a declared link");
78 }
79 if (mesh.path.empty()) {
80 addIssue(result, prefix + ".path", "mesh collision geometry path must not be empty");
81 }
82 if (mesh.scaleToMeters <= 0.0) {
83 addIssue(result, prefix + ".scaleToMeters", "mesh collision geometry scaleToMeters must be positive");
84 }
85 if (mesh.margin_m < 0.0) {
86 addIssue(result, prefix + ".margin_m", "mesh collision geometry margin must be non-negative");
87 }
88 if (mesh.quality.triangleCount.has_value() && *mesh.quality.triangleCount == 0) {
89 addIssue(result, prefix + ".quality.triangleCount", "mesh collision geometry triangleCount must be positive when provided");
90 }
91 if (mesh.quality.maxSimplificationError_m.has_value() &&
92 *mesh.quality.maxSimplificationError_m < 0.0) {
93 addIssue(result,
94 prefix + ".quality.maxSimplificationError_m",
95 "mesh collision geometry maxSimplificationError_m must be non-negative when provided");
96 }
97 if (mesh.quality.mode == MeshQualityMode::Simplified &&
98 (!mesh.quality.simplifiedFrom.has_value() || mesh.quality.simplifiedFrom->empty())) {
99 addIssue(result,
100 prefix + ".quality.simplifiedFrom",
101 "simplified mesh collision geometry must record simplifiedFrom");
102 }
103 }
104
105 for (std::size_t i = 0; i < profile.disabledPairs.size(); ++i) {
106 const DisabledCollisionPair& pair = profile.disabledPairs[i];
107 const std::string prefix = "disabledPairs[" + std::to_string(i) + "]";
108
109 if (pair.geometryA.empty() || meshIds.find(pair.geometryA) == meshIds.end()) {
110 addIssue(result, prefix + ".geometryA",
111 "disabled mesh collision pair geometryA must refer to an existing mesh id");
112 }
113 if (pair.geometryB.empty() || meshIds.find(pair.geometryB) == meshIds.end()) {
114 addIssue(result, prefix + ".geometryB",
115 "disabled mesh collision pair geometryB must refer to an existing mesh id");
116 }
117 }
118
119 return result;
120}
