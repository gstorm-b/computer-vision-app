137void AppSettings::load() {
138 QFile f(filePath());
139 if (!f.open(QFile::ReadOnly)) return;
140
141 QVariantMap loaded;
142 if (!decode(f.readAll(), loaded)) return; // corrupt or wrong magic → keep defaults
143
144 // Merge: values from disk override defaults, but unknown new-version keys
145 // that don't exist in the file retain their default values.
146 for (auto it = loaded.cbegin(); it != loaded.cend(); ++it)
147 m_data[it.key()] = it.value();
148}

169QByteArray AppSettings::encode(const QVariantMap& map) {
170 const QByteArray cbor = QCborValue::fromVariant(map).toCbor();
171 const QByteArray hash = QCryptographicHash::hash(cbor, QCryptographicHash::Sha256);
172 const QByteArray obf = obfuscate(cbor);
173
174 QByteArray out;
175 out.reserve(kHeaderSize + obf.size());
176
177 auto appendU32 = [&](quint32 v) {
178 out.append(char((v >> 24) & 0xFF));
179 out.append(char((v >> 16) & 0xFF));
180 out.append(char((v >> 8) & 0xFF));
181 out.append(char( v & 0xFF));
182 };
183
184 appendU32(kMagic);
185 appendU32(kVersion);
186 out.append(hash); // 32 bytes SHA-256
187 out.append(obf);
188 return out;
189}

194bool AppSettings::decode(const QByteArray& raw, QVariantMap& out) {
195 if (raw.size() < kHeaderSize) return false;
196
197 const quint32 magic = (quint8(raw[0]) << 24) | (quint8(raw[1]) << 16)
198 | (quint8(raw[2]) << 8) | quint8(raw[3]);
199 if (magic != kMagic) return false;
200
201 // Version field reserved for future migration branches:
202 // const quint32 fileVersion = (quint8(raw[4]) << 24) | ... | quint8(raw[7]);
203
204 const QByteArray storedHash = raw.mid(8, 32);
205 const QByteArray cbor = obfuscate(raw.mid(kHeaderSize)); // XOR is symmetric
206
207 // Integrity check — any bit flip or manual edit will fail here.
208 if (QCryptographicHash::hash(cbor, QCryptographicHash::Sha256) != storedHash)
209 return false;
210
211 QCborParserError err;
212 const QCborValue val = QCborValue::fromCbor(cbor, &err);
213 if (err.error != QCborError::NoError || !val.isMap()) return false;
214
215 out = val.toVariant().toMap();
216 return true;
217}

221QByteArray AppSettings::obfuscate(const QByteArray& data) {
222 static constexpr int keyLen = static_cast<int>(sizeof(kObfKey));
223 QByteArray result = data;
224 for (int i = 0; i < result.size(); ++i)
225 result[i] = char(quint8(result[i]) ^ kObfKey[i % keyLen]);
226 return result;
227}
