126 QPixmap pixmap(const QSize &size, QIcon::Mode mode, QIcon::State state) override
127 {
128 const QString resolvedPath = themedPath();
129 const QSize requestedSize = size.isValid() ? size : QSize(m_intent, m_intent);
130 const QString cacheKey = QStringLiteral("themed-svg:%1:%2:%3:%4:%5x%6")
131 .arg(styleKey(),
132 resolvedPath,
133 QString::number(static_cast<int>(mode)),
134 QString::number(static_cast<int>(state)),
135 QString::number(requestedSize.width()),
136 QString::number(requestedSize.height()));
137
138 QPixmap cached;
139 if (QPixmapCache::find(cacheKey, &cached))
140 return cached;
141
142 QIcon icon(resolvedPath);
143 icon.addPixmap(icon.pixmap(m_intent));
144 const QPixmap pm = icon.pixmap(requestedSize, mode, state);
145 QPixmapCache::insert(cacheKey, pm);
146 return pm;
147 }
148
151 void paint(QPainter *painter, const QRect &rect,
152 QIcon::Mode mode, QIcon::State state) override
153 {
154 if (!painter)
155 return;
156 painter->drawPixmap(rect, pixmap(rect.size(), mode, state));
157 }
158
161 QSize actualSize(const QSize &size, QIcon::Mode mode, QIcon::State state) override
162 {
163 const QIcon icon(themedPath());
164 return icon.actualSize(size, mode, state);
165 }
166
167private:
170 QString themedPath() const
171 {
172 if (!qApp)
173 return m_basePath;
174 return ThemeManager::instance()->themedIcon(m_basePath);
175 }

180 QString styleKey() const
181 {
182 if (!qApp)
183 return QStringLiteral("no-app");
184 const QString currentId = ThemeManager::instance()->currentStyleId();
185 return currentId.isEmpty() ? QStringLiteral("uninitialized") : currentId;
186 }
