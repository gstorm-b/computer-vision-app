121void GripperRegisterDialog::onSave() {
122 // Rebuild from scratch so the store's own uniqueness rule is what decides, rather
123 // than a second copy of that rule living here.
124 vc::model::GripperPresetStore rebuilt;
125
126 for (int row = 0; row < ui->tbl_presets->rowCount(); ++row) {
127 auto *nameItem = ui->tbl_presets->item(row, kColName);
128 const QString name = nameItem ? nameItem->text().trimmed() : QString();
129
130 if (name.isEmpty()) {
131 ui->tbl_presets->selectRow(row);
132 showMessage(tr("Row %1 has no name. Every gripper needs a unique name.")
133 .arg(row + 1));
134 return;
135 }
136
137 vc::model::GripperPreset preset;
138 preset.name = name;
139 preset.boxes.size = cv::Size2f(
140 static_cast<float>(spinValue(ui->tbl_presets, row, kColWidth)),
141 static_cast<float>(spinValue(ui->tbl_presets, row, kColHeight)));
142 preset.boxes.distance = spinValue(ui->tbl_presets, row, kColDistance);
143
144 if (!rebuilt.add(preset)) {
145 ui->tbl_presets->selectRow(row);
146 showMessage(tr("Row %1 repeats the name \"%2\". Names must be unique.")
147 .arg(row + 1).arg(name));
148 return;
149 }
150 }
151
152 m_presets = rebuilt;
153 accept();
154}
