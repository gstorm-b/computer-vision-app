129void SystemLogForm::rerenderLogs() {
130 auto *scrollBar = ui->log_view->verticalScrollBar();
131 const int previousValue = scrollBar ? scrollBar->value() : 0;
132 const bool wasAtBottom = scrollBar
133 ? previousValue >= scrollBar->maximum() - 4
134 : true;
135
136 const bool dark = ThemeManager::instance()->isDark();
137 QStringList rows;
138 rows.reserve(m_entries.size());
139 for (const LogMessage &msg : m_entries) {
140 if (shouldShow(msg))
141 rows << renderLogEntry(msg, dark);
142 }
143
144 const QString html = QStringLiteral(
145 "<html><body style='margin:0; color:%1; font-family:\"JetBrains Mono\",\"Consolas\",monospace; "
146 "font-size:10pt;'>%2</body></html>")
147 .arg(tokenValue("text.primary", dark), rows.join(QString()));
148 ui->log_view->setHtml(html);
149
150 if (!scrollBar)
151 return;
152 if (wasAtBottom) {
153 scrollBar->setValue(scrollBar->maximum());
154 } else {
155 scrollBar->setValue(qMin(previousValue, scrollBar->maximum()));
156 }
157}
