66bool ShellHandoff::launchSibling(ShellKind self, QString *error)
67{
68 const QString path = siblingPath(self);
69
70 // Checked before launching rather than relying on startDetached's return: an install
71 // missing one of the two executables is a real field case (a partial copy), and
72 // "ncr_runtime.exe was not found next to this application" is a diagnosis, whereas
73 // "could not start the other application" is not.
74 if (!QFileInfo::exists(path)) {
75 if (error != nullptr) {
76 *error = QCoreApplication::translate(
77 "ShellHandoff",
78 "%1 was not found next to this application:\n%2\n\n"
79 "Both executables must be installed in the same folder.")
80 .arg(executableName(siblingOf(self)), path);
81 }
82 LOG_USER_ERR << "Shell hand-off failed: sibling executable is missing." << path;
83 return false;
84 }
85
86 qint64 pid = 0;
87 const bool started = QProcess::startDetached(
88 path,
89 QStringList{ QLatin1String(kHandoffFlag) },
90 QCoreApplication::applicationDirPath(),
91 &pid);
92
93 if (!started) {
94 if (error != nullptr) {
95 *error = QCoreApplication::translate(
96 "ShellHandoff",
97 "%1 could not be started:\n%2")
98 .arg(executableName(siblingOf(self)), path);
99 }
100 LOG_USER_ERR << "Shell hand-off failed: sibling could not be started." << path;
101 return false;
102 }
103
104#ifdef Q_OS_WIN
105 // Hand the foreground to the incoming process before this one gives it up.
106 //
107 // Windows only lets the process that currently owns the foreground grant that right to
108 // another. Without this the new shell comes up BEHIND whatever is on screen and only
109 // flashes its taskbar button — the operator switches applications and appears to get
110 // nothing, then has to hunt for the window. The permission is granted to the process
111 // and survives until it is used, so it does not matter that the child has no window
112 // yet. The incoming shell still has to ask for the foreground; see
113 // vc::ui::presentShellWindow().
114 if (pid > 0) {
115 ::AllowSetForegroundWindow(static_cast<DWORD>(pid));
116 }
117#endif
118
119 LOG_USER_INFO << "Shell hand-off: sibling started."
120 << "target=" << applicationName(siblingOf(self))
121 << "pid=" << pid;
122 return true;
123}
