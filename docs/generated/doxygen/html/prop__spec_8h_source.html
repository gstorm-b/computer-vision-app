3// ============================================================================
4// prop_spec.h — Generic property-spec framework for QtPropertyBrowser
5//
6// Design goal: define a new editable parameter by adding ONE entry to a
7// spec list. Building, refreshing, and dispatching changes are handled
8// entirely by the helper functions below. Nothing else needs to change.
9//
10// Typical usage pattern:
11//
12// // 1. Define specs (usually as a static / file-level variable)
13// static const QList<PropSpec<MyConfig>> kMySpecs = {
14// { "speed", "Speed (m/s)", "Maximum linear velocity",
15// QMetaType::Double, 0.0, 10.0, 0.01, 2, false,
16// [](const MyConfig &c){ return c.speed; },
17// [](MyConfig &c, const QVariant &v){ c.speed = v.toDouble(); } },
18// { "enabled", "Enabled", nullptr,
19// QMetaType::Bool, {}, {}, {}, -1, false,
20// [](const MyConfig &c){ return c.enabled; },
21// [](MyConfig &c, const QVariant &v){ c.enabled = v.toBool(); } },
22// };
23//
24// // 2. Build (once, inside your "build" method)
25// auto *grp = mgr->addProperty(QtVariantPropertyManager::groupTypeId(), "My Group");
26// m_propMap = PropSpecHelper::buildGroup(mgr, grp, kMySpecs, cfg, m_propKeys);
27// browser->addProperty(grp);
28//
29// // 3. Refresh (on external cfg changes)
30// PropSpecHelper::refresh(mgr, kMySpecs, cfg, m_propMap);
31//
32// // 4. Dispatch (in valueChanged slot)
33// if (PropSpecHelper::dispatch(kMySpecs, key, val, cfg))
34// emit configModified();
35//
36// ============================================================================

87template<typename Config>
88inline void applyAttributes(QtVariantPropertyManager *mgr,
89 QtVariantProperty *prop,
90 const PropSpec<Config> &spec)
91{
92 if (spec.min.isValid()) mgr->setAttribute(prop, QStringLiteral("minimum"), spec.min);
93 if (spec.max.isValid()) mgr->setAttribute(prop, QStringLiteral("maximum"), spec.max);
94 if (spec.step.isValid()) mgr->setAttribute(prop, QStringLiteral("singleStep"), spec.step);
95 if (spec.decimals >= 0) mgr->setAttribute(prop, QStringLiteral("decimals"), spec.decimals);
96 if (spec.description) prop->setDescriptionToolTip(QString::fromUtf8(spec.description));
97 prop->setEnabled(!spec.readOnly);
98}

106template<typename Config>
107inline QtVariantProperty *buildOne(QtVariantPropertyManager *mgr,
108 const PropSpec<Config> &spec,
109 const Config &cfg,
110 QMap<QString, QtVariantProperty *> *propMap = nullptr,
111 QMap<QtProperty *, QString> *propKeyMap = nullptr)
112{
113 auto *prop = mgr->addProperty(spec.propType, QString::fromUtf8(spec.label));
114 if (!prop) return nullptr;
115
116 applyAttributes(mgr, prop, spec);
117 prop->setValue(spec.read(cfg));
118
119 const QString key = QString::fromUtf8(spec.key);
120 if (propMap) (*propMap)[key] = prop;
121 if (propKeyMap) (*propKeyMap)[prop] = key;
122 return prop;
123}

128template<typename Config>
129inline QMap<QString, QtVariantProperty *>
130buildGroup(QtVariantPropertyManager *mgr,
131 QtVariantProperty *group,
132 const QList<PropSpec<Config>> &specs,
133 const Config &cfg,
134 QMap<QtProperty *, QString> &propKeyMap)
135{
136 QMap<QString, QtVariantProperty *> propMap;
137 for (const auto &spec : specs) {
138 auto *prop = buildOne(mgr, spec, cfg, &propMap, &propKeyMap);
139 if (prop && group) group->addSubProperty(prop);
140 }
141 return propMap;
142}

147template<typename Config>
148inline QMap<QString, QtVariantProperty *>
149buildGroups(QtVariantPropertyManager *mgr,
150 QtAbstractPropertyBrowser *browser,
151 const QList<PropGroup<Config>> &groups,
152 const Config &cfg,
153 QMap<QtProperty *, QString> &propKeyMap)
154{
155 QMap<QString, QtVariantProperty *> combined;
156 for (const auto &grp : groups) {
157 auto *groupProp = mgr->addProperty(QtVariantPropertyManager::groupTypeId(),
158 QString::fromUtf8(grp.label));
159 auto sub = buildGroup(mgr, groupProp, grp.specs, cfg, propKeyMap);
160 combined.insert(sub);
161 browser->addProperty(groupProp);
162 }
163 return combined;
164}

170inline void refresh(QtVariantPropertyManager *mgr,
171 const QList<PropSpec<Config>> &specs,
172 const Config &cfg,
173 const QMap<QString, QtVariantProperty *> &propMap)
174{
175 const QSignalBlocker blocker(mgr);
176 for (const auto &spec : specs) {
177 const QString key = QString::fromUtf8(spec.key);
178 if (auto *p = propMap.value(key)) p->setValue(spec.read(cfg));
179 }

196template<typename Config>
197inline bool dispatch(const QList<PropSpec<Config>> &specs,
198 const QString &key,
199 const QVariant &val,
200 Config &cfg)
201{
202 for (const auto &spec : specs) {
203 if (QLatin1String(spec.key) == key) {
204 spec.write(cfg, val);
205 return true;
206 }
207 }
208 return false;
209}
