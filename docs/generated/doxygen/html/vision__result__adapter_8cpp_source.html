33VisionResultObject makeResultObject(const mtc::MatchedObject &object,
34 const QPointF &offset)
35{
36 VisionResultObject resultObject;
37 resultObject.patternName = QString::fromStdWString(object.pattern_name);
38 resultObject.patternIndex = object.pattern_index;
39 resultObject.score = object.matched_Score;
40 resultObject.center = QPointF(object.point_Center.x + offset.x(),
41 object.point_Center.y + offset.y());
42 resultObject.corners = {
43 QPointF(object.point_LT.x + offset.x(), object.point_LT.y + offset.y()),
44 QPointF(object.point_RT.x + offset.x(), object.point_RT.y + offset.y()),
45 QPointF(object.point_RB.x + offset.x(), object.point_RB.y + offset.y()),
46 QPointF(object.point_LB.x + offset.x(), object.point_LB.y + offset.y()),
47 };
48 const auto appendPickingPolygon = [&resultObject, &offset](const std::vector<cv::Point2f> &points) {
49 if (points.size() != 4) return;
50 QVector<QPointF> polygon;
51 polygon.reserve(4);
52 for (const cv::Point2f &point : points) {
53 polygon.append(QPointF(point.x + offset.x(), point.y + offset.y()));
54 }
55 resultObject.pickingBoxPolygons.append(polygon);
56 };
57
58 const mtc::MatchedObject::GripperBox &gripperBox = object.gripperBox();
59 appendPickingPolygon(gripperBox.box_left_pts);
60 appendPickingPolygon(gripperBox.box_right_pts);
61
62 if (resultObject.pickingBoxPolygons.isEmpty()
63 && object.pickingBox.size.width > 0.0f
64 && object.pickingBox.size.height > 0.0f) {
65 cv::Point2f boxPoints[4];
66 object.pickingBox.points(boxPoints);
67 QVector<QPointF> polygon;
68 polygon.reserve(4);
69 for (const cv::Point2f &point : boxPoints) {
70 polygon.append(QPointF(point.x + offset.x(), point.y + offset.y()));
71 }
72 resultObject.pickingBoxPolygons.append(polygon);
73 }
74 resultObject.matchedAngleDeg = object.matched_Angle;
75 resultObject.pointAngleDeg = object.point_angle;
76 resultObject.possibleToPick = object.isPossibleToPick();
77 resultObject.hasCollision = object.hasCollision();
78 resultObject.outsideConditionRoi = object.isOutsideConditionRoi();
79 resultObject.rejected = object.hasCollision()
80 || object.isOutsideConditionRoi()
81 || !object.isPossibleToPick();
82 resultObject.faulted = object.hasCollision() || object.isOutsideConditionRoi();
83 return resultObject;
84}

118VisionResultOverlay VisionResultAdapter::fromMatchResult(
119 const mtc::MatchResult &result,
120 const QSize &sourceImageSize,
121 const vc::model::CameraWorkspace *workspace,
122 const QMap<QString, QVariant> *runtimeSignalValues)
123{
124 VisionResultOverlay overlay;
125 overlay.sourceImageSize = sourceImageSize;
126 if (runtimeSignalValues) {
127 overlay.runtimeSignalValues = *runtimeSignalValues;
128 }
129
130 const QPointF offset(result.cropOffsetPoint.x, result.cropOffsetPoint.y);
131 int nextIndex = 1;
132 for (const mtc::MatchedObject &object : result.Objects) {
133 VisionResultObject resultObject = makeResultObject(object, offset);
134 resultObject.index = nextIndex++;
135 if (resultObject.rejected) {
136 overlay.rejectedObjects.append(resultObject);
137 } else {
138 overlay.acceptedObjects.append(resultObject);
139 overlay.pickingPoints.append(resultObject.center);
140 }
141 }
142
143 appendWorkspaceOverlays(&overlay, workspace);
144 return overlay;
145}

147VisionResultOverlay VisionResultAdapter::fromCycleResult(
148 const vc::model::LocalizationRuntimeController::CycleResult &result,
149 const vc::model::CameraWorkspace *workspace,
150 const QMap<QString, QVariant> *runtimeSignalValues)
151{
152 QSize imageSize;
153 if (!result.rawImage.empty()) {
154 imageSize = QSize(result.rawImage.cols, result.rawImage.rows);
155 } else if (!result.matchResult.Image.empty()) {
156 imageSize = QSize(result.matchResult.Image.cols, result.matchResult.Image.rows);
157 } else {
158 imageSize = result.displayImage.empty()
159 ? QSize()
160 : QSize(result.displayImage.cols, result.displayImage.rows);
161 }
162
163 VisionResultOverlay overlay = fromMatchResult(result.matchResult, imageSize,
164 workspace, runtimeSignalValues);
165
166 for (int index = 0; index < result.rows.size(); ++index) {
167 const auto &row = result.rows.at(index);
168 const bool sent = row.status.startsWith(QStringLiteral("Sent"));
169 if (index < overlay.acceptedObjects.size()) {
170 overlay.acceptedObjects[index].sentToOutput = sent;
171 }
172 }
173
174 return overlay;
