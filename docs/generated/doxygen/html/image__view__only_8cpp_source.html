11inline QPixmap cvMatToQPixmap(const cv::Mat& mat) {
12 QImage qimg;
13 if (mat.type() == CV_8UC1) {
14 // Grayscale
15 qimg = QImage(mat.data,
16 mat.cols,
17 mat.rows,
18 static_cast<int>(mat.step),
19 QImage::Format_Grayscale8);
20 } else if (mat.type() == CV_8UC3) {
21 // OpenCV uses BGR, Qt uses RGB
22 cv::Mat rgb;
23 cv::cvtColor(mat, rgb, cv::COLOR_BGR2RGB);
24 qimg = QImage(rgb.data,
25 rgb.cols,
26 rgb.rows,
27 static_cast<int>(rgb.step),
28 QImage::Format_RGB888).copy();
29 } else if (mat.type() == CV_8UC4) {
30 // BGRA → RGBA
31 cv::Mat rgba;
32 cv::cvtColor(mat, rgba, cv::COLOR_BGRA2RGBA);
33 qimg = QImage(rgba.data,
34 rgba.cols,
35 rgba.rows,
36 static_cast<int>(rgba.step),
37 QImage::Format_RGBA8888).copy();
38 } else {
39 return QPixmap();
40 }
41
42 return QPixmap::fromImage(qimg);
43}

137void ImageViewOnly::loadImage(QPixmap &pixmap, bool fitsize) {
138 if (!m_pixmap_item) {
139 m_pixmap_item = m_scene->addPixmap(pixmap);
140 // using FastTransformation to avoid blur image
141 m_pixmap_item->setTransformationMode(Qt::FastTransformation);
142 } else {
143 m_pixmap_item->setPixmap(pixmap);
144 }
145 m_scene->setSceneRect(pixmap.rect());

148 if (fitsize || (!m_first_time_image_set)) {
