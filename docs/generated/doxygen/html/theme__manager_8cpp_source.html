174QString ThemeManager::tokenValue(const QString& name, bool dark) {
175 const auto it = tokenTable().constFind(name);

181QString ThemeManager::resolveTokens(const QString& qss, bool dark) {
182 // Match "@{group.token}" placeholders. Token names use lower-case words joined
183 // by '.' or '-'; the brace boundary keeps the match unambiguous next to QSS
184 // punctuation (';', ')', '}').
185 static const QRegularExpression rx(QStringLiteral(R"(@\{([a-z0-9.\-]+)\})"));
186 static QSet<QString> warnedUnknown;
187
188 QString out;
189 out.reserve(qss.size());
190 qsizetype last = 0;
191 auto it = rx.globalMatch(qss);
192 while (it.hasNext()) {
193 const QRegularExpressionMatch m = it.next();
194 out += qss.mid(last, m.capturedStart() - last);
195 const QString name = m.captured(1);
196 const QString value = tokenValue(name, dark);
197 if (value.isEmpty()) {
198 // Unknown token: leave the placeholder verbatim so the defect is
199 // visible rather than silently rendering a wrong colour.
200 out += m.captured(0);
201 if (!warnedUnknown.contains(name)) {
202 warnedUnknown.insert(name);
203 LOG_DEV_ERR << "ThemeManager: unknown QSS design token" << name;
204 }
205 } else {
206 out += value;
207 }
208 last = m.capturedEnd();
209 }
210 out += qss.mid(last);
211 return out;
212}

218QString ThemeManager::themedIcon(const QString& basePath) const {
219 if (isDark()) return basePath;
220 const int dot = basePath.lastIndexOf('.');
221 if (dot < 0) return basePath;
222 const QString darkPath = basePath.left(dot) + "_dark" + basePath.mid(dot);
223 return QFile::exists(darkPath) ? darkPath : basePath;

226void ThemeManager::apply(const ThemeStyle& style) {
227 if (style.palette.has_value())
228 qApp->setPalette(*style.palette);
229
230 QString qss;
231 if (!style.qssPath.isEmpty()) {
232 QFile f(style.qssPath);
233 if (f.open(QFile::ReadOnly | QFile::Text))
234 qss = QString::fromUtf8(f.readAll());
235 }
236 // Resolve §5 design tokens for the style being installed. style.isDark is used
237 // (not isDark()) because m_currentId may not yet point at this style when
238 // registerStyle() applies an updated definition.
239 qApp->setStyleSheet(resolveTokens(qss, style.isDark));
240}
