12bool LocalizationPipeline::loadModel(mtc::ImageMatcher &matcher,
13 const std::shared_ptr<mtc::MatchGroup> &group) const {
14 if (!group) {
15 LOG_USER_ERR << "Localization matching failed: match group is null.";
16 return false;
17 }
18
19 // Expects a freshly-constructed matcher (empty model). Callers reload by
20 // recreating the matcher on group change, so no model reset is needed here.
21 auto *model = matcher.getModel();
22 group->cloneConfigTo(*model);
23
24 for (const auto &patternPtr : group->patterns()) {
25 if (!patternPtr)
26 continue;
27
28 mtc::MatchPatternConfig cfg = patternPtr->config();
29 if (cfg.m_rawImage.empty())
30 continue;
31
32 auto r = model->addPattern(cfg);
33 if (!r)
34 continue;
35
36 auto last = model->lastPatternAccess();
37 if (!last)
38 continue;
39
40 cv::Mat trainImg = cfg.m_rawImage.clone();
41 if (!last->learnPattern(trainImg)) {
42 LOG_USER_ERR << "Localization matching failed: learnPattern failed for"
43 << QString::fromStdWString(cfg.m_patternName);
44 }
45 }
46
47 if (model->isEmpty()) {
48 LOG_USER_ERR << "Localization matching failed: matching model is empty.";
49 return false;
50 }
51 return true;
52}

54mtc::MatchResult LocalizationPipeline::runMatchOn(mtc::ImageMatcher &matcher,
55 const CameraWorkspace &workspace,
56 const cv::Mat &image,
57 const mtc::IRobotPickingChecker *pickingChecker,
58 int maxObjects) const {
59 mtc::MatchResult emptyResult;
60 if (image.empty()) {
61 LOG_USER_ERR << "Localization matching failed: source image is empty.";
62 return emptyResult;
63 }
64
65 if (workspace.useWorkspace) {
66 matcher.setMatchingROI(workspace.roi.tl(), workspace.roi.br());
67 }
68 if (workspace.useConditionWorkspace) {
69 matcher.setMatchingConditionROI(workspace.conditionRoi.tl(), workspace.conditionRoi.br());
70 }
71 matcher.setRobotPickingChecker(pickingChecker);
72 matcher.setImageSource(image.clone());
73 matcher.matching(true, maxObjects, workspace.useWorkspace, workspace.useConditionWorkspace);
74 return matcher.match_result;
75}
