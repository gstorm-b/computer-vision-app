168QJsonObject groupToJson(const MatchGroup &group) {

178 if (auto *ecfg = g.edgeConfig())

184 o["patterns"] = patterns;

538ManagerResult PatternGroupManager::validateGroupConfig(
539 const MatchGroupConfig &cfg, const QString &excludeName) const
540{
541 if (cfg.m_groupName.empty())
542 return ManagerResult::fail(QStringLiteral("Group name must not be empty.").toStdWString());
543
544 for (std::shared_ptr<MatchGroup> g : m_groups) {
545 if (QString::fromStdWString(g->name()) == excludeName) continue; // skip self
546 if (g->name() == cfg.m_groupName)
547 return ManagerResult::fail(
548 QStringLiteral("A group named \"%1\" already exists.").arg(cfg.m_groupName).toStdWString());
549 if (g->number() == cfg.m_groupIndex)
550 return ManagerResult::fail(
551 QStringLiteral("Group number %1 already exists.").arg(cfg.m_groupIndex).toStdWString());
552 }
553 return ManagerResult::success();
554}

574bool PatternGroupManager::fromJson(const QJsonObject &obj) {
575 // Reset to empty state. No clearAll() exists, so iterate by number;
576 // groups() returns a snapshot copy, so removing during iteration is safe.
577 const auto existing = m_groups;
578 for (const auto &g : existing)
579 if (g) removeGroupByNumber(g->number());
580
581 bool allOk = true;
582 const QJsonArray groupsArr = obj["groups"].toArray();
583 for (const QJsonValue &gv : groupsArr) {
584 const QJsonObject gObj = gv.toObject();
585 const MatchGroupConfig gcfg = groupConfigFromJson(gObj);
586
587 if (auto r = addGroup(gcfg); !r) {
588 LOG_DEV_ERR << "PatternGroupManager::fromJson – addGroup failed: "
589 << QString::fromStdWString(r.error);
590 allOk = false;
591 continue;
592 }
593
594 const QString groupName = QString::fromStdWString(gcfg.m_groupName);
595 for (const QJsonValue &pv : gObj["patterns"].toArray()) {
596 const MatchPatternConfig pcfg = patternConfigFromJson(pv.toObject());
597 if (auto r = addPattern(groupName, pcfg); !r) {
598 LOG_DEV_ERR << "PatternGroupManager::fromJson – addPattern failed: "
599 << QString::fromStdWString(r.error);
600 allOk = false;
601 }
602 }
603 }
604 return allOk;
605}
