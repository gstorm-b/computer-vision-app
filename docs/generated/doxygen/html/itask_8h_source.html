238 virtual bool fromJson(const QJsonObject& obj) {
239 if (!obj.contains("id") ||
240 !obj.contains("name") ||
241 !obj.contains("taskType") ||
242 !obj.contains("taskConfig")) {
243 LOG_DEV_ERR << "Cannot convert task from json, wrong json format.";
244 LOG_DEV_ERR << QJsonDocument(obj).toJson();
245 return false;
246 }
247
248 TaskType task_type = stringToQEnum(obj["taskType"].toString(), TaskType::UndefineTask);
249 if (task_type != this->taskType()) {
250 LOG_DEV_ERR << "Cannot convert task from json, wrong task type.";
251 LOG_DEV_ERR << obj["taskType"].toString();
252 return false;
253 }
254
255 m_id = obj["id"].toString();
256 m_name = obj["name"].toString();
257
258 if (obj.contains("assignedDeviceIds")) {
259 m_assignedDeviceIds.clear();
260 for (const QJsonValue &v : obj["assignedDeviceIds"].toArray())
261 m_assignedDeviceIds.append(v.toString());
262 }
263
264 if (m_abstract_cfg) {
265 return m_abstract_cfg->fromJson(obj["taskConfig"].toObject());
266 }
267 return true;
268 }
