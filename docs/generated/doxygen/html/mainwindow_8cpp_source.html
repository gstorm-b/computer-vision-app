60QStringList phase5PatternSignatures(mtc::PatternGroupManager *mgr) {
61 QStringList out;
62 if (!mgr) return out;
63 for (const auto &group : mgr->groups()) {
64 if (!group) continue;
65 for (const auto &pattern : group->patterns()) {
66 if (!pattern) continue;
67 const mtc::MatchPatternConfig *c = pattern->patternConfigPtr();
68 if (!c) continue;
69 out << QString::fromStdWString(group->name()) + "/"
70 + QString::fromStdWString(c->m_patternName) + ":"
71 + QString::number(c->m_gripperBoxes.size.width, 'f', 4) + ","
72 + QString::number(c->m_gripperBoxes.size.height, 'f', 4) + ","
73 + QString::number(c->m_gripperBoxes.distance, 'f', 4) + ","
74 + QString::number(c->m_pickingBoxAngle, 'f', 4) + ","
75 + (c->m_usePickingBox ? "1" : "0") + ","
76 + QString::number(c->m_pickingRotationOffset.x, 'f', 4) + ","
77 + QString::number(c->m_pickingRotationOffset.y, 'f', 4) + ","
78 + QString::number(c->m_pickingRotationOffset.z, 'f', 4);
79 }
80 }
81 out.sort();
82 return out;
83}

87void phase5VerifyPatternSchema(const std::shared_ptr<vc::model::Project> &project) {
88 if (!project) return;
89
90 for (const auto &task : project->getCurrentTasks()) {
91 auto *loc = dynamic_cast<vc::model::TaskLocalization *>(task.get());
92 if (!loc) continue;
93 mtc::PatternGroupManager *mgr = loc->patternManager();
94 if (!mgr) continue;
95
96 const QStringList loaded = phase5PatternSignatures(mgr);
97 LOG_USER_INFO << QStringLiteral("[Phase5] Task '%1': %2 pattern(s) loaded from disk.")
98 .arg(loc->name()).arg(loaded.size());
99 for (const QString &line : loaded)
100 LOG_USER_INFO << QStringLiteral("[Phase5] ") + line;
101
102 // Round-trip through the current schema.
103 const QJsonObject doc = mgr->toJson();
104 mtc::PatternGroupManager reloaded;
105 if (!reloaded.fromJson(doc)) {
106 LOG_USER_INFO << QStringLiteral("[Phase5] Task '%1': ROUND-TRIP FAILED — "
107 "fromJson() rejected the document.").arg(loc->name());
108 continue;
109 }
110
111 const QStringList after = phase5PatternSignatures(&reloaded);
112 if (loaded == after) {
113 LOG_USER_INFO << QStringLiteral("[Phase5] Task '%1': round-trip OK "
114 "(schema version %2, %3 pattern(s) identical).")
115 .arg(loc->name())
116 .arg(doc.value("version").toInt(0))
117 .arg(after.size());
118 } else {
119 LOG_USER_INFO << QStringLiteral("[Phase5] Task '%1': ROUND-TRIP MISMATCH.")
120 .arg(loc->name());
121 LOG_USER_INFO << QStringLiteral("[Phase5] before: ") + loaded.join(" | ");
122 LOG_USER_INFO << QStringLiteral("[Phase5] after : ") + after.join(" | ");
123 }
124 }
125}

411void MainWindow::saveToFile(const QString &path, bool saveAs) {
412 const bool isNew = (!QFile::exists(path) || saveAs);
413 const QString timeStr = QDateTime::currentDateTime().toString(Qt::ISODate);
414
415 if (isNew) {
416 if (!ProjectRepo::createNew(path, m_project->name(), timeStr)) {
417 QMessageBox::critical(this,
418 tr("Save Project error"),
419 tr("Cannot create Project file: %1")
420 .arg(ProjectRepo::lastMsg()));
421 return;
422 }
423 }
424
425 QFileInfo finfo(path);
426 if (finfo.completeSuffix() != QLatin1String("vproj") || !finfo.isAbsolute()) {
427 QMessageBox::critical(this, tr("Save Project error"), tr("File format invalid!"));
428 return;
429 }
430
431 m_project->setUpdatedAt(timeStr);
432 if (!ProjectRepo::save(path, *m_project)) {
433 QMessageBox::critical(this,
434 tr("Save Project error"),
435 tr("Cannot save Project: %1").arg(ProjectRepo::lastMsg()));
436 return;
437 }
438
439 LOG_USER_INFO << "Project saved: " << path;
440 m_projectFilePath = path;
441 m_isDirty = false;
442 updateWindowTitle();
443 addRecentPath(path);
444}

549void MainWindow::closeAllDockTabs() {
550 // remove all devices docks
551 QMap<QString, UIDeviceEntry>::const_iterator devIt = m_deviceEntries.cbegin();
552 while (devIt != m_deviceEntries.cend()) {
553 if (devIt.value().dockWidget) {
554 m_dockManager->removeDockWidget(devIt.value().dockWidget);
555 }
556 ++devIt;
557 }
558 m_deviceEntries.clear();
559
560 // remove all tasks docks and its children docks
561 QMap<QString, UITaskEntry>::const_iterator taskIt = m_taskEntries.cbegin();
562 while (taskIt != m_taskEntries.cend()) {
563 QMap<QString, UITaskEntry>::const_iterator childIt = taskIt.value().children.cbegin();
564 while (childIt != taskIt.value().children.cend()) {
565 if (childIt.value().dockWidget) {
566 m_dockManager->removeDockWidget(childIt.value().dockWidget);
567 }
568 ++childIt;
569 }
570
571 if (taskIt.value().dockWidget) {
572 m_dockManager->removeDockWidget(taskIt.value().dockWidget);
573 }
574 ++taskIt;
575 }
576 m_taskEntries.clear();
577}

942void MainWindow::onMoveDeviceRequested(const QString &taskId,
943 const QString &deviceId)
944{
945 if (!m_project) return;
946
947 // Build list of candidate tasks (all except current)
948 QStringList taskNames;
949 QStringList taskIds;
950 for (const auto &pair : m_project->getCurrentTasks()) {
951 if (pair->id() != taskId) {
952 taskNames.append(pair->name());
953 taskIds.append(pair->id());
954 }
955 }
956
957 if (taskIds.isEmpty()) {
958 QMessageBox::information(this, tr("Move Device"),
959 tr("No other tasks available."));
960 return;
961 }
962
963 bool ok = false;
964 const QString chosen = QInputDialog::getItem(
965 this, tr("Move Device"),
966 tr("Select destination task:"),
967 taskNames, 0, false, &ok);
968
969 if (!ok || chosen.isEmpty()) return;
970
971 const QString targetId = taskIds.value(taskNames.indexOf(chosen));
972 if (m_project->moveDeviceToTask(deviceId, taskId, targetId)) {
973 LOG_USER_INFO << QString("Device %1 moved from task %2 to %3")
974 .arg(deviceId, taskId, targetId);
975 }

982void MainWindow::onDeleteDeviceRequested(const QString &taskId,

1008void MainWindow::onDeleteTaskRequested(const QString &taskId)
1009{
1010 if (!m_project) return;
1011
1012 auto task = m_project->taskById(taskId);
1013 if (!task) return;
1014
1015 const auto reply = QMessageBox::question(
1016 this, tr("Delete Task"),
1017 tr("Delete task '%1' and all its devices?\nThis cannot be undone.")
1018 .arg(task->name()),
1019 QMessageBox::Yes | QMessageBox::No);
1020
1021 if (reply != QMessageBox::Yes) return;
1022
1023 // Release all devices owned by this task first
1024 const QStringList devIds = task->assignedDeviceIds();
1025 for (const QString &devId : devIds) {
1026 // task->unassignDevice(devId);
1027 m_project->unassignDeviceFromTask(devId, taskId);
1028 m_project->deviceManager()->releaseDevice(devId);
1029 }
1030
1031 m_project->removeTask(taskId); // emits taskDeleted → onTaskDeleted + tasksChanged → refreshTree
1032 LOG_USER_INFO << QString("Task %1 deleted").arg(taskId);
1033}
