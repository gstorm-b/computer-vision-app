13bool ITask::transitionTaskState(TaskState targetState, const QString &reason)
14{
15 if (m_taskState == targetState) {
16 return true;
17 }
18
19 if (!canTransitionTaskState(m_taskState, targetState)) {

23 }

159void ITask::syncRunnersWithDevices()
160{
161 if (!m_proj) return;
162
163 auto dm = m_proj->deviceManager();
164 if (!dm) return;
165
166 // 1. Register runners for newly assigned devices.
167 // TaskRunner::registerDevice() auto-starts + attaches the runner
168 // if we're already past Idle (Commission / Runtime), so the new
169 // device is immediately ready for the Widget to drive.
170 for (const QString &id : m_assignedDeviceIds) {
171 if (!m_taskRunner->hasRunner(id)) {
172 auto device = dm->deviceById(id);
173 if (device) {
174 m_taskRunner->registerDevice(id, device);
175 }
176 }
177 }
178
179 // 2. Unregister runners for devices that are no longer assigned.
180 // (User clicked "remove device" → free its thread, drop the runner.)
181 for (const QString &id : m_taskRunner->registeredDeviceIds()) {
182 if (!m_assignedDeviceIds.contains(id)) {
183 m_taskRunner->unregisterDevice(id);
184 }
185 }
186}
