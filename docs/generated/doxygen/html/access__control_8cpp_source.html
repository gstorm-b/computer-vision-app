59bool AccessControl::elevate(const QString &password)
60{
61 if (!provider()->verify(password)) {
62 // Logged at USER level on purpose: a run of failed attempts is something the
63 // person reading the event log afterwards should be able to see.
64 LOG_USER_WARN << "Access level change refused: incorrect admin password.";
65 return false;
66 }
67
68 if (m_role == AccessRole::Admin) {
69 return true; // Already there; nothing changed, so no signal.
70 }
71
72 m_role = AccessRole::Admin;
73 LOG_USER_INFO << "Access level changed." << "role=" << roleName(m_role);
74 emit roleChanged(m_role);
75 return true;
76}
