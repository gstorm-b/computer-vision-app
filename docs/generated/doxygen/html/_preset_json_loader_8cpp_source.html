30bool hasOnlyKnownTopLevelFields(const QJsonObject& object)
31{
32 const std::set<QString> known = {
33 "schema", "identity", "units", "topology", "links", "joints", "frames",
34 "tools", "defaultTool", "posture", "solver", "sources", "metadata",
35 };
36 for (auto it = object.begin(); it != object.end(); ++it) {
37 if (known.find(it.key()) == known.end()) {
38 return false;
39 }
40 }
41 return true;
42}

79Joint parseJoint(const QJsonObject& object)
80{
81 Joint joint;
82 joint.id = stringField(object, "id");
83 joint.type = jointTypeFromString(object.value("type").toString());
84 joint.parentLinkId = stringField(object, "parent");
85 joint.childLinkId = stringField(object, "child");
86 joint.origin = poseFromObject(object.value("origin").toObject());
87
88 const QJsonArray axis = object.value("axis").toArray();
89 joint.axis = Eigen::Vector3d(axis.at(0).toDouble(), axis.at(1).toDouble(), axis.at(2).toDouble());
90
91 if (object.contains("limits") && object.value("limits").isObject()) {
92 const QJsonObject limits = object.value("limits").toObject();
93 JointLimits parsed;
94 parsed.lower = limits.value("lower").toDouble();
95 parsed.upper = limits.value("upper").toDouble();
96 if (limits.contains("velocity") && !limits.value("velocity").isNull()) {
97 parsed.velocity = limits.value("velocity").toDouble();
98 }
99 if (limits.contains("acceleration") && !limits.value("acceleration").isNull()) {
100 parsed.acceleration = limits.value("acceleration").toDouble();
101 }
102 joint.limits = parsed;
103 }
104 joint.home = object.value("home").toDouble();
105 for (const QJsonValue& alias : object.value("aliases").toArray()) {
106 joint.aliases.push_back(alias.toString().toStdString());
107 }
108 return joint;
109}

148Result<SerialRobotConfig> PresetJsonLoader::loadJson(const std::string& json)
149{
150 QJsonParseError error;
151 const QJsonDocument document = QJsonDocument::fromJson(QByteArray::fromStdString(json), &error);
152 if (error.error != QJsonParseError::NoError || !document.isObject()) {
153 return invalid("preset JSON is not a valid object");
154 }
155
156 const QJsonObject root = document.object();
157 if (!hasOnlyKnownTopLevelFields(root)) {
158 return invalid("preset contains unknown top-level fields");
159 }
160 if (root.value("schema").toString() != "robot-kinematics-preset/v1") {
161 return invalid("unsupported preset schema");
162 }
163 const QJsonObject units = root.value("units").toObject();
164 if (units.value("length").toString() != "m" || units.value("angle").toString() != "rad") {
165 return invalid("preset units must be m/rad");
166 }
167
168 SerialRobotConfig config;
169 const QJsonObject identity = root.value("identity").toObject();
170 config.identity = RobotIdentity{
171 stringField(identity, "vendor"),
172 stringField(identity, "model"),
173 stringField(identity, "name"),
174 stringField(identity, "revision"),
175 };
176
177 const QJsonObject topology = root.value("topology").toObject();
178 config.topology = RobotTopologyType::Serial;
179 config.dof = topology.value("dof").toInt();
180
181 for (const QJsonValue& link : root.value("links").toArray()) {
182 config.links.push_back(Link{stringField(link.toObject(), "id")});
183 }
184 for (const QJsonValue& joint : root.value("joints").toArray()) {
185 config.joints.push_back(parseJoint(joint.toObject()));
186 }
187
188 const QJsonObject frames = root.value("frames").toObject();
189 config.frames.baseLinkId = stringField(frames, "base");
190 config.frames.flangeLinkId = stringField(frames, "flange");
191 for (const QJsonValue& frameValue : frames.value("userFrames").toArray()) {
192 const QJsonObject frame = frameValue.toObject();
193 config.frames.userFrames.push_back(UserFrame{
194 stringField(frame, "id"),
195 stringField(frame, "parent"),
196 poseFromObject(frame.value("transform").toObject()),
197 });
198 }
199
200 for (const QJsonValue& toolValue : root.value("tools").toArray()) {
201 const QJsonObject tool = toolValue.toObject();
202 config.tools.push_back(Tool{
203 stringField(tool, "id"),
204 stringField(tool, "name"),
205 poseFromObject(tool.value("flangeToTcp").toObject()),
206 });
207 }
208 config.defaultToolId = root.value("defaultTool").toString().toStdString();
209
210 parsePosture(root.value("posture").toObject(), config);
211
212 const QJsonObject solver = root.value("solver").toObject();
213 config.solver.defaultSolver = stringField(solver, "default");
214 const QJsonObject parameters = solver.value("parameters").toObject();
215 for (auto it = parameters.begin(); it != parameters.end(); ++it) {
216 if (it.value().isDouble()) {
217 config.solver.numericParameters[it.key().toStdString()] = it.value().toDouble();
218 }
219 }
220
221 for (const QJsonValue& sourceValue : root.value("sources").toArray()) {
222 const QJsonObject source = sourceValue.toObject();
223 SourceReference parsed;
224 parsed.type = stringField(source, "type");
225 parsed.title = stringField(source, "title");
226 parsed.reference = stringField(source, "reference");
227 for (const QJsonValue& appliesTo : source.value("appliesTo").toArray()) {
228 parsed.appliesTo.push_back(appliesTo.toString().toStdString());
229 }
230 config.sources.push_back(parsed);
231 }
232
233 const QJsonObject metadata = root.value("metadata").toObject();
234 for (auto it = metadata.begin(); it != metadata.end(); ++it) {
235 if (it.value().isString()) {
236 config.metadata[it.key().toStdString()] = it.value().toString().toStdString();
237 }
238 }
239
240 const ModelValidationResult validation = RobotModelValidator::validateSerialRobotConfig(config);
241 if (!validation.ok()) {
242 return invalid(validation.issues.front().message);
243 }
244 return Result<SerialRobotConfig>::success(config);
245}
