110void VisionResultViewerWidget::setOverlay(const VisionResultOverlay &overlay)
111{
112 m_overlay = overlay;
113 // The toolbar toggles are the source of truth for overlay visibility. A
114 // freshly produced result overlay carries default visibility, so keep the
115 // user's current selection instead of resetting the toolbar (which would
116 // force the user to re-toggle to see the overlay items again).
117 m_overlay.visibility = visibilityFromActions();
118 m_canvas->clearSelectedResultObject();
119 m_canvas->setResultOverlay(m_overlay);
120}
