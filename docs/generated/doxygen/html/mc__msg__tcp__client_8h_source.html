151 const MsgErrorState SendMsg(QByteArray &buffer) override {
152 if (buffer.isEmpty()) {
153 m_error_state = MsgErrorState::BufferEmpty;
154 return m_error_state;
155 }
156
157 try {
158 m_socket->write(buffer);
159 if (!m_socket->waitForBytesWritten(m_config.m_writeTimeout)) {
160 m_error_state = MsgErrorState::WriteTimeout;
161 return m_error_state;

163 // OLOG_INFO << "sent string:" << cmd;
164 } catch (const std::exception& e){
165 // OLOG_CRITICAL << "send string" << cmd << ", fail:" << e.what();
166 m_error_state = MsgErrorState::ErrorOcurred;
167 m_error_description = QString::fromStdString(e.what());
168 return m_error_state;

172 return m_error_state;

184 const MsgErrorState ReceiveMsg(QByteArray &buffer, int wait_buffer = 1) override {
185 m_socket->waitForReadyRead(wait_buffer);
186
187 if (m_socket->bytesAvailable() < 1) {
188 m_total_wait_time += wait_buffer;
189 if (m_total_wait_time > m_config.m_responseTimeout) {
190 m_error_state = MsgErrorState::ResponseTimeout;
191 return m_error_state;
192 }
193
194 m_error_state = MsgErrorState::BufferEmpty;
195 return m_error_state;
196 }
197
198 m_total_wait_time = 0;
199 QByteArray read_bytes = m_socket->readAll();
200 buffer.append(read_bytes);
201 m_error_state = MsgErrorState::NoError;
202 return m_error_state;
203 }
