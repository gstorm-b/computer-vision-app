45cv::Mat CalibrationBoard::binarize(const cv::Mat& image, double* usedThreshold) const
46{
47 cv::Mat gray;
48 if (image.channels() == 1) {
49 gray = image;
50 } else {
51 cv::cvtColor(image, gray, cv::COLOR_BGR2GRAY);
52 }
53
54 cv::Mat binary;
55 const int t = binarizeThreshold();
56 double applied = 0.0;
57 if (t < 0) {
58 // Auto: Otsu picks the threshold; cv::threshold returns the value used.
59 applied = cv::threshold(gray, binary, 0, 255,
60 cv::THRESH_BINARY_INV | cv::THRESH_OTSU);
61 } else {
62 applied = static_cast<double>(t);
63 cv::threshold(gray, binary, applied, 255, cv::THRESH_BINARY_INV);
64 }
65 if (usedThreshold) *usedThreshold = applied;
66 return binary;
67}

101uint32_t pngCrc32(const uint8_t* data, size_t len)
102{
103 static uint32_t table[256];
104 static bool ready = false;
105 if (!ready) {
106 for (uint32_t i = 0; i < 256; ++i) {
107 uint32_t c = i;
108 for (int k = 0; k < 8; ++k)
109 c = (c & 1u) ? (0xEDB88320u ^ (c >> 1)) : (c >> 1);
110 table[i] = c;
111 }
112 ready = true;
113 }
114 uint32_t c = 0xFFFFFFFFu;
115 for (size_t i = 0; i < len; ++i)
116 c = table[(c ^ data[i]) & 0xFFu] ^ (c >> 8);
117 return c ^ 0xFFFFFFFFu;
118}

136bool writePrintablePng(const std::string& path,
137 const cv::Mat& img,
138 double pxPerMm)
139{
140 if (img.empty() || pxPerMm <= 0.0) return false;
141
142 std::vector<uchar> encoded;
143 if (!cv::imencode(".png", img, encoded)) return false;
144
145 static const uint8_t sig[8] = { 0x89, 'P','N','G', 0x0D, 0x0A, 0x1A, 0x0A };
146 if (encoded.size() < 16 ||
147 !std::equal(sig, sig + 8, encoded.begin())) return false;
148
149 size_t insertPos = 0;
150 size_t pos = 8;
151 while (pos + 12 <= encoded.size()) {
152 const uint32_t len = (uint32_t(encoded[pos ]) << 24) |
153 (uint32_t(encoded[pos + 1]) << 16) |
154 (uint32_t(encoded[pos + 2]) << 8) |
155 uint32_t(encoded[pos + 3]);
156 const char* type = reinterpret_cast<const char*>(&encoded[pos + 4]);
157 if (std::memcmp(type, "IDAT", 4) == 0) { insertPos = pos; break; }
158 pos += 12u + len;
159 }
160 if (insertPos == 0) return false;
161
162 const uint32_t pxPerMeter = static_cast<uint32_t>(std::round(pxPerMm * 1000.0));
163 std::vector<uint8_t> chunk;
164 chunk.reserve(21);
165 writeU32BE(chunk, 9);
166 const size_t typeBegin = chunk.size();
167 chunk.push_back('p'); chunk.push_back('H');
168 chunk.push_back('Y'); chunk.push_back('s');
169 writeU32BE(chunk, pxPerMeter);
170 writeU32BE(chunk, pxPerMeter);
171 chunk.push_back(0x01); // unit = meter
172 const uint32_t crc = pngCrc32(chunk.data() + typeBegin, chunk.size() - typeBegin);
173 writeU32BE(chunk, crc);
174
175 std::ofstream f(path, std::ios::binary);
176 if (!f) return false;
177 f.write(reinterpret_cast<const char*>(encoded.data()),
178 static_cast<std::streamsize>(insertPos));
179 f.write(reinterpret_cast<const char*>(chunk.data()),
180 static_cast<std::streamsize>(chunk.size()));
181 f.write(reinterpret_cast<const char*>(encoded.data() + insertPos),
182 static_cast<std::streamsize>(encoded.size() - insertPos));
183 return f.good();
184}
