736void EditPatternWizard::goToStep(int step) {
737 if (step < 0 || step > STEP_LAST) return;
738 if (step > m_currentStep && !currentStepValid()) return;
739
740 m_currentStep = step;
741 m_stack->setCurrentIndex(step);
742
743 if (step == STEP_PICK && m_pickCanvas) {
744 m_pickCanvas->setPick({m_new.pickX, m_new.pickY});
745 m_pickCanvas->setPickAngle(m_new.pickAngle);
746 }
747 if (step == STEP_BOX && m_boxCanvas) {
748 m_boxCanvas->setPick({m_new.pickX, m_new.pickY});
749 m_boxCanvas->setBoxConfig(m_new.pickBoxW, m_new.pickBoxH,
750 m_new.pickBoxDist, m_new.pickBoxAngle);
751 }
752 if (step == STEP_FINISH && m_finishCanvas) {
753 m_finishCanvas->setPick({m_new.pickX, m_new.pickY});
754 m_finishCanvas->setPickAngle(m_new.pickAngle);
755 m_finishCanvas->setBoxConfig(m_new.pickBoxW, m_new.pickBoxH,
756 m_new.pickBoxDist, m_new.pickBoxAngle);
757 refreshDiff();
758 }
759
760 updateStepRail();
761 updateFooterStatus();
762 m_btnBack->setEnabled(step > 0);
763
764 if (step == STEP_LAST) {
765 m_btnNext->setText("✓ " + tr("Apply Changes"));
766 disconnect(m_btnNext, nullptr, this, nullptr);
767 connect(m_btnNext, &QPushButton::clicked, this, &EditPatternWizard::onApply);
768 } else {
769 m_btnNext->setText(tr("Next") + " →");
770 disconnect(m_btnNext, nullptr, this, nullptr);
771 connect(m_btnNext, &QPushButton::clicked, this, &EditPatternWizard::onNext);
772 }
773
774 if (m_subtitleLabel) {
775 m_subtitleLabel->setText(QString("%1 · %2 %3 of %4 — %5")
776 .arg(tr("Group:"), m_groupName)
777 .arg(step + 1)
778 .arg(STEP_COUNT)
779 .arg(stepSublines().at(step)));
780 }
781
782 // Park focus on Next on every step, including the initial goToStep(0) from the
783 // constructor. Whatever holds focus is what a stray Return or Space activates, so the
784 // only safe resting place is the button that moves the user forward.
785 m_btnNext->setFocus(Qt::OtherFocusReason);
786}
