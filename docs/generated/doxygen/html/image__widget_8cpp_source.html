137 } else {
138 m_pixmapItem->setPixmap(pixmap);

140
141 m_scene->setSceneRect(pixmap.rect().adjusted(-15, -15, 10, 10));
142 // fit image with current window size
143 m_pixmap_bounding_rect = m_pixmapItem->boundingRect();
144 this->fitInView(m_pixmap_bounding_rect, Qt::KeepAspectRatio);
