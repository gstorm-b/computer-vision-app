47Eigen::VectorXd clampToLimits(const SerialRobotConfig& config, const Eigen::VectorXd& q)
48{
49 Eigen::VectorXd clamped = q;
50 int index = 0;
51 for (const Joint& joint : config.joints) {
52 if (joint.type != JointType::Revolute && joint.type != JointType::Prismatic) {
53 continue;
54 }
55 if (joint.limits.has_value()) {
56 clamped[index] = clamp(clamped[index], joint.limits->lower, joint.limits->upper);
57 }
58 ++index;
59 }
60 return clamped;
61}

78JointVector midpointJoint(const SerialRobotConfig& config)
79{
80 Eigen::VectorXd q(ForwardKinematics::movableJointCount(config));
81 int index = 0;

83 if (joint.type != JointType::Revolute && joint.type != JointType::Prismatic) {

115double signedHalfRangeValue(const Joint& joint, int sign)
116{
117 if (!joint.limits.has_value()) {
118 return joint.home;
119 }
120 if (sign < 0) {
121 return 0.5 * (joint.limits->lower + std::min(0.0, joint.limits->upper));
122 }
123 return 0.5 * (std::max(0.0, joint.limits->lower) + joint.limits->upper);

132void applyPostureBranchSeedValue(Eigen::VectorXd& q,
133 const SerialRobotConfig& config,
134 int movableIndex,
135 const std::optional<int>& branch)
136{
137 if (!branch.has_value() || movableIndex < 0 || movableIndex >= q.size()) {
138 return;
139 }
140
141 int index = 0;
142 for (const Joint& joint : config.joints) {
143 if (joint.type != JointType::Revolute && joint.type != JointType::Prismatic) {
144 continue;
145 }
146 if (index == movableIndex) {
147 q[index] = signedHalfRangeValue(joint, *branch);
148 return;
149 }

159std::optional<JointVector> postureSeed(const SerialRobotConfig& config, const ArmPosture& posture)
160{
161 if (config.posture.resolver != "serial_6dof_shoulder_elbow_wrist"
162 || ForwardKinematics::movableJointCount(config) < 6) {
163 return std::nullopt;

165
166 Eigen::VectorXd q = midpointJoint(config).values();

168 applyPostureBranchSeedValue(q, config, 2, posture.elbow);

181std::vector<JointVector> buildSeeds(const SerialRobotConfig& config,
182 const IKRequest& request,
183 const NumericalIKDefaults& defaults)
184{
185 std::vector<JointVector> seeds;
186 if (request.previousJoint.has_value()) {
187 appendUniqueSeed(seeds, *request.previousJoint, defaults.duplicateJointTolerance);
188 }
189 if (request.seedJoint.has_value()) {
190 appendUniqueSeed(seeds, *request.seedJoint, defaults.duplicateJointTolerance);
191 }
192 if (request.posture.has_value()) {
193 const std::optional<JointVector> seed = postureSeed(config, *request.posture);
194 if (seed.has_value()) {
195 appendUniqueSeed(seeds, *seed, defaults.duplicateJointTolerance);
196 }
197 }
198
199 appendUniqueSeed(seeds, homeJoint(config), defaults.duplicateJointTolerance);
200 appendUniqueSeed(seeds, midpointJoint(config), defaults.duplicateJointTolerance);
201
202 const int dof = ForwardKinematics::movableJointCount(config);
203 for (int s = 0; static_cast<int>(seeds.size()) < defaults.maxSeeds && s < defaults.maxSeeds * 4; ++s) {
204 Eigen::VectorXd q(dof);
205 int index = 0;
206 for (const Joint& joint : config.joints) {
207 if (joint.type != JointType::Revolute && joint.type != JointType::Prismatic) {
208 continue;
209 }
210 const double lower = joint.limits.has_value() ? joint.limits->lower : -1.0;
211 const double upper = joint.limits.has_value() ? joint.limits->upper : 1.0;
212 const double fraction = std::fmod((s + 1) * (index + 1) * 0.6180339887498948, 1.0);
213 q[index] = lower + fraction * (upper - lower);
214 ++index;
215 }
216 appendUniqueSeed(seeds, JointVector(q), defaults.duplicateJointTolerance);
217 }
218 return seeds;
219}

227Residual computeResidual(const Pose& current, const Pose& target, const NumericalIKDefaults& defaults)
228{
229 Residual residual;
230 const Eigen::Vector3d position = target.translation_m() - current.translation_m();
231 const Eigen::Matrix3d rotationError = target.isometry().linear() * current.isometry().linear().transpose();
232 const Eigen::AngleAxisd angleAxis(rotationError);
233 Eigen::Vector3d orientation = Eigen::Vector3d::Zero();
234 if (std::abs(angleAxis.angle()) > 1e-14) {
235 orientation = angleAxis.axis() * angleAxis.angle();
236 }
237
238 residual.positionNorm = position.norm();
239 residual.orientationNorm = std::abs(angleAxis.angle());
240 residual.vector.template segment<3>(0) = defaults.positionResidualWeight * position;
241 residual.vector.template segment<3>(3) = defaults.orientationResidualWeight * orientation;
242 return residual;
243}

250Eigen::MatrixXd geometricJacobian(const FkChain& chain, const Pose& currentTool)
251{
252 Eigen::MatrixXd jacobian(6, static_cast<Eigen::Index>(chain.joints.size()));
253 const Eigen::Vector3d end = currentTool.translation_m();
254
255 for (Eigen::Index i = 0; i < static_cast<Eigen::Index>(chain.joints.size()); ++i) {
256 const JointFrameData& joint = chain.joints[static_cast<std::size_t>(i)];
257 if (joint.type == JointType::Revolute) {
258 jacobian.block<3, 1>(0, i) = joint.axisInBase.cross(end - joint.originInBase);
259 jacobian.block<3, 1>(3, i) = joint.axisInBase;
260 } else if (joint.type == JointType::Prismatic) {
261 jacobian.block<3, 1>(0, i) = joint.axisInBase;
262 jacobian.block<3, 1>(3, i).setZero();
263 } else {
264 jacobian.block<6, 1>(0, i).setZero();
265 }
266 }
267 return jacobian;
268}

307SolveAttempt solveFromSeed(const SerialRobotConfig& config,
308 const IKSolveContext& context,
309 const JointVector& seed,
310 const NumericalIKDefaults& defaults)
311{
312 const int dof = ForwardKinematics::movableJointCount(config);
313 Eigen::VectorXd q = clampToLimits(config, seed.values());
314 double damping = defaults.initialDamping;
315 double previousCost = std::numeric_limits<double>::infinity();
316
317 for (int iteration = 0; iteration < defaults.maxIterations; ++iteration) {
318 const JointVector currentJoints(q);
319 const FkChain chain = ForwardKinematics::computeChain(config, currentJoints);
320 const Pose current = chain.flangeInBase;
321 const Residual residual = computeResidual(current, context.targetFlangeInBase, defaults);
322
323 if (residual.positionNorm <= context.request.options.maxPositionError_m
324 && residual.orientationNorm <= context.request.options.maxOrientationError_rad) {
325 IKSolution solution = scoreSolution(config, context.request, currentJoints,
326 residual.positionNorm, residual.orientationNorm, defaults);
327 if (context.request.options.requirePosture && context.request.posture.has_value()
328 && !IKSolutionRanker::postureMatches(solution.posture, *context.request.posture)) {
329 return SolveAttempt{false, {}, KinematicsStatus::PostureConstraintUnsatisfied};
330 }
331
332 SolveAttempt attempt;
333 attempt.converged = true;
334 attempt.solution = std::move(solution);
335 return attempt;
336 }
337
338 const double cost = residual.vector.squaredNorm();
339 if (std::abs(previousCost - cost) <= defaults.costTolerance) {
340 damping = std::min(defaults.maxDamping, damping * defaults.dampingIncreaseFactor);
341 }
342 previousCost = cost;
343
344 const Eigen::MatrixXd jacobian = geometricJacobian(chain, current);
345 const Eigen::Matrix<double, 6, 6> normal =
346 jacobian * jacobian.transpose()
347 + Eigen::Matrix<double, 6, 6>::Identity() * damping * damping;
348 Eigen::VectorXd step = jacobian.transpose() * normal.ldlt().solve(residual.vector);
349 if (!step.allFinite()) {
350 return SolveAttempt{false, {}, KinematicsStatus::NumericalError};
351 }
352 const double stepNorm = step.norm();
353 if (stepNorm <= defaults.stepTolerance) {
354 return SolveAttempt{false, {}, KinematicsStatus::NoConvergedSolution};
355 }
356 if (stepNorm > defaults.maxJointStepNorm) {
357 step *= defaults.maxJointStepNorm / stepNorm;
358 }
359
360 const Eigen::VectorXd candidate = clampToLimits(config, q + step);
361 const Pose candidatePose = ForwardKinematics::flangePose(config, JointVector(candidate));
362 const Residual candidateResidual = computeResidual(candidatePose, context.targetFlangeInBase, defaults);
363 if (candidateResidual.vector.squaredNorm() < cost) {
364 q = candidate;
365 damping = std::max(defaults.minDamping, damping * defaults.dampingDecreaseFactor);
366 } else {
367 damping = std::min(defaults.maxDamping, damping * defaults.dampingIncreaseFactor);
368 }
369 }
370
371 if (dof == 0) {
372 return SolveAttempt{false, {}, KinematicsStatus::InvalidRobotConfig};
373 }
374 return SolveAttempt{false, {}, KinematicsStatus::MaxIterationsReached};
375}

424IKResult NumericalIKSolver::solveAll(const SerialRobotConfig& config, const IKSolveContext& context) const
425{
426 const ModelValidationResult model = RobotModelValidator::validateSerialRobotConfig(config);
427 if (!model.ok()) {
428 return IKResult{model.status(), {}, model.issues.front().message};
429 }
430 if (!finitePose(context.targetFlangeInBase)) {
431 return IKResult{IKStatus::InvalidRequest, {}, "target pose contains non-finite values"};
432 }
433
434 const int dof = ForwardKinematics::movableJointCount(config);
435 if ((context.request.seedJoint.has_value() && context.request.seedJoint->size() != dof)
436 || (context.request.previousJoint.has_value() && context.request.previousJoint->size() != dof)) {
437 return IKResult{IKStatus::JointDimensionMismatch, {}, "seed or previous joint dimension mismatch"};
438 }
439
440 IKResult result;
441 result.status = IKStatus::NoConvergedSolution;
442 KinematicsStatus lastFailure = KinematicsStatus::NoConvergedSolution;
443
444 const std::vector<JointVector> seeds = buildSeeds(config, context.request, defaults_);
445 for (const JointVector& seed : seeds) {
446 const JointLimitCheck seedLimits = JointLimitValidator::validate(config, seed);
447 if (seedLimits.status == KinematicsStatus::JointDimensionMismatch) {
448 return IKResult{IKStatus::JointDimensionMismatch, {}, "generated seed dimension mismatch"};
449 }
450
451 const SolveAttempt attempt = solveFromSeed(config, context, seed, defaults_);
452 if (attempt.converged) {
453 if (!duplicateSolution(result.solutions, attempt.solution, defaults_.duplicateJointTolerance)) {
454 result.solutions.push_back(attempt.solution);
455 }
456 if (context.request.options.maxSolutions > 0
457 && static_cast<int>(result.solutions.size()) >= context.request.options.maxSolutions) {
458 break;
459 }
460 } else {
461 lastFailure = attempt.failureStatus;
462 }
463 }
464
465 if (!result.solutions.empty()) {
466 std::sort(result.solutions.begin(), result.solutions.end(), [](const IKSolution& a, const IKSolution& b) {
467 return a.score.totalCost < b.score.totalCost;
468 });
469 result.status = IKStatus::Ok;
470 return result;
471 }
472
473 result.status = lastFailure;
474 result.message = "no converged numerical IK solution";
475 return result;
476}
