180 CollisionState checkCollisionObject2(const cv::Mat &contourMask) {
181 m_collisionState = CollisionState::Outside;
182 m_hasCollision = false;
183 if (contourMask.empty()) {
184 return m_collisionState;
185 }
186
187 // Union bounding box of both jaw quads, clamped to the mask. Jaws may
188 // legitimately extend off-image; the off-image part cannot overlap any
189 // contour, so clamping is safe.
190 std::vector<cv::Point2f> allPts;
191 allPts.reserve(m_gripperBox.box_left_pts.size() + m_gripperBox.box_right_pts.size());
192 allPts.insert(allPts.end(), m_gripperBox.box_left_pts.begin(), m_gripperBox.box_left_pts.end());
193 allPts.insert(allPts.end(), m_gripperBox.box_right_pts.begin(), m_gripperBox.box_right_pts.end());
194 if (allPts.empty()) {
195 return m_collisionState;
196 }
197
198 cv::Rect bbox = cv::boundingRect(allPts) & cv::Rect(0, 0, contourMask.cols, contourMask.rows);
199 if (bbox.width <= 0 || bbox.height <= 0) {
200 return m_collisionState;
201 }
202
203 // Rasterise the two jaws into a bbox-sized mask, shifted into
204 // crop-local coordinates. Each jaw is a convex quad.
205 cv::Mat gripperCrop = cv::Mat::zeros(bbox.size(), CV_8UC1);
206 const cv::Point shift = bbox.tl();
207 fillJawInto(gripperCrop, m_gripperBox.box_left_pts, shift);
208 fillJawInto(gripperCrop, m_gripperBox.box_right_pts, shift);
209
210 // contourMask(bbox) is a view (no copy); AND it with the jaw mask.
211 cv::Mat inter;
212 cv::bitwise_and(gripperCrop, contourMask(bbox), inter);
213
214 const int interCount = cv::countNonZero(inter);
215 const int gripperCount = cv::countNonZero(gripperCrop);
216
217 if (interCount == 0) {
218 m_collisionState = CollisionState::Outside;
219 } else if (gripperCount > 0 && interCount == gripperCount) {
220 m_collisionState = CollisionState::Inside;
221 } else {
222 m_collisionState = CollisionState::Collision;
223 }
224
225 m_hasCollision = (m_collisionState != CollisionState::Outside);
226 return m_collisionState;
227 }

281 static void fillJawInto(cv::Mat &mask, const std::vector<cv::Point2f> &pts,
282 const cv::Point &shift) {
283 if (pts.size() < 3) {
284 return;
285 }
286 std::vector<cv::Point> ipts;
287 ipts.reserve(pts.size());

289 ipts.emplace_back(cvRound(p.x) - shift.x, cvRound(p.y) - shift.y);
290 }
291 cv::fillConvexPoly(mask, ipts, cv::Scalar(255));

298 void drawMaskIntoImage(cv::Mat &image, cv::Mat &mask, cv::Scalar color) {
299 if (mask.empty() || image.empty()) {
300 return;
301 }
302
303 // wrap always counter clockwise
304 double angle = -matched_Angle;
305
306 // rotate mask
307 cv::Mat rotMat = cv::getRotationMatrix2D(cv::Point2f(0,0), angle, 1.0);
308
309 // get mask bounding box
310 std::vector<cv::Point2f> corners = {
311 cv::Point2f(0,0),
312 cv::Point2f(mask.cols,0),
313 cv::Point2f(mask.cols, mask.rows),
314 cv::Point2f(0,mask.rows)
315 };
316
317 // transform rotated box
318 std::vector<cv::Point2f> rotatedCorners;
319 cv::transform(corners, rotatedCorners, rotMat);
320
321 cv::Rect2f bbox = cv::boundingRect(rotatedCorners);
322
323 rotMat.at<double>(0,2) -= bbox.x;
324 rotMat.at<double>(1,2) -= bbox.y;
325
326 cv::Mat rotatedMask;
327 cv::warpAffine(mask, rotatedMask, rotMat, bbox.size(),
328 cv::INTER_NEAREST, cv::BORDER_CONSTANT, cv::Scalar(0));
329
330 // get raw bouding box
331 std::vector<cv::Point2f> raw_corners = {
332 cv::Point2f(point_LT),
333 cv::Point2f(point_LB),
334 cv::Point2f(point_RT),
335 cv::Point2f(point_RB)
336 };
337 cv::Rect raw_bouding = cv::boundingRect(raw_corners);
338
339 cv::Rect roi(raw_bouding.x, raw_bouding.y, rotatedMask.cols, rotatedMask.rows);
340 roi = roi & cv::Rect(0, 0, image.cols, image.rows);
341
342 cv::Mat roiImg = image(roi);
343 cv::Point mask_new_tf((raw_bouding.x < 0) ? -raw_bouding.x : 0,
344 (raw_bouding.y < 0) ? -raw_bouding.y : 0);
345 cv::Mat maskROI = rotatedMask(cv::Rect(mask_new_tf.x, mask_new_tf.y,roi.width,roi.height));
346 roiImg.setTo(color, maskROI);
