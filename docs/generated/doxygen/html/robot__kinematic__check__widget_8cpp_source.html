92Eigen::Vector3d rpyDegFromPose(const Pose &pose) {
93 const Eigen::Matrix3d r = pose.isometry().rotation();
94 const double pitch = std::atan2(-r(2, 0), std::sqrt(r(0, 0) * r(0, 0) + r(1, 0) * r(1, 0)));
95 const double yaw = std::atan2(r(1, 0), r(0, 0));
96 const double roll = std::atan2(r(2, 1), r(2, 2));
97 return Eigen::Vector3d(units::toDeg(roll), units::toDeg(pitch), units::toDeg(yaw));
98}

136QString resolveAssetPath(const QString &relative) {
137 const QFileInfo direct(relative);
138 if (direct.exists()) return direct.absoluteFilePath();
139
140 QStringList roots;
141 roots << QDir::currentPath() << QCoreApplication::applicationDirPath();
142 QDir walker(QCoreApplication::applicationDirPath());
143 for (int i = 0; i < 8 && walker.cdUp(); ++i)
144 roots << walker.absolutePath();
145 roots.removeDuplicates();
146
147 for (const QString &root : roots) {
148 const QFileInfo candidate(QDir(root).filePath(relative));
149 if (candidate.exists()) return candidate.absoluteFilePath();
150 }
151 return {};

158const MeshCollisionProfile *simplifiedMeshProfile() {
159 static const std::optional<MeshCollisionProfile> profile =
160 []() -> std::optional<MeshCollisionProfile> {
161 QString path;
162 const QFileInfo deployed(QDir(QCoreApplication::applicationDirPath())
163 .filePath(QString::fromLatin1(kSimplifiedProfileDeployedRel)));
164 if (deployed.exists())
165 path = deployed.absoluteFilePath();
166 else
167 path = resolveAssetPath(QString::fromLatin1(kSimplifiedProfileSourceRel));
168 if (path.isEmpty())
169 return std::nullopt;
170 const Result<MeshCollisionProfile> res =
171 MeshCollisionProfileJsonLoader::loadFile(path.toStdString());
172 if (!res.ok())
173 return std::nullopt;
174 return res.value;
175 }();
176 return profile ? &(*profile) : nullptr;
177}

181enum class CollisionState { Disabled, Unavailable, Error, Free, Colliding };

190CollisionState checkSelfCollision(const vc::device::RobotKinematicCheckConfig &cfg,
191 const SerialRobotConfig &robot,
192 const JointVector &joints) {
193 if (!cfg.collisionCheckEnabled) return CollisionState::Disabled;
194 const MeshCollisionProfile *profile = simplifiedMeshProfile();
195 if (!profile) return CollisionState::Unavailable;
196
197 MeshCollisionCheckRequest req;
198 req.joints = joints;
199 const CollisionCheckResult res = CollisionBackends::checkMesh(robot, *profile, req);
200 if (!res.ok()) return CollisionState::Error;
201 return res.hasCollision ? CollisionState::Colliding : CollisionState::Free;
202}

212 case CollisionState::Colliding: return QObject::tr(" | SELF-COLLISION");
213 }

471void RobotKinematicCheckWidget::onComputeForward() {
472 const vc::device::RobotKinematicCheckConfig cfg = config();
473 SerialRobotConfig robot;
474 if (!buildRobotConfig(cfg, robot)) {
475 ui->lbl_tester_status->setText(tr("FK: select a robot preset first"));
476 return;
477 }
478 if (ForwardKinematics::movableJointCount(robot) != 6) {
479 ui->lbl_tester_status->setText(tr("FK: preset is not a 6-axis arm"));
480 return;
481 }
482
483 const JointVector q = JointVector::fromDegrees({
484 ui->spb_j1->value(), ui->spb_j2->value(), ui->spb_j3->value(),
485 ui->spb_j4->value(), ui->spb_j5->value(), ui->spb_j6->value() });
486
487 const Pose tcpPose = ForwardKinematics::toolPose(robot, q, flangeToTcpPose(cfg));
488 const Eigen::Vector3d t = tcpPose.translation_m();
489 const Eigen::Vector3d rpy = rpyDegFromPose(tcpPose);
490
491 ui->spb_pose_x->setValue(units::toMm(t.x()));
492 ui->spb_pose_y->setValue(units::toMm(t.y()));
493 ui->spb_pose_z->setValue(units::toMm(t.z()));
494 ui->spb_pose_roll->setValue(rpy.x());
495 ui->spb_pose_pitch->setValue(rpy.y());
496 ui->spb_pose_yaw->setValue(rpy.z());
497
498 const CollisionState collision = checkSelfCollision(cfg, robot, q);
499 ui->lbl_tester_status->setText(tr("FK: ok") + collisionSuffix(collision));
500}

507void RobotKinematicCheckWidget::onComputeInverse() {
508 const vc::device::RobotKinematicCheckConfig cfg = config();
509 SerialRobotConfig robot;
510 if (!buildRobotConfig(cfg, robot)) {
511 ui->lbl_tester_status->setText(tr("IK: select a robot preset first"));
512 return;
513 }
514
515 IKRequest req;
516 req.targetPose = Pose::fromXYZRPY_mm_deg(
517 ui->spb_pose_x->value(), ui->spb_pose_y->value(), ui->spb_pose_z->value(),
518 ui->spb_pose_roll->value(), ui->spb_pose_pitch->value(), ui->spb_pose_yaw->value());
519 req.tool = ToolId{kPickingToolId};
520
521 const IKResult ik = SerialRobotKinematics(robot).solve(req);
522 if (!ik.ok()) {
523 ui->lbl_tester_status->setText(tr("IK: %1").arg(statusToString(ik.status)));
524 return;
525 }
526
527 // Show the best solution's joints (degrees).
528 const std::vector<double> sol = ik.best().joints.toDegrees();
529 QDoubleSpinBox *joints[6] = { ui->spb_j1, ui->spb_j2, ui->spb_j3,
530 ui->spb_j4, ui->spb_j5, ui->spb_j6 };
531 for (int i = 0; i < 6 && i < static_cast<int>(sol.size()); ++i)
532 joints[i]->setValue(sol[i]);
533
534 const CollisionState collision = checkSelfCollision(cfg, robot, ik.best().joints);
535 ui->lbl_tester_status->setText(
536 tr("IK: %1 solution(s) — %2")
537 .arg(static_cast<int>(ik.solutions.size()))
538 .arg(statusToString(ik.status))
539 + collisionSuffix(collision));
