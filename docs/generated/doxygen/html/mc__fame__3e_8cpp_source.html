153bool Frame3E::checkErrorStatus(QByteArray &data) {
154 const int resp_status_index = 9;
155 quint16 end_code = convert_uint16_FromBytes(data, resp_status_index, true);
156
157 if (end_code == 0x0000) {
158 m_last_error = "Status OK";
159 return true;
160 }
161
162 m_last_error = QString("Reponse error, error code: 0x%1, frame: %2")

171void Frame3E::make_send_data(Context_Mc3E* ctx, QByteArray &data) {
172 QByteArray send_frame;
173 send_frame.clear();
174 // sub header -> big endian - 2 bytes
175 appendToByteArray_uint16(send_frame, ctx->m_subHeader, false);
176 // network - 1 byte
177 send_frame.append(ctx->m_network);
178 // pc - ff - 1 byte
179 send_frame.append(ctx->m_pc);
180 // module io number - little endian - 2 byte
181 appendToByteArray_uint16(send_frame, ctx->m_destModuleIo, true);
182 // module station number - 1 byte
183 send_frame.append(ctx->m_multiStation);
184
185 // add request size
186 // (monitoring time + request data) length - little endian - 2 byte
187 // m_current_data_len = 2 + data.length();
188 appendToByteArray_uint16(send_frame, 2 + data.length(), true);
189 // add monitoring time
190 appendToByteArray_uint16(send_frame, ctx->m_monitoringTime, true);
191
192 // append request data
193 send_frame.append(data);
194 data.clear();
195 data.append(send_frame);
196}

314RtCode Frame3E::parse_read_bit(vc::device::MCRequest *request, Context_Mc3E* ctx, QByteArray &data) {
315 if (data.size() < 9) {
316 m_last_error = "total receive len under 9 bytes";
317 return RtCode::ResponseInvalid;
318 }
319
320 if (!checkErrorStatus(data)) {
321 return RtCode::ResponseError;
322 }
323
324 const int resp_data_index = 11;
325 quint16 data_len = convert_uint16_FromBytes(data, 7, true) - 2;
326
327 if ((data.size() - 11) < data_len) {
328 LOG_USER_ERR << "parse read bit waiting for receive full frame" << data_len << data.size();
329 return RtCode::WaitingReceive;
330 }
331
332 QList<quint8> value;
333 int device_count = 0;
334 for (int index=0;index<data_len;index++) {
335 quint8 value_byte = data.at(index + resp_data_index);
336 value.append( ( (value_byte & 0x10) >> 4 ) ); // 0x10 = 0001 0000
337 device_count += 2;
338 if (device_count >= request->m_amount) {
339 break;
340 }
341 value.append( (value_byte & 0x01) ); // 0x01 = 0000 0001
342 }
343
344 McDeviceMap *dv_map = ctx->getDeviceMap();
345 if (request->m_device_type == 'M') {
346 for (int idx = 0; idx < value.size(); idx++) {
347 dv_map->device_map_m[request->m_start_address + idx] = value.at(idx);
348 // qInfo() << "Value at" << QString::number(request->m_start_address + idx)
349 // << "is " << value.at(idx);
350 }
351 }
352
353 return RtCode::ResponseOk;

411 if (data.size() < 9) {

443 // dv_map->device_map_d[request->m_start_address + idx] = static_cast<qint16>(value.at(idx));

456RtCode Frame3E::parse_write(vc::device::MCRequest *request, Context_Mc3E* ctx, QByteArray &data) {
457 // A complete 3E write response is 11 bytes: 9-byte header + the 2-byte end
458 // code that checkErrorStatus() reads at offset 9. Accepting a 9-10 byte
459 // fragment would silently read as end_code 0x0000 (success) past the
460 // available bytes, desyncing the request/response pairing.
461 if (data.size() < 11) {
462 m_last_error = "total receive len under 11 bytes";
463 return RtCode::ResponseInvalid;
464 }
465
466 if (!checkErrorStatus(data)) {
467 return RtCode::ResponseError;
468 }
469
470 return RtCode::ResponseOk;
471}
