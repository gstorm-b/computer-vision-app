97 // Posture mapping per the Nachi official manual (docs/preset_references/nachi-mz04d.md).
98 // The generic resolver classifies branches by sign: shoulder=sign(J1), elbow=sign(J3),
99 // wrist=sign(J5), with branchSign(v) = (v < 0 ? -1 : +1). The label axes below map that
100 // sign to the Nachi names:
101 // shoulder: J1 < 0 -> righty (-1), J1 > 0 -> lefty (+1)
102 // wrist: J5 < 0 -> flip (-1), J5 > 0 -> non-flip (+1)
103 // elbow: J3 < 0 -> above (-1), J3 > 0 -> below (+1)
104 // The elbow below-side (J3 > 0) is confirmed by the Arm-config ground-truth measurements
105 // in docs/preset_references/nachi-mz04d.md; the above-side (J3 < 0) is inferred by symmetry.
106 config.posture.resolver = "serial_6dof_shoulder_elbow_wrist";
107 config.posture.labels["shoulder"] = PostureLabelAxis{"righty", "lefty"};
108 config.posture.labels["elbow"] = PostureLabelAxis{"above", "below"};
109 config.posture.labels["wrist"] = PostureLabelAxis{"flip", "non-flip"};

113 config.solver.numericParameters["maxSeeds"] = 32.0;
