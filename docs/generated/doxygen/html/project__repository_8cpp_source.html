70static QByteArray serializeImage(const cv::Mat& image) {
71 if (image.empty()) {
72 return QByteArray();
73 }
74
75 std::vector<uchar> buffer;
76 std::vector<int> params = {cv::IMWRITE_PNG_COMPRESSION, 3};
77 cv::imencode(".bmp", image, buffer, params);
78
79 // convert vector std::uchar to QByteArray
80 return QByteArray(reinterpret_cast<const char*>(buffer.data()), static_cast<int>(buffer.size()));
81}

87static cv::Mat deserializeImage(const QByteArray& byteArray) {
88 if (byteArray.isEmpty()) {
89 return cv::Mat();
90 }
91
92 // convert from QByteArray to vector
93 std::vector<uchar> buffer(byteArray.constData(), byteArray.constData() + byteArray.size());
94 return cv::imdecode(buffer, cv::IMREAD_UNCHANGED);
95}

104bool ProjectRepository::createSchema(const QString& connName) {
105 auto db = QSqlDatabase::database(connName);
106 for (const QString& stmt :
107 QString(kSchema).split(';', Qt::SkipEmptyParts)) {
108 const QString s = stmt.trimmed();

172 if (!openDB(path, conn)) {

174 closeDB(conn);

178 {

181
182 // 1. Upsert project_info

184 q.prepare("INSERT OR REPLACE INTO project_info VALUES (?,?)");

186 { "name", project.name() },

199 QJsonDocument(project.toJson()).toJson(QJsonDocument::Compact);
200
201 qDebug() << "Project respository:" << json;
202
203 q.prepare("INSERT OR REPLACE INTO project_data (id, json) "
204 "VALUES (1, ?)");
205 q.addBindValue(QString::fromUtf8(json));
206 ok = q.exec() && ok;

235bool ProjectRepository::load(const QString& path, Project& out) {
236 const QString conn = "vc_load_" + path;
237 bool ok = true;
238 bool is_empty = false;
239 QMap<QString,QString> info;
240 QByteArray json;
241
242 if (!openDB(path, conn)) {
243 m_lastMsg = QSqlDatabase::database(conn).lastError().text();
244 closeDB(conn);
245 return false;
246 }
247
248 {
249 auto db = QSqlDatabase::database(conn);
250
251 // 1. Load project_info
252 QSqlQuery q(db);
253 q.exec("SELECT key, value FROM project_info");
254
255 while (q.next())
256 info[q.value(0).toString()] = q.value(1).toString();
257 // 2. Parse JSON -> Project struct
258 q.exec("SELECT json FROM project_data WHERE id = 1");
259 if (!q.next()) {
260 m_lastMsg = "project_data is empty";
261 is_empty = true;
262 }
263
264 if (!is_empty) {
265 json = q.value(0).toString().toUtf8();
266 }
267
268 const QJsonObject root =
269 QJsonDocument::fromJson(json).object();
270 out.fromJson(root);
271 out.setName(info.value("name"));
272 out.setDescription(info.value("description"));
273 out.setCreatedAt(info.value("created_at"));
274 out.setUpdatedAt(info.value("updated_at"));
275
276 // 3. Inject image BLOBs to task
277 QVector<ITask*> tasks;
278 const QMap<QString, std::shared_ptr<vc::model::ITask>>& proj_tasks = out.getCurrentTasks();
279 auto task_it = proj_tasks.cbegin();
280 while (task_it != proj_tasks.cend()) {
281 tasks.append(task_it.value().get());
282 task_it++;
283 }
284 ok = loadImages(conn, tasks);
285 if (!ok) m_lastMsg = db.lastError().text();
286 }
287
288 if (is_empty) {
289 closeDB(conn);

291 }
292
293 closeDB(conn);

307bool ProjectRepository::saveImages(const QString& connName,
308 const QVector<ITask*>& tasks) {
309 auto db = QSqlDatabase::database(connName);
310 QSqlQuery q(db);
311 q.exec("DELETE FROM project_images");
312
313 q.prepare("INSERT INTO project_images "
314 "(task_id, image_name, image_data) "
315 "VALUES (?, ?, ?)");
316
317 for (const auto& task : tasks) {
318 QMap<QString, cv::Mat> image_map = task->getTaskImageMap();
319
320 auto map_it = image_map.cbegin();
321 while (map_it != image_map.cend()) {
322 q.addBindValue(task->id());
323 q.addBindValue(map_it.key());
324 q.addBindValue(serializeImage(map_it.value()));
325 if (!q.exec()) {
326 m_lastMsg = q.lastError().text();
327 return false;
328 }
329 map_it++;
330 }
331 }
332 return true;

339bool ProjectRepository::loadImages(const QString& connName,

342 QSqlQuery q(db);
343 q.prepare("SELECT image_name, image_data "
344 "FROM project_images WHERE task_id = ?");
345
346 for (auto& task : tasks) {
347 q.addBindValue(task->id());
348 if (!q.exec()) {
349 m_lastMsg = q.lastError().text();
350 return false;
351 }
352
353 QMap<QString, cv::Mat> iamge_map;
354 while (q.next()) {
355 const QString image_name = q.value(0).toString();
356 const auto blob = q.value(1).toByteArray();
357 cv::Mat image = deserializeImage(blob);
358 iamge_map.insert(image_name, image);
359 }
360
361 task->loadTaskImageMap(iamge_map);
362 }
