35
36 if (m_active) {

38 syncRuntimeState();

41
42 m_active = true;

88bool VisionTcpipDeviceBase::pushRequest(IRequest *request) {
89 if (!request || request->type() != RequestType::Request_VisionOutput) {
90 return false;
91 }
92 if (!m_mainSocket || m_mainSocket->state() != QAbstractSocket::ConnectedState) {
93 LOG_DEV_INFO << "VisionTcpipDeviceBase: drop request, no main link";
94 return false;
95 }
96
97 VisionOutputRequest *vreq = static_cast<VisionOutputRequest*>(request);
98
99 // Phase 2: advisory robot-kinematics reachability check on the pick poses.
100 // Runs before the write; it only logs / emits, never blocks the send.
101 if (vreq->kind() == VisionOutputRequest::Result) {
102 runKinematicCheck(vreq->positions());
103 }
104
105 QByteArray payload = vreq->buildPayload();
106
107 qint64 written = m_mainSocket->write(payload);
108 if (written != payload.size()) {
109 LOG_DEV_ERR << "VisionTcpipDeviceBase: short write" << written << "/" << payload.size();
110 return false;
111 }
112 m_mainSocket->flush();
113
114 ++m_diagnostics.resultPayloadsSent;
115 emit resultSent(payload);
116 return true;
117}

146QString resolveAssetPath(const QString &relative) {
147 const QFileInfo direct(relative);
148 if (direct.exists()) return direct.absoluteFilePath();
149
150 QStringList roots;
151 roots << QDir::currentPath() << QCoreApplication::applicationDirPath();
152 QDir walker(QCoreApplication::applicationDirPath());
153 for (int i = 0; i < 8 && walker.cdUp(); ++i)
154 roots << walker.absolutePath();
155 roots.removeDuplicates();
156
157 for (const QString &root : roots) {
158 const QFileInfo candidate(QDir(root).filePath(relative));
159 if (candidate.exists()) return candidate.absoluteFilePath();
160 }
161 return {};
162}

191const RobotKinematics::MeshCollisionProfile *nachiSimplifiedMeshProfile() {
192 static const std::optional<RobotKinematics::MeshCollisionProfile> profile =
193 []() -> std::optional<RobotKinematics::MeshCollisionProfile> {
194 // Prefer the copy deployed next to the binary; fall back to the source tree.
195 QString path;
196 const QFileInfo deployed(QDir(QCoreApplication::applicationDirPath())
197 .filePath(QString::fromLatin1(kSimplifiedMeshProfileDeployedRel)));
198 if (deployed.exists())

216 }();
217 return profile ? &(*profile) : nullptr;

230void VisionTcpipDeviceBase::runKinematicCheck(const QVector<VisionOutputPosition> &positions) {
231 const RobotKinematicCheckConfig kc = kinematicCheckConfig();
232 if (!kc.enabled) return;
233
234 RobotKinematics::SerialRobotConfig robotCfg;
235 if (!buildRobotConfig(kc, robotCfg)) {
236 LOG_DEV_ERR << "VisionOutput kinematic check: unknown preset"
237 << kc.presetName << "- check skipped";
238 return;
239 }
240
241 const RobotKinematics::SerialRobotKinematics solver(robotCfg);
242
243 // Mesh self-collision is optional and only attempted when both the operator
244 // enabled it and the Nachi mesh profile is available.
245 const RobotKinematics::MeshCollisionProfile *meshProfile =
246 kc.collisionCheckEnabled ? nachiSimplifiedMeshProfile() : nullptr;
247
248 int reachable = 0;
249 int colliding = 0;
250 for (const VisionOutputPosition &p : positions) {
251 RobotKinematics::IKRequest req;
252 req.targetPose = RobotKinematics::Pose::fromXYZRPY_mm_deg(p.x, p.y, p.z, 180.0, 0.0, p.r);
253 req.tool = RobotKinematics::ToolId{kPickingToolId};
254
255 const RobotKinematics::IKResult ik = solver.solve(req);
256 if (!ik.ok()) {
257 LOG_USER_WARN << "VisionOutput kinematic check: pose not reachable"
258 << "(" << p.toString() << ")";
259 continue;
260 }
261 ++reachable;
262
263 if (!meshProfile) continue;
264
265 RobotKinematics::MeshCollisionCheckRequest collisionReq;
266 collisionReq.joints = ik.best().joints;
267 const RobotKinematics::CollisionCheckResult collision =
268 RobotKinematics::CollisionBackends::checkMesh(robotCfg, *meshProfile, collisionReq);
269 if (!collision.ok()) {
270 LOG_DEV_ERR << "VisionOutput collision check: backend error -"
271 << QString::fromStdString(collision.message);
272 meshProfile = nullptr; // stop retrying for this batch
273 } else if (collision.hasCollision) {
274 ++colliding;
275 LOG_USER_WARN << "VisionOutput collision check: self-collision at pose"
276 << "(" << p.toString() << ")";
277 }
278 }
279
280 if (kc.collisionCheckEnabled) {
281 LOG_DEV_DEBUG << "VisionOutput kinematic check:" << reachable << "/" << positions.size()
282 << "reachable," << colliding << "in self-collision";
283 }
284 emit kinematicCheckResult(positions.size(), reachable);
285}

444void VisionTcpipDeviceBase::handleHeartbeatPayload(const QByteArray &payload) {
445 // Valid form: "ack,{msg_count}."
446 if (!payload.startsWith(VISION_OUTPUT_HB_ACK_PREFIX) || !payload.endsWith('.')) {
447 declareLostConnection(QString("Invalid heartbeat reply format: %1")
448 .arg(QString::fromUtf8(payload)));
449 return;
450 }
451
452 QByteArray middle = payload.mid(qstrlen(VISION_OUTPUT_HB_ACK_PREFIX),
453 payload.size() - qstrlen(VISION_OUTPUT_HB_ACK_PREFIX) - 1);
454 bool ok = false;
455 int parsed = middle.toInt(&ok);
456 if (!ok || parsed < 0 || parsed >= VISION_OUTPUT_MSG_COUNT_LIMIT) {
457 declareLostConnection(QString("Invalid heartbeat msg_count: %1")
458 .arg(QString::fromUtf8(middle)));
459 return;
460 }
461
462 if (static_cast<quint16>(parsed) != m_expectedAckCount) {
463 declareLostConnection(QString("Heartbeat msg_count mismatch, expected %1 got %2")
464 .arg(m_expectedAckCount).arg(parsed));
465 return;
466 }
467
468 m_lastAckCount = static_cast<quint16>(parsed);
469 m_hbAwaitingReply = false;
470 m_hbLastReplyTimer.restart();
471
472 // Advance the counter, wrap at 2^16.
473 m_expectedAckCount = static_cast<quint16>((m_expectedAckCount + 1) % VISION_OUTPUT_MSG_COUNT_LIMIT);
474 syncRuntimeState();
475}

481void VisionTcpipDeviceBase::onHeartbeatTick() {
482 if (!m_hbSocket || m_hbSocket->state() != QAbstractSocket::ConnectedState) {
483 return;
484 }
485
486 // Silent for too long since the last valid reply -> lost connection.
487 if (m_hbAwaitingReply && m_hbLastReplyTimer.isValid()
488 && m_hbLastReplyTimer.elapsed() > cfgHeartbeatTimeoutMs()) {
489 ++m_diagnostics.heartbeatTimeoutCount;
490 declareLostConnection(QString("Heartbeat reply timeout (%1 ms)")
491 .arg(cfgHeartbeatTimeoutMs()));
492 return;
493 }
494
495 sendHeartbeatProbe();
496}

502void VisionTcpipDeviceBase::sendDisconnectNotice() {
503 // Bounded wait so the notice clears the socket buffer before the imminent
504 // detachHeartbeatSocket()->abort(), which would otherwise discard it.
505 constexpr int kDisconnectFlushMs = 150;
506
507 if (!m_hbSocket || m_hbSocket->state() != QAbstractSocket::ConnectedState) {
508 return;
509 }
510
511 QByteArray notice(VISION_OUTPUT_HB_DISCONNECT);
512 if (m_hbSocket->write(notice) != notice.size()) {
513 LOG_DEV_ERR << "VisionTcpip: short write on disconnect notice" << name();
514 return;
515 }
516 m_hbSocket->flush();
517 m_hbSocket->waitForBytesWritten(kDisconnectFlushMs);
518 LOG_DEV_INFO << "VisionTcpip sent disconnect notice" << name();
519}

525void VisionTcpipDeviceBase::sendHeartbeatProbe() {
526 if (!m_hbSocket) return;
527
528 QByteArray probe(VISION_OUTPUT_HB_MESSAGE);
529 m_hbSocket->write(probe);
530 m_hbSocket->flush();
531
532 m_hbAwaitingReply = true;
533 // The timer measures "time since the last valid reply" (or since the first
534 // probe when no reply has arrived yet). It must NOT be restarted on every
535 // probe, otherwise elapsed() never exceeds the timeout.
536 if (!m_hbLastReplyTimer.isValid()) {
537 m_hbLastReplyTimer.start();
538 }
539 syncRuntimeState();
540}
