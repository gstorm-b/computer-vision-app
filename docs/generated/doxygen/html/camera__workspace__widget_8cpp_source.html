124 // Render an ROI rect as a short status string, or "Not set" when empty.

126 if (r.width() <= 0.0 || r.height() <= 0.0)
127 return tr("Not set");
128 QString text = tr("ROI %1×%2 @ (%3, %4)")
129 .arg(qRound(r.width())).arg(qRound(r.height()))
130 .arg(qRound(r.x())).arg(qRound(r.y()));
131 if (!hasImage)
132 text += tr(" (no image)");
133 return text;
134 };
135
136 {

138 row.useCheck->setChecked(useWorkspace);

140 row.statusLabel->setText(roiText(roi));
