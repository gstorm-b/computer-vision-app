185 static QtVariantProperty *addPropertyToBrowser(
186 const QMetaObject &meta,
187 QMetaProperty &prop,
188 QVariant &value,
189 QtVariantPropertyManager *manager,
190 QtTreePropertyBrowser *browser)
191 {
192 const QString propName = QString::fromLatin1(prop.name());
193 if (propName == QLatin1String("objectName")) return nullptr;
194
195 QtVariantProperty *variantProp = nullptr;
196
197 if (prop.isEnumType()) {
198 variantProp = manager->addProperty(QtVariantPropertyManager::enumTypeId(), propName);
199
200 QStringList enumNames;
201 const QMetaEnum metaEnum = prop.enumerator();
202 enumNames.reserve(metaEnum.keyCount());
203 for (int j = 0; j < metaEnum.keyCount(); ++j) {
204 enumNames << QCoreApplication::translate(meta.className(),
205 metaEnum.key(j));
206 }
207 variantProp->setAttribute(QLatin1String("enumNames"), enumNames);
208 variantProp->setValue(value.toInt());
209 } else {
210 variantProp = manager->addProperty(value.userType(), propName);
211 if (variantProp) variantProp->setValue(value);
212 }
213
214 if (!variantProp) return nullptr;
215
216 // Display name override
217 const int nameIdx = meta.indexOfClassInfo(
218 QStringLiteral("%1_name").arg(propName).toUtf8());
219 if (nameIdx != -1) {
220 const QString displayName = QString::fromUtf8(meta.classInfo(nameIdx).value());
221 if (!displayName.isEmpty())
222 variantProp->setDisplayName(displayName);
223 }
224
225 // Numeric range (stored as strings in Q_CLASSINFO)
226 const int minIdx = meta.indexOfClassInfo(
227 QStringLiteral("%1_min").arg(propName).toUtf8());
228 if (minIdx != -1)
229 variantProp->setAttribute(QStringLiteral("minimum"),
230 QString::fromUtf8(meta.classInfo(minIdx).value()).toDouble());
231
232 const int maxIdx = meta.indexOfClassInfo(
233 QStringLiteral("%1_max").arg(propName).toUtf8());
234 if (maxIdx != -1)
235 variantProp->setAttribute(QStringLiteral("maximum"),
236 QString::fromUtf8(meta.classInfo(maxIdx).value()).toDouble());
237
238 // Tooltip / description
239 const int tipIdx = meta.indexOfClassInfo(
240 QStringLiteral("%1_desc").arg(propName).toUtf8());
241 if (tipIdx != -1)
242 variantProp->setDescriptionToolTip(
243 QString::fromUtf8(meta.classInfo(tipIdx).value()));
244
245 if (!prop.isWritable())
246 variantProp->setEnabled(false);
247
248 if (browser)
249 browser->addProperty(variantProp);
250
251 return variantProp;
252 }
