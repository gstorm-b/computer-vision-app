29[[maybe_unused]] static QString CameraTypeToString(CameraType t) {
30 switch(t) {
31 case vc::device::CameraType::Realsense:
32 return CAM_TYPE_REALSENSE;
33 case vc::device::CameraType::BaslerGigE:
34 return CAM_TYPE_BASLER_GIGE;

45[[maybe_unused]] static CameraType CameraTypeFromString(QString t) {
