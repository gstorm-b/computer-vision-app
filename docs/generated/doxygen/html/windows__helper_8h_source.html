111 QPixmap pixmap(const QSize &size, QIcon::Mode mode, QIcon::State state) override
112 {
113 const QString resolvedPath = themedPath();
114 const QSize requestedSize = size.isValid() ? size : QSize(m_intent, m_intent);
115 const QString cacheKey = QStringLiteral("themed-svg:%1:%2:%3:%4:%5x%6")
116 .arg(styleKey(),
117 resolvedPath,
118 QString::number(static_cast<int>(mode)),
119 QString::number(static_cast<int>(state)),
120 QString::number(requestedSize.width()),
121 QString::number(requestedSize.height()));
122
123 QPixmap cached;
124 if (QPixmapCache::find(cacheKey, &cached))
125 return cached;
126
127 QIcon icon(resolvedPath);
128 icon.addPixmap(icon.pixmap(m_intent));
129 const QPixmap pm = icon.pixmap(requestedSize, mode, state);
130 QPixmapCache::insert(cacheKey, pm);
131 return pm;
132 }

136 void paint(QPainter *painter, const QRect &rect,
137 QIcon::Mode mode, QIcon::State state) override
138 {
139 if (!painter)
140 return;
141 painter->drawPixmap(rect, pixmap(rect.size(), mode, state));
142 }
143
146 QSize actualSize(const QSize &size, QIcon::Mode mode, QIcon::State state) override
147 {
148 const QIcon icon(themedPath());

150 }
