494void CameraMappingWidget::applyDuplicateWarnings()
495{
496 QSet<QString> seenCameraIds;
497
498 for (int i = 0; i < count(); ++i) {
499 QListWidgetItem *item = this->item(i);
500 if (item == m_addRowItem) continue;
501
502 CameraRowWidget *rowWidget = qobject_cast<CameraRowWidget*>(itemWidget(item));
503 if (!rowWidget) continue;
504
505 const QString cameraId = rowWidget->nameWidget->userData();
506 const bool duplicate = !cameraId.isEmpty() && seenCameraIds.contains(cameraId);
507 const QString tooltip = duplicate
508 ? tr("This camera is already mapped by an earlier row. "
509 "Choose a different camera or remove the duplicate entry.")
510 : QString();
511
512 setDuplicateWarning(rowWidget, duplicate, tooltip);
513
514 if (!cameraId.isEmpty() && !duplicate)
515 seenCameraIds.insert(cameraId);
516 }
517}

560QList<int> CameraMappingWidget::getAvailableNumbers() const {
561 QList<int> available;
562 QMap<int, QString> currentMap = getCurrentMapping();
563 QSet<int> usedNumbers;
564 for (auto it = currentMap.constBegin(); it != currentMap.constEnd(); ++it) {
565 usedNumbers.insert(it.key());
566 }
567
568 for (int i = 1; i <= m_limitNumber; ++i) {
569 if (!usedNumbers.contains(i)) {
570 available.append(i);
571 }
572 }
573 return available;
574}
