43void PatternThumbnailView::setPattern(const QPixmap &image, const QPointF &pick,
44 double pickAngleDeg,
45 double boxW, double boxH, double boxDist,
46 double boxAngleDeg, bool showBoxes) {
47 m_pick = pick;
48 m_pickAngle = pickAngleDeg;
49 m_boxW = boxW;
50 m_boxH = boxH;
51 m_boxDist = boxDist;
52 m_boxAngle = boxAngleDeg;
53 // A jaw pair with no width or height is unset rather than degenerate; drawing it would
54 // put two labelled dots on the image and imply geometry the pattern does not have.
55 m_showBoxes = showBoxes && boxW > 0.0 && boxH > 0.0;
56
57 m_pixmapItem->setPixmap(image);
58 m_scene->setSceneRect(image.isNull() ? QRectF() : QRectF(image.rect()));
59
60 m_autoFit = true;
61 resetView();
62}

81void PatternThumbnailView::drawForeground(QPainter *painter, const QRectF &rect) {
82 QGraphicsView::drawForeground(painter, rect);
83 if (m_pixmapItem->pixmap().isNull()) return;
84
85 painter->save();
86 // Drop the view transform so the overlay is drawn in viewport pixels. This is what
87 // keeps the gizmo a constant on-screen size at any zoom, exactly as on the wizard
88 // canvas — leaving the transform in place would scale the arrows with the image.
89 painter->resetTransform();
90
91 const QPointF pw = mapFromScene(m_pick);
92 const double scale = transform().m11();
93
94 // The jaw boxes are image geometry, so they are pre-scaled and DO grow with the zoom.
95 if (m_showBoxes) {
96 pick_overlay::drawJawPair(*painter, pw,
97 m_boxW * scale, m_boxH * scale,
98 m_boxDist * scale, m_boxAngle);
99 }
100 // Gizmo before the centre marker: both axes run through the pick point and the marker
101 // is what covers the crossing.
102 pick_overlay::drawOrientationGizmo(*painter, pw, m_pickAngle,
103 /*showKnob=*/false, /*knobHeld=*/false);
104 pick_overlay::drawPickCrosshair(*painter, pw);
105
106 painter->restore();
107}

109void PatternThumbnailView::wheelEvent(QWheelEvent *event) {
110 if (m_pixmapItem->pixmap().isNull()) return;
111
112 const double factor = (event->angleDelta().y() > 0) ? kZoomStep : 1.0 / kZoomStep;
113 const double current = transform().m11();
114 const double target = current * factor;
115 if (target < kMinZoom || target > kMaxZoom) return;
116
117 m_autoFit = false; // the user has chosen a framing; stop refitting on resize
118 scale(factor, factor);
119 event->accept();
120}
