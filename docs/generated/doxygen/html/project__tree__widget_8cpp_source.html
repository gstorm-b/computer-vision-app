36static ChipInfo chipForDevice(const vc::device::IDevice *device) {
37 if (vc::device::isVirtualDevice(device)) {
38 return { "VIRT", QStringLiteral("state.warning") };
39 }
40
41 switch (device ? device->deviceType() : vc::device::DeviceType::UserType) {
42 case vc::device::DeviceType::Camera: return { "CAM", QStringLiteral("device.camera") };
43 case vc::device::DeviceType::PLC: return { "PLC", QStringLiteral("device.plc") };
44 default: return { "DEV", QStringLiteral("device.robot") };
45 }

69void ProjectTreeDelegate::paint(QPainter *painter,
70 const QStyleOptionViewItem &option,
71 const QModelIndex &index) const
72{
73 const QString chip = index.data(TreeItemRole::ChipText).toString();
74 const QString chipToken = index.data(TreeItemRole::ChipColor).toString();
75
76 QStyleOptionViewItem opt = option;
77 initStyleOption(&opt, index);
78
79 if (chip.isEmpty()) {
80 QStyledItemDelegate::paint(painter, opt, index);
81 return;
82 }
83
84 // Shrink text rect to leave room for the chip on the right
85 opt.rect.adjust(0, 0, -(kChipWidth + kChipMargin * 2), 0);
86 QStyledItemDelegate::paint(painter, opt, index);
87
88 // Draw chip badge
89 const int cy = option.rect.top() + (option.rect.height() - kChipHeight) / 2;
90 QRect chipRect(option.rect.right() - kChipWidth - kChipMargin,
91 cy, kChipWidth, kChipHeight);
92
93 painter->save();
94 painter->setRenderHint(QPainter::Antialiasing);
95
96 const QString tokenValue = ThemeManager::tokenValue(
97 chipToken, ThemeManager::instance()->isDark());
98 QColor base(tokenValue);
99 if (!base.isValid())
100 base = option.palette.color(QPalette::Highlight);
101 QColor bg = base;
102 bg.setAlpha(30);
103
104 painter->setPen(Qt::NoPen);
105 painter->setBrush(bg);
106 painter->drawRoundedRect(chipRect, 3, 3);
107
108 QPen pen(base);
109 pen.setWidthF(0.8);
110 painter->setPen(pen);
111 painter->setBrush(Qt::NoBrush);
112 painter->drawRoundedRect(chipRect, 3, 3);
113
114 painter->setPen(base);
115 QFont f = painter->font();
116 f.setPointSizeF(6.5);
117 f.setWeight(QFont::Bold);
118 f.setLetterSpacing(QFont::AbsoluteSpacing, 0.5);
119 painter->setFont(f);
120 painter->drawText(chipRect, Qt::AlignCenter, chip);
121
122 painter->restore();
123}

448void ProjectTreeWidget::onContextNewTask()
449{
450 if (!m_project) return;
451
452 bool ok = false;
453 QString name;
454 while (true) {
455 name = QInputDialog::getText(this, tr("New Task"),
456 tr("Task name:"), QLineEdit::Normal,
457 QString(), &ok);
458 if (!ok) return;
459
460 name = name.trimmed();
461 if (name.isEmpty()) {
462 QMessageBox::warning(this, tr("Error"), tr("Task name cannot be empty."));
463 continue;
464 }
465 if (m_project->isTaskNameOccupied(name)) {
466 QMessageBox::warning(this, tr("Error"), tr("Task name already exists."));
467 continue;
468 }
469 break;
470 }
471
472 auto *task = new vc::model::TaskLocalization(name);
473 m_project->addTask(task); // triggers tasksChanged → refreshTree via MainWindow
474 LOG_USER_INFO << QString("New task created: %1 (%2)").arg(name, task->id());
475}
