140 const MsgErrorState SendMsg(QByteArray &buffer) override {

170 const MsgErrorState ReceiveMsg(QByteArray &buffer, int wait_buffer = 1) override {
171 m_socket->waitForReadyRead(wait_buffer);
172
173 if (m_socket->bytesAvailable() < 1) {
174 m_total_wait_time += wait_buffer;
175 if (m_total_wait_time > m_config.m_responseTimeout) {
176 m_error_state = MsgErrorState::ResponseTimeout;
177 return m_error_state;
178 }
179
180 m_error_state = MsgErrorState::BufferEmpty;
181 return m_error_state;
182 }
183
184 m_total_wait_time = 0;
185 QByteArray read_bytes = m_socket->readAll();
186 buffer.append(read_bytes);
187 m_error_state = MsgErrorState::NoError;
188 return m_error_state;
189 }
