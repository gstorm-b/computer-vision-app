322void MainWindow::saveToFile(const QString &path, bool saveAs) {
323 const bool isNew = (!QFile::exists(path) || saveAs);
324 const QString timeStr = QDateTime::currentDateTime().toString(Qt::ISODate);
325
326 if (isNew) {
327 if (!ProjectRepo::createNew(path, m_project->name(), timeStr)) {
328 QMessageBox::critical(this,
329 tr("Save Project error"),
330 tr("Cannot create Project file: %1")
331 .arg(ProjectRepo::lastMsg()));
332 return;
333 }
334 }
335
336 QFileInfo finfo(path);
337 if (finfo.completeSuffix() != QLatin1String("vproj") || !finfo.isAbsolute()) {
338 QMessageBox::critical(this, tr("Save Project error"), tr("File format invalid!"));
339 return;
340 }
341
342 m_project->setUpdatedAt(timeStr);
343 if (!ProjectRepo::save(path, *m_project)) {
344 QMessageBox::critical(this,
345 tr("Save Project error"),
346 tr("Cannot save Project: %1").arg(ProjectRepo::lastMsg()));
347 return;
348 }
349
350 LOG_USER_INFO << "Project saved: " << path;
351 m_projectFilePath = path;
352 m_isDirty = false;
353 updateWindowTitle();
354 addRecentPath(path);
355}

366bool MainWindow::maybeSave() {
367 if (!(m_isDirty || (m_projectFilePath.isEmpty() && m_project)))
368 return true;
369
370 const auto ret = QMessageBox::question(
371 this,
372 tr("Would you want to change?"),
373 tr("Project [%1] has some changed.\nWould you want to save the project?")
374 .arg(m_project ? m_project->name() : "error"),
375 QMessageBox::Save | QMessageBox::Discard | QMessageBox::Cancel);
376
377 if (ret == QMessageBox::Save)
378 return (onSaveProject(), !m_isDirty);
379
380 if (ret == QMessageBox::Discard)
381 return true;
382
383 return false;
384}

460void MainWindow::closeAllDockTabs() {
461 // remove all devices docks
462 QMap<QString, UIDeviceEntry>::const_iterator devIt = m_deviceEntries.cbegin();
463 while (devIt != m_deviceEntries.cend()) {
464 if (devIt.value().dockWidget) {
465 m_dockManager->removeDockWidget(devIt.value().dockWidget);
466 }
467 ++devIt;
468 }
469 m_deviceEntries.clear();
470
471 // remove all tasks docks and its children docks
472 QMap<QString, UITaskEntry>::const_iterator taskIt = m_taskEntries.cbegin();
473 while (taskIt != m_taskEntries.cend()) {
474 QMap<QString, UITaskEntry>::const_iterator childIt = taskIt.value().children.cbegin();
475 while (childIt != taskIt.value().children.cend()) {
476 if (childIt.value().dockWidget) {
477 m_dockManager->removeDockWidget(childIt.value().dockWidget);
478 }
479 ++childIt;
480 }
481
482 if (taskIt.value().dockWidget) {
483 m_dockManager->removeDockWidget(taskIt.value().dockWidget);
484 }
485 ++taskIt;
486 }
487 m_taskEntries.clear();
488}
