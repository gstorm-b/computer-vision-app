30static QString readyLampStateForTask(vc::model::TaskState state)
31{
32 switch (state) {
33 case vc::model::TaskState::Ready:
34 case vc::model::TaskState::RunningCycle:
35 return QStringLiteral("on");
36 case vc::model::TaskState::CommissionStarting:
37 case vc::model::TaskState::Commission:
38 case vc::model::TaskState::RuntimeStarting:
39 case vc::model::TaskState::Recovering:
40 case vc::model::TaskState::Stopping:
41 return QStringLiteral("warn");
42 case vc::model::TaskState::Faulted:
43 return QStringLiteral("error");
44 case vc::model::TaskState::Idle:
45 default:
46 return QStringLiteral("off");
47 }
48}

360void LocalizationTaskWidget::updateStatusLamps()
361{
362 // States: [0]=READY, [1]=CAM, [2]=PLC, [3]=OUT
363 // The localization task has no robot; the fourth lamp tracks the assigned
364 // Vision Output device instead.
365 bool hasCam = false, hasPlc = false, hasOutput = false;
366 QString readyLampState = QStringLiteral("off");
367
368 if (m_localizeTask && m_localizeTask->project()) {
369 auto *mgr = m_localizeTask->project()->deviceManager().get();
370 for (const QString &id : m_localizeTask->assignedDeviceIds()) {
371 auto dev = mgr->deviceById(id);
372 if (!dev) continue;
373 switch (dev->deviceType()) {
374 case vc::device::DeviceType::Camera: hasCam = true; break;
375 case vc::device::DeviceType::PLC: hasPlc = true; break;
376 case vc::device::DeviceType::VisionOutput: hasOutput = true; break;
377 default: break;
378 }
379 }
380 }
381
382 auto repolish = [](QWidget *w) {
383 w->style()->unpolish(w);
384 w->style()->polish(w);
385 w->update();
386 };
387
388 if (m_localizeTask) {
389 readyLampState = readyLampStateForTask(m_localizeTask->taskState());
390 }
391
392 for (int i = 0; i < m_statusLamps.size(); ++i) {
393 QString state = QStringLiteral("off");
394 if (i == 0) {
395 state = readyLampState;
396 } else if (i == 1) {
397 state = hasCam ? QStringLiteral("on") : QStringLiteral("off");
398 } else if (i == 2) {
399 state = hasPlc ? QStringLiteral("on") : QStringLiteral("off");
400 } else if (i == 3) {
401 state = hasOutput ? QStringLiteral("on") : QStringLiteral("off");
402 }
403 m_statusLamps[i].dot->setProperty("lampState", state);
404 m_statusLamps[i].label->setProperty("lampState", state);
405 repolish(m_statusLamps[i].dot);
406 repolish(m_statusLamps[i].label);
407 }
408}

460void LocalizationTaskWidget::onTaskDevicesChanged() {
461 QStringList deviceIds = m_localizeTask->assignedDeviceIds();
462 // Iterate from the top down: removeWidget() shifts higher indices, so a
463 // forward loop would skip the page that slides into a just-vacated slot
464 // (two consecutive removed device pages left a stale widget behind).
465 for (int idx = ui->content_stack->count() - 1; idx >= 0; --idx) {
466 QWidget *w = ui->content_stack->widget(idx);
467 IDeviceWidget *dw = qobject_cast<IDeviceWidget*>(w);
468 if (dw) {
469 const QString deviceId = dw->deviceId();
470 // Only tear down pages for devices that are no longer assigned.
471 // These calls used to run for every page found — including still-live
472 // ones — desyncing m_devicePages from the stack and orphaning pages.
473 if (!deviceIds.contains(deviceId)) {
474 ui->content_stack->removeWidget(w);
475 w->deleteLater();
476 removePropertyBrowserWidget(dw->getPropertyBrowser());
477 m_devicePages.remove(deviceId);
478 }
479 }
480 }
481
482 // Sync runners with the new assigned-device set. TaskRunner will:
483 // - register + start a runner for any newly added device
484 // - unregister + stop the runner for any removed device
485 if (m_localizeTask) m_localizeTask->resyncDevices();
486
487 rebuildDeviceNav();
488}

490// ──────────────────────────────────────────────────────────────────────────────
491// Device nav rebuild
492// ──────────────────────────────────────────────────────────────────────────────
499void LocalizationTaskWidget::rebuildDeviceNav()
500{
501 // Clear existing items
502 QLayout *lay = m_devListWidget->layout();
503 while (QLayoutItem *it = lay->takeAt(0)) {
504 if (QWidget *w = it->widget()) w->deleteLater();
505 delete it;
506 }
507 m_navItems.clear();
508
509 if (!m_localizeTask || !m_localizeTask->project()) return;
510
511 auto *mgr = m_localizeTask->project()->deviceManager().get();
512
513 // Resolve and sort assigned devices by type (Camera → PLC → VisionOutput →
514 // Robot, matching the DeviceType enum order), then by name within a type, so
515 // the nav order is grouped and stable across rebuilds.
516 QList<std::shared_ptr<vc::device::IDevice>> devices;
517 for (const QString &devId : m_localizeTask->assignedDeviceIds()) {
518 if (auto device = mgr->deviceById(devId))
519 devices.append(device);
520 }
521 std::sort(devices.begin(), devices.end(),
522 [](const std::shared_ptr<vc::device::IDevice> &a,
523 const std::shared_ptr<vc::device::IDevice> &b) {
524 if (a->deviceType() != b->deviceType())
525 return a->deviceType() < b->deviceType();
526 return a->name().localeAwareCompare(b->name()) < 0;
527 });
528
529 const bool anyDevice = !devices.isEmpty();
530
531 for (const auto &device : devices) {
532 const QString devId = device->id();
533
534 auto *item = new DeviceNavItemWidget(m_devListWidget);
535 item->setDevice(device);
536 connect(item, &DeviceNavItemWidget::activated,
537 this, &LocalizationTaskWidget::onDeviceNavClicked);
538
539 m_navItems.insert(devId, item);
540 lay->addWidget(item);
541 }
542
543 if (!anyDevice) {
544 auto *emptyLbl = new QLabel(tr("No devices assigned.\nUse [+] to add."),
545 m_devListWidget);
546 emptyLbl->setProperty("navPart", QStringLiteral("empty"));
547 emptyLbl->setAlignment(Qt::AlignCenter);
548 QFont f = emptyLbl->font();
549 f.setPointSizeF(8.5);
550 emptyLbl->setFont(f);
551 emptyLbl->setWordWrap(true);
552 emptyLbl->setContentsMargins(4, 8, 4, 8);
553 lay->addWidget(emptyLbl);
554 }
555
556 static_cast<QVBoxLayout*>(lay)->addStretch();
557 refreshNavItemStyles();
558 updateStatusLamps();
559 wireDeviceNavDots();
560}

562// ──────────────────────────────────────────────────────────────────────────────
563// Device nav connection dots
564// ──────────────────────────────────────────────────────────────────────────────
572void LocalizationTaskWidget::wireDeviceNavDots()
573{
574 // Drop previous subscriptions before re-binding (dots may have been
575 // recreated by a rebuild, or runners replaced by a phase change).
576 for (const auto &conn : std::as_const(m_navDotConns))
577 QObject::disconnect(conn);
578 m_navDotConns.clear();
579
580 auto *taskRunner = m_localizeTask ? m_localizeTask->taskRunner() : nullptr;
581
582 for (auto it = m_navItems.cbegin(); it != m_navItems.cend(); ++it) {
583 const QString &devId = it.key();
584 DeviceNavItemWidget *item = it.value();
585 if (!item) continue;
586
587 auto *runner = taskRunner ? taskRunner->runnerFor(devId) : nullptr;
588 if (!runner || !runner->device()) {
589 item->setIndicatorState(QStringLiteral("off"));
590 continue;
591 }
592
593 // Seed from current status (plain enum read; corrected by the next
594 // forwarded change). The runner forwards connectStatusChanged onto the
595 // GUI thread, and the nav item is the connection context so it
596 // auto-detaches when the item is destroyed.
597 item->setConnectStatus(runner->device()->connectStatus());
598 m_navDotConns.append(connect(
599 runner, &vc::runtime::IDeviceRunner::connectStatusChanged, item,
600 [item](vc::device::ConnectStatus status) {
601 item->setConnectStatus(status);
602 }));
603 }
604}

835QWidget *LocalizationTaskWidget::getOrCreateDeviceConfigPage(const QString &deviceId)
836{
837 if (m_devicePages.contains(deviceId))
838 return m_devicePages.value(deviceId);
839
840 if (!m_localizeTask || !m_localizeTask->project()) return nullptr;
841
842 auto device = m_localizeTask->project()->deviceById(deviceId);
843 if (!device) return nullptr;
844
845 // ── Resolve the typed runner from TaskRunner ─────────────────────────
846 // The Widget never creates a worker / thread. TaskRunner owns the
847 // runner (and its QThread); we just look it up here and inject it.
848 auto *taskRunner = m_localizeTask->taskRunner();
849 vc::runtime::IDeviceRunner *runner =
850 taskRunner ? taskRunner->runnerFor(deviceId) : nullptr;
851
852 QWidget *page = DeviceWidgetFactory::createDeviceWidget(device, runner, nullptr, this);
853 if (!page) {
854 page = new QWidget(this);
855 auto *l = new QVBoxLayout(page);
856 auto *lbl = new QLabel(tr("No configuration panel available for this device."), page);
857 lbl->setAlignment(Qt::AlignCenter);
858 lbl->setStyleSheet(QStringLiteral("font-size: 11pt;"));
859 l->addWidget(lbl);
860 }
861
862 ui->content_stack->addWidget(page);
863 m_devicePages.insert(deviceId, page);
864 return page;
865}
