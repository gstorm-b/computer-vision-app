25static QtVariantProperty* addPropertyToBrowser(const QMetaObject &meta, QMetaProperty &prop, QVariant &value,
26 QtVariantPropertyManager *manager, QtTreePropertyBrowser *browser) {
27
28 QString propName = prop.name();
29 if (propName == "objectName") {
30 // ignore objectName
31 return nullptr;
32 }
33
34 QtVariantProperty *variantProp = nullptr;
35 // enum type Check
36 if (prop.isEnumType()) {
37 variantProp = manager->addProperty(QtVariantPropertyManager::enumTypeId(), propName);
38
39 // get enum names
40 QStringList enumNames;
41 QMetaEnum metaEnum = prop.enumerator();
42 for (int j = 0; j < metaEnum.keyCount(); ++j) {
43 const char* key = metaEnum.key(j);
44 QString translatedName = QCoreApplication::translate(meta.className(), key);
45 enumNames << translatedName;
46 }
47 variantProp->setAttribute(QLatin1String("enumNames"), enumNames);
48 // set enum value
49 variantProp->setValue(value.toInt());
50
51 } else {
52 // normal data types handle (int, QString, bool, QColor, v.v.)
53 int typeId = value.userType();
54 variantProp = manager->addProperty(typeId, propName);
55 if (variantProp) {
56 variantProp->setValue(value);
57 }
58 }
59
60 if (!variantProp) {
61 return nullptr;
62 }
63
64 int displayNameIdx = meta.indexOfClassInfo(QString("%1_name").arg(propName).toUtf8());
65 if (displayNameIdx != -1) {
66 const QString displayName = meta.classInfo(displayNameIdx).value();
67 if (!displayName.isEmpty()) {
68 variantProp->setDisplayName(displayName);
69 }
70 }
71
72 // --- set attributes---
73 int minIdx = meta.indexOfClassInfo(QString("%1_min").arg(propName).toUtf8());
74 if (minIdx != -1) {
75 variantProp->setAttribute("minimum", QString(meta.classInfo(minIdx).value()).toInt());
76 }
77
78 int maxIdx = meta.indexOfClassInfo(QString("%1_max").arg(propName).toUtf8());
79 if (maxIdx != -1) {
80 variantProp->setAttribute("maximum", QString(meta.classInfo(maxIdx).value()).toInt());
81 }
82
83 if (!prop.isWritable()) {
84 variantProp->setEnabled(false);
85 }
86
87 return variantProp;
88}

206void MitsubishiMcDeviceWidget::onPropertyValueChanged(QtProperty *property, const QVariant &variant) {
207 if (m_populating_browser) {
208 return;
209 }
210
211 vc::device::McContext *context = this->m_config.context();
212 vc::device::McMsgItfConfig *msg_cfg = context->msgConfig();
213
214 QString propName = property->propertyName();
215
216 const QMetaObject *idevice_meta = m_device->metaObject();
217 const QMetaObject &meta_context = context->getMetaObject();
218 const QMetaObject &meta_msg = msg_cfg->getMetaObject();
219
220 // abstract device
221 int index = idevice_meta->indexOfProperty(propName.toUtf8());
222 if (index != -1) {
223 if (propName == "name") {
224 QString new_name = variant.toString();
225 // avoid loop made by signal valueChanged
226 if (m_device->name() == new_name) {

231 LOG_USER_WARN << tr("Cannot rename device to \"%1\": the name is "
232 "already in use.").arg(new_name);
233 m_variantManager->setValue(property, m_device->name());
234 }
235 }
236 return;
237 }
238
239 // mc context
240 index = meta_context.indexOfProperty(propName.toUtf8());
241 if (index != -1) {
242 QMetaProperty mProp = meta_context.property(index);
243 mProp.writeOnGadget(context, variant);
244 // qDebug() << "Updated context: " << propName << "to" << variant;
245 this->saveConfig();
246 rebuildMonitorRanges();
247 refreshMetaSummary();
248 return;

251 // interface config
252 index = meta_msg.indexOfProperty(propName.toUtf8());
253 if (index != -1) {
254 QMetaProperty mProp = meta_msg.property(index);
255 mProp.writeOnGadget(msg_cfg, variant);

290void MitsubishiMcDeviceWidget::onConnectionStateChanged(vc::device::ConnectStatus state) {
291 updateConnectionVisual(state);
292 switch (state) {
293 // Lost/closed link → clear the cached monitor values.
294 case vc::device::ConnectStatus::Disconnected:
295 case vc::device::ConnectStatus::LostConnected:
296 case vc::device::ConnectStatus::ConnectFailed:
297 if (m_monitor_m) m_monitor_m->clearAllStatuses();
298 if (m_monitor_d) m_monitor_d->clearAllStatuses();
299 break;
300 // Enumerate the rest explicitly (no default:) so adding a ConnectStatus
301 // value surfaces a -Wswitch / C4062 warning here.
302 case vc::device::ConnectStatus::NoConnection:
303 case vc::device::ConnectStatus::Connected:
304 case vc::device::ConnectStatus::Connecting:
305 break;
306 }
307}

545void MitsubishiMcDeviceWidget::onBitWriteRequested(int address, quint8 value) {
